import { HttpClient, HttpContext, HttpParams } from '@angular/common/http';
import { inject, Injectable } from '@angular/core';
import { Observable } from 'rxjs';
import { SapTriggerRequest, SapTriggerResponse } from '@models/sap/sap-trigger.model';
import { SKIP_AUTH } from '@interceptors/auth.interceptor';
import { SKIP_AUTO_LOGOUT, SUPPRESS_GLOBAL_ERROR } from '@interceptors/error.interceptor';
import { ExternalConfigService } from 'src/app/shared/services/external-config.service';

@Injectable({ providedIn: 'root' })
export class SapService {
  private readonly http = inject(HttpClient);
  private readonly extConfigService = inject(ExternalConfigService);

  private get apiUrl(): string {
    return `${this.extConfigService.settings.sapServiceUrl}/trigger`;
  }

  /**
   * SAP AAE is a third party on a different origin, so none of this app's own request conventions
   * apply to it — all three opt-outs below are load-bearing, each for a failure confirmed live:
   *
   * - SKIP_AUTH: authInterceptor otherwise attaches this app's own AGWA JWT to every outgoing
   *   request, which sent our internal bearer token to an external host on every submit.
   * - SKIP_AUTO_LOGOUT: errorInterceptor treats any 401 as "this app's session expired" and calls
   *   authStore.logout(). SAP rejecting a request it was never going to accept is not our session
   *   expiring — without this, one SAP 401 signs the user out of AGWA mid-submit (and, because
   *   that branch returns EMPTY rather than rethrowing, WwellIrStore's own error handler never
   *   runs, so the workflow id is never recorded and the Submit spinner never clears).
   * - SUPPRESS_GLOBAL_ERROR: WwellIrStore surfaces its own sapTriggerError toast; without this the
   *   user gets two toasts for the same failure.
   */
  trigger(request: SapTriggerRequest): Observable<SapTriggerResponse> {
    const params = new HttpParams().set('native_response', 'false');
    const context = new HttpContext()
      .set(SKIP_AUTH, true)
      .set(SKIP_AUTO_LOGOUT, true)
      .set(SUPPRESS_GLOBAL_ERROR, true);
    return this.http.post<SapTriggerResponse>(this.apiUrl, request, { params, context });
  }
}

import { JwtService } from 'src/app/core/services/jwt/jwt.service';

export interface AuthUser {
  readonly upn?: string;
  readonly name?: string;
  readonly preferred_username?: string;
  readonly unique_name?: string;
  readonly exp?: number;
  readonly last_login?: number | string;
  readonly groups?: string[];
  readonly [key: string]: unknown;
}

export interface AuthState {
  readonly token: string | null;
  readonly isAuthenticated: boolean;
  readonly user: AuthUser | null;
  readonly isLoading: boolean;
  readonly error: string | null;
  readonly sessionExpired: boolean;
}

export const initialState: AuthState = {
  token: null,
  isAuthenticated: false,
  user: null,
  isLoading: false,
  error: null,
  sessionExpired: false,
};

export function selectIsTokenExpired(token: string | null, jwt: JwtService): boolean {
  return token ? jwt.isExpired(token) : true;
}

export function selectDisplayUsername(user: AuthUser | null): string {
  if (!user) return 'User';

  const username =
    user.name ||
    user.preferred_username ||
    user.upn ||
    user.unique_name;

  return typeof username === 'string' && username.trim().length > 0
    ? username.trim()
    : 'User';
}

export function selectLastLogin(user: AuthUser | null): Date | null {
  const ts = user?.last_login;
  if (!ts) return null;
  // JWT timestamps are Unix seconds; multiply to get ms for the Date constructor
  return new Date(Number(ts) * 1000);
}

/** The bare network ID (e.g. "junedalam"), stripped of the "@aramco.com" domain `upn` sometimes
 *  carries — see selectUserEmail's own doc comment on that inconsistency. Falls back to
 *  preferred_username/unique_name for tokens that don't carry a upn claim at all. */
export function selectNetworkId(user: AuthUser | null): string | null {
  const identifier = user?.upn ?? user?.preferred_username ?? user?.unique_name;

  if (typeof identifier !== 'string') return null;

  const trimmed = identifier.trim();
  if (!trimmed) return null;

  return trimmed.includes('@') ? trimmed.split('@')[0] : trimmed;
}

export function selectUserEmail(user: AuthUser | null): string | null {
  const upn = user?.upn;

  if (typeof upn !== 'string') {
    return null;
  }

  const normalizedUpn = upn.trim();

  if (!normalizedUpn) {
    return null;
  }

  return normalizedUpn.includes('@')
    ? normalizedUpn
    : `${normalizedUpn}@aramco.com`;
}

import { computed, inject } from '@angular/core';
import { HttpClient, HttpContext, HttpErrorResponse } from '@angular/common/http';
import { Router } from '@angular/router';
import { patchState, signalStore, withComputed, withMethods, withState } from '@ngrx/signals';

import { JwtService } from 'src/app/core/services/jwt/jwt.service';
import { LoaderService } from 'src/app/shared/components/global-loader/loader.service';
import { ExternalConfigService } from 'src/app/shared/services/external-config.service';
import { RbacStore } from '@store/rbac/rbac.store';
import { SKIP_AUTO_LOGOUT } from '@interceptors/error.interceptor';
import { extractHttpErrorMessage } from '@core/utils/http-error.util';
import { AuthState, initialState, selectDisplayUsername, selectIsTokenExpired, selectLastLogin, selectNetworkId, selectUserEmail } from './auth.selectors';

export const AuthStore = signalStore(
  { providedIn: 'root' },
  withState<AuthState>(initialState),
  withComputed(({ token, user }) => ({
    isTokenExpired: computed(() => selectIsTokenExpired(token(), inject(JwtService))),
    displayUsername: computed(() => selectDisplayUsername(user())),
    userEmail: computed(() => selectUserEmail(user())),
    networkId: computed(() => selectNetworkId(user())),
    lastLogin: computed(() => selectLastLogin(user())),
  })),
  withMethods((
    store,
    http = inject(HttpClient),
    router = inject(Router),
    jwt = inject(JwtService),
    loading = inject(LoaderService),
    extAppConfigService = inject(ExternalConfigService),
    rbacStore = inject(RbacStore),
  ) => ({
    async login() {
      loading.startLogin('Contacting identity service...');
      patchState(store, { isLoading: true, error: null });

      http.get(`${extAppConfigService.settings.tokenUrl}`, {
        responseType: 'text',
        withCredentials: true,
        // A failed login must not trigger the interceptor's auto-logout branch (a 401 here
        // means "bad credentials", not "session expired") — but the backend's own message
        // should still reach both the global toast and the inline error below the login button.
        context: new HttpContext().set(SKIP_AUTO_LOGOUT, true),
      })
        .subscribe({
          next: async (token) => {
            const normalizedToken = token.trim();
            const decoded = jwt.decode(normalizedToken) as Record<string, unknown>;
            localStorage.setItem('agwa_token', normalizedToken);
            patchState(store, { token: normalizedToken, isAuthenticated: true, user: decoded, isLoading: false });
            const baseUrl = extAppConfigService.settings.dailyOperationServiceUrl;
            await rbacStore.loadPermissions(`${baseUrl}/daily-operations/api/v1/rbac/permissions`, normalizedToken);
            void router.navigate(['main']).finally(() => loading.completeLogin());
          },
          error: (err: HttpErrorResponse) => {
            localStorage.removeItem('agwa_token');
            patchState(store, { ...initialState, error: extractHttpErrorMessage(err, 'Login Failed') });
            loading.fail();
          },
        });
    },

    logout() {
      localStorage.removeItem('agwa_token');
      patchState(store, initialState);
      rbacStore.reset();
      void router.navigate(['/login']);
    },

    // Token Refresh — called at app bootstrap via APP_INITIALIZER
    autoLogin() {
      const token = localStorage.getItem('agwa_token');

      if (token && !jwt.isExpired(token)) {
        const decoded = jwt.decode(token);
        patchState(store, { token, isAuthenticated: true, user: decoded, sessionExpired: false });
        return;
      }

      localStorage.removeItem('agwa_token');

      // If a token existed but is now expired, flag it so the guard can
      // redirect to login with a session-expired message.
      patchState(store, { ...initialState, sessionExpired: !!token });
    },
  })),
);

/** Prefers the backend's own `{ message }` (see ApiResponse<T> — every exad_agwa endpoint here
 *  returns that envelope on both success and error) over Angular's generic "Http failure
 *  response for ..." text, so a specific backend validation error (e.g. an Oracle constraint
 *  message) reaches the user instead of a vague fallback. */
function errorMessage(err: unknown, fallback: string): string {
  if (err instanceof HttpErrorResponse) {
    const body: unknown = err.error;
    const backendMessage = body && typeof body === 'object' ? (body as { message?: unknown }).message : undefined;
    if (typeof backendMessage === 'string' && backendMessage) return backendMessage;
    return err.message || fallback;
  }
  return err instanceof Error ? err.message : fallback;
}

/** Like errorMessage above, but prefers the SAP trigger endpoint's own `{ detail: [{ msg }] }`
 *  422 validation shape (SapValidationError) over the generic `{ message }` envelope every other
 *  exad_agwa endpoint here uses — so a real "field X is required" from SAP reaches the user
 *  instead of a vague fallback. */
function sapErrorMessage(err: unknown, fallback: string): string {
  if (err instanceof HttpErrorResponse) {
    const body: unknown = err.error;
    const detail = body && typeof body === 'object' ? (body as { detail?: unknown }).detail : undefined;
    if (Array.isArray(detail)) {
      const messages = detail
        .map(item => (item && typeof item === 'object' ? (item as { msg?: unknown }).msg : undefined))
        .filter((msg): msg is string => typeof msg === 'string' && msg.length > 0);
      if (messages.length > 0) return messages.join('; ');
    }
  }
  return errorMessage(err, fallback);
}

/** A repeat "Save as Draft" on the same IR reuses its ir_seq (see wwell-ir-payload.mapper.ts), so
 *  the response describes the *same* record, not a new one — replace it in place instead of
 *  prepending a duplicate. Falls back to prepend for a genuinely new IR (irSeq not seen yet). */
function markSubmittedForReview(list: readonly IntegratedRequirement[], id: string): IntegratedRequirement[] {
  return list.map(item => (item.id === id ? { ...item, status: 'submit-for-review' as const } : item));
}

function upsertByIrSeq(list: readonly IntegratedRequirement[], created: IntegratedRequirement): IntegratedRequirement[] {
  const existingIndex = created.irSeq != null ? list.findIndex(item => item.irSeq === created.irSeq) : -1;
  if (existingIndex === -1) return [created, ...list];
  // A brand-new record's `created` (see createIr's own buildIntegratedRequirementFromPayload
  // fallback) has no way to know the record's real approval chain (that's a separate, still-mocked
  // endpoint; see loadIrApprovalTimeline), so it always starts a fresh "just initiated" chain.
  // Harmless for a first-time INSERT (nothing to preserve yet), but would otherwise wipe out an
  // already-in-progress/approved record's real chain on every subsequent save. Keep whichever
  // chain this record already had.
  return list.map((item, index) => (index === existingIndex ? { ...created, chain: item.chain } : item));
}

      /** Trimmed {value,label} view of linkedRequestWells for the "Link to Request" picker itself —
       *  see wwell-ir-editor.component.ts's `applyLinkedRequestWell` for where the full record
       *  (looked up from linkedRequestWells by this same value) drives the auto-populate. */
      linkedRequestWellOptions: computed(() => linkedRequestWells().map(well => ({ value: String(well.prewapSeq), label: well.wellName }))),
      /** FULL_NAME option list for the Geologist Contact "Primary"/"Secondary" pickers — value and
       *  label are the same string; the real EMPNUM to persist is resolved separately via
       *  `geologistEmpNumByName` below at save time. */
      geologistOptions: computed(() => gwdOpsMembers().map(member => member.fullName)),
      /** FULL_NAME → EMPNUM, threaded into buildWwellIrBulkSavePayload's WwellIrSaveContext so
       *  cont_geologist/sub_cont_geologist persist the real employee number, not a display name. */
      geologistEmpNumByName: computed<Readonly<Record<string, number>>>(() =>
        Object.fromEntries(gwdOpsMembers().map(member => [member.fullName, member.empNum])),
      ),
      /** Inverse of geologistEmpNumByName - EMPNUM → FULL_NAME, so a detail/list fetch that comes
       *  back with exad_gwd_ir_header.cont_geologist/sub_cont_geologist (a raw EMPNUM) can be
       *  redisplayed as the name the Geologist Contact picker showed when it was saved. Without
       *  this, mergeIrDetailBlocksIntoIntegratedRequirement has no way to turn that EMPNUM back
       *  into a name and the field renders blank even though the correct value is in the DB. */
      geologistNameByEmpNum: computed<Readonly<Record<number, string>>>(() =>
        Object.fromEntries(gwdOpsMembers().map(member => [member.empNum, member.fullName])),
      ),
      activeErrors: computed(() =>
        [
          irsError(),
          linkedRequestWellsError(),
          gwdOpsMembersError(),
          createIrError(),
          sapTriggerError(),
          irDetailError(),
          irApprovalTimelineError(),
          deleteIrError(),
          attachmentError(),
        ].filter((message): message is string => message !== null),
      ),
      hasError: computed(
        () =>
          irsError() !== null ||
          linkedRequestWellsError() !== null ||
          gwdOpsMembersError() !== null ||
          createIrError() !== null ||
          sapTriggerError() !== null ||
          irDetailError() !== null ||
          irApprovalTimelineError() !== null ||
          deleteIrError() !== null ||
          attachmentError() !== null,
      ),
    }),
  ),

  withMethods((store, svc = inject(WwellIrService), sap = inject(SapService), authStore = inject(AuthStore)) => {
    // Step 3, reached only after SAP has accepted the workflow: stamp the id and status onto the
    // IR header (update_ir_sap_wf_prc — `seq` lands in sap_wf_id, `status` in sap_wf_status).
    //
    // Deliberately not called on a failed trigger: sap_wf_id/sap_wf_status describe a workflow
    // that exists in SAP, so recording one for a push SAP never accepted would leave the header
    // pointing at an id no SAP workflow was ever created under.
    function persistSapWorkflow(ir: IntegratedRequirement, irSeq: number, seq: number): void {
      const finish = (persistError: string | null) =>
        patchState(store, {
          updatingIrStatus: false,
          sapTriggerError: persistError,
          // Only a clean run — SAP accepted the workflow AND the header recorded it — moves the
          // record to submitted. This is a local reflection of the sap_wf_status just written, not
          // a separate persisted status: exad_gwd_ir_header has no status column, and the list read
          // still derives status from publish_flg alone (wwell-ir-summary.mapper.ts), so this
          // reverts on refresh until that read path surfaces sap_wf_status.
          irs: persistError === null ? markSubmittedForReview(store.irs(), ir.id) : store.irs(),
        });

      svc.updateIrSapWorkflow(irSeq, seq, WWELL_IR_SAP_WF_STATUS_SUBMITTED).subscribe({
        next: () => finish(null),
        error: (err: unknown) => finish(errorMessage(err, 'SAP workflow triggered, but failed to record it on the integrated requirement')),
      });
    }

    // Steps 1-2: mint a workflow id (get_wf_next_id_fnc), then push the workflow to SAP under it.
    // workflow_id must be minted fresh per submission and formatted as
    // `{NETWORK_ID}_{SCENARIO_ID}_{seq}` — see buildSapWorkflowId.
    function triggerSapWorkflow(ir: IntegratedRequirement, irSeq: number): void {
      const networkId = authStore.networkId() ?? '';
      svc.getNextSapWorkflowSeq().subscribe({
        next: seq => {
          sap
            .trigger({
              workflow_title: `Integrated Requirement ${ir.code} - ${ir.requestReference.wellName}`,
              initiator_network_id: networkId,
              requester_network_id: networkId,
              process_id: WWELL_IR_SAP_PROCESS_ID,
              scenario_id: WWELL_IR_SAP_SCENARIO_ID,
              workflow_id: buildSapWorkflowId(networkId, seq),
              body: [],
            })
            .subscribe({
              next: () => persistSapWorkflow(ir, irSeq, seq),
              // SAP refused the workflow, so there is nothing to stamp — surface the failure and
              // leave the header exactly as it was.
              error: (err: unknown) =>
                patchState(store, {
                  updatingIrStatus: false,
                  sapTriggerError: sapErrorMessage(err, 'Failed to trigger the SAP review workflow'),
                }),
            });
        },
        // Nothing to trigger or persist without a real id — never call SAP with a fabricated one.
        error: (err: unknown) =>
          patchState(store, { updatingIrStatus: false, sapTriggerError: errorMessage(err, 'Failed to generate a SAP workflow id') }),
      });
    }

    const loadIrs = rxMethod<void>(
      pipe(
        tap(() => patchState(store, { irsLoading: true, irsError: null })),
        switchMap(() =>
          svc.getIrs().pipe(
            tapResponse({
              next: irs => patchState(store, { irs, irsLoading: false }),
              error: (err: unknown) => patchState(store, { irsLoading: false, irsError: errorMessage(err, 'Failed to load integrated requirements') }),
            }),
          ),
        ),
      ),
    );

    const loadLinkedRequestWells = rxMethod<void>(
      pipe(
        tap(() => {
          console.log('[IR:link] loadLinkedRequestWells — fetching GET /wells/eligible-for-ir');
          patchState(store, { linkedRequestWellsLoading: true, linkedRequestWellsError: null });
        }),
        switchMap(() =>
          svc.getEligibleWellsForIr().pipe(
            tapResponse({
              next: linkedRequestWells => {
                console.log('[IR:link] loadLinkedRequestWells — response received', { count: linkedRequestWells.length, linkedRequestWells });
                patchState(store, { linkedRequestWells, linkedRequestWellsLoading: false });
              },
              error: (err: unknown) => {
                console.log('[IR:link] loadLinkedRequestWells — request failed', err);
                patchState(store, {
                  linkedRequestWellsLoading: false,
                  linkedRequestWellsError: errorMessage(err, 'Failed to load wells eligible for IR initiation'),
                });
              },
            }),
          ),
        ),
      ),
    );

    // Stable reference data (not tied to a specific IR selection, unlike loadLinkedRequestWells) —
    // loaded once at list-page init, see WwellIrComponent.ngOnInit.
    const loadGwdOpsMembers = rxMethod<void>(
      pipe(
        tap(() => patchState(store, { gwdOpsMembersLoading: true, gwdOpsMembersError: null })),
        switchMap(() =>
          svc.getGwdOpsMembers().pipe(
            tapResponse({
              next: gwdOpsMembers => patchState(store, { gwdOpsMembers, gwdOpsMembersLoading: false }),
              error: (err: unknown) =>
                patchState(store, { gwdOpsMembersLoading: false, gwdOpsMembersError: errorMessage(err, 'Failed to load GWD ops members') }),
            }),
          ),
        ),
      ),
    );

    // Keyed by ir_seq (the real primary key once an IR is saved) — switchMap cancels a still-in-
    // flight fetch for a previously-selected row if the user picks another one before it resolves,
    // so a slow first response can never land after (and clobber) a faster later selection.
    const loadIrDetail = rxMethod<number>(
      pipe(
        tap(() => patchState(store, { irDetailLoading: true, irDetailError: null })),
        switchMap(irSeq =>
          svc.getIrDetailBlocks(irSeq).pipe(
            tapResponse({
              next: blocks => {
                const existing = store.irs().find(item => item.irSeq === irSeq);
                // Nothing to merge onto if the summary row isn't loaded (shouldn't happen from the
                // normal select-a-row flow) — just clear the loading flag and leave state as-is.
                if (!existing) {
                  patchState(store, { irDetailLoading: false });
                  return;
                }
                const merged = mergeIrDetailBlocksIntoIntegratedRequirement(existing, blocks, store.geologistNameByEmpNum());
                patchState(store, { irs: upsertByIrSeq(store.irs(), merged), irDetailLoading: false });
              },
              error: (err: unknown) =>
                patchState(store, { irDetailLoading: false, irDetailError: errorMessage(err, 'Failed to load integrated requirement detail') }),
            }),
          ),
        ),
      ),
    );

    // Separate call from loadIrDetail above — mirrors the future real API shape (approval process
    // gets its own endpoint), mocked client-side in the meantime. Same switchMap-cancels-a-stale-
    // fetch reasoning applies when the selection changes quickly.
    const loadIrApprovalTimeline = rxMethod<number>(
      pipe(
        tap(() => patchState(store, { irApprovalTimelineLoading: true, irApprovalTimelineError: null })),
        switchMap(irSeq =>
          svc.getIrApprovalTimeline(irSeq).pipe(
            tapResponse({
              next: chain => {
                patchState(store, {
                  irs: store.irs().map(item => (item.irSeq === irSeq ? { ...item, chain } : item)),
                  irApprovalTimelineLoading: false,
                });
              },
              error: (err: unknown) =>
                patchState(store, { irApprovalTimelineLoading: false, irApprovalTimelineError: errorMessage(err, 'Failed to load approval timeline') }),
            }),
          ),
        ),
      ),
    );

    // Attachments are files on the backend's filesystem, not exad_gwd_ir_* columns, so they load
    // through their own endpoint and get merged onto the record the same way the chain above does.
    const loadIrAttachments = rxMethod<number>(
      pipe(
        tap(() => patchState(store, { attachmentsLoading: true, attachmentError: null })),
        switchMap(irSeq =>
          svc.getIrAttachments(irSeq).pipe(
            tapResponse({
              next: attachments => {
                patchState(store, {
                  irs: store.irs().map(item => (item.irSeq === irSeq ? { ...item, attachments } : item)),
                  attachmentsLoading: false,
                });
              },
              error: (err: unknown) =>
                patchState(store, { attachmentsLoading: false, attachmentError: errorMessage(err, 'Failed to load attachments') }),
            }),
          ),
        ),
      ),
    );

    /** Replaces just this slot's entry on the record, leaving the other slot untouched. */
    const withAttachment = (irSeq: number, slot: IrAttachmentSlot, attachment: IrAttachment | null) =>
      store.irs().map(item => {
        if (item.irSeq !== irSeq) return item;
        const others = item.attachments.filter(existing => existing.slot !== slot);
        return { ...item, attachments: attachment ? [...others, attachment] : others };
      });

    return {
      loadIrs(): void {
        loadIrs();
      },

      loadIrDetail(irSeq: number): void {
        loadIrDetail(irSeq);
      },

      loadIrAttachments(irSeq: number): void {
        loadIrAttachments(irSeq);
      },

      /**
       * Stages a picked file. Attaching never uploads on its own — the user sends it with the
       * slot's own upload action (`uploadStagedAttachment`), or, for a New IR that has no ir_seq to
       * upload against yet, the first save flushes it via `uploadPendingAttachments`.
       */
      setPendingAttachment(slot: IrAttachmentSlot, file: File): void {
        patchState(store, { pendingAttachments: { ...store.pendingAttachments(), [slot]: file }, attachmentError: null });
      },

      /**
       * Sends the file staged for one slot. Clears the staging only after the server confirms, so a
       * failed upload leaves the file in the slot to retry rather than silently dropping the user's
       * pick — the one thing that would actually lose their work.
       */
      uploadStagedAttachment(irSeq: number, slot: IrAttachmentSlot): void {
        const file = store.pendingAttachments()[slot];
        if (!file) return;

        patchState(store, { busyAttachmentSlot: slot, attachmentUploadPercent: 0, attachmentError: null });
        svc.uploadIrAttachment(irSeq, slot, file).subscribe({
          next: ({ percent, attachment }) => {
            if (!attachment) {
              patchState(store, { attachmentUploadPercent: percent });
              return;
            }
            const remaining = Object.fromEntries(
              Object.entries(store.pendingAttachments()).filter(([key]) => key !== slot),
            ) as Partial<Record<IrAttachmentSlot, File>>;
            patchState(store, {
              irs: withAttachment(irSeq, slot, attachment),
              pendingAttachments: remaining,
              busyAttachmentSlot: null,
              attachmentUploadPercent: null,
            });
          },
          error: (err: unknown) =>
            patchState(store, {
              busyAttachmentSlot: null,
              attachmentUploadPercent: null,
              attachmentError: errorMessage(err, 'Failed to upload the attachment'),
            }),
        });
      },

      removePendingAttachment(slot: IrAttachmentSlot): void {
        const remaining = Object.fromEntries(
          Object.entries(store.pendingAttachments()).filter(([key]) => key !== slot),
        ) as Partial<Record<IrAttachmentSlot, File>>;
        patchState(store, { pendingAttachments: remaining });
      },

      /** Drops every held file — the draft was abandoned, or a different record was selected. */
      clearPendingAttachments(): void {
        patchState(store, { pendingAttachments: {} });
      },

      /**
       * Uploads everything held for a draft that has just been assigned `irSeq`, one slot at a
       * time. Sequential rather than parallel because `busyAttachmentSlot`/`attachmentUploadPercent`
       * describe a single in-flight upload — two at once would fight over both.
       *
       * State is cleared up front so a second save can't re-upload the same files, and a failure on
       * one slot doesn't abort the other: each reports its own error and the chain continues.
       */
      uploadPendingAttachments(irSeq: number): void {
        const pending = Object.entries(store.pendingAttachments()) as [IrAttachmentSlot, File][];
        if (!pending.length) return;
        patchState(store, { pendingAttachments: {} });

        from(pending)
          .pipe(
            concatMap(([slot, file]) => {
              patchState(store, { busyAttachmentSlot: slot, attachmentUploadPercent: 0, attachmentError: null });
              return svc.uploadIrAttachment(irSeq, slot, file).pipe(
                tap(({ percent, attachment }) => {
                  if (!attachment) {
                    patchState(store, { attachmentUploadPercent: percent });
                    return;
                  }
                  patchState(store, {
                    irs: withAttachment(irSeq, slot, attachment),
                    busyAttachmentSlot: null,
                    attachmentUploadPercent: null,
                  });
                }),
                catchError((err: unknown) => {
                  patchState(store, {
                    busyAttachmentSlot: null,
                    attachmentUploadPercent: null,
                    attachmentError: errorMessage(err, `Failed to upload the ${slot} attachment`),
                  });
                  return of(null);
                }),
              );
            }),
          )
          .subscribe();
      },

      /** Percent stays null throughout: a delete has no measurable progress to report. */
      deleteIrAttachment(irSeq: number, slot: IrAttachmentSlot): void {
        patchState(store, { busyAttachmentSlot: slot, attachmentError: null, attachmentUploadPercent: null });
        svc.deleteIrAttachment(irSeq, slot).subscribe({
          next: () => patchState(store, { irs: withAttachment(irSeq, slot, null), busyAttachmentSlot: null }),
          error: (err: unknown) =>
            patchState(store, { busyAttachmentSlot: null, attachmentError: errorMessage(err, 'Failed to remove the attachment') }),
        });
      },

      loadIrApprovalTimeline(irSeq: number): void {
        loadIrApprovalTimeline(irSeq);
      },

      loadLinkedRequestWells(): void {
        loadLinkedRequestWells();
      },

      loadGwdOpsMembers(): void {
        loadGwdOpsMembers();
      },

      /**
       * No optimistic echo: the backend only ever confirms an `ir_seq` (see WwellIrSaveResult),
       * so this waits out a real follow-up GET (getIrDetailBlocks) and merges *that* onto the
       * record's last known-real state before `irs` ever changes — never onto whatever was just
       * typed. A field with no backing column at all (client, targetAquifer, targetDepthFt... see
       * mergeIrDetailBlocksIntoIntegratedRequirement's own doc comment) simply won't reflect the
       * edit once this settles, since there is nowhere real to read it back from — better an
       * honest revert than a value that only looks saved until the next reload.
       *
       * `creatingIr` stays true for both round-trips (POST, then GET), so the spinner is what
       * covers the extra latency — nothing renders in between.
       */
      createIr(payload: CreateIntegratedRequirementPayload, irBulkPayload: WwellIrBulkSavePayload): void {
        patchState(store, { creatingIr: true, createIrError: null });
        svc
          .createIr(irBulkPayload)
          .pipe(
            delay(WWELL_SAVE_DELAY_MS),
            switchMap(irSeq => {
              if (irSeq === null) return of({ irSeq, blocks: null });
              return svc.getIrDetailBlocks(irSeq).pipe(
                map(blocks => ({ irSeq, blocks })),
                // The save itself already succeeded here — a hiccup on this second, read-only
                // call must not read as "the save failed" (see the `blocks === null` branch below).
                catchError(() => of({ irSeq, blocks: null })),
              );
            }),
          )
          .subscribe({
            next: ({ irSeq, blocks }) => {
              if (irSeq === null) {
                // No identity to file this under at all — shouldn't happen for a genuine backend
                // success, so there's nothing to merge or select; just stop spinning.
                patchState(store, { creatingIr: false });
                return;
              }
              if (blocks === null) {
                // Confirmed saved (a real ir_seq came back) but the follow-up detail fetch failed
                // — record the real identity so a resave correctly UPDATEs instead of re-INSERTing,
                // but leave `irs` untouched rather than show anything unconfirmed.
                patchState(store, {
                  draftIrSeq: irSeq,
                  creatingIr: false,
                  createIrError: 'Saved, but the latest data failed to reload — refresh to see it.',
                });
                return;
              }
              const existing = store.irs().find(item => item.irSeq === irSeq);
              const base =
                existing ?? buildIntegratedRequirementFromPayload(payload, { id: `IR-${irSeq}`, code: `IR-${irSeq}`, irSeq });
              const merged = mergeIrDetailBlocksIntoIntegratedRequirement(base, blocks, store.geologistNameByEmpNum());
              patchState(store, {
                irs: upsertByIrSeq(store.irs(), merged),
                draftIrSeq: irSeq,
                creatingIr: false,
              });
            },
            error: (err: unknown) => patchState(store, { creatingIr: false, createIrError: errorMessage(err, 'Failed to create integrated requirement') }),
          });
      },

      /**
       * "Submit for Review" is the SAP workflow submission, in three steps against two backends:
       * mint a workflow id (agwa-planning's get_wf_next_id_fnc) → push the workflow to SAP AAE
       * under that id (SapService) → record the outcome on the IR header (update_ir_sap_wf_prc).
       *
       * There is no separate IR status-transition call: exad_gwd_ir_header has no status column of
       * its own, so the SAP workflow columns this writes ARE the record of having submitted.
       * `updatingIrStatus` stays true across all three so the editor's Submit button spins for the
       * whole action rather than flickering between steps.
       */
      submitIrForReview(ir: IntegratedRequirement): void {
        // A draft that was never saved has no ir_seq to file the workflow under, so there would be
        // nothing to record the outcome against — refuse before calling SAP rather than after.
        if (ir.irSeq == null) {
          patchState(store, { sapTriggerError: 'Save the integrated requirement before submitting it for review.' });
          return;
        }
        patchState(store, { updatingIrStatus: true, sapTriggerError: null });
        triggerSapWorkflow(ir, ir.irSeq);
      },

      /** Deletes the whole IR (header row, cascading to casing/tops/hydrogeology/water — see
       *  WwellIrService.deleteIr) and drops it from `irs` on success. `irSeq` must already be a
       *  persisted record's — there's nothing to delete server-side for a local-only draft. */
      deleteIr(irSeq: number): void {
        patchState(store, { deletingIr: true, deleteIrError: null });
        svc
          .deleteIr(irSeq)
          .pipe(delay(WWELL_DELETE_DELAY_MS))
          .subscribe({
            next: () => patchState(store, { irs: store.irs().filter(item => item.irSeq !== irSeq), deletingIr: false }),
            error: (err: unknown) => patchState(store, { deletingIr: false, deleteIrError: errorMessage(err, 'Failed to delete integrated requirement') }),
          });
      },
    };
  }),
);

/** Every IrXxxRow interface in wwell-ir-payload.model.ts (IrHeaderRow, IrCasingRow, IrTopsRow,
 *  IrHydrogeologyRow, IrWaterRow) uses lowercase snake_case field names mirroring the Oracle
 *  column names — but Oracle's own default column casing (and whatever exad_agwa_queries_pkg's
 *  get_ir_data_prc happens to emit) isn't guaranteed to be lowercase, so a row can come back with
 *  the same fields in UPPERCASE or MixedCase. Since every field this file reads is lowercase by
 *  contract, lowercasing every key up front makes each row match regardless of how the backend
 *  cased it — the same fix as `rowsFor`'s table-name comparison below, applied to row field names. */
function normalizeRowKeys<TRow>(row: Record<string, unknown>): TRow {
  const normalized: Record<string, unknown> = {};
  for (const key of Object.keys(row)) {
    normalized[key.toLowerCase()] = row[key];
  }
  return normalized as TRow;
}

/** Case-insensitive on `candidate.table`: the GET /{irSeq} detail endpoint echoes back
 *  UPPERCASE table names (EXAD_GWD_IR_HEADER...) per exad_agwa_queries_pkg.get_ir_data_prc,
 *  while WwellIrTableName (and the bulk-save write path) uses lowercase — a strict `===` here
 *  never matched, so every block silently resolved to empty and wiped casing/formationTops on
 *  every detail fetch. */
function rowsFor<TRow>(blocks: readonly WwellIrDetailTableBlock[], table: WwellIrTableName): readonly TRow[] {
  const block = blocks.find(candidate => candidate.table.toLowerCase() === table);
  return (block?.data ?? []).map(row => normalizeRowKeys<TRow>(row));
}

/** Inverse of resolveGeologistId in wwell-ir-payload.mapper.ts: cont_geologist/sub_cont_geologist
 *  come back as a raw EMPNUM, not the name the picker showed - resolve it through the same
 *  FULL_NAME/EMPNUM reference data (WwellIrStore.geologistNameByEmpNum) used to build the picker's
 *  own options. Falls back to `fallback` (the caller's `existing` value) when the header has no
 *  value for this column yet, or the EMPNUM isn't found in the current ops-members list (e.g. a
 *  fetch that hasn't completed, or a since-removed employee) - same as every other lookup below,
 *  an unresolved value should keep whatever was last known rather than go blank. */
function resolveGeologistName(
  empNum: number | null | undefined,
  geologistNameByEmpNum: Readonly<Record<number, string>>,
  fallback: string | null,
): string | null {
  if (empNum == null) return fallback;
  return geologistNameByEmpNum[empNum] ?? fallback;
}

/** Inverse of the `h2s: '${ppm} ppm'` formatting in wwell-ir-payload.mapper.ts. */
function parseH2sPpm(h2s: string | null | undefined): number | null {
  if (!h2s) return null;
  const value = parseFloat(h2s);
  return Number.isFinite(value) ? value : null;
}

/** exad_gwd_well_design's flag columns come back as `'Y'|'N'|null` in practice, but the raw block
 *  is read as loosely-typed `Record<string, unknown>` (see rowsFor above) — narrow defensively to
 *  the two real values rather than trust the cast, falling back to 'N' for anything else (an
 *  absent/garbled column reads the same as "not set", matching how deriveCompletionElementFlags
 *  itself never produces a third value). */
function normalizeFlag(value: unknown): 'Y' | 'N' {
  return value === 'Y' ? 'Y' : 'N';
}

function normalizeRemarks(value: unknown): string | null {
  return typeof value === 'string' && value ? value : null;
}

/** Reads the eight completion-element columns off a fetched exad_gwd_well_design row — see
 *  IntegratedRequirementWellDesignFlags' own doc comment. `null` when no row came back at all
 *  (no exad_gwd_well_design record yet for this well), not `undefined` — `undefined` is reserved
 *  for "never fetched" (a still-local draft), which this function is never called for. */
function mapWellDesignFlags(row: IrWellDesignRow | undefined): IntegratedRequirementWellDesignFlags | null {
  if (!row) return null;
  return {
    ohFlg: normalizeFlag(row.oh_flg),
    ohRemarks: normalizeRemarks(row.oh_remarks),
    gpFlg: normalizeFlag(row.gp_flg),
    gpRemarks: normalizeRemarks(row.gp_remarks),
    perforatedFlg: normalizeFlag(row.perforated_flg),
    perforatedRemarks: normalizeRemarks(row.perforated_remarks),
    lsFlg: normalizeFlag(row.ls_flg),
    lsRemarks: normalizeRemarks(row.ls_remarks),
  };
}

/** `stripLegacyIrHtml` version of the `raw ?? fallback` pattern used throughout this merge: falls
 *  back to `fallback` only when the backend genuinely sent nothing (`null`/`undefined`) for this
 *  column, same as a bare `??` would — an explicitly-cleared column (`''`) is a real value and
 *  must still win over `fallback`, not be treated as "missing" just because stripping an empty
 *  string also yields `''`. */
function resolveLegacyText(raw: string | null | undefined, fallback: string): string {
  return raw != null ? stripLegacyIrHtml(raw) : fallback;
}

const isBlankCasingRow = (row: WwellIrRawCasingRow): boolean =>

 *  EXAD_AGWA_IUD_PKG.save_casing/save_tops in 05-agwa-iud-package.sql) — a row whose key isn't
 *  already present in `existing` is genuinely new and must be sent as INSERT, not UPDATE, or the
 *  backend's `WHERE <key match>` finds nothing and silently drops it. */
const casingKey = (row: { readonly type: string; readonly sizeIn: string }): string => `${row.type} ${row.sizeIn}`;
const topKey = (row: { readonly name: string }): string => row.name;

/** Only the columns save_casing's/save_tops' UPDATE branch actually writes (type/size are the
 *  match key itself, not a writable column) — comparing anything else could flag a row "changed"
 *  when nothing the backend would touch actually differs. */
const casingRowUnchanged = (existingRow: IntegratedRequirementCasingRow, row: WwellIrRawCasingRow): boolean =>
  existingRow.depthFtBgl === (row.depthFtBgl ?? 0) && (existingRow.remarks ?? null) === (row.remarks || null);

const topRowUnchanged = (existingRow: IntegratedRequirementFormationTop, row: WwellIrRawFormationTop): boolean =>
  existingRow.depthFt === (row.depthFt ?? 0) && (existingRow.subSeaFtAsl ?? null) === (row.subSeaFtAsl ?? null);

function buildCasingRows(
  raw: readonly WwellIrRawCasingRow[],
  irSeq: number | null,
  existing: readonly IntegratedRequirementCasingRow[] | null | undefined,
): IrCasingRow[] {
  const currentRows = raw.filter(row => !isBlankCasingRow(row));

  if (existing == null) {
    const action = resolveRowAction(irSeq);
    return currentRows.map(row => ({
      action,
      ir_seq: irSeq,
      csg_size: row.sizeIn,
      csg_type: row.type,
      csg_depth: row.depthFtBgl ?? 0,
      csg_remarks: row.remarks || null,
    }));
  }

  const existingByKey = new Map(existing.map(row => [casingKey(row), row]));
  const keptKeys = new Set<string>();

  // A row this save didn't actually change is sent as '' (no-op), not a redundant UPDATE — see
  // IrRowAction's doc comment: the backend's per-row action switch then skips the DML entirely
  // instead of re-writing a row with the exact values it already has on every single save.
  const upserts: IrCasingRow[] = currentRows.map(row => {
    const key = casingKey(row);
    const existingRow = existingByKey.get(key);
    if (existingRow) keptKeys.add(key);
    const action: IrRowAction = existingRow ? (casingRowUnchanged(existingRow, row) ? '' : 'UPDATE') : 'INSERT';
    return {
      action,
      ir_seq: existingRow ? irSeq : null,
      csg_size: row.sizeIn,
      csg_type: row.type,
      csg_depth: row.depthFtBgl ?? 0,
      csg_remarks: row.remarks || null,
    };
  });

  const deletes: IrCasingRow[] = existing
    .filter(row => !keptKeys.has(casingKey(row)))
    .map(row => ({
      action: 'DELETE',
      ir_seq: irSeq,
      csg_size: row.sizeIn,
      csg_type: row.type,
      csg_depth: row.depthFtBgl,
      csg_remarks: row.remarks,
    }));

  return [...upserts, ...deletes];
}

      est_productivity: null,
      flowing: null,
      siwhp: null,
      h2s: null,

  if (existing == null) {
    const action = resolveRowAction(irSeq);
    return currentRows.map(row => toWaterRow(row, action, irSeq));
  }

  const existingByKey = new Map(existing.map(row => [waterKey(row.epANum), row]));
  const keptKeys = new Set<string>();

  const upserts: IrWaterRow[] = currentRows.map(row => {
    const key = waterKey(row.epANum ?? 0);
    const existingRow = existingByKey.get(key);
    if (existingRow) keptKeys.add(key);
    const action: IrRowAction = existingRow ? (waterRowUnchanged(existingRow, row) ? '' : 'UPDATE') : 'INSERT';
    return toWaterRow(row, action, existingRow ? irSeq : null);
  });

  const deletes: IrWaterRow[] = existing
    .filter(row => !keptKeys.has(waterKey(row.epANum)))
    .map(row => toWaterRow(offsetWellToRaw(row), 'DELETE', irSeq));

  return [...upserts, ...deletes];
}

/**

 * Completion/Open Hole Size via deriveCompletionElementFlags — see IrWellDesignRow's own doc
 * comment) are mapped onto exad_gwd_well_design. Unlike every other block above, this table is
 * keyed by wd_seq (not ir_seq): with no wd_seq yet known (no detail fetch has resolved one), a row
 * is only worth INSERTing once there's an actual value to save (a pump depth, or a real Completion
 * selection) and a linked request to hang it off of — otherwise there is nothing to persist yet.
 *
 * DATA-LOSS CAVEAT: update_ir_data_prc's exad_gwd_well_design MERGE (docs/ir/update_ir_data_prc.sql)
 * is a blanket `WHEN MATCHED THEN UPDATE SET <every column the proc's JSON_TABLE declares>` off
 * this same JSON row — a column this app's payload omits (design_rpt, status, design_phase,
 * w_prim_hid, w_num, sta_water_lvl, tds, completion_type, notes, publish_dt/flg/by, curr_flg) comes
 * back NULL from JSON_TABLE and gets written as NULL, not left alone. If a wd_seq this UPDATE
 * targets already carries real data from that other Well Design workflow, this silently wipes it.
 * That risk already existed for pump_lvl-only saves before this change; sending Completion more
 * often (it's now the primary driver, not just an occasional pump-depth edit) makes it fire more
 * often too. Not fixed here — would need either the backend proc to COALESCE(t.col, h.col) instead
 * of overwriting, or this app reading back and re-sending every other column untouched.
 */
function buildWellDesignRow(
  programMain: WwellIrRawProgramMain,
  pumpDepthFt: number | null,
  prewapSeq: number | null,
  existingWellDesignSeq: number | null | undefined,
  existingPumpDepthFt: number | null | undefined,
  existingProgramDetails: Pick<IntegratedRequirementProgramDetails, 'completion' | 'openHoleSizeIn'> | null | undefined,
): IrWellDesignRow | null {
  const wdSeq = existingWellDesignSeq ?? null;
  const flags = deriveCompletionElementFlags(programMain.completion, programMain.openHoleSizeIn || null);
  // IntegratedRequirementWellDesignFlags is camelCase (domain-model shape, shared with the
  // wellbore-diagram mapper); the wire payload below needs update_ir_data_prc's own snake_case
  // column names (see docs/ir/update_ir_data_prc.sql) — translate rather than rename one side.
  const flagColumns = {
    oh_flg: flags.ohFlg,
    oh_remarks: flags.ohRemarks,
    gp_flg: flags.gpFlg,
    gp_remarks: flags.gpRemarks,
    perforated_flg: flags.perforatedFlg,
    perforated_remarks: flags.perforatedRemarks,
    ls_flg: flags.lsFlg,
    ls_remarks: flags.lsRemarks,
  };

  if (wdSeq != null) {
    const unchanged =
      (existingPumpDepthFt ?? null) === pumpDepthFt &&
      (existingProgramDetails?.completion ?? '') === programMain.completion &&
      (existingProgramDetails?.openHoleSizeIn ?? null) === (programMain.openHoleSizeIn || null);
    const action: IrRowAction = unchanged ? '' : 'UPDATE';
    return { action, wd_seq: wdSeq, prewap_seq: prewapSeq, pump_lvl: pumpDepthFt, ...flagColumns };
  }

  if ((pumpDepthFt == null && !programMain.completion) || prewapSeq == null) return null;
  return { action: 'INSERT', wd_seq: null, prewap_seq: prewapSeq, pump_lvl: pumpDepthFt, ...flagColumns };
}

/** Coarse lifecycle split for the master list's top toggle — "Active" is every status before the

 *  finer-grained `statusFilter` chips below (which only make sense to show *within* Active, since
 *  Inactive only ever has one possible status). */

  'in-progress',
  'submit-for-review',

  protected readonly isLoading = this.store.irsLoading;
  protected readonly loadError = this.store.irsError;
  protected readonly isLoadingDetail = this.store.irDetailLoading;

  protected readonly searchTerm = signal('');
  // ir_seq is the real primary key once an IR is saved (see integrated-requirement.model.ts) —
  // selection is keyed on it directly, not the display-only `code`/`id` string, so it can be
  // handed straight to loadIrDetail() below without a lookup.
  protected readonly selectedIrSeq = signal<number | null>(null);

  /** Sticky across selection changes — see WwellIrDetailMode's own doc comment. */
  protected readonly detailMode = signal<WwellIrDetailMode>('edit');

  protected readonly lifecycleFilter = signal<WwellIrLifecycleFilter>('active');
  protected readonly statusFilter = signal<IntegratedRequirementStatus | 'all'>('all');
  // Sub-filter chips shown only in the Active bucket — Inactive is exactly one status
  // (approved), so there's nothing left to narrow further once that's selected.
  protected readonly statusChips: readonly (IntegratedRequirementStatus | 'all')[] = ['all', ...WWELL_IR_ACTIVE_STATUSES];

  // Scoped by the lifecycle toggle only (not the status chip or search term) so chip counts stay
  // stable while the user narrows further — each chip shows how many rows *it* would reveal, not
  // how many remain after some other filter already trimmed the set.
  private readonly lifecycleScopedIrs = computed(() => {
    const all = this.store.irs();
    return this.lifecycleFilter() === 'inactive'
      ? all.filter(ir => ir.status === 'approved')
      : all.filter(ir => WWELL_IR_ACTIVE_STATUSES.includes(ir.status));
  });

  protected readonly rows = computed(() => {
    const term = this.searchTerm().trim().toLowerCase();
    const status = this.statusFilter();
    const scoped = this.lifecycleScopedIrs().filter(ir => status === 'all' || ir.status === status);
    if (!term) return scoped;

    return scoped.filter(
      ir =>
        ir.code.toLowerCase().includes(term) ||
        ir.requestReference.wellName.toLowerCase().includes(term) ||
        (ir.requestReference.linkedRequestId ?? '').toLowerCase().includes(term) ||
        ir.requestReference.client.toLowerCase().includes(term),
    );
  });

  // A brand-new IR created via createNewIr() below — never in `store.irs()` (no backend call has
  // happened yet, so there's nothing to add it to) and takes priority over `selectedIrSeq` here.
  // Cleared once its first save actually persists it (see the creatingIr effect below) or the
  // user selects an existing row instead (see selectIrRow), discarding whatever was unsaved.
  protected readonly pendingDraftIr = signal<IntegratedRequirement | null>(null);

  // Deliberately looks up the *unfiltered* store list (not `rows()`) so narrowing the search
  // never drops the current selection just because it fell outside the filtered results.
  // Water Well Name/Coordinates/Surface Elevation arrive on the record itself once its detail has
  // been fetched (see mergeIrDetailBlocksIntoIntegratedRequirement's exad_rcd_prewap block) — no
  // separate hydration pass needed here.
  protected readonly selectedIr = computed<IntegratedRequirement | null>(
    () => this.pendingDraftIr() ?? this.store.irs().find(ir => ir.irSeq === this.selectedIrSeq()) ?? null,
  );

  // Identity/order of the *visible set* of rows, not the row objects themselves — selecting a row
  // patches that one IR's `chain` in place (loadIrDetail / loadIrApprovalTimeline both replace
  // `irs` with a new array reference), which would otherwise look identical to "the result set changed" to a plain
  // `rows()` reference check below and replay the list-reveal animation across every row on every
  // click. This string only changes on an actual add/remove/reorder/filter.
  protected readonly rowsSignature = computed(() => this.rows().map(ir => ir.irSeq ?? ir.id).join('|'));

  protected readonly statusMeta = WWELL_IR_STATUS_META;

  protected readonly skeletonRows = Array.from({ length: SKELETON_ROW_COUNT }, (_, index) => index);

  // Bound to the viewport's [itemSize] and each row-item's fixed height in the template — see
  // WWELL_IR_ROW_ITEM_SIZE_PX's own comment for why these must stay pinned together.
  protected readonly rowItemSize = WWELL_IR_ROW_ITEM_SIZE_PX;

  private readonly pendingSelection = createPendingSelection<number>(WWELL_LOAD_DELAY_MS);
  protected readonly pendingId = this.pendingSelection.pendingId;
  // Shows the loading UI for the cosmetic selection transition *and* the real detail fetch it
  // now kicks off — otherwise the shimmer could resolve before the fetch does, briefly exposing
  // stale summary-only fields (blank casing/tops) before the enriched detail lands. Deliberately
  // excludes the approval-timeline flag: that fetch never blocks this panel, including the one
  // real fetch a brand-new record needs right after its first save (see the creatingIr effect
  // below) — re-showing the whole panel as a loading skeleton the instant after a save completes
  // would undo the "seamless save" this effect exists to provide.
  protected readonly isSelecting = computed(() => this.pendingSelection.isSelecting() || this.isLoadingDetail());

  @ViewChild('detailHost') private readonly detailHost?: ElementRef<HTMLElement>;
  // `{ read: ElementRef }` is required now that #listHost sits on <cdk-virtual-scroll-viewport>
  // (a component) — an unqualified @ViewChild query on a component host resolves to that
  // component instance by default, not its native element, which used to be the (unstated)
  // default back when this ref lived on a plain <ul>.
  @ViewChild('listHost', { read: ElementRef }) private readonly listHost?: ElementRef<HTMLElement>;
  // Only rendered while detailMode() === 'edit' (see the template's `@else` branch) — undefined
  // whenever the detail panel is showing the read-only view stepper or the empty state, which is
  // exactly when there's nothing open to lose (see canDeactivate()'s guard below).
  @ViewChild(WwellIrEditorComponent) private readonly editorRef?: WwellIrEditorComponent;
  private previousSelectedIrSeq: number | null = null;
  private wasCreatingIr = false;
  private wasDeletingIr = false;

  constructor() {
    effect(() => {
      const message = this.store.createIrError();
      if (message) this.notify.error(message);
    });

    // Reports the real outcome of a save once it settles — the editor deliberately shows nothing
    // at click time (see saveMergedRaw), since the request is still in flight there (its own Save
    // button shows a spinner instead — see wwell-ir-editor.component.html). A failure is already
    // toasted by the createIrError effect above, so only success is announced here.
    //
    // On success, a save that completes while pendingDraftIr is set means *this* draft just got
    // its first real backend row — swap from the local-only draft over to the now-real,
    // ir_seq-keyed record. By the time `creatingIr` flips false, store.createIr has already
    // waited out a real follow-up GET and merged it into `irs` (no optimistic echo — see that
    // method's own doc comment), so this is already the confirmed, persisted record. Future saves
    // then UPDATE instead of re-INSERTing.
    effect(() => {
      const creating = this.store.creatingIr();
      const settled = this.wasCreatingIr && !creating;
      this.wasCreatingIr = creating;
      if (!settled || this.store.createIrError()) return;

      const wasNewIr = this.pendingDraftIr() !== null;
      const savedIrSeq = wasNewIr ? this.store.draftIrSeq() : this.selectedIrSeq();

      if (wasNewIr && savedIrSeq !== null) {
        this.pendingDraftIr.set(null);
        // Deliberately sets `previousSelectedIrSeq` before selecting, so the "selection changed"
        // effect below no-ops for this transition instead of calling loadIrDetail (redundant —
        // store.createIr's own follow-up GET already covers it — and it would flip
        // isLoadingDetail(), swapping the whole detail panel to the loading skeleton right after
        // the user just watched this exact record save).
        this.previousSelectedIrSeq = savedIrSeq;
        this.selectedIrSeq.set(savedIrSeq);
        // The one fetch a brand-new record still needs: its local draft never had a real approval
        // chain, only whatever buildIntegratedRequirementFromPayload defaulted it to. Doesn't gate
        // isSelecting() (see that computed's comment), so this never reintroduces the skeleton.
        this.store.loadIrApprovalTimeline(savedIrSeq);
        // A brand-new record has an ir_seq for the first time here, so this is the first moment
        // its attachment slots can be fetched (and the first moment uploads become possible).
        this.store.loadIrAttachments(savedIrSeq);
        // Anything the user attached while this was still an unsaved draft had nowhere to go until
        // now — send it against the id the save just assigned.
        this.store.uploadPendingAttachments(savedIrSeq);
      }

      const savedCode = savedIrSeq !== null ? this.store.irs().find(ir => ir.irSeq === savedIrSeq)?.code : undefined;
      const subject = savedCode ?? 'Integrated requirement';
      this.notify.info(wasNewIr ? `${subject} created successfully.` : `${subject} updated successfully.`);
    });

    effect(() => {
      const message = this.store.sapTriggerError();
      if (message) this.notify.error(message);
    });

    effect(() => {
      const message = this.store.irDetailError();
      if (message) this.notify.error(message);
    });

    effect(() => {
      const message = this.store.irApprovalTimelineError();
      if (message) this.notify.error(message);
    });

    effect(() => {
      const message = this.store.deleteIrError();
      if (message) this.notify.error(message);
    });

    // Reports success once a delete settles — the editor's own Delete IR button shows a spinner
    // while in flight (see wwell-ir-editor.component.html's isDeleting()), so nothing needs
    // announcing at click time. A failure is already toasted by the effect above.
    effect(() => {
      const deleting = this.store.deletingIr();
      const settled = this.wasDeletingIr && !deleting;
      this.wasDeletingIr = deleting;
      if (!settled || this.store.deleteIrError()) return;

      this.notify.info('Integrated requirement deleted.');

      // The deleted record is already gone from store.irs() by this point (WwellIrStore.deleteIr
      // removes it on success) — if it was the one open, fall back to null so the "default to
      // first visible record" effect below picks whatever the list now shows first (or leaves the
      // panel empty if nothing remains), the same path a fresh session start already uses.
      if (this.selectedIrSeq() !== null && !this.store.irs().some(ir => ir.irSeq === this.selectedIrSeq())) {
        this.selectedIrSeq.set(null);
        this.previousSelectedIrSeq = null;
      }
    });

    // Default to the first *visible* record once the list loads — mirrors the reference app,
    // which always has something selected in the detail pane rather than starting on an empty
    // state. Reads `rows()` (lifecycle/status/search-scoped), not the raw store list, so the
    // default selection is never a row the Active/Inactive toggle is currently hiding. Skipped
    // while a pendingDraftIr is open — selectedIrSeq is intentionally left null throughout
    // drafting (the draft isn't in `rows()`), and this effect would otherwise immediately fetch
    // and select an unrelated real record's detail out from under the still-open draft.
    effect(() => {
      if (this.pendingDraftIr()) return;
      const firstIrSeq = this.rows()[0]?.irSeq;
      if (firstIrSeq != null && this.selectedIrSeq() === null) {
        this.selectedIrSeq.set(firstIrSeq);
      }
    });

    // Selection changed → play the reveal transition and fetch that record's real detail
    // (casing/tops/hydrogeology) by ir_seq, its primary key.
    effect(() => {
      const irSeq = this.selectedIrSeq();
      if (irSeq === null || irSeq === this.previousSelectedIrSeq) return;

      const host = this.detailHost?.nativeElement;
      if (host) playIfMotionAllowed(buildDetailRevealTimeline(host));
      this.previousSelectedIrSeq = irSeq;
      this.store.loadIrDetail(irSeq);
      // Separate call — the approval-process API doesn't exist yet (mocked for now), but the
      // request shape already mirrors the real future one: one fetch per selected IR.
      this.store.loadIrApprovalTimeline(irSeq);
      // Also its own fetch: attachments are files on the backend's filesystem keyed by ir_seq, not
      // exad_gwd_ir_* columns, so they can't ride along in the detail blocks above.
      this.store.loadIrAttachments(irSeq);
      // Files held for an abandoned New IR belong to that draft alone — drop them rather than let
      // them surface on whichever record was just selected. (This effect deliberately no-ops on the
      // save transition, see the comment above, so it can't discard files mid-upload.)
      this.store.clearPendingAttachments();
      // Re-fetched on every selection, not just once at ngOnInit — this backs the editor's own
      // "Link to Request" dropdown and its live auto-populate-on-pick (see
      // wwell-ir-editor.component.ts's applyLinkedRequestWell / wwell-ir-linked-well.util.ts),
      // used when the user picks a NEW link on the opened record. Eligibility can change
      // server-side between selections in a long-lived session (another user initiates an IR for
      // a well, a well newly becomes eligible...) — no cached/one-time snapshot, always the
      // current server state for whichever record is being opened.
      this.store.loadLinkedRequestWells();
    });

    // Re-plays whenever the visible rows actually change (initial load finishing, or the
    // search term narrowing/widening the result set) — keyed on `rowsSignature()` (identity/order
    // only), not `rows()` itself, so a selected row's detail/timeline arriving in place never
    // replays this across the whole list — see `rowsSignature`'s comment above.
    effect(() => {
      const signature = this.rowsSignature();
      if (this.isLoading() || signature === '') return;

      const host = this.listHost?.nativeElement;
      if (!host) return;

      const items = Array.from(host.querySelectorAll<HTMLElement>('.wwell-ir__row'));
      if (items.length) playIfMotionAllowed(buildListRowsRevealTimeline(items));
    });
  }

  ngOnInit(): void {
    this.store.loadIrs();
    // Started here (not lazily in the editor once a record is selected) so the "Link to Request"
    // dropdown/picker (see loadLinkedRequestWells's other call site, on selection change below)
    // has options ready by the time the first record's editor renders, instead of racing it.
    this.store.loadLinkedRequestWells();
    // Geologist Contact picker's option list — stable reference data, unlike linkedRequestWells
    // above, so it's only loaded once here rather than re-fetched on every IR selection.
    this.store.loadGwdOpsMembers();
  }

  ngOnDestroy(): void {
    this.pendingSelection.clear();
  }

  protected onSearchChange(term: string): void {
    this.searchTerm.set(term);
  }

  /** Switching lifecycle bucket resets the status chip back to "All" — a chip selected in the old
   *  bucket (e.g. "In Review") would otherwise carry over into Inactive, where it can never match
   *  anything, silently showing an empty list. */
  protected setLifecycleFilter(filter: WwellIrLifecycleFilter): void {
    this.lifecycleFilter.set(filter);
    this.statusFilter.set('all');
  }

  protected toggleLifecycleFilter(): void {
    this.setLifecycleFilter(this.lifecycleFilter() === 'active' ? 'inactive' : 'active');
  }

  protected setStatusFilter(status: IntegratedRequirementStatus | 'all'): void {
    this.statusFilter.set(status);
  }

  // Chips only ever render within the Active lifecycle bucket (see the template's
  // `@if (lifecycleFilter() === 'active')` gate) — "All" here already only ever means "every
  // active-status record", never the inactive/approved ones the toggle hides. Labeling it
  // "Active" instead of the bare, easy-to-misread "All" makes that scope explicit.
  protected statusChipLabel(status: IntegratedRequirementStatus | 'all'): string {
    return status === 'all' ? 'Active' : this.statusMeta[status].label;
  }

  protected statusChipCount(status: IntegratedRequirementStatus | 'all'): number {
    const scoped = this.lifecycleScopedIrs();
    return status === 'all' ? scoped.length : scoped.filter(ir => ir.status === status).length;
  }

  protected selectIr(irSeq: number): void {
    this.guardNavigation(() => this.performSelectIr(irSeq));
  }

  /** Selecting a real row discards whatever unsaved local draft was open — nothing was ever
   *  persisted for it, so there's nothing to preserve (see guardNavigation()'s own comment for why
   *  a section *with* real edits is asked about before this ever runs). */
  private performSelectIr(irSeq: number): void {
    this.pendingDraftIr.set(null);
    this.pendingSelection.select(irSeq, this.selectedIrSeq(), resolved => this.selectedIrSeq.set(resolved));
  }

  /** Normalizes the model's `irSeq?: number | null` to a plain `number | null` for the template
   *  (which can't express an `== null` check under the eqeqeq lint rule). */
  protected rowIrSeq(ir: IntegratedRequirement): number | null {
    return ir.irSeq ?? null;
  }

  protected selectIrRow(ir: IntegratedRequirement): void {
    const irSeq = this.rowIrSeq(ir);
    if (irSeq !== null) this.selectIr(irSeq);
  }

  /** The row's "view" (eye) icon — selects the record exactly like selectIrRow above, plus
   *  switches the detail panel to the read-only stepper. Staying in view mode afterwards (rather
   *  than resetting on every selection change) is what lets the master list double as a way to
   *  browse several records' read-only views in a row, without an extra click back into view
   *  mode each time. */
  protected viewIr(irSeq: number): void {
    // Switching to the read-only stepper destroys this editor instance (see the template's
    // `@if (detailMode() === 'view')` branch) even when `irSeq` is the record already open here —
    // so this needs the same guard as an actual selection change, not just performSelectIr's own
    // "discard a blank draft" case.
    this.guardNavigation(() => {
      this.detailMode.set('view');
      this.performSelectIr(irSeq);
    });
  }

  /** WwellIrViewComponent's own "Edit" button — switches the currently-open record's detail
   *  panel back to the inline-edit surface. */
  protected setDetailMode(mode: WwellIrDetailMode): void {
    this.detailMode.set(mode);
  }

  /** `cdkVirtualFor`'s identity function — same identity rule as the old `@for`'s `track`
   *  expression, just spelled as a function since `*cdkVirtualFor` (a plain structural directive,
   *  not the `@for` built-in control flow) takes `trackBy` instead of an inline `track` expression. */
  protected trackByIrRow(_index: number, ir: IntegratedRequirement): number | string {
    return ir.irSeq ?? ir.id;
  }

  /** Opens a blank draft in the detail pane and selects it — purely client-side, no backend call,
   *  since there's nothing to save yet (see this class's `pendingDraftIr`). WwellIrEditorComponent
   *  binds to it exactly like any persisted record; its first real Save (a section's own, or the
   *  global "Save" button) is what actually creates the backend row. */
  protected createNewIr(): void {
    this.guardNavigation(() => this.performCreateNewIr());
  }

  private performCreateNewIr(): void {
    const raw = buildWwellDynamicForm(WWELL_IR_FORM_SCHEMA, this.fb).getRawValue() as WwellIrFormRawValue;
    const { payload } = mapIrFormValueToSubmission(raw, 'draft');
    const draft = buildIntegratedRequirementFromPayload(payload, { id: 'draft', code: 'New IR', irSeq: null });
    // Start the new draft with empty slots — a previous draft's held files must not carry over.
    this.store.clearPendingAttachments();

    this.pendingDraftIr.set(draft);
    this.selectedIrSeq.set(null);
    this.previousSelectedIrSeq = null;
    // A brand-new draft has nothing committed to view read-only yet — always open it for editing,
    // regardless of whatever mode the previously-open record was left in.
    this.detailMode.set('edit');

    const host = this.detailHost?.nativeElement;
    if (host) playIfMotionAllowed(buildDetailRevealTimeline(host));
  }

  protected chainBorderColor(chain: readonly ApprovalChainStep[]): string {
    return chainCompletionColor(chain);
  }

  /** Powers the approval-workflow sidebar's "N of M approved" progress line (view mode only). */
  protected approvedStepCount(chain: readonly ApprovalChainStep[]): number {
    return chain.filter(step => step.status === 'approved').length;
  }

  // ── Unsaved-changes guard — backs three separate "leaving the currently-open record" paths with
  // one prompt: switching the selected row / opening "New IR" / clicking the view (eye) icon (all
  // via guardNavigation below), and a real Angular route change away from this page entirely (via
  // canDeactivate(), wired up as the 'integrated-requirements' route's CanDeactivate guard — see
  // app.routes.ts and unsaved-changes.guard.ts). Browser tab close/refresh is covered separately by
  // the beforeunload listener below, which can't run this same async dialog flow.

  /** Router entry point (ComponentCanDeactivate) — resolves `true` once it's safe to actually leave
   *  the route: immediately if nothing is unsaved, after a confirmed Discard, or after a confirmed
   *  Save has settled. `false` means stay (Cancel, or a save that failed). */
  canDeactivate(): Observable<boolean> {
    const editor = this.editorRef;
    if (!editor?.hasUnsavedChanges()) return of(true);

    return this.confirmDialog
      .openWithResult({
        title: 'Unsaved changes',
        subtitle: 'This Integrated Requirement has section edits that haven’t been saved yet. Save them before leaving?',
        confirmLabel: 'Save & Leave',
        discardLabel: 'Discard Changes',
        cancelLabel: 'Stay Here',
        severity: 'warning',
      })
      .pipe(
        switchMap(result => {
          if (result === 'cancel') return of(false);
          if (result === 'discard') {
            editor.discardUnsavedChanges();
            return of(true);
          }
          return from(editor.saveUnsavedChangesAndWait());
        }),
      );
  }

  /** In-page equivalent of canDeactivate() — same prompt/save/discard flow, but for switching what
   *  this page itself shows (selecting another row, opening a blank "New IR", or the view-icon's
   *  mode switch) rather than leaving the route. `action` only runs once it's actually safe to
   *  proceed. */
  private guardNavigation(action: () => void): void {
    this.canDeactivate()
      .pipe(take(1))
      .subscribe(canLeave => {
        if (canLeave) action();
      });
  }

  /** Native browser navigation (tab close, refresh, typed URL) bypasses the Router entirely, so
   *  canDeactivate() never runs for it — this is the one place that has to use the browser's own
   *  confirmation prompt instead (no custom dialog is possible here; per spec, `returnValue` must be
   *  set and the browser supplies its own generic copy). */
  @HostListener('window:beforeunload', ['$event'])
  protected onBeforeUnload(event: BeforeUnloadEvent): void {
    if (!this.editorRef?.hasUnsavedChanges()) return;
    event.preventDefault();
    event.returnValue = '';
  }
}

 * succeed while the HTTP response still comes back as a 500, with the real success payload
 * embedded inside the error envelope's own `message` field instead of a proper 2xx — e.g.
 * `{"statusCode":500,"error":true,"message":"{\"status\":\"SUCCESS\",\"ir_seq\":9024}","data":[]}`.
 * Matches only that exact signature (a JSON-parseable message shaped like
 * `{status:"SUCCESS", ir_seq:number}`) and returns the REAL ir_seq the backend actually
 * assigned — never a client-minted placeholder — so the record's identity stays correct and
 * stable across resaves. Any other failure (a genuine validation error, network failure, a
 * differently-shaped message) falls through unmatched and is re-thrown as a real error.
 * Remove once the backend returns a proper 2xx for a save that actually succeeded.
 */
function extractIrSeqFromMisreportedSuccess(err: unknown): number | null {
  if (!(err instanceof HttpErrorResponse)) return null;
  const body: unknown = err.error;
  const message = body && typeof body === 'object' ? (body as { message?: unknown }).message : undefined;
  if (typeof message !== 'string') return null;

  try {
    const parsed: unknown = JSON.parse(message);
    const status = parsed && typeof parsed === 'object' ? (parsed as { status?: unknown }).status : undefined;
    const irSeq = parsed && typeof parsed === 'object' ? (parsed as { ir_seq?: unknown }).ir_seq : undefined;
    return status === 'SUCCESS' && typeof irSeq === 'number' ? irSeq : null;
  } catch {
    return null;
  }
}

@Injectable({ providedIn: 'root' })
export class WwellIrService {
  private readonly http = inject(HttpClient);

   * "Link to Request" picker. Returns the full record (not just {value,label}) so selecting one
   * can also auto-populate Water Well Name/Surface Elevation/Coordinates — see WwellIrStore's
   * `linkedRequestWellOptions` computed for the trimmed picker-option view derived from this.

  getEligibleWellsForIr(): Observable<IrEligibleWell[]> {

  getGwdOpsMembers(): Observable<GwdOpsMember[]> {

   * here to fetch the full { table, data } detail (casing/tops/hydrogeology) for that one record.
   * No offline-mock fallback here (unlike the other reads above): faking detail for a specific,
   * already-known-real record would misrepresent it rather than degrade gracefully, so a failure
   * surfaces as a real error instead — see WwellIrStore.loadIrDetail.
   */
  getIrDetailBlocks(irSeq: number): Observable<readonly WwellIrDetailTableBlock[]> {
    return this.http
      .get<ApiResponse<WwellIrDetailTableBlock>>(`${this.apiUrl}/integrated-requirements/${irSeq}`)
      .pipe(map(res => res.data));
  }

   * Downloads the formal "Integrated Requirements" letter as a PDF — rendered server-side
   * (Qute + openhtmltopdf, agwa-planning's IrLetterService) from the same detail data
   * getIrDetailBlocks() above already reads, so the PDF always reflects the last-saved record
   * rather than whatever the Letter tab's own client-side view currently has in memory.
   */
  /** POST (not GET) because the wellbore diagram is rendered client-side (the same D3/SVG
   *  components the Letter tab already shows) and its PNG snapshot rides along as the request
   *  body — the backend has no drawing logic of its own for it anymore. `wellboreImageBase64` is
   *  omitted when the IR has no casing/tops data or the client-side capture failed; the letter
   *  still renders, just without that section. */
  downloadLetterPdf(irSeq: number, wellboreImageBase64?: string): Observable<Blob> {

      `${this.apiUrl}/integrated-requirements/${irSeq}/letter-pdf`,
      { wellboreImageBase64: wellboreImageBase64 ?? null },
      { responseType: 'blob' },
    );
  }

  /**
   * The IR's two named attachment slots. A separate call from the detail fetch — these are real
   * files on the backend's filesystem, not exad_gwd_ir_* columns — so an IR with nothing uploaded
   * simply comes back with an empty list rather than an error.
   */
  getIrAttachments(irSeq: number): Observable<readonly IrAttachment[]> {
    return this.http
      .get<ApiResponse<IrAttachment>>(`${this.apiUrl}/integrated-requirements/${irSeq}/attachments`)
      .pipe(map(res => res.data ?? []));
  }

  /**
   * Uploads (or replaces) one slot's file, reporting real byte-level progress as it goes.
   *
   * Emits `{ percent, attachment: null }` while the body is still being sent, then one final
   * `{ percent: 100, attachment }` carrying the stored record. Uses `HttpRequest` +
   * `reportProgress` rather than a plain `post()` so the editor's progress bar tracks actual bytes
   * transferred instead of animating a guess — on a slow link or a 20 MB scan that difference is
   * the whole point of showing a bar at all.
   *
   * Multipart, so no explicit Content-Type header — the browser must set it itself to include the
   * multipart boundary.
   *
   * SUPPRESS_GLOBAL_ERROR: the editor surfaces a per-slot inline message (see WwellIrStore's
   * attachmentError), so the global interceptor toast would double up on the same failure.
   */
  uploadIrAttachment(irSeq: number, slot: IrAttachmentSlot, file: File): Observable<IrAttachmentUploadEvent> {
    const body = new FormData();
    body.append('file', file);
    const request = new HttpRequest<FormData>(
      'POST',
      `${this.apiUrl}/integrated-requirements/${irSeq}/attachments/${slot}`,
      body,
      {
        reportProgress: true,
        context: new HttpContext().set(SUPPRESS_GLOBAL_ERROR, true),
      },
    );

    return this.http.request<ApiResponse<IrAttachment>>(request).pipe(
      // Only these two event types say anything the UI can use — the rest (Sent, ResponseHeader,
      // DownloadProgress) would otherwise map to a bogus 0 and visibly reset the bar mid-upload.
      filter(event => event.type === HttpEventType.UploadProgress || event.type === HttpEventType.Response),
      map(event => {
        if (event.type === HttpEventType.UploadProgress) {
          // `total` is absent on some servers/proxies; report 0 rather than dividing by undefined,
          // which the editor renders as an indeterminate bar instead of a wrong number.
          const percent = event.total ? Math.round((100 * event.loaded) / event.total) : 0;
          return { percent, attachment: null };
        }
        const response = event as HttpResponse<ApiResponse<IrAttachment>>;
        return { percent: 100, attachment: response.body?.data?.[0] ?? null };
      }),
    );
  }

  /** Empties one slot. 404s when it was already empty — the caller treats that as "already gone". */
  deleteIrAttachment(irSeq: number, slot: IrAttachmentSlot): Observable<void> {
    return this.http
      .delete<ApiResponse<unknown>>(`${this.apiUrl}/integrated-requirements/${irSeq}/attachments/${slot}`, {
        context: new HttpContext().set(SUPPRESS_GLOBAL_ERROR, true),
      })
      .pipe(map(() => undefined));
  }

  /**
   * Pulls a stored attachment back as a `File` so it can be handed to FilePreviewDialogComponent,
   * which previews from a File (it builds its own object URL) rather than from a remote URL.
   * Fetching through HttpClient also means the JWT interceptor authenticates it — a plain
   * `<a href>` to the same endpoint would not carry the token.
   */
  getIrAttachmentFile(irSeq: number, slot: IrAttachmentSlot, fileName: string): Observable<File> {
    return this.http
      .get(`${this.apiUrl}/integrated-requirements/${irSeq}/attachments/${slot}`, {
        responseType: 'blob',
        context: new HttpContext().set(SUPPRESS_GLOBAL_ERROR, true),
      })
      .pipe(map(blob => new File([blob], fileName, { type: blob.type })));
  }

  /** Absolute URL of a stored file, for opening it in a viewer tab. */
  irAttachmentUrl(irSeq: number, slot: IrAttachmentSlot): string {
    return `${this.apiUrl}/integrated-requirements/${irSeq}/attachments/${slot}`;
  }

  /**
   * The approval-process API is a separate call from the detail fetch above and doesn't exist yet
   * — selecting a row fires both, this one just has nothing real to hit. Mocked client-side (with
   * an artificial delay so it reads as its own network round-trip) until a real
   * `GET .../integrated-requirements/{irSeq}/approval-timeline` endpoint replaces it.
   */
  getIrApprovalTimeline(irSeq: number): Observable<readonly ApprovalChainStep[]> {
    return of(mockApprovalChainForIrSeq(irSeq)).pipe(delay(300));
  }

  /**
   * Returns only the real, backend-confirmed `ir_seq` — never a client-fabricated
   * `IntegratedRequirement` (that was this method's old `buildIrFromPayload`-based "optimistic
   * echo," removed because it kept showing whatever the user had just typed for fields with no
   * backing column at all — client, targetAquifer, targetDepthFt... (see
   * mergeIrDetailBlocksIntoIntegratedRequirement's doc comment for the full list) — with nothing to
   * ever correct it except a full page reload. WwellIrStore.createIr now follows this up with a
   * real GET (getIrDetailBlocks) and merges that onto the record's last known-real state instead.
   */
  createIr(irBulkPayload: WwellIrBulkSavePayload): Observable<number | null> {
    // The backend only accepts the raw exad_gwd_ir_* bulk-save array as the request body (see
    // wwell-ir-payload.model.ts / IntegratedRequirementResource.java's @Consumes(APPLICATION_JSON))
    // — it has no multipart handling and no columns for the richer domain fields below (client,
    // wellName, contacts, attachments...), so those stay client-side only until exad_agwa_iud_pkg's
    // header table grows to carry them. Sending them today just 415s (wrong Content-Type/body shape).
    //
    // No offline-fallback here (unlike getIrs()/getEligibleWellsForIr() above) — a failed save
    // must surface as a real, visible error (WwellIrStore.createIrError), not a fabricated
    // success. A genuine POST failure with no recoverable ir_seq (see extractIrSeqFromMisreportedSuccess
    // below for the one confirmed-defect exception) always re-throws rather than minting a random
    // ir_seq — a client-minted identity on real failure would let a resave silently re-INSERT under
    // a new fake id instead of UPDATE-ing the original row, orphaning the real backend record and
    // duplicating it in the list.
    //
    // SUPPRESS_GLOBAL_ERROR: WwellIrStore.createIr's own error handler (→ createIrError →
    // WwellIrComponent's notify.error) already shows a toast for a genuine failure — without this,
    // errorInterceptor's app-wide catchError would show a second, identical one for the same error.
    // The header row's own `ir_seq` is the identity this save was submitted under — `null` for a
    // brand-new IR (the backend mints one), otherwise the already-known ir_seq of the record being
    // updated. exad_agwa_iud_pkg.update_ir_data_prc (docs/ir/update_ir_data_prc.sql) only echoes
    // back a fresh ir_seq on that INSERT branch; an UPDATE's `p_return_value` is just the literal
    // `SUCCESS`, no ir_seq, because the caller already has it — see WwellIrSaveResult. Falling back
    // to `knownIrSeq` (not a random mint) below is what keeps a resaved record's identity stable.
    const knownIrSeq = irBulkPayload[0].data?.[0]?.ir_seq ?? null;

    return this.http
      .post<ApiResponse<WwellIrSaveResult>>(`${this.apiUrl}/integrated-requirements`, irBulkPayload, {
        context: new HttpContext().set(SUPPRESS_GLOBAL_ERROR, true),
      })
      .pipe(
        map(res => res.data[0]?.irSeq ?? knownIrSeq),
        catchError((err: unknown) => {
          const realIrSeq = extractIrSeqFromMisreportedSuccess(err);
          return realIrSeq !== null ? of(realIrSeq) : throwError(() => err);
        }),
      );
  }

  /**
   * Deletes the whole IR (header row) by ir_seq. exad_agwa_iud_pkg.save_header's DELETE arm
   * removes just that one row; casing/tops/hydrogeology/water cascade automatically via their
   * `ir_seq ... ON DELETE CASCADE` FK (01-schema.sql), so nothing else needs to be sent — no
   * per-block bulk-save payload here, unlike createIr() above.
   */
  deleteIr(irSeq: number): Observable<void> {
    return this.http
      .delete<ApiResponse<never>>(`${this.apiUrl}/integrated-requirements/${irSeq}`, {
        context: new HttpContext().set(SUPPRESS_GLOBAL_ERROR, true),
      })
      .pipe(map(() => undefined));
  }

  /**
   * Next unique SAP workflow sequence (EXAD_AGWA_QUERIES_PKG.get_wf_next_id_fnc,
   * docs/ir/get_wf_next_id_fnc.sql) — a raw DB sequence value the caller both formats into the
   * composite `workflow_id` SAP expects (see buildSapWorkflowId) and stores as-is via
   * updateIrSapWorkflow below. Live in agwa-planning (SapWorkflowResource) — no offline fallback:
   * a failure here must stop WwellIrStore.submitIrForReview before it mints a fabricated id, not
   * silently hand back one that was never really reserved.
   */
  getNextSapWorkflowSeq(): Observable<number> {
    return this.http.get<ApiResponse<number>>(`${this.apiUrl}/sap-workflow/next-id`).pipe(map(res => res.data[0]));
  }

  /**
   * Persists the SAP workflow id/status onto the IR header
   * (EXAD_AGWA_IUD_PKG.update_ir_sap_wf_prc, docs/ir/update_ir_sap_wf_prc.sql) once the SAP AAE
   * trigger call has been attempted — called on both outcomes (see WwellIrStore.submitIrForReview)
   * so the header always reflects whether the push to SAP actually succeeded. Live in
   * agwa-planning (IntegratedRequirementResource.updateSapWorkflow) — no offline fallback: a
   * failure here is a real, visible error (WwellIrStore's sapTriggerError), not a silently
   * swallowed one, since it would otherwise mask the header never actually getting stamped.
   */
  updateIrSapWorkflow(irSeq: number, wfId: number, wfStatus: string): Observable<void> {
    return this.http
      .patch<ApiResponse<never>>(`${this.apiUrl}/integrated-requirements/${irSeq}/sap-workflow`, { wfId, wfStatus })
      .pipe(map(() => undefined));
  }
}
