    effect(onCleanup => {

      const subscription = linkedRequestIdControl?.valueChanges.subscribe((linkedRequestId: string) => {

        if (linkedRequestId) this.applyLinkedRequestWell(ctrl, linkedRequestId);
      });
      onCleanup(() => subscription?.unsubscribe());

    // (selectedIr(); as ir)` branch never changes just because the *bound* record does (switching
    // list rows, or opening a blank "New IR" draft), so nothing here gets recreated for free. A
    // section already open at that moment keeps the FormGroup it already built — SectionEditState
    // .open() is a no-op once editing (see its own doc comment) — so without this, an open
    // section keeps showing the *previous* record's values after the identity underneath it
    // changes. Closing everything forces every section back to its read display, which re-derives
    // fresh from whatever `ir()` now is on next open. Compares `id` (not the object reference,
    // which also changes on every in-place update from a save) so an update to the *same* record
    // never closes sections the user still has open mid-edit.
    effect(() => {
      const id = this.ir().id;
      if (this.previousIrId !== null && id !== this.previousIrId) this.closeAllSections();
      this.previousIrId = id;
    });

   *  Aquifer/Target Depth have no source there (exad_rcd_prewap doesn't carry them) and stay
   *  manually entered. Purely a client-side lookup against the already-loaded eligible-wells
   *  list (see store.loadLinkedRequestWells()) — selecting a link never itself triggers a
   *  network call.
   *
   *  Whichever fields the link actually supplies are then locked read-only: they track the source
   *  request record, so letting someone edit or clear them here would silently diverge from (or
   *  desync) that source. Also reused at initial section-open for an already-linked record (see
   *  requestReferenceEdit above) — re-locks the same fields to the same values, so it's a no-op
   *  data-wise there. */
  private applyLinkedRequestWell(referenceGroup: FormGroup, linkedRequestId: string): void {
    console.log('[IR:link] applyLinkedRequestWell — start', { linkedRequestId, eligibleWellsLoaded: this.store.linkedRequestWells().length });
    const well = findLinkedWell(linkedRequestId, this.store.linkedRequestWells());
    if (!well) {
      console.log('[IR:link] applyLinkedRequestWell — no matching eligible well, aborting patch', { linkedRequestId });
      return;
    }

    const hasElevation = linkedWellHasElevation(well);
    const hasCoordinates = linkedWellHasCoordinates(well);

    const patch: Record<string, string | number> = { wellName: well.wellName };

   *  now would claim success before the backend has answered (and would still show next to the
   *  error toast when the save actually fails). WwellIrComponent reports the real outcome once
   *  store.creatingIr() settles.

   *  Status handling: the save payload type only ever accepts 'draft' | 'in-progress' (see
   *  CreateIntegratedRequirementPayload) — there's no contract for "edit this field but keep it
   *  Approved/In Review". Editing a draft/in-progress record keeps its current status; editing
   *  anything further along pulls it back to 'in-progress' (not 'draft', since it was already
   *  worked past that point) so it's clear the record needs to move through review again. */
  private saveMergedRaw(raw: WwellIrFormRawValue, attachmentNames?: readonly string[]): void {

    const status: Extract<IntegratedRequirementStatus, 'draft' | 'in-progress'> =
      current.status === 'draft' || current.status === 'in-progress' ? current.status : 'in-progress';
    const { payload } = mapIrFormValueToSubmission(raw, status);
    const irBulkPayload = buildWwellIrBulkSavePayload(raw, { irSeq: current.irSeq ?? null, existing: current });
    this.store.createIr(payload, irBulkPayload, attachmentNames ?? current.attachmentNames);

  /** Global footer "Save" — merges every currently-open section's live form value (plus
   *  attachments, if that's open too) into a single raw value and saves it in one call, instead of
   *  requiring the user to click each open section's own Save button separately. A section that
   *  isn't open contributes its existing saved value unchanged, exactly as it would for any one
   *  section's own save (see saveMergedRaw / currentRawFormValue). This is also what turns a brand
   *  new local-only draft (ir().irSeq === null) into a real backend record on the very first click,
   *  even if the user never opened a single section. */
  protected saveAll(): void {
    let raw = this.currentRawFormValue();

    const referenceCtrl = this.requestReferenceEdit.control();
    if (referenceCtrl) raw = { ...raw, reference: { main: referenceCtrl.getRawValue() } };

    const programMainCtrl = this.programMainEdit.control();
    if (programMainCtrl) raw = { ...raw, programDetails: { ...raw.programDetails, main: programMainCtrl.getRawValue() } };

    const programDevCtrl = this.programDevEdit.control();
    if (programDevCtrl) {
      raw = { ...raw, programDetails: { ...raw.programDetails, developmentTesting: programDevCtrl.getRawValue() } };
    }

    const geologistCtrl = this.geologistContactEdit.control();
    if (geologistCtrl) raw = { ...raw, contacts: { ...raw.contacts, geologist: geologistCtrl.getRawValue() } };

    const hydrogeologyCtrl = this.hydrogeologyEdit.control();
    if (hydrogeologyCtrl) raw = { ...raw, hydrogeology: { main: hydrogeologyCtrl.getRawValue() } };

    const formationTopsCtrl = this.formationTopsEdit.control();
    if (formationTopsCtrl) raw = { ...raw, subsurface: { ...raw.subsurface, formationTops: formationTopsCtrl.getRawValue() } };

    const casingCtrl = this.casingEdit.control();
    if (casingCtrl) raw = { ...raw, subsurface: { ...raw.subsurface, casing: casingCtrl.getRawValue() } };

    const offsetWellsCtrl = this.offsetWellsEdit.control();
    if (offsetWellsCtrl) {
      raw = {
        ...raw,
        contacts: {
          ...raw.contacts,
          offsetWells: this.offsetWellsArrayControl(offsetWellsCtrl).getRawValue(),
          remarks: this.offsetWellsRemarksControl(offsetWellsCtrl).getRawValue(),
        },
      };
    }

    const attachmentNames = this.attachmentsEditing()
      ? [...this.attachmentsDraftNames(), ...this.attachmentsNewFiles().map(file => file.name)]
      : undefined;

    this.saveMergedRaw(raw, attachmentNames);
    this.closeAllSections();
  }

  private closeAllSections(): void {
    this.requestReferenceEdit.close();
    this.programMainEdit.close();
    this.programDevEdit.close();
    this.geologistContactEdit.close();
    this.hydrogeologyEdit.close();
    this.formationTopsEdit.close();
    this.casingEdit.close();
    this.offsetWellsEdit.close();
    this.closeAttachmentsEdit();
  }

  protected openAttachmentsEdit(): void {
    this.attachmentsDraftNames.set(this.ir().attachmentNames);
    this.attachmentsNewFiles.set([]);
    this.attachmentsEditing.set(true);
  }

  protected closeAttachmentsEdit(): void {
    this.attachmentsEditing.set(false);
  }

  protected removeAttachmentName(name: string): void {
    this.attachmentsDraftNames.update(names => names.filter(existing => existing !== name));
  }

  protected removeNewAttachmentFile(file: File): void {
    this.attachmentsNewFiles.update(files => files.filter(existing => existing !== file));
  }

  protected onAttachmentFilesPicked(event: Event): void {
    const input = event.target as HTMLInputElement;
    const picked = input.files ? Array.from(input.files) : [];
    this.attachmentsNewFiles.update(files => [...files, ...picked]);
    input.value = '';
  }

  protected submitForReview(): void {
    this.store.submitIrForReview(this.ir());
  }
}

 * (presentation/well-bore-view) expects — so both the wizard (live, still-being-edited raw form
 * value) and the read-only IR detail page (a submitted `IntegratedRequirement`) can embed that
 * component unmodified: the wizard as an always-animated "Well Design View" step
 * (`buildWellboreDiagramData`), the detail page as a static as-submitted diagram
 * (`buildWellboreDiagramDataFromIntegratedRequirement`).
 *
 * An Integrated Requirement is always a plan, never an as-drilled record, so no `drlgCasings`
 * entry produced here ever carries a real bottom depth (`wCsgBotDpth`/`wLnrBotDepth`).
 * `well-bore-view`'s renderers only switch a casing/liner to the green "actual" gradient when

 * with the steel/silver "planned" gradient, even though `drlgCasings` is no longer always

const LINER_TYPE = 'Liner';

/**
 * A freshly added (or still-mid-edit) Casing/Formation Top row starts out with empty/null
 * field values — but `mat-select`/number inputs still exist and produce SOME value the moment
 * a user makes any single selection (e.g. just picking "Liner" from the Type dropdown before
 * touching Size or Depth). Mapping every row unconditionally meant a half-filled row already
 * rendered a shape/label on the diagram (a "Liner @ 0" annotation with no real size or depth
 * behind it) as soon as it existed, rather than once it actually described something real. These
 * gates require every field the diagram actually draws from before a row counts — reused by both
 * the Subsurface & Construction step's live preview and the Well Design View step (both call
 * `buildWellboreDiagramData`).
 */
function isRenderableCasingRow(row: WwellIrRawCasingRow): boolean {
  return !!row.type && !!row.sizeIn && row.depthFtBgl !== null && row.depthFtBgl !== undefined;
}

function isRenderableFormationTop(row: WwellIrRawFormationTop): boolean {
  return !!row.name && row.depthFt !== null && row.depthFt !== undefined;
}

const COMPLETION_OPEN_HOLE = 'Open Hole';
const COMPLETION_SCREENED = 'Screened';
const COMPLETION_SCREENED_GRAVEL_PACK = 'Screened + Gravel Pack';
const COMPLETION_LINER_PERFORATED = 'Liner + Perforated';

/** Anchors the depth scale to a stable minimum so a shallow/partial entry doesn't stretch
 * across the full drawing height, and so the scale's domain is never degenerate ([0, 0]) before the
 * (required, but non-linear-stepper-skippable) target depth field is filled in. */
const MIN_SCALE_DEPTH_FT = 500;

/**
 * Never used to match a real casing — `casing.renderer.ts` / `wellbore-depth.helpers.ts`
 * match `drlgCasings` rows to casings by `wCsgOdSz === csg.csgSize`, so this sentinel must be
 * a value no user-entered casing size could ever equal. Only `buildLinerTopAnchors`' rows use
 * this; they're read exclusively through `wLnrOdSz`, never `wCsgOdSz`.
 */
const LINER_ANCHOR_SENTINEL = '__wwell_ir_liner_anchor__';

/**
 * `casing.renderer.ts`'s `getCasingTier` derives each casing's on-screen half-width purely
 * from its position in this array via `indexOf` (tier 0 = narrowest, each later tier wider) —
 * it never reads `csgSize` for width, only for the text label. It also paints casings in this
 * same array order, later entries on top, so whichever entry ends up widest is also drawn last
 * — the standard "wide, shallow casing painted over a narrow, deep one" telescoping look.
 *
 * Sorting by SIZE (ascending — smallest first, so the largest entered size lands last/widest)
 * is what makes width actually track the number the user typed, rather than depth order. For
 * well-formed data (size decreasing with depth, real telescoping construction) this produces
 * the exact same order as depth-descending would — size and depth are inversely correlated
 * there, so there's no behavior change for the common case. It only diverges for physically
 * inverted rows (a deeper casing entered wider than a shallower one, flagged separately by
 * `findInvertedCasingSizeRows` in wwell-ir-casing-warnings.util.ts) — there, the wider-but-
 * deeper entry now paints last and can fully cover the narrower-but-shallower one in their
 * overlapping depth range, since two casings can't actually nest that way in reality either.
 * That's an unavoidable consequence of a genuinely invalid combination, not a rendering bug —
 * see `casing.renderer.ts` (off-limits to edit) for why: `buildCasingPath` always draws a
 * casing as a full tube from ground level (0) to its own depth, so the only way to reflect its
 * real entered size is via this array's order.
 *
 * Unparseable sizes sort as 0in (narrowest/first) — the safest fallback, since
 * `isRenderableCasingRow` already guarantees `sizeIn` is non-empty text, just not necessarily
 * a `parseInchSize`-recognizable number.
 */
function mapCasings(rows: readonly WwellIrRawCasingRow[]): ICasingIR[] {
  const casings: ICasingIR[] = rows.map(row => ({
    csgSize: row.sizeIn,
    csgType: row.type,
    csgDepth: row.depthFtBgl ?? 0,
    csgRemarks: row.remarks || null,
  }));
  return [...casings].sort((a, b) => (parseInchSize(a.csgSize) ?? 0) - (parseInchSize(b.csgSize) ?? 0));
}

/**
 * The wizard's "Top Of Liner (TOL)" field (`tolFtBgl`, shown only for Liner rows) is optional —
 * most IR liners never get one filled in. `liner.renderer.ts`'s `rawLinersFromIR` only treats
 * `wLnrTopDepth` as a genuine top-of-liner value (and shows the "TOL @ ..." label/tooltip row)
 * when `LinerEntry.hasActualTop` is true, which it derives straight from whether this anchor's
 * `wLnrTopDepth` is a real positive number — so this must leave `wLnrTopDepth` unset (`null`)
 * whenever the user hasn't entered one, rather than silently filling it with a computed guess
 * that would then get mislabeled as a real measured depth.
 *
 * This builds synthetic `IDrlgCsg` "anchor" entries carrying ONLY `wLnrTopDepth` (when
 * provided) per liner — never `wLnrBotDepth` or a real `wCsgOdSz` — because `rawLinersFromIR`
 * gates `hasActual` (and therefore the green-vs-silver color choice) strictly on
 * `wLnrBotDepth`, and gates the casing-actual lookup strictly on `wCsgOdSz`. Leaving both unset
 * keeps every casing and liner silver.
 *
 * When TOL is left blank, `rawLinersFromIR` falls back to its own `chainTop` chaining instead:
 * the shallowest liner hangs from the deepest structural casing's shoe; each deeper liner hangs
 * from the liner immediately above it (matching the physical assumption already baked into
 * `renderLinerShapes`'s `hangerY`). Casing rows whose type is Conductor, Surface, Intermediate,
 * or Production are all "structural" here — anything that isn't `'Liner'`, exactly matching how
 * casing.renderer.ts itself distinguishes casing from liner (`csgType !== 'Liner'`).
 *
 * An explicit `tolFtBgl` doesn't otherwise perturb that chain: the next liner's fallback top is
 * still this row's own `depthFtBgl` (its shoe), not its `tolFtBgl`, since a liner physically
 * hangs from the casing/liner above it regardless of how far down its own top was entered.
 */
function buildLinerTopAnchors(rows: readonly WwellIrRawCasingRow[]): IDrlgCsg[] {
  const linersShallowToDeep = rows
    .filter(row => row.type === LINER_TYPE)
    .slice()
    .sort((a, b) => (a.depthFtBgl ?? 0) - (b.depthFtBgl ?? 0));

  return linersShallowToDeep.map(row => ({
    wCsgOdSz: LINER_ANCHOR_SENTINEL,
    wCsgBotDpth: 0,
    wLnrOdSz: row.sizeIn,
    wLnrTopDepth: row.tolFtBgl ?? null,
    wLnrBotDepth: null,
  }));
}

  return Math.max(...depths, MIN_SCALE_DEPTH_FT);
}

/** Shared tail for both entry points below — takes already-normalized raw rows so the actual
 *  gating/mapping (isRenderable*, mapCasings, buildLinerTopAnchors, mapGeologicTops,
 *  buildWellDesignFromForm) is written once regardless of whether the source was a live,
 *  still-being-edited form value or a submitted `IntegratedRequirement`. */
function assembleWellboreDiagramData(
  wellName: string,
  targetDepthFt: number,
  casingRows: readonly WwellIrRawCasingRow[],
  formationRows: readonly WwellIrRawFormationTop[],
  surfaceElevationFtAsl: number | null,
  programMain: WwellIrRawProgramMain,
  hydrology: WwellIrRawHydrology,
): WellboreDiagramData {
  const renderableCasings = casingRows.filter(isRenderableCasingRow);
  const renderableFormationTops = formationRows.filter(isRenderableFormationTop);

  return {
    wellName,
    totalDepth: targetDepthFt,
    displayDepth: computeDisplayDepthFt(targetDepthFt, casingRows, formationRows),
    casings: mapCasings(renderableCasings),
    drlgCasings: buildLinerTopAnchors(renderableCasings),
    geologicTops: mapGeologicTops(renderableFormationTops, surfaceElevationFtAsl),
    hydrogeology: null,
    prewap: null,
    rigActivity: null,
    currentDepth: 0,
    targetDepth: targetDepthFt,
    mudCirculation: [],
    wellDesign: buildWellDesignFromForm(programMain, hydrology),
  };
}

/**
 * Gates the *entire* diagram — depth scale, ground level line, and total-depth label included,
 * not just individual casing/formation-top shapes (see `isRenderableCasingRow`/
 * `isRenderableFormationTop` above). Before any row is complete, `well-bore-view` still draws
 * that scaffold around nothing: a bare 0–500ft ruler, a "GROUND LEVEL" line, and a
 * "TOTAL DEPTH: 0 FT" label with no actual construction data behind any of it. Callers should
 * check this first and render an empty state instead of the diagram until it's true.
 */
export function hasWellboreDiagramData(raw: WwellIrFormRawValue): boolean {
  return (
    raw.subsurface.casing.some(isRenderableCasingRow) ||
    raw.subsurface.formationTops.some(isRenderableFormationTop)
  );
}

export function buildWellboreDiagramData(raw: WwellIrFormRawValue): WellboreDiagramData {
  return assembleWellboreDiagramData(
    raw.reference.main.wellName,
    raw.reference.main.targetDepthFt ?? 0,
    raw.subsurface.casing,
    raw.subsurface.formationTops,
    raw.reference.main.surfaceElevationFtAsl,
    raw.programDetails.main,
    raw.hydrogeology.main,
  );
}

/** `IntegratedRequirementCasingRow`/`FormationTop`/`ProgramDetails` are already-resolved
 *  (non-nullable) counterparts of the wizard's raw row types — these just fill in the raw
 *  shape's optional-text fields (`remarks`/`description`/`openHoleSizeIn`) with `''` in place
 *  of `null`, since a submitted IR's `null` here means the same "nothing entered" as the
 *  wizard's `''`. */
function toRawCasingRows(rows: readonly IntegratedRequirementCasingRow[]): WwellIrRawCasingRow[] {
  return rows.map(row => ({
    type: row.type,
    sizeIn: row.sizeIn,
    depthFtBgl: row.depthFtBgl,
    tolFtBgl: row.tolFtBgl,
    remarks: row.remarks ?? '',
  }));
}

function toRawFormationRows(rows: readonly IntegratedRequirementFormationTop[]): WwellIrRawFormationTop[] {
  return rows.map(row => ({
    name: row.name,
    depthFt: row.depthFt,
    subSeaFtAsl: row.subSeaFtAsl ?? null,
    description: row.description ?? '',
  }));
}

function toRawProgramMain(ir: IntegratedRequirement): WwellIrRawProgramMain {
  return {
    completion: ir.programDetails.completion,
    openHoleSizeIn: ir.programDetails.openHoleSizeIn ?? '',
    acidization: ir.programDetails.acidization,
    transmission: ir.programDetails.transmission,
  };
}

/** `IntegratedRequirementHydrogeology`/`TestProgram` are two separate domain groups that the
 *  wizard's single "Estimated Hydrology" step (and its one flat `WwellIrRawHydrology` raw shape —
 *  see wwell-ir-form.types.ts) collects together — recombine them here the same way. */
function toRawHydrology(ir: IntegratedRequirement): WwellIrRawHydrology {
  return {
    testType: ir.testProgram.testType,
    aquifer: ir.hydrogeology.aquifer,
    siwhpPsi: ir.testProgram.siwhpPsi ?? null,
    estimatedTdsPpm: ir.hydrogeology.estimatedTdsPpm,
    estimatedRateGpm: ir.testProgram.estimatedRateGpm,
    h2sPpm: ir.hydrogeology.h2sPpm,
    estimatedWaterLevelFt: ir.testProgram.estimatedWaterLevelFt,
    estimatedPumpDepthFt: ir.hydrogeology.estimatedPumpDepthFt,
  };
}

/** Same emptiness gate as `hasWellboreDiagramData`, for the IR editor/detail page. `liveCasingRows`/
 *  `liveFormationRows` let the editor substitute its Casing/Formation Tops sections' in-progress
 *  FormArray values while those sections are open for editing — pass `null`/omit for either one
 *  still reads straight from `ir`. */
export function hasWellboreDiagramDataForIntegratedRequirement(
  ir: IntegratedRequirement,
  liveCasingRows?: readonly WwellIrRawCasingRow[] | null,
  liveFormationRows?: readonly WwellIrRawFormationTop[] | null,
): boolean {
  const casingRows = liveCasingRows ?? toRawCasingRows(ir.casing);
  const formationRows = liveFormationRows ?? toRawFormationRows(ir.formationTops);
  return casingRows.some(isRenderableCasingRow) || formationRows.some(isRenderableFormationTop);
}

/** Detail/editor-page counterpart of `buildWellboreDiagramData` — same live `app-well-bore-view`
 *  diagram, fed from a submitted `IntegratedRequirement` instead of the in-progress wizard form,
 *  so those pages show the real engineering rendering instead of the lighter
 *  `wwell-ir-design-preview` sketch. See `hasWellboreDiagramDataForIntegratedRequirement` above
 *  for what `liveCasingRows`/`liveFormationRows` are for — the editor passes its Casing/Formation
 *  Tops sections' live FormArray values while they're open, so the preview updates as rows are
 *  added/edited instead of only once that section is saved.
 *
 *  `liveProgramMain`/`liveHydrology` are the same idea for Program Details (Completion — the
 *  element flags in `buildWellDesignFromForm`) and Estimated Hydrology (static water level / SIWHP
 *  / pump depth): without these, picking a different Completion or test type while that section is
 *  open wouldn't move the diagram until the whole record was saved, unlike casing/formation-top
 *  edits, which already updated it live. */
export function buildWellboreDiagramDataFromIntegratedRequirement(
  ir: IntegratedRequirement,
  liveCasingRows?: readonly WwellIrRawCasingRow[] | null,
  liveFormationRows?: readonly WwellIrRawFormationTop[] | null,
  liveProgramMain?: WwellIrRawProgramMain | null,
  liveHydrology?: WwellIrRawHydrology | null,
): WellboreDiagramData {
  return assembleWellboreDiagramData(
    ir.requestReference.wellName,
    ir.requestReference.targetDepthFt,
    liveCasingRows ?? toRawCasingRows(ir.casing),
    liveFormationRows ?? toRawFormationRows(ir.formationTops),
    ir.requestReference.surfaceElevationFtAsl,
    liveProgramMain ?? toRawProgramMain(ir),
    liveHydrology ?? toRawHydrology(ir),
  );
}

import { ChangeDetectionStrategy, Component, computed, input } from '@angular/core';
import { MatIconModule } from '@angular/material/icon';
import { ApprovalChainStep } from '@models/wwell-planning/wwell-planning.model';
import { IntegratedRequirement } from '@models/wwell-planning/integrated-requirement.model';

const OPEN_HOLE_COMPLETION = 'Open Hole';

/** "Letter" step of the read-only IR view — a printable-looking rendition of the GWD Integrated
 *  Requirements Letter, populated straight from the submitted record (never a separate document
 *  fetched from the backend — there is no such endpoint yet). Section numbering/labels follow
 *  the reference GWD letter template; every value comes directly off `IntegratedRequirement`
 *  fields rather than the editor's `KvRow` builders, since the letter's own labels ("Completion
 *  Design", "Logging Program"…) intentionally read as prose, not as the wizard's field names.
 *  The attachment list is appended at the end, per the same record's `attachmentNames`. */
@Component({
  selector: 'app-wwell-ir-view-letter',
  standalone: true,
  imports: [MatIconModule],
  templateUrl: './wwell-ir-view-letter.component.html',
  styleUrl: './wwell-ir-view-letter.component.scss',
  changeDetection: ChangeDetectionStrategy.OnPush,
})
export class WwellIrViewLetterComponent {
  readonly ir = input.required<IntegratedRequirement>();

  protected readonly isOpenHole = computed(() => this.ir().programDetails.completion === OPEN_HOLE_COMPLETION);
  protected readonly isFlowTest = computed(() => this.ir().testProgram.testType === 'flow-test');

  /** The chain's last (most senior) step stands in for the letter's sign-off line — see
   *  WWELL_IR_CHAIN_ROLES (Hydrogeologist → Lead Hydrogeologist → Manager). Mocked for now, same
   *  as the approval-workflow panel next to this stepper (loadIrApprovalTimeline). */
  protected readonly finalApprover = computed<ApprovalChainStep | null>(() => {
    const chain = this.ir().chain;
    return chain.length ? chain[chain.length - 1] : null;
  });

  protected subSeaFt(depthFt: number): number | null {
    const elevation = this.ir().requestReference.surfaceElevationFtAsl;
    return elevation === null ? null : elevation - depthFt;
  }
}

        // Deliberately sets `previousSelectedIrSeq` before selecting, so the "selection changed"
        // effect below no-ops for this transition instead of calling loadIrDetail (redundant —
        // refreshIrDetail above already covers it — and it would flip isLoadingDetail(), swapping
        // the whole detail panel to the loading skeleton right after the user just watched this
        // exact record save).
        this.previousSelectedIrSeq = savedIrSeq;
        this.selectedIrSeq.set(savedIrSeq);
        // The one fetch a brand-new record still needs: its local draft never had a real approval
        // chain, only whatever buildIntegratedRequirementFromPayload defaulted it to. Doesn't gate
        // isSelecting() (see that computed's comment), so this never reintroduces the skeleton.
        this.store.loadIrApprovalTimeline(savedIrSeq);
      }

      const savedCode = savedIrSeq !== null ? this.store.irs().find(ir => ir.irSeq === savedIrSeq)?.code : undefined;
      const subject = savedCode ?? 'Integrated requirement';
      this.notify.info(wasNewIr ? `${subject} created successfully.` : `${subject} updated successfully.`);
    });

    effect(() => {
      const message = this.store.updateIrStatusError();
      if (message) this.notify.error(message);
    });

    effect(() => {
      const message = this.store.irDetailError();
      if (message) this.notify.error(message);
    });

    effect(() => {
      const message = this.store.irApprovalTimelineError();
      if (message) this.notify.error(message);
    });

    effect(() => {
      const message = this.store.deleteIrError();
      if (message) this.notify.error(message);
    });

    // Reports success once a delete settles — the editor's own Delete IR button shows a spinner
    // while in flight (see wwell-ir-editor.component.html's isDeleting()), so nothing needs
    // announcing at click time. A failure is already toasted by the effect above.
    effect(() => {
      const deleting = this.store.deletingIr();
      const settled = this.wasDeletingIr && !deleting;
      this.wasDeletingIr = deleting;
      if (!settled || this.store.deleteIrError()) return;

      this.notify.info('Integrated requirement deleted.');

      // The deleted record is already gone from store.irs() by this point (WwellIrStore.deleteIr
      // removes it on success) — if it was the one open, fall back to null so the "default to
      // first visible record" effect below picks whatever the list now shows first (or leaves the
      // panel empty if nothing remains), the same path a fresh session start already uses.
      if (this.selectedIrSeq() !== null && !this.store.irs().some(ir => ir.irSeq === this.selectedIrSeq())) {
        this.selectedIrSeq.set(null);
        this.previousSelectedIrSeq = null;
      }
    });

    // Default to the first *visible* record once the list loads — mirrors the reference app,
    // which always has something selected in the detail pane rather than starting on an empty
    // state. Reads `rows()` (lifecycle/status/search-scoped), not the raw store list, so the
    // default selection is never a row the Active/Inactive toggle is currently hiding. Skipped
    // while a pendingDraftIr is open — selectedIrSeq is intentionally left null throughout
    // drafting (the draft isn't in `rows()`), and this effect would otherwise immediately fetch
    // and select an unrelated real record's detail out from under the still-open draft.
    effect(() => {
      if (this.pendingDraftIr()) return;
      const firstIrSeq = this.rows()[0]?.irSeq;
      if (firstIrSeq != null && this.selectedIrSeq() === null) {
        this.selectedIrSeq.set(firstIrSeq);
      }
    });

    // Selection changed → play the reveal transition and fetch that record's real detail
    // (casing/tops/hydrogeology) by ir_seq, its primary key.
    effect(() => {
      const irSeq = this.selectedIrSeq();
      if (irSeq === null || irSeq === this.previousSelectedIrSeq) return;

      const host = this.detailHost?.nativeElement;
      if (host) playIfMotionAllowed(buildDetailRevealTimeline(host));
      this.previousSelectedIrSeq = irSeq;
      this.store.loadIrDetail(irSeq);
      // Separate call — the approval-process API doesn't exist yet (mocked for now), but the
      // request shape already mirrors the real future one: one fetch per selected IR.
      this.store.loadIrApprovalTimeline(irSeq);
      // Re-fetched on every selection, not just once at ngOnInit — this backs the editor's own
      // "Link to Request" dropdown and its live auto-populate-on-pick (see
      // wwell-ir-editor.component.ts's applyLinkedRequestWell / wwell-ir-linked-well.util.ts),
      // used when the user picks a NEW link on the opened record. Eligibility can change
      // server-side between selections in a long-lived session (another user initiates an IR for
      // a well, a well newly becomes eligible...) — no cached/one-time snapshot, always the
      // current server state for whichever record is being opened.
      this.store.loadLinkedRequestWells();
    });

    // Re-plays whenever the visible rows actually change (initial load finishing, or the
    // search term narrowing/widening the result set) — keyed on `rowsSignature()` (identity/order
    // only), not `rows()` itself, so a selected row's detail/timeline arriving in place never
    // replays this across the whole list — see `rowsSignature`'s comment above.
    effect(() => {
      const signature = this.rowsSignature();
      if (this.isLoading() || signature === '') return;

      const host = this.listHost?.nativeElement;
      if (!host) return;

      const items = Array.from(host.querySelectorAll<HTMLElement>('.wwell-ir__row'));
      if (items.length) playIfMotionAllowed(buildListRowsRevealTimeline(items));
    });
  }

  ngOnInit(): void {
    this.store.loadIrs();
    // Started here (not lazily in the editor once a record is selected) so the "Link to Request"
    // dropdown/picker (see loadLinkedRequestWells's other call site, on selection change below)
    // has options ready by the time the first record's editor renders, instead of racing it.
    this.store.loadLinkedRequestWells();
  }

  ngOnDestroy(): void {
    this.pendingSelection.clear();
  }

  protected onSearchChange(term: string): void {
    this.searchTerm.set(term);
  }

  /** Switching lifecycle bucket resets the status chip back to "All" — a chip selected in the old
   *  bucket (e.g. "In Review") would otherwise carry over into Inactive, where it can never match
   *  anything, silently showing an empty list. */
  protected setLifecycleFilter(filter: WwellIrLifecycleFilter): void {
    this.lifecycleFilter.set(filter);
    this.statusFilter.set('all');
  }

  protected toggleLifecycleFilter(): void {
    this.setLifecycleFilter(this.lifecycleFilter() === 'active' ? 'inactive' : 'active');
  }

  protected setStatusFilter(status: IntegratedRequirementStatus | 'all'): void {
    this.statusFilter.set(status);
  }

  // Chips only ever render within the Active lifecycle bucket (see the template's
  // `@if (lifecycleFilter() === 'active')` gate) — "All" here already only ever means "every
  // active-status record", never the inactive/approved ones the toggle hides. Labeling it
  // "Active" instead of the bare, easy-to-misread "All" makes that scope explicit.
  protected statusChipLabel(status: IntegratedRequirementStatus | 'all'): string {
    return status === 'all' ? 'Active' : this.statusMeta[status].label;
  }

  protected statusChipCount(status: IntegratedRequirementStatus | 'all'): number {
    const scoped = this.lifecycleScopedIrs();
    return status === 'all' ? scoped.length : scoped.filter(ir => ir.status === status).length;
  }

  protected selectIr(irSeq: number): void {
    // Selecting a real row discards whatever unsaved local draft was open — nothing was ever
    // persisted for it, so there's nothing to preserve.
    this.pendingDraftIr.set(null);
    this.pendingSelection.select(irSeq, this.selectedIrSeq(), resolved => this.selectedIrSeq.set(resolved));
  }

  /** Normalizes the model's `irSeq?: number | null` to a plain `number | null` for the template
   *  (which can't express an `== null` check under the eqeqeq lint rule). */
  protected rowIrSeq(ir: IntegratedRequirement): number | null {
    return ir.irSeq ?? null;
  }

  protected selectIrRow(ir: IntegratedRequirement): void {
    const irSeq = this.rowIrSeq(ir);
    if (irSeq !== null) this.selectIr(irSeq);
  }

  /** The row's "view" (eye) icon — selects the record exactly like selectIrRow above, plus
   *  switches the detail panel to the read-only stepper. Staying in view mode afterwards (rather
   *  than resetting on every selection change) is what lets the master list double as a way to
   *  browse several records' read-only views in a row, without an extra click back into view
   *  mode each time. */
  protected viewIr(irSeq: number): void {
    this.detailMode.set('view');
    this.selectIr(irSeq);
  }

  /** WwellIrViewComponent's own "Edit" button — switches the currently-open record's detail
   *  panel back to the inline-edit surface. */
  protected setDetailMode(mode: WwellIrDetailMode): void {
    this.detailMode.set(mode);
  }

  /** `cdkVirtualFor`'s identity function — same identity rule as the old `@for`'s `track`
   *  expression, just spelled as a function since `*cdkVirtualFor` (a plain structural directive,
   *  not the `@for` built-in control flow) takes `trackBy` instead of an inline `track` expression. */
  protected trackByIrRow(_index: number, ir: IntegratedRequirement): number | string {
    return ir.irSeq ?? ir.id;
  }

  /** Opens a blank draft in the detail pane and selects it — purely client-side, no backend call,
   *  since there's nothing to save yet (see this class's `pendingDraftIr`). WwellIrEditorComponent
   *  binds to it exactly like any persisted record; its first real Save (a section's own, or the
   *  global "Save" button) is what actually creates the backend row. */
  protected createNewIr(): void {
    const raw = buildWwellDynamicForm(WWELL_IR_FORM_SCHEMA, this.fb).getRawValue() as WwellIrFormRawValue;
    const { payload } = mapIrFormValueToSubmission(raw, 'draft');
    const draft = buildIntegratedRequirementFromPayload(payload, { id: 'draft', code: 'New IR', irSeq: null }, []);

    this.pendingDraftIr.set(draft);
    this.selectedIrSeq.set(null);
    this.previousSelectedIrSeq = null;
    // A brand-new draft has nothing committed to view read-only yet — always open it for editing,
    // regardless of whatever mode the previously-open record was left in.
    this.detailMode.set('edit');

    const host = this.detailHost?.nativeElement;
    if (host) playIfMotionAllowed(buildDetailRevealTimeline(host));
  }

  protected chainBorderColor(chain: readonly ApprovalChainStep[]): string {
    return chainCompletionColor(chain);
  }

  /** Powers the approval-workflow sidebar's "N of M approved" progress line (view mode only). */
  protected approvedStepCount(chain: readonly ApprovalChainStep[]): number {
    return chain.filter(step => step.status === 'approved').length;
  }
}
