/** Prefers the backend's own `{ message }` (see ApiResponse<T> — every exad_agwa endpoint here
 *  returns that envelope on both success and error) over Angular's generic "Http failure
 *  response for ..." text, so a specific backend validation error (e.g. an Oracle constraint
 *  message) reaches the user instead of a vague fallback. */
function errorMessage(err: unknown, fallback: string): string {
  if (err instanceof HttpErrorResponse) {
    // status 0 means the request never reached a server at all (refused connection, DNS failure,
    // CORS rejection, offline). There is no ApiResponse envelope to prefer, and `err.error` holds
    // the transport's own TypeError — surfacing its "Failed to fetch" as though it were a backend
    // message tells the user nothing, so the caller's own copy wins here.
    if (err.status === 0) return fallback;
    const body: unknown = err.error;
    const backendMessage = body && typeof body === 'object' ? (body as { message?: unknown }).message : undefined;
    if (typeof backendMessage === 'string' && backendMessage) return backendMessage;
    return err.message || fallback;
  }
  return err instanceof Error ? err.message : fallback;
}

/** Like errorMessage above, but prefers the SAP trigger endpoint's own `{ detail: [{ msg }] }`
 *  422 validation shape (SapValidationError) over the generic `{ message }` envelope every other
 *  exad_agwa endpoint here uses — so a real "field X is required" from SAP reaches the user
 *  instead of a vague fallback. */
function sapErrorMessage(err: unknown, fallback: string): string {
  if (err instanceof HttpErrorResponse) {
    const body: unknown = err.error;
    const detail = body && typeof body === 'object' ? (body as { detail?: unknown }).detail : undefined;
    if (Array.isArray(detail)) {
      const messages = detail
        .map(item => (item && typeof item === 'object' ? (item as { msg?: unknown }).msg : undefined))
        .filter((msg): msg is string => typeof msg === 'string' && msg.length > 0);
      if (messages.length > 0) return messages.join('; ');
    }
  }
  return errorMessage(err, fallback);
}

 *  exactly what `update_ir_sap_wf_prc` just wrote to sap_wf_id/sap_wf_status

 *  a local placeholder, not SAP's real text). `chain` is deliberately left untouched: nothing was
 *  fetched to refresh it with here — the next time this record is *viewed*

 *  reads SAP's real current status and chain back over this placeholder, everywhere it's shown
 *  (list badge, detail header badge, Approval Workflow sidebar) at once. */
function markSubmittedForReview(list: readonly IntegratedRequirement[], id: string, sapWfId: number, sapWfStatusText: string): IntegratedRequirement[] {
  return list.map(item => (item.id === id ? { ...item, status: 'submit-for-review' as const, sapWfId, sapWfStatusText } : item));
}

function upsertByIrSeq(list: readonly IntegratedRequirement[], created: IntegratedRequirement): IntegratedRequirement[] {
  const existingIndex = created.irSeq != null ? list.findIndex(item => item.irSeq === created.irSeq) : -1;
  if (existingIndex === -1) return [created, ...list];
  // A brand-new record's `created` (see createIr's own buildIntegratedRequirementFromPayload
  // fallback) has no way to know the record's real approval chain (that's a separate fetch; see
  // loadIrApprovalTimeline), so it always starts a fresh "just initiated" chain. Harmless for a
  // first-time INSERT (nothing to preserve yet), but would otherwise wipe out an already-in-
  // progress/approved record's real chain on every subsequent save. Keep whichever chain this
  // record already had.
  return list.map((item, index) => (index === existingIndex ? { ...created, chain: item.chain } : item));
}

    // `recallingIr` deliberately stays true across both calls, not just the first — it's what the
    // editor's footer button reads for its spinner, and flipping it false the moment the recall ack
    // lands would stop the spinner mid-action, right as the follow-up status fetch starts.
    function recallSapWorkflow(irSeq: number, sapWfId: number): void {
      const networkId = authStore.networkId() ?? '';
      sap.recall(WWELL_IR_SAP_PROCESS_ID, sapWfId, networkId).subscribe({
        // SAP applies a recall asynchronously (~5-7s), so one immediate GET /display still reads the
        // old status. Poll until it reports Recalled (each read is shown as it arrives), capped at
        // WWELL_IR_SAP_RECALL_POLL_MAX_ATTEMPTS; a failed read is just retried on the next tick.
        next: () => {
          let sawStatus = false;
          timer(WWELL_IR_SAP_RECALL_POLL_MS, WWELL_IR_SAP_RECALL_POLL_MS)
            .pipe(
              take(WWELL_IR_SAP_RECALL_POLL_MAX_ATTEMPTS),
              concatMap(() => sap.getWorkflowStatus(WWELL_IR_SAP_PROCESS_ID, sapWfId).pipe(catchError(() => of(null)))),
              tap(status => {
                if (!status) return;
                sawStatus = true;
                patchState(store, {
                  irs: store.irs().map(item =>
                    item.irSeq === irSeq
                      ? { ...item, chain: mapSapWorkflowStatusToApprovalChain(status), sapWfStatusText: status.header.wf_status_text }
                      : item,
                  ),
                });
              }),
              takeWhile(status => !status || !isSapWorkflowRecalled(status.header.wf_status_text), true),
            )
            .subscribe({
              complete: () =>
                patchState(store, {
                  recallingIr: false,
                  sapRecallError: sawStatus ? null : 'Workflow recalled, but failed to refresh its latest status',
                }),
            });
        },
        // SAP refused the recall — surface the failure and leave the header exactly as it was, same
        // as a refused trigger.
        error: (err: unknown) =>
          patchState(store, {
            recallingIr: false,
            sapRecallError: sapErrorMessage(err, 'Failed to recall the SAP review workflow'),
          }),
      });
    }

    const loadIrs = rxMethod<void>(
      pipe(
        tap(() => patchState(store, { irsLoading: true, irsError: null })),
        switchMap(() =>
          svc.getIrs().pipe(
            tapResponse({
              next: irs => patchState(store, { irs, irsLoading: false }),
              error: (err: unknown) => patchState(store, { irsLoading: false, irsError: errorMessage(err, 'Failed to load integrated requirements') }),
            }),
          ),
        ),
      ),
    );

    // exhaustMap, not switchMap: this fires once at list init and again on the first selection —
    // milliseconds apart whenever the list is already in the store — and switchMap aborted the first
    // request (a red "(canceled)" row in the network tab) in favor of an identical second one.
    const loadLinkedRequestWells = rxMethod<void>(
      pipe(
        tap(() => {
          console.log('[IR:link] loadLinkedRequestWells — fetching GET /wells/eligible-for-ir');
          patchState(store, { linkedRequestWellsLoading: true, linkedRequestWellsError: null });
        }),
        exhaustMap(() =>
          svc.getEligibleWellsForIr().pipe(
            tapResponse({
              next: linkedRequestWells => {
                console.log('[IR:link] loadLinkedRequestWells — response received', { count: linkedRequestWells.length, linkedRequestWells });
                patchState(store, { linkedRequestWells, linkedRequestWellsLoading: false });
              },
              error: (err: unknown) => {
                console.log('[IR:link] loadLinkedRequestWells — request failed', err);
                patchState(store, {
                  linkedRequestWellsLoading: false,
                  linkedRequestWellsError: errorMessage(err, 'Failed to load wells eligible for IR initiation'),
                });
              },
            }),
          ),
        ),
      ),
    );

    // Stable reference data (not tied to a specific IR selection, unlike loadLinkedRequestWells) —
    // loaded once at list-page init, see WwellIrComponent.ngOnInit.
    const loadGwdOpsMembers = rxMethod<void>(
      pipe(
        tap(() => patchState(store, { gwdOpsMembersLoading: true, gwdOpsMembersError: null })),
        switchMap(() =>
          svc.getGwdOpsMembers().pipe(
            tapResponse({
              next: gwdOpsMembers => patchState(store, { gwdOpsMembers, gwdOpsMembersLoading: false }),
              error: (err: unknown) =>
                patchState(store, { gwdOpsMembersLoading: false, gwdOpsMembersError: errorMessage(err, 'Failed to load GWD ops members') }),
            }),
          ),
        ),
      ),
    );

    const loadAquifers = rxMethod<void>(
      pipe(
        switchMap(() =>
          svc.getAquifers().pipe(
            tapResponse({
              next: aquifers => patchState(store, { aquifers }),
              error: (err: unknown) => notify.error(errorMessage(err, 'Failed to load aquifer list')),
            }),
          ),
        ),
      ),
    );

    const loadDropdownMenus = rxMethod<void>(
      pipe(
        switchMap(() =>
          svc.getDropdownMenus().pipe(
            tapResponse({
              next: dropdownMenus => patchState(store, { dropdownMenus }),
              // Non-fatal: the dropdowns just stay empty (no arrow) until a reload succeeds.
              error: () => undefined,
            }),
          ),
        ),
      ),
    );

    // Keyed by ir_seq (the real primary key once an IR is saved) — switchMap cancels a still-in-
    // flight fetch for a previously-selected row if the user picks another one before it resolves,
    // so a slow first response can never land after (and clobber) a faster later selection.
    const loadIrDetail = rxMethod<number>(
      pipe(
        tap(() => patchState(store, { irDetailLoading: true, irDetailError: null })),
        switchMap(irSeq =>
          svc.getIrDetailBlocks(irSeq).pipe(
            tapResponse({
              next: blocks => {
                const existing = store.irs().find(item => item.irSeq === irSeq);
                // Nothing to merge onto if the summary row isn't loaded (shouldn't happen from the
                // normal select-a-row flow) — just clear the loading flag and leave state as-is.
                if (!existing) {
                  patchState(store, { irDetailLoading: false });
                  return;
                }
                const merged = mergeIrDetailBlocksIntoIntegratedRequirement(existing, blocks, store.geologistNameByEmpNum());
                patchState(store, { irs: upsertByIrSeq(store.irs(), merged), irDetailLoading: false });
                // Only the detail fetch's exad_gwd_ir_header block carries sap_wf_id (the list
                // endpoint doesn't — see wwell-ir-summary.mapper.ts), so this is the first point a
                // real SAP lookup becomes possible for a freshly-selected record.
                loadIrApprovalTimeline({ irSeq, sapWfId: merged.sapWfId });
              },
              error: (err: unknown) =>
                patchState(store, { irDetailLoading: false, irDetailError: errorMessage(err, 'Failed to load integrated requirement detail') }),
            }),
          ),
        ),
      ),
    );

    // Feeds the Approval Workflow sidebar straight from SAP's own GET /display, looked up by the
    // IR's sapWfId (== wf_obj_id, the raw id minted by Submit for Review — see
    // wwell-ir-sap.config.ts). An IR never submitted has no sapWfId and therefore no SAP workflow
    // to look up — that case clears the chain locally instead of calling SAP with nothing to ask
    // about. Same switchMap-cancels-a-stale-fetch reasoning as loadIrDetail applies when the
    // selection changes quickly.
    const loadIrApprovalTimeline = rxMethod<{ irSeq: number; sapWfId: number | null }>(
      pipe(
        tap(() => patchState(store, { irApprovalTimelineLoading: true, irApprovalTimelineError: null })),
        switchMap(({ irSeq, sapWfId }) => {
          // Nothing to look up yet — leave `chain` exactly as it was (the initial local preview;
          // see buildIntegratedRequirementFromPayload) rather than blank it, since chainBorderColor
          // reads it for the detail panel's accent strip in both view and edit mode, not only the
          // view-mode sidebar this gates (wwell-ir.component.html).
          if (sapWfId == null) {
            patchState(store, { irApprovalTimelineLoading: false });
            return EMPTY;
          }
          return sap.getWorkflowStatus(WWELL_IR_SAP_PROCESS_ID, sapWfId).pipe(
            tapResponse({
              next: status => {
                patchState(store, {
                  // header.wf_status_text is the real workflow status SAP is showing right now —
                  // surfaced on the list badge (wwell-ir.component.html) in place of the local
                  // "Submit for Review" stand-in markSubmittedForReview sets optimistically, so a
                  // freshly-submitted record doesn't keep showing that stale label once SAP's own
                  // status is known.
                  irs: store.irs().map(item =>
                    item.irSeq === irSeq
                      ? { ...item, chain: mapSapWorkflowStatusToApprovalChain(status), sapWfStatusText: status.header.wf_status_text }
                      : item,
                  ),
                  irApprovalTimelineLoading: false,
                });
              },
              error: (err: unknown) =>
                patchState(store, {
                  irApprovalTimelineLoading: false,
                  irApprovalTimelineError: sapErrorMessage(err, 'Failed to load the SAP approval workflow status'),
                }),
            }),
          );
        }),
      ),
    );

    /** Every object stored under this IR's prefix (trailing slash so IR 900's query can never match
     *  IR 9006's objects — see S3Service.search), newest-first from the proxy and reversed here to
     *  oldest-uploaded-first (matches `irAttachmentsFor`'s doc comment). A key that doesn't parse as
     *  one of this app's own slots (see `slotFromObjectKey`) is silently skipped rather than shown
     *  with a broken slot. */
    const loadIrAttachments = rxMethod<number>(
      pipe(
        tap(() => patchState(store, { attachmentsLoading: true, attachmentError: null })),
        switchMap(irSeq =>
          s3.search(`ir-attachments/ir-${irSeq}/`).pipe(
            tapResponse({
              next: ({ objects }) => {
                const attachments: IrAttachment[] = [...objects]
                  .reverse()
                  .flatMap(object => {
                    const slot = slotFromObjectKey(object.OBJ_KEY);
                    if (!slot) return [];
                    return [
                      {
                        slot,
                        objectKey: object.OBJ_KEY,
                        label: labelForSlot(slot),
                        fileName: object.ORIGINAL_FILENAME,
                        sizeBytes: object.OBJ_SIZE,
                        contentType: object.CONTENT_TYPE_CD,
                      },
                    ];
                  });
                patchState(store, {
                  irs: store.irs().map(item => (item.irSeq === irSeq ? { ...item, attachments } : item)),
                  attachmentsLoading: false,
                });
              },
              error: (err: unknown) =>
                patchState(store, { attachmentsLoading: false, attachmentError: errorMessage(err, 'Failed to load attachments') }),
            }),
          ),
        ),
      ),
    );

    /** Appends a newly-uploaded attachment onto the record, leaving every existing one (in this
     *  slot or any other) untouched — a slot accumulates attachments rather than holding just one. */
    const appendAttachment = (irSeq: number, attachment: IrAttachment) =>
      store.irs().map(item => (item.irSeq === irSeq ? { ...item, attachments: [...item.attachments, attachment] } : item));

    /** Drops one specific stored attachment (matched by its object key, its only identity) off the
     *  record, leaving every other attachment in place. */
    const withoutAttachment = (irSeq: number, objectKey: string) =>
      store.irs().map(item =>
        item.irSeq === irSeq ? { ...item, attachments: item.attachments.filter(existing => existing.objectKey !== objectKey) } : item,
      );

    /** Marks one pending file as mid-upload (or clears that marking) — shared by the single-item
     *  and sweep upload paths below so both track busy/progress identically per file id. */
    function setPendingUploadBusy(id: string, percent: number | null): void {
      patchState(store, {
        uploadingPendingIds: new Set(store.uploadingPendingIds()).add(id),
        uploadProgressByPendingId: { ...store.uploadProgressByPendingId(), [id]: percent },
      });
    }

    function clearPendingUploadBusy(id: string): void {
      const uploadingPendingIds = new Set(store.uploadingPendingIds());
      uploadingPendingIds.delete(id);
      const uploadProgressByPendingId = { ...store.uploadProgressByPendingId() };
      delete uploadProgressByPendingId[id];
      patchState(store, { uploadingPendingIds, uploadProgressByPendingId });
    }

    /** Drops one staged file (by id) from its slot's pending list, leaving any others staged there untouched. */
    function withoutPendingAttachment(slot: IrAttachmentSlot, id: string): Partial<Record<IrAttachmentSlot, readonly IrPendingAttachment[]>> {
      const remaining = (store.pendingAttachments()[slot] ?? []).filter(item => item.id !== id);
      return { ...store.pendingAttachments(), [slot]: remaining };
    }

    /**
     * Uploads one staged file straight to the S3 object-store proxy — the sole upload call now,
     * with no agwa-planning involvement at all (see ir-attachment.model.ts's class doc comment).
     * Independent of every other file in flight — each call tracks its own busy/progress by the
     * pending item's id (see `uploadingPendingIds`/`uploadProgressByPendingId`), so clicking several
     * rows' Upload buttons in a row fires that many concurrent requests rather than queuing behind
     * one another. Clears the staging only after the proxy confirms, so a failed upload leaves the
     * file in its slot to retry rather than silently dropping the user's pick.
     */
    function uploadOne(irSeq: number, slot: IrAttachmentSlot, item: IrPendingAttachment): void {
      setPendingUploadBusy(item.id, null);
      patchState(store, { attachmentError: null });
      const networkId = authStore.networkId() ?? '';
      const prefix = `ir-attachments/ir-${irSeq}/${slot}`;
      s3.store(item.file, prefix, networkId, item.file.name).subscribe({
        next: ({ percent, response }) => {
          if (!response) {
            setPendingUploadBusy(item.id, percent);
            return;
          }
          const attachment: IrAttachment = {
            slot,
            objectKey: response.key,
            label: labelForSlot(slot),
            fileName: item.file.name,
            sizeBytes: item.file.size,
            contentType: response.content_type,
          };
          clearPendingUploadBusy(item.id);
          patchState(store, {
            irs: appendAttachment(irSeq, attachment),
            pendingAttachments: withoutPendingAttachment(slot, item.id),
          });
          notify.info(`${attachment.label} uploaded.`);
        },
        error: (err: unknown) => {
          clearPendingUploadBusy(item.id);
          patchState(store, { attachmentError: errorMessage(err, `Failed to upload the ${slot} attachment`) });
        },
      });
    }

    return {
      loadIrs(): void {
        loadIrs();
      },

      loadIrDetail(irSeq: number): void {
        loadIrDetail(irSeq);
      },

      loadIrAttachments(irSeq: number): void {
        loadIrAttachments(irSeq);
      },

      /**
       * Stages a picked file alongside whatever else is already staged for that slot — attaching
       * never replaces, only adds. Attaching never uploads on its own either: the user sends it with
       * the file's own upload action (`uploadPendingAttachment`), or, for a New IR with no ir_seq to
       * upload against yet, the first save flushes everything staged via `uploadPendingAttachments`.
       */
      addPendingAttachment(slot: IrAttachmentSlot, file: File): string {
        const item: IrPendingAttachment = { id: crypto.randomUUID(), file };
        const existing = store.pendingAttachments()[slot] ?? [];
        patchState(store, {
          pendingAttachments: { ...store.pendingAttachments(), [slot]: [...existing, item] },
          attachmentError: null,
        });
        return item.id;
      },

      /** Sends one staged file. See `uploadOne` above for why this is safe to call for several
       *  files at once without them fighting over shared progress/busy state. */
      uploadPendingAttachment(irSeq: number, slot: IrAttachmentSlot, id: string): void {
        const item = (store.pendingAttachments()[slot] ?? []).find(candidate => candidate.id === id);
        if (!item) return;
        uploadOne(irSeq, slot, item);
      },

      /** Un-stages one file without sending it — the user changed their mind before uploading. */
      removePendingAttachment(slot: IrAttachmentSlot, id: string): void {
        patchState(store, { pendingAttachments: withoutPendingAttachment(slot, id) });
      },

      /** Drops every held file — the draft was abandoned, or a different record was selected. */
      clearPendingAttachments(): void {
        patchState(store, { pendingAttachments: {} });
      },

      /**
       * Uploads everything held for a draft that has just been assigned `irSeq` — every staged file
       * in every slot, each independently (see `uploadOne`). State is cleared up front so a second
       * save can't re-upload the same files, and a failure on one file doesn't abort the others:
       * each reports its own error and the rest continue.
       */
      uploadPendingAttachments(irSeq: number): void {
        const bySlot = store.pendingAttachments();
        patchState(store, { pendingAttachments: {} });
        (Object.entries(bySlot) as [IrAttachmentSlot, readonly IrPendingAttachment[]][]).forEach(([slot, items]) =>
          items.forEach(item => uploadOne(irSeq, slot, item)),
        );
      },

      /** Deletes one specific stored attachment by its object key — its only identity now. Tracked
       *  by that same key in `deletingAttachmentKeys` so only this one row shows busy, not the rest
       *  of the slot's list. Calls the S3 object-store proxy directly — there is no backend
       *  round-trip for attachments at all anymore (see ir-attachment.model.ts's class doc comment). */
      deleteIrAttachment(irSeq: number, objectKey: string): void {
        const networkId = authStore.networkId() ?? '';
        patchState(store, { deletingAttachmentKeys: new Set(store.deletingAttachmentKeys()).add(objectKey), attachmentError: null });
        s3.delete(objectKey, networkId).subscribe({
          next: () => {
            const deletingAttachmentKeys = new Set(store.deletingAttachmentKeys());
            deletingAttachmentKeys.delete(objectKey);
            patchState(store, { irs: withoutAttachment(irSeq, objectKey), deletingAttachmentKeys });
          },
          error: (err: unknown) => {
            const deletingAttachmentKeys = new Set(store.deletingAttachmentKeys());
            deletingAttachmentKeys.delete(objectKey);
            patchState(store, { deletingAttachmentKeys, attachmentError: errorMessage(err, 'Failed to remove the attachment') });
          },
        });
      },

      loadLinkedRequestWells(): void {
        loadLinkedRequestWells();
      },

      loadGwdOpsMembers(): void {
        loadGwdOpsMembers();
      },

      loadAquifers(): void {
        loadAquifers();
      },

      loadDropdownMenus(): void {
        loadDropdownMenus();
      },

      /**
       * No optimistic echo: the backend only ever confirms an `ir_seq` (see WwellIrSaveResult),
       * so this waits out a real follow-up GET (getIrDetailBlocks) and merges *that* onto the
       * record's last known-real state before `irs` ever changes — never onto whatever was just
       * typed. A field with no backing column at all (client, targetAquifer, targetDepthFt... see
       * mergeIrDetailBlocksIntoIntegratedRequirement's own doc comment) simply won't reflect the
       * edit once this settles, since there is nowhere real to read it back from — better an
       * honest revert than a value that only looks saved until the next reload.
       *
       * `creatingIr` stays true for both round-trips (POST, then GET), so the spinner is what
       * covers the extra latency — nothing renders in between.
       */
      createIr(payload: CreateIntegratedRequirementPayload, irBulkPayload: WwellIrBulkSavePayload): void {
        patchState(store, { creatingIr: true, createIrError: null });
        svc
          .createIr(irBulkPayload)
          .pipe(
            delay(WWELL_SAVE_DELAY_MS),
            switchMap(irSeq => {
              if (irSeq === null) return of({ irSeq, blocks: null });
              return svc.getIrDetailBlocks(irSeq).pipe(
                map(blocks => ({ irSeq, blocks })),
                // The save itself already succeeded here — a hiccup on this second, read-only
                // call must not read as "the save failed" (see the `blocks === null` branch below).
                catchError(() => of({ irSeq, blocks: null })),
              );
            }),
          )
          .subscribe({
            next: ({ irSeq, blocks }) => {
              if (irSeq === null) {
                // No identity to file this under at all — shouldn't happen for a genuine backend
                // success, so there's nothing to merge or select; just stop spinning.
                patchState(store, { creatingIr: false });
                return;
              }
              if (blocks === null) {
                // Confirmed saved (a real ir_seq came back) but the follow-up detail fetch failed
                // — record the real identity so a resave correctly UPDATEs instead of re-INSERTing,
                // but leave `irs` untouched rather than show anything unconfirmed.
                patchState(store, {
                  draftIrSeq: irSeq,
                  creatingIr: false,
                  createIrError: 'Saved, but the latest data failed to reload — refresh to see it.',
                });
                return;
              }
              const existing = store.irs().find(item => item.irSeq === irSeq);
              const base =
                existing ?? buildIntegratedRequirementFromPayload(payload, { id: `IR-${irSeq}`, code: `IR-${irSeq}`, irSeq });
              const merged = mergeIrDetailBlocksIntoIntegratedRequirement(base, blocks, store.geologistNameByEmpNum());
              patchState(store, {
                irs: upsertByIrSeq(store.irs(), merged),
                draftIrSeq: irSeq,
                creatingIr: false,
              });
              // A brand-new record's local draft never had a real sapWfId — this GET's own
              // exad_gwd_ir_header block is the first point one could exist (a resave of an
              // already-submitted IR).
              loadIrApprovalTimeline({ irSeq, sapWfId: merged.sapWfId });
            },
            error: (err: unknown) => patchState(store, { creatingIr: false, createIrError: errorMessage(err, 'Failed to create integrated requirement') }),
          });
      },

      /**
       * "Submit for Review" is the SAP workflow submission, in three steps against two backends:
       * mint a workflow id (agwa-planning's get_wf_next_id_fnc) → push the workflow to SAP AAE
       * under that id (SapService.trigger) → record the id and a local "in progress" placeholder
       * on the IR header (update_ir_sap_wf_prc). Deliberately does not also call SAP's own GET
       * /display to ask for a real status — see WWELL_IR_SAP_WF_STATUS_IN_PROGRESS's own doc
       * comment — so submitting stays fast and only ever costs one round-trip to SAP itself.
       *
       * There is no separate IR status-transition call: exad_gwd_ir_header has no status column of
       * its own, so the SAP workflow columns this writes ARE the record of having submitted.
       * `updatingIrStatus` stays true across all three so the editor's Submit button spins for the
       * whole action rather than flickering between steps.
       */
      submitIrForReview(ir: IntegratedRequirement): void {
        // A draft that was never saved has no ir_seq to file the workflow under, so there would be
        // nothing to record the outcome against — refuse before calling SAP rather than after.
        if (ir.irSeq == null) {
          patchState(store, { sapTriggerError: 'Save the integrated requirement before submitting it for review.' });
          return;
        }
        if (!ir.attachments.some(attachment => attachment.slot === 'location-map')) {
          patchState(store, { sapTriggerError: 'Attach a Location Map before submitting for review.' });
          return;
        }
        patchState(store, { updatingIrStatus: true, sapTriggerError: null });
        triggerSapWorkflow(ir, ir.irSeq);
      },

      /** "Recall" pulls back an already-submitted workflow — the editor only shows the button once
       *  `sapWfId` is set (see WwellIrEditorComponent.canRecallIr), but this guards it again since
       *  the store method is the one place that actually matters: there is no workflow to recall
       *  without a real `irSeq`/`sapWfId` pair to act on. */
      recallIr(ir: IntegratedRequirement): void {
        if (ir.irSeq == null || ir.sapWfId == null) {
          patchState(store, { sapRecallError: 'This integrated requirement has not been submitted for review.' });
          return;
        }
        patchState(store, { recallingIr: true, sapRecallError: null });
        recallSapWorkflow(ir.irSeq, ir.sapWfId);
      },

      /** On-demand counterpart to the automatic loadIrDetail → loadIrApprovalTimeline refresh: lets
       *  the UI (the Approval Workflow sidebar's refresh button, the detail header's status badge)
       *  pull SAP's real current status/chain for an already-selected record without needing a
       *  reselect. A no-op for a record with no `sapWfId` — there is no SAP workflow to ask about
       *  yet, same guard `loadIrApprovalTimeline` itself already applies. */
      refreshIrApprovalStatus(irSeq: number, sapWfId: number | null): void {
        if (sapWfId == null) return;
        loadIrApprovalTimeline({ irSeq, sapWfId });
      },

      /** Deletes the whole IR (header row, cascading to casing/tops/hydrogeology/water — see
       *  WwellIrService.deleteIr) and drops it from `irs` on success. `irSeq` must already be a
       *  persisted record's — there's nothing to delete server-side for a local-only draft. */
      deleteIr(irSeq: number): void {
        patchState(store, { deletingIr: true, deleteIrError: null });
        svc
          .deleteIr(irSeq)
          .pipe(delay(WWELL_DELETE_DELAY_MS))
          .subscribe({
            next: () => patchState(store, { irs: store.irs().filter(item => item.irSeq !== irSeq), deletingIr: false }),
            error: (err: unknown) => patchState(store, { deletingIr: false, deleteIrError: errorMessage(err, 'Failed to delete integrated requirement') }),
          });
      },
    };
  }),
);

import { filter, merge } from 'rxjs';
import { takeUntilDestroyed } from '@angular/core/rxjs-interop';
import { ChangeDetectionStrategy, Component, DestroyRef, computed, effect, inject, signal } from '@angular/core';
import { MatButtonModule } from '@angular/material/button';
import { MAT_DIALOG_DATA, MatDialogModule, MatDialogRef } from '@angular/material/dialog';
import { MatIconModule } from '@angular/material/icon';
import { MatProgressBarModule } from '@angular/material/progress-bar';
import { DomSanitizer, SafeUrl } from '@angular/platform-browser';
import { IrAttachmentSlot } from '@models/wwell-planning/ir-attachment.model';
import { WwellIrStore } from '../store/wwell-ir.store';

export interface WwellIrAttachmentUploadDialogData {
  /** Passed in rather than injected: the editor's store is provided on a parent component, which a
   *  dialog opened from the root injector can't see. */
  readonly store: InstanceType<typeof WwellIrStore>;
  readonly slot: IrAttachmentSlot;
  readonly slotLabel: string;
  readonly slotIcon: string;
  /** The staged file's id (see WwellIrStore.addPendingAttachment). */
  readonly pendingId: string;
  /** Null for a New IR that hasn't been saved yet — there's no ir_seq to upload under. */
  readonly irSeq: number | null;
}

/** How long the success state stays up before the dialog closes itself. */
const SUCCESS_LINGER_MS = 900;

/**
 * Opens right after a file is attached: previews the file, offers Upload and, once clicked, shows
 * the real byte-progress, then a brief success state before closing itself.
 *
 * Cancel (button, Esc or backdrop) discards the staged file entirely — nothing half-attached is left
 * in the list. The one exception is an IR that hasn't been saved yet: it has no ir_seq to upload
 * under, so the primary action is "Attach", which keeps the file staged to upload with the first save.
 */
@Component({
  selector: 'app-wwell-ir-attachment-upload-dialog',
  standalone: true,
  imports: [MatButtonModule, MatDialogModule, MatIconModule, MatProgressBarModule],
  templateUrl: './wwell-ir-attachment-upload-dialog.component.html',
  styleUrl: './wwell-ir-attachment-upload-dialog.component.scss',
  changeDetection: ChangeDetectionStrategy.OnPush,
})
export class WwellIrAttachmentUploadDialogComponent {
  protected readonly data = inject<WwellIrAttachmentUploadDialogData>(MAT_DIALOG_DATA);
  private readonly dialogRef = inject(MatDialogRef<WwellIrAttachmentUploadDialogComponent>);
  private readonly sanitizer = inject(DomSanitizer);
  private readonly store = this.data.store;

  /** Captured once: the store drops the staged item the moment the upload succeeds, but the success
   *  state still needs the file's name. */
  protected readonly file = (this.store.pendingAttachments()[this.data.slot] ?? []).find(entry => entry.id === this.data.pendingId)?.file;
  protected readonly isImage = this.file?.type.startsWith('image/') ?? false;
  protected readonly isPdf = this.file?.type === 'application/pdf';
  protected readonly sizeLabel = formatBytes(this.file?.size ?? 0);
  protected readonly typeLabel = this.file?.name.split('.').pop()?.toUpperCase() ?? 'FILE';

  private readonly objectUrl = this.isImage && this.file ? URL.createObjectURL(this.file) : null;
  protected readonly previewUrl: SafeUrl | null = this.objectUrl ? this.sanitizer.bypassSecurityTrustUrl(this.objectUrl) : null;

  protected readonly busy = computed(() => this.store.uploadingPendingIds().has(this.data.pendingId));
  protected readonly percent = computed(() => this.store.uploadProgressByPendingId()[this.data.pendingId] ?? null);
  protected readonly error = this.store.attachmentError;
  protected readonly canUpload = this.data.irSeq !== null;
  protected readonly done = signal(false);
  private cancelled = false;

  constructor() {
    inject(DestroyRef).onDestroy(() => {
      if (this.objectUrl) URL.revokeObjectURL(this.objectUrl);
    });

    // Esc / backdrop behave like Cancel, except mid-upload when dismissal is ignored.
    this.dialogRef.disableClose = true;
    merge(
      this.dialogRef.backdropClick(),
      this.dialogRef.keydownEvents().pipe(filter(event => event.key === 'Escape')),
    )
      .pipe(takeUntilDestroyed())
      .subscribe(() => this.cancel());

    // The store drops the staged item only after the proxy confirms the upload.
    effect(() => {
      if (this.cancelled) return;
      const staged = (this.store.pendingAttachments()[this.data.slot] ?? []).some(entry => entry.id === this.data.pendingId);
      if (staged || this.done()) return;
      this.done.set(true);
      setTimeout(() => this.dialogRef.close(true), SUCCESS_LINGER_MS);
    });
  }

  protected cancel(): void {
    if (this.busy() || this.done()) return;
    this.cancelled = true;
    this.store.removePendingAttachment(this.data.slot, this.data.pendingId);
    this.dialogRef.close(false);
  }

  /** Unsaved IR only: keep the file staged and let the first save upload it. */
  protected attachForLater(): void {
    this.dialogRef.close(true);
  }

  protected upload(): void {
    if (this.busy() || this.done() || this.data.irSeq === null) return;
    this.store.uploadPendingAttachment(this.data.irSeq, this.data.slot, this.data.pendingId);
  }
}

function formatBytes(bytes: number): string {
  if (bytes < 1024) return `${bytes} B`;
  if (bytes < 1024 * 1024) return `${(bytes / 1024).toFixed(1)} KB`;
  return `${(bytes / (1024 * 1024)).toFixed(1)} MB`;
}

export const WWELL_IR_ATTACHMENT_SLOTS: readonly WwellIrAttachmentSlotConfig[] = [

    icon: 'map',

 * Restricted to what the backend can actually embed as PDF pages (see agwa-planning's
 * IrAttachmentPdfAppender). Office formats would need a headless converter, so they're refused at
 * upload rather than stored as something that silently never appears in the letter.
 */
export const WWELL_IR_ATTACHMENT_ACCEPT = '.pdf,.jpg,.jpeg,.png';

/** Purely a client-side guard now — the S3 object-store proxy enforces no size cap of its own, so
 *  this exists only to fail fast before spending time on an upload that's almost certainly wrong. */
export const WWELL_IR_ATTACHMENT_MAX_BYTES = 25 * 1024 * 1024;

/** Every file occupying a slot, oldest-uploaded first — shared by the editor, letter and details
 *  views. A slot can hold any number of attachments, not just one, so callers render this as a list
 *  rather than picking a single result. */
export function irAttachmentsFor(attachments: readonly IrAttachment[], slot: IrAttachmentSlot): readonly IrAttachment[] {
  return attachments.filter(attachment => attachment.slot === slot);
}

/**
 * Recovers the slot from one of this app's own object keys
 * (`ir-attachments/ir-<irSeq>/<slot>/<uuid>`). Anchored on the literal `ir-attachments` segment
 * rather than the key's start, because the real S3 proxy prefixes every key with its own
 * environment segment (e.g. `dev/`) that this app never generates and must simply tolerate.
 * Returns null for a key that doesn't look like one of this app's own.
 */
export function slotFromObjectKey(objectKey: string): IrAttachmentSlot | null {
  const segments = objectKey.split('/');
  const anchor = segments.indexOf('ir-attachments');
  const candidate = anchor >= 0 ? segments[anchor + 2] : undefined;
  const config = WWELL_IR_ATTACHMENT_SLOTS.find(entry => entry.slot === candidate);
  return config ? config.slot : null;
}

/** Display label for a slot ("Location Map") — used to build an `IrAttachment` straight from a
 *  search result, since the proxy itself has no notion of this app's slot labels. */
export function labelForSlot(slot: IrAttachmentSlot): string {
  return WWELL_IR_ATTACHMENT_SLOTS.find(entry => entry.slot === slot)?.label ?? slot;
}

    live: readonly TRow[] | null,
    saved: () => readonly TRow[],
    draftRows: readonly TRow[] | null,
    keyOf: (row: TRow) => string,
  ): readonly TRow[] | null {
    if (!draftRows?.length) return live;
    const rows = live ?? saved();
    const keys = new Set(rows.map(keyOf));
    // "Add" refuses a row whose natural key is already taken (it would overwrite that row) — including by
    // another staged draft row — so a colliding draft is not drawn either.
    const additions = draftRows.filter(draft => {
      const key = keyOf(draft);
      if (keys.has(key)) return false;
      keys.add(key);
      return true;
    });
    return additions.length ? [...rows, ...additions] : live;
  }

  /** Commits every row staged in a "+" add-form's own FormArray into the section's live FormArray — the
   *  same array inline edits and deletes write to, so Save diffs it like any other change. All-or-nothing:
   *  Save matches rows on their natural key, so if ANY staged row's key is already taken (by an existing
   *  row, or by another row in the same batch) the whole commit is refused rather than silently overwriting
   *  that row, or committing some rows and leaving the colliding one stranded in the add-form. */
  private commitNewRows<TRow>(
    add: SectionEditState<FormArray>,
    section: WwellFormSectionConfig,
    commit: {
      readonly existingKeys: readonly string[];
      readonly keyOf: (row: TRow) => string;
      readonly liveArray: () => FormArray | null;
      readonly duplicate: (key: string) => string;
      readonly added: (count: number) => string;
    },
  ): void {
    const form = add.control();
    if (!form || form.invalid || form.length === 0) return;

    const rows = form.getRawValue() as TRow[];
    const keys = new Set(commit.existingKeys);
    for (const row of rows) {
      const key = commit.keyOf(row);
      if (keys.has(key)) {
        this.notify.warn(commit.duplicate(key));
        return;
      }
      keys.add(key);
    }

    const array = commit.liveArray();
    if (!array) return;
    rows.forEach(row => {
      addRepeatableRow(array, section.repeatable?.rowFields ?? [], this.fb);
      array.at(array.length - 1).patchValue(row as object);
    });
    add.close();
    this.notify.info(commit.added(rows.length));
  }

  private saveSettledResolvers: ((succeeded: boolean) => void)[] = [];
  private wasSavingForNavigation = false;

  constructor() {
    effect(onCleanup => {
      const ctrl = this.requestReferenceEdit.control();
      if (!ctrl) return;
      const linkedRequestIdControl = ctrl.get('linkedRequestId');
      const subscription = linkedRequestIdControl?.valueChanges.subscribe((linkedRequestId: string) => {
        console.log('[IR:link] "Link to Request" select — valueChanges fired', { linkedRequestId });
        if (linkedRequestId) this.applyLinkedRequestWell(ctrl, linkedRequestId);
      });
      onCleanup(() => subscription?.unsubscribe());
    });

    effect(() => {
      const id = this.ir().id;
      if (this.previousIrId !== null && id !== this.previousIrId) this.closeAllSections();
      this.previousIrId = id;
    });

    effect(() => {
      if (this.ir().irSeq != null) return;
      untracked(() => this.requestReferenceEdit.open());
    });

    effect(() => {
      const saving = this.store.creatingIr();
      const settled = this.wasSavingForNavigation && !saving;
      this.wasSavingForNavigation = saving;
      if (!settled || this.saveSettledResolvers.length === 0) return;

      const succeeded = !this.store.createIrError();
      const resolvers = this.saveSettledResolvers;
      this.saveSettledResolvers = [];
      resolvers.forEach(resolve => resolve(succeeded));
    });
  }

  public saveUnsavedChangesAndWait(): Promise<boolean> {

      if (this.store.creatingIr()) {
        this.saveSettledResolvers.push(resolve);
        return;

      if (this.saveAll()) {
        this.saveSettledResolvers.push(resolve);

    });

  private applyLinkedRequestWell(referenceGroup: FormGroup, linkedRequestId: string): void {
    console.log('[IR:link] applyLinkedRequestWell — start', { linkedRequestId, eligibleWellsLoaded: this.store.linkedRequestWells().length });
    const well = findLinkedWell(linkedRequestId, this.store.linkedRequestWells());
    if (!well) {

    const hasTargetDepth = linkedWellHasTargetDepth(well);

    if (hasElevation) patch['surfaceElevationFtAsl'] = well.surfaceElevation as number;
    patch['coordinates'] = hasCoordinates ? formatLinkedWellCoordinates(well) : '';
    if (hasTargetDepth) patch['targetDepthFt'] = well.targetDepth as number;

      hasElevation,
      hasCoordinates,
      hasTargetDepth,
      hasTargetAquifer,
      hasSupportingBusiness,

    referenceGroup.get('wellName')?.disable();

  /** Returns whether a save was actually kicked off — `false` on every early-return branch below
   *  (nothing to save, or a blocked validation). `saveUnsavedChangesAndWait` needs this to settle
   *  its own promise immediately in those cases instead of waiting forever for a `creatingIr`
   *  transition that will now never happen. */
  protected saveAll(): boolean {
    // A record whose stored remarks still carry legacy HTML is saved even with nothing edited — the form already
    // holds the cleaned text, and saving is what writes it back.
    if (this.ir().irSeq !== null && !this.hasUnsavedChanges() && !this.ir().hasLegacyMarkup) {
      this.notify.info('No changes to save.');
      return false;
    }

    let raw = this.currentRawFormValue();

    const referenceCtrl = this.requestReferenceEdit.control();
    if (referenceCtrl) raw = { ...raw, reference: { main: referenceCtrl.getRawValue() } };

    // `letterNumber` carries a `Validators.required` in the schema (wwell-ir-form.schema.ts), but
    // nothing elsewhere in this method ever checks form validity before saving — it only gathers
    // raw values. Without this explicit gate, a blank Letter Number saved silently; the asterisk
    // and mat-error only ever showed once the field itself had been touched.
    if (!raw.reference.main.letterNumber?.trim()) {
      this.requestReferenceEdit.open();
      this.requestReferenceEdit.control()?.get('letterNumber')?.markAsTouched();
      this.notify.error('Letter Number is required before saving.');
      return false;
    }

    const programCtrl = this.programEdit.control();
    if (programCtrl) {
      const flat = programCtrl.getRawValue() as WwellIrRawProgramDetails;
      raw = {
        ...raw,
        programDetails: {
          main: {
            completion: flat.completion,
            openHoleSizeIn: flat.openHoleSizeIn,
            acidization: flat.acidization,
            transmission: flat.transmission,
          },
          developmentTesting: {
            logging: flat.logging,
            mudLogging: flat.mudLogging,
            sampling: flat.sampling,
            flowTest: flat.flowTest,
            remarks: flat.remarks,
          },
        },
      };
    }

    const geologistCtrl = this.geologistContactEdit.control();
    if (geologistCtrl) raw = { ...raw, contacts: { ...raw.contacts, geologist: geologistCtrl.getRawValue() } };

    const hydrogeologyCtrl = this.hydrogeologyEdit.control();
    if (hydrogeologyCtrl) raw = { ...raw, hydrogeology: { main: hydrogeologyCtrl.getRawValue() } };

    const formationTopsCtrl = this.formationTopsEdit.control();
    if (formationTopsCtrl) raw = { ...raw, subsurface: { ...raw.subsurface, formationTops: formationTopsCtrl.getRawValue() } };

    const topsRemarksCtrl = this.topsRemarksEdit.control();
    if (topsRemarksCtrl) raw = { ...raw, subsurface: { ...raw.subsurface, remarks: topsRemarksCtrl.getRawValue() } };

    const casingCtrl = this.casingEdit.control();
    if (casingCtrl) raw = { ...raw, subsurface: { ...raw.subsurface, casing: casingCtrl.getRawValue() } };

    const offsetWellsCtrl = this.offsetWellsEdit.control();
    if (offsetWellsCtrl) {
      raw = {
        ...raw,
        contacts: {
          ...raw.contacts,
          offsetWells: this.offsetWellsArrayControl(offsetWellsCtrl).getRawValue(),
          remarks: this.offsetWellsRemarksControl(offsetWellsCtrl).getRawValue(),
        },
      };
    }

    this.saveMergedRaw(raw);
    this.closeAllSections();
    return true;
  }

import { WWELL_IR_ATTACHMENT_SLOTS } from '../wwell-ir-attachments.config';
import { IntegratedRequirement } from '@models/wwell-planning/integrated-requirement.model';

export type WwellIrTimelineStepStatus = 'completed' | 'pending';

/** Per-rail-group completion status for the Timeline view (see WWELL_IR_SECTION_RAIL) — a group
 *  counts as "completed" once it holds any meaningful data, so the timeline can tell work already
 *  captured apart from sections still waiting on input. `groupKey` matches a top-level
 *  WwellIrRailGroup.key; anything else falls back to 'pending'. */
export function wwellIrTimelineStepStatus(ir: IntegratedRequirement, groupKey: string): WwellIrTimelineStepStatus {
  switch (groupKey) {
    case 'wellbore':
      return ir.requestReference.wellName?.trim() ? 'completed' : 'pending';

    case 'general': {
      const { completion, acidization, transmission, openHoleSizeIn, developmentTesting } = ir.programDetails;
      const developmentFilled = Object.values(developmentTesting).some(value => value?.trim());
      const geologistFilled = !!ir.contacts.geologistPrimary?.trim();
      return completion || acidization || transmission || openHoleSizeIn || developmentFilled || geologistFilled
        ? 'completed'
        : 'pending';
    }

    case 'horizons':
      return ir.formationTops.length > 0 || !!ir.topsRemarks?.trim() ? 'completed' : 'pending';

    case 'casing':
      return ir.casing.length > 0 ? 'completed' : 'pending';

    case 'hydrogeology': {
      const { aquifer, estimatedTdsPpm, h2sPpm, estimatedPumpDepthFt } = ir.hydrogeology;
      return aquifer?.trim() || estimatedTdsPpm != null || h2sPpm != null || estimatedPumpDepthFt != null
        ? 'completed'
        : 'pending';
    }

    case 'areaWaterWells':
      return ir.offsetWells.length > 0 || !!ir.offsetWellsRemarks?.trim() ? 'completed' : 'pending';

    // Both slots are required by the form schema (see wwell-ir-form.schema.ts), so the step is
    // complete only once each one holds at least one file.
    case 'attachments':
      return WWELL_IR_ATTACHMENT_SLOTS.every(({ slot }) => ir.attachments.some(attachment => attachment.slot === slot))
        ? 'completed'
        : 'pending';

    default:
      return 'pending';
  }
}

export function buildWwellIrFormSchema(
  linkedRequestOptions: Signal<readonly WwellFieldOption[]>,
  geologistOptions: Signal<readonly string[]>,
  dropdownMenus: Signal<Readonly<Record<string, readonly string[]>>> = signal({}),
  aquiferOptions: Signal<readonly WwellFieldOption[]> = signal([]),
): readonly WwellFormStepConfig[] {
  return WWELL_IR_FORM_SCHEMA.map(step => {
    if (step.key === 'subsurface') {
      return {
        ...step,
        sections: step.sections.map(section =>
          section.key === 'formationTops' && section.repeatable
            ? {
                ...section,
                repeatable: {
                  ...section.repeatable,
                  rowFields: section.repeatable.rowFields.map(field => (field.key === 'name' ? { ...field, options: aquiferOptions } : field)),
                },
              }
            : section,
        ),
      };
    }
    if (step.key === 'hydrogeology') {
      return {
        ...step,
        sections: step.sections.map(section => ({
          ...section,
          fields: section.fields?.map(field => (field.key === 'aquifer' ? { ...field, options: aquiferOptions } : field)),
        })),
      };
    }
    if (step.key === 'programDetails') {
      return {
        ...step,
        sections: step.sections.map(section => ({
          ...section,
          fields: section.fields?.map(field => withDropdownMenuOptions(field, dropdownMenus)),
        })),
      };
    }
    if (step.key === 'reference') {
      return {
        ...step,
        sections: step.sections.map(section => {
          if (section.key !== 'main' || !section.fields) return section;
          return {
            ...section,
            fields: section.fields.map(field =>
              field.key === 'linkedRequestId' ? { ...field, options: linkedRequestOptions } : field,
            ),
          };
        }),
      };
    }
    if (step.key === 'contacts') {
      return {
        ...step,
        sections: step.sections.map(section => {
          if (section.key !== 'geologist' || !section.fields) return section;
          return {
            ...section,
            fields: section.fields.map(field =>
              field.key === 'geologistPrimary' || field.key === 'geologistSecondary' ? { ...field, options: geologistOptions } : field,
            ),
          };
        }),
      };
    }
    return step;
  });
}

/** Drives a Program Details dropdown purely from its EXAD_DHUB_LKUP_DATA menu — empty until loaded,
 *  and empty if the table has no rows for it (any fallback lives in WwellIrStore.loadDropdownMenus). */
function withDropdownMenuOptions(field: WwellFieldConfig, dropdownMenus: Signal<Readonly<Record<string, readonly string[]>>>): WwellFieldConfig {
  const menu = WWELL_IR_DROPDOWN_MENU_BY_FIELD[field.key];
  if (!menu) return field;
  return { ...field, options: computed(() => dropdownMenus()[menu] ?? []) };
}

import { IntegratedRequirement, IntegratedRequirementStatus } from '@models/wwell-planning/integrated-requirement.model';
import { WWELL_STATUS_COLORS, WwellStatusMeta, wwellStatusMeta } from '../shared/wwell-status.config';

/** SAP text that means the workflow was actively killed rather than merely progressing — matched
 *  case-insensitively, the same way wwell-ir-approval.mapper.ts's stepStatus() reads a single
 *  step's own `response`. SAP's exact wording for these isn't part of our contract (the /display
 *  response only documents `wf_status_text` as free text), so this matches on the words themselves
 *  rather than a fixed value list. */
const SAP_TERMINATED_STATUS_TEXT = /reject|cancel|recall/i;
const SAP_APPROVED_STATUS_TEXT = /approve/i;
const SAP_RECALLED_STATUS_TEXT = /recall/i;

/** Every status renders the same view+inline-edit surface (`WwellIrEditorComponent`) now — this
 *  list instead gates the one status-transition action that still needs a distinct status ("Submit
 *  for Review" only makes sense while the record hasn't already been submitted). */
export const WWELL_IR_SUBMITTABLE_STATUSES: readonly IntegratedRequirementStatus[] = ['draft', 'in-progress'];

export const WWELL_IR_STATUS_META: Record<IntegratedRequirementStatus, WwellStatusMeta> = {
  draft: wwellStatusMeta('Draft', WWELL_STATUS_COLORS.slate),
  'in-progress': wwellStatusMeta('In Progress', WWELL_STATUS_COLORS.blue),
  'submit-for-review': { label: 'Submit for Review', bg: 'var(--wp-accent-fill)', color: 'var(--wp-on-accent)', dot: 'var(--wp-on-accent)' },
  'in-review': wwellStatusMeta('In Review', WWELL_STATUS_COLORS.purple),
  approved: wwellStatusMeta('Approved', WWELL_STATUS_COLORS.green),
};

/** True once a record's status badge has something real to show: `status === 'approved'` (a real
 *  fact off publish_flg) or a fetched `sapWfStatusText` (SAP's own current text, or the local
 *  "in progress" placeholder submitIrForReview writes — see WwellIrStore.loadIrApprovalTimeline /
 *  WWELL_IR_SAP_WF_STATUS_IN_PROGRESS). Everything else is the list endpoint's blind "in-progress"
 *  default (no sap_wf_id column to tell a genuine draft from an already-submitted record apart —
 *  see wwell-ir-summary.mapper.ts) and must not be shown as though it were real. Shared by the
 *  list badge, the detail header badge (editor + view), and nowhere else — so all three agree. */
export function hasKnownIrStatus(ir: IntegratedRequirement): boolean {
  return ir.status === 'approved' || ir.sapWfStatusText !== null;
}

/** The status badge's actual colour+label — same fallback rule as the label alone used to follow
 *  (`ir.sapWfStatusText ?? statusMeta[ir.status].label`), but the colour now follows the SAP text
 *  too instead of always trailing `ir.status`. `IntegratedRequirementStatus` has no slot for SAP's
 *  own terminal states (Rejected/Cancelled/Recalled all just leave `ir.status` at whatever it was
 *  pre-submission — see WwellIrStore.loadIrApprovalTimeline), so without this a rejected/cancelled/
 *  recalled record kept showing its old "In Review"/"Submit for Review" colour with only the label
 *  text changing underneath it. Recognised SAP text is shown red (terminated) or green (approved);
 *  anything else keeps riding on `ir.status`'s own colour, only the label swaps, same as before. */
export function irBadgeMeta(ir: IntegratedRequirement): WwellStatusMeta {
  const text = ir.sapWfStatusText;
  const fallback = WWELL_IR_STATUS_META[ir.status];
  if (!text) return fallback;
  if (SAP_TERMINATED_STATUS_TEXT.test(text)) return wwellStatusMeta(text, WWELL_STATUS_COLORS.red);
  if (SAP_APPROVED_STATUS_TEXT.test(text)) return wwellStatusMeta(text, WWELL_STATUS_COLORS.green);
  return { ...fallback, label: text };
}

/** True once SAP's own text says the workflow is done moving — rejected, cancelled, recalled, or
 *  approved — same two patterns irBadgeMeta colours red/green. Gates the editor's "Recall" button
 *  (WwellIrEditorComponent.canRecallIr): a workflow already at a terminal SAP state has nothing left
 *  to recall, same as a real SAP recall would refuse it. */
export function isSapWorkflowTerminal(sapWfStatusText: string | null): boolean {
  return sapWfStatusText !== null && (SAP_TERMINATED_STATUS_TEXT.test(sapWfStatusText) || SAP_APPROVED_STATUS_TEXT.test(sapWfStatusText));
}

/** True only for the recalled subset of `isSapWorkflowTerminal` — unlike Approved/Rejected/Cancelled,
 *  a recall was the user's own action, so the record isn't done: `WwellIrEditorComponent` brings
 *  "Submit for Review" back for it (`canSubmitForReview`), letting them resubmit fresh under a new
 *  SAP workflow id. Deliberately its own predicate rather than reusing `isSapWorkflowTerminal` — the
 *  other three terminal outcomes stay terminal, with no way back from the editor. */
export function isSapWorkflowRecalled(sapWfStatusText: string | null): boolean {
  return sapWfStatusText !== null && SAP_RECALLED_STATUS_TEXT.test(sapWfStatusText);
}

/** Classifies the live SAP status text into the IR letter PDF's three stamp states — the exact
 *  same two patterns `irBadgeMeta` uses to colour the badge red/green, reused here so this is the
 *  one place that decision gets made. Sent directly to agwa-planning (see
 *  `WwellIrService.downloadLetterPdf`'s `stampStatus` param / `LetterPdfRequest.stampStatus`) so
 *  the backend just stamps whatever it's told rather than keeping its own copy of this
 *  classification. `null` when the IR was never submitted (no SAP text at all) — the backend
 *  skips the stamp image entirely in that case instead of defaulting to "Pending". */
export function resolveLetterStampStatus(ir: IntegratedRequirement): 'APPROVED' | 'REJECTED' | 'PENDING' | null {
  const text = ir.sapWfStatusText;
  if (!text) return null;
  if (SAP_TERMINATED_STATUS_TEXT.test(text)) return 'REJECTED';
  if (SAP_APPROVED_STATUS_TEXT.test(text)) return 'APPROVED';
  return 'PENDING';
}

/** Every field the IR form schema marks `required` (wwell-ir-form.schema.ts), outside the
 *  repeatable rows — Casing/Formation Tops/Offset Wells only validate a row once one exists, so
 *  there's no top-level requirement that those sections hold any rows at all. Gates "Submit for
 *  Review": checked against the already-saved `ir` itself, not a live, possibly-still-open form,
 *  because submitting sends exactly this record (WwellIrEditorComponent.submitForReview calls
 *  `store.submitIrForReview(this.ir())`) — an in-progress edit elsewhere doesn't change whether
 *  the saved record is actually ready. The one attachment checked is the Location Map: it must be
 *  uploaded before the record can go for review (Initiation Request is not part of this gate). */
export function wwellIrHasRequiredSubmitFields(ir: IntegratedRequirement): boolean {
  return (
    !!ir.requestReference.linkedRequestId &&
    !!ir.requestReference.letterNumber?.trim() &&
    !!ir.requestReference.wellName.trim() &&
    !!ir.programDetails.completion.trim() &&
    !!ir.contacts.geologistPrimary.trim() &&
    !!ir.hydrogeology.aquifer.trim() &&
    !!ir.testProgram.testType &&
    ir.attachments.some(attachment => attachment.slot === 'location-map')
  );
}
