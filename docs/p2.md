```typescript
import { computed, inject } from '@angular/core';
import { patchState, signalStore, withComputed, withMethods, withState } from '@ngrx/signals';
import { rxMethod } from '@ngrx/signals/rxjs-interop';
import { tapResponse } from '@ngrx/operators';
import { pipe, switchMap, tap } from 'rxjs';
import { ApprovalDecision, CreateWwellPlanningRequestPayload } from '@models/wwell-planning/wwell-planning.model';
import { WwellPlanningService } from '../wwell-planning.service';
import { WwellPlanningStoreState, initialWwellPlanningState } from './wwell-planning.state';

function errorMessage(err: unknown, fallback: string): string {
  return err instanceof Error ? err.message : fallback;
}

export const WwellPlanningStore = signalStore(
  withState<WwellPlanningStoreState>(initialWwellPlanningState),

  withComputed(({ approvals, requestsLoading, approvalsLoading, integratedRequirementsLoading, error }) => ({
    isLoading: computed(() => requestsLoading() || approvalsLoading() || integratedRequirementsLoading()),
    hasError: computed(() => error() !== null),
    pendingApprovalsCount: computed(() => approvals().filter(a => a.decision === 'pending').length),
  })),

  withMethods((store, svc = inject(WwellPlanningService)) => {
    const loadRequests = rxMethod<void>(
      pipe(
        tap(() => patchState(store, { requestsLoading: true, error: null })),
        switchMap(() =>
          svc.getRequests().pipe(
            tapResponse({
              next: requests => patchState(store, { requests, requestsLoading: false }),
              error: (err: unknown) =>
                patchState(store, { requestsLoading: false, error: errorMessage(err, 'Failed to load requests') }),
            }),
          ),
        ),
      ),
    );

    const loadApprovals = rxMethod<void>(
      pipe(
        tap(() => patchState(store, { approvalsLoading: true, error: null })),
        switchMap(() =>
          svc.getApprovals().pipe(
            tapResponse({
              next: approvals => patchState(store, { approvals, approvalsLoading: false }),
              error: (err: unknown) =>
                patchState(store, { approvalsLoading: false, error: errorMessage(err, 'Failed to load approvals') }),
            }),
          ),
        ),
      ),
    );

    const loadIntegratedRequirements = rxMethod<void>(
      pipe(
        tap(() => patchState(store, { integratedRequirementsLoading: true, error: null })),
        switchMap(() =>
          svc.getIntegratedRequirements().pipe(
            tapResponse({
              next: integratedRequirements => patchState(store, { integratedRequirements, integratedRequirementsLoading: false }),
              error: (err: unknown) =>
                patchState(store, {
                  integratedRequirementsLoading: false,
                  error: errorMessage(err, 'Failed to load integrated requirements'),
                }),
            }),
          ),
        ),
      ),
    );

    return {
      loadRequests(): void {
        loadRequests();
      },

      loadApprovals(): void {
        loadApprovals();
      },

      loadIntegratedRequirements(): void {
        loadIntegratedRequirements();
      },

      decideApproval(approvalId: string, decision: ApprovalDecision, comments: string | null = null): void {
        svc.decideApproval(approvalId, decision, comments).subscribe({
          next: updated =>
            patchState(store, {
              approvals: store.approvals().map(a => (a.id === updated.id ? updated : a)),
            }),
          error: (err: unknown) => patchState(store, { error: errorMessage(err, 'Failed to update approval') }),
        });
      },

      createRequest(payload: CreateWwellPlanningRequestPayload, files: readonly File[] = []): void {
        patchState(store, { creatingRequest: true, error: null });
        svc.createRequest(payload, files).subscribe({
          next: created =>
            patchState(store, {
              requests: [created, ...store.requests()],
              creatingRequest: false,
            }),
          error: (err: unknown) =>
            patchState(store, { creatingRequest: false, error: errorMessage(err, 'Failed to create request') }),
        });
      },

      setUiError(message: string | null): void {
        patchState(store, { error: message });
      },
    };
  }),
);
```

```typescript
export class WwellApprovalsComponent implements OnInit {

  protected readonly hasError = this.store.hasError;

  protected readonly selectedApproval = signal<WwellPlanningApproval | null>(null);
  protected readonly comments = signal('');

  private gridApi: GridApi | null = null;

  ngOnInit(): void {
    this.store.loadApprovals();
  }

  protected onGridReady(api: GridApi): void {
    this.gridApi = api;
    this.gridApi.sizeColumnsToFit();
  }

    const [selected] = (this.gridApi?.getSelectedRows() as WwellPlanningApproval[]) ?? [];
    this.selectedApproval.set(selected ?? null);
    this.comments.set('');

    this.decide('approved');
  }

  protected reject(): void {
    this.decide('rejected');
```

```typescript
import { Injectable, inject } from '@angular/core';
import { HttpClient } from '@angular/common/http';
import { Observable, map } from 'rxjs';
import {
  ApprovalDecision,
  CreateWwellPlanningRequestPayload,
  IntegratedRequirement,
  WwellPlanningApproval,
  WwellPlanningRequest,
} from '@models/wwell-planning/wwell-planning.model';
import { ApiResponse } from 'src/app/shared/models/wwell/api-response.model';
import { ExternalConfigService } from 'src/app/shared/services/external-config.service';

@Injectable({ providedIn: 'root' })
export class WwellPlanningService {
  private readonly http = inject(HttpClient);
  private readonly extConfigService = inject(ExternalConfigService);

  private get apiUrl(): string {
    const serverUrl = this.extConfigService.settings.planningServiceUrl ?? '';
    return `${serverUrl}/planning/api/v1`;
  }

  getRequests(): Observable<WwellPlanningRequest[]> {
    return this.http
      .get<ApiResponse<WwellPlanningRequest>>(`${this.apiUrl}/requests`)
      .pipe(map(res => res.data));
  }

  createRequest(
    payload: CreateWwellPlanningRequestPayload,
    files: readonly File[] = [],
  ): Observable<WwellPlanningRequest> {
    const formData = new FormData();

    Object.entries(payload).forEach(([key, value]) => {
      if (value !== null && value !== undefined) {
        formData.append(key, String(value));
      }
    });
    files.forEach(file => formData.append('attachments', file, file.name));

    return this.http
      .post<ApiResponse<WwellPlanningRequest>>(`${this.apiUrl}/requests`, formData)
      .pipe(map(res => res.data[0]));
  }

  getApprovals(): Observable<WwellPlanningApproval[]> {
    return this.http
      .get<ApiResponse<WwellPlanningApproval>>(`${this.apiUrl}/approvals`)
      .pipe(map(res => res.data));
  }

  getIntegratedRequirements(): Observable<IntegratedRequirement[]> {
    return this.http
      .get<ApiResponse<IntegratedRequirement>>(`${this.apiUrl}/integrated-requirements`)
      .pipe(map(res => res.data));
  }

  decideApproval(approvalId: string, decision: ApprovalDecision, comments: string | null): Observable<WwellPlanningApproval> {
    return this.http
      .patch<ApiResponse<WwellPlanningApproval>>(`${this.apiUrl}/approvals/${approvalId}`, { decision, comments })
      .pipe(map(res => res.data[0]));
  }
}
```

```typescript
  selector: 'app-wwell-request-form',
  standalone: true,
  imports: [
    ReactiveFormsModule,
    MatButtonModule,

    MatSelectModule,
  ],
  templateUrl: './wwell-request-form.component.html',
  styleUrl: './wwell-request-form.component.scss',
  changeDetection: ChangeDetectionStrategy.OnPush,
  encapsulation: ViewEncapsulation.None,

export class WwellRequestFormComponent {
  private readonly fb = inject(FormBuilder);
  private readonly dialogRef = inject(MatDialogRef<WwellRequestFormComponent, WwellRequestFormResult | undefined>);
  protected readonly data = inject<WwellRequestFormDialogData>(MAT_DIALOG_DATA);

  protected readonly isViewMode = this.data.mode === 'view';
  protected readonly isDragOver = signal(false);
  protected readonly attachments = signal<AttachmentItem[]>([]);

  protected readonly form = this.fb.nonNullable.group({
    client: [this.data.request?.client ?? '', Validators.required],
    supporting: [this.data.request?.supporting ?? '', Validators.required],
    coordinates: [

      [Validators.required, Validators.pattern(WWELL_REQUEST_COORDINATES_PATTERN)],

    if (this.form.invalid) {
      this.form.markAllAsTouched();
      return;

    const value = this.form.getRawValue();
    const payload: CreateWwellPlanningRequestPayload = {
      client: value.client,
      supporting: value.supporting,
      coordinates: value.coordinates,
      targetAquifer: value.targetAquifer,
      proposedDepth: value.proposedDepth,
      businessUse: value.businessUse,
      assignedTo: value.assignedTo,
      priority: value.priority,
      notes: value.notes?.trim() || null,
      status,
    };

    this.dialogRef.close({ payload, files: this.attachments().map(a => a.file) });
  }

  protected onDragOver(event: DragEvent): void {
    event.preventDefault();
    event.stopPropagation();
    if (!this.isViewMode) this.isDragOver.set(true);
  }

  protected onDragLeave(event: DragEvent): void {
    event.preventDefault();
    event.stopPropagation();
    this.isDragOver.set(false);
  }

  protected onDrop(event: DragEvent): void {
    event.preventDefault();
    event.stopPropagation();
    this.isDragOver.set(false);
    if (this.isViewMode) return;
    this.addFiles(Array.from(event.dataTransfer?.files ?? []));
  }

  protected onFileInputChange(event: Event): void {
    const input = event.target as HTMLInputElement | null;
    this.addFiles(Array.from(input?.files ?? []));
    if (input) input.value = '';
  }

  protected removeAttachment(id: string): void {
    this.attachments.update(items => items.filter(item => item.id !== id));
  }

  protected fileSizeLabel(size: number): string {
    if (size < 1024) return `${size} B`;
    if (size < 1024 * 1024) return `${(size / 1024).toFixed(1)} KB`;
    return `${(size / (1024 * 1024)).toFixed(1)} MB`;
  }

  private addFiles(selected: File[]): void {
    if (!selected.length) return;

    const valid = selected.filter(
      file =>
        this.hasAllowedExtension(file) && file.size <= WWELL_REQUEST_MAX_ATTACHMENT_SIZE_BYTES,
    );
    if (!valid.length) return;

    this.attachments.update(existing => [
      ...existing,
      ...valid.map((file, index) => ({ file, id: `${file.name}-${file.size}-${existing.length + index}` })),
    ]);
  }

  private hasAllowedExtension(file: File): boolean {
    const ext = file.name.split('.').pop()?.toLowerCase() ?? '';
    return (WWELL_REQUEST_ALLOWED_ATTACHMENT_EXTENSIONS as readonly string[]).includes(ext);
  }
}
```

```typescript
#!/usr/bin/env -S npx tsx
/**
 * Decodes docs/agwa-user-manual.docx.base64.txt back into a .docx file.
 *
 * Usage:
 *   npx tsx scripts/decode-manual-base64.ts
 *   npx tsx scripts/decode-manual-base64.ts <input.base64.txt> <output.docx>
 */
import { readFileSync, writeFileSync } from 'fs';
import { resolve } from 'path';

interface DecodeOptions {
  readonly inputPath: string;
  readonly outputPath: string;
}

function parseArgs(argv: readonly string[]): DecodeOptions {
  const [inputArg, outputArg] = argv;
  const inputPath = resolve(inputArg ?? 'docs/agwa-user-manual.docx.base64.txt');
  const outputPath = resolve(outputArg ?? 'docs/agwa-user-manual.docx');
  return { inputPath, outputPath };
}

function decodeBase64File({ inputPath, outputPath }: DecodeOptions): void {
  const base64 = readFileSync(inputPath, 'utf8').trim();
  const buffer = Buffer.from(base64, 'base64');
  writeFileSync(outputPath, buffer);
  console.log(`Decoded ${base64.length} base64 chars → ${buffer.length} bytes`);
  console.log(`Wrote ${outputPath}`);
}

function main(): void {
  const options = parseArgs(process.argv.slice(2));
  try {
    decodeBase64File(options);
  } catch (error: unknown) {
    const message = error instanceof Error ? error.message : 'Unknown error';
    console.error(`Failed to decode: ${message}`);
    process.exit(1);
  }
}

main();
```
