import { FormControl } from '@angular/forms';
import { MatCheckboxChange, MatCheckboxModule } from '@angular/material/checkbox';
import { MatIconModule } from '@angular/material/icon';
import { MatMenuModule } from '@angular/material/menu';
import { Subscription } from 'rxjs';
import {
  WwellFieldConfig,
  WwellFieldOption,
  resolveFieldOptions,
  wwellFieldOptionLabel,
  wwellFieldOptionValue,
} from '../wwell-dynamic-form.types';

/**
 * "Dropdown + big textarea" control for the IR Program Details remarks fields (Acidization,
 * Transmission, Logging, Mud Logging, Sampling, Flow Test, Development/Testing) — several of
 * these carry long, multi-line legacy remarks (see wwell-ir-legacy-html.util.ts), too much for the
 * single-line `combo` autocomplete input the rest of the dynamic-form engine uses.
 *
 * The dropdown is a checkbox list, not a single-pick select: any number of the field's canned
 * options can be checked at once, and each checked option becomes its own line in the box below
 * (see `toggleOption`) — unchecking one removes exactly that line. A box line that doesn't match
 * any option's label (free-typed text) is left alone either way. Checked state is derived live
 * from the box's own content (`checkedLabels`), not tracked separately, so typing a line that
 * happens to match an option's label also shows it checked, and deleting that line unchecks it.
 *
 * The `FormControl<string>` this binds to always holds plain text with real `\n` line breaks —
 * never markup. Deliberately no rich-formatting toolbar (bold/italic/font-size were all tried and
 * dropped) — the IR "no HTML in storage" fix (wwell-ir-legacy-html.util.ts) intentionally keeps
 * these columns plain-text-only, so there was never anything for formatting to actually persist.
 *
 * Not a `ControlValueAccessor` — `DynamicFieldComponent` (this control's only caller) already
 * hands every field its own live `FormControl` instance directly rather than binding through
 * `formControlName`/`[formControl]`, so this only needs to read/write that instance's `.value`
 * and listen to its `valueChanges`, not implement the full CVA contract.
 */
@Component({
  selector: 'app-wwell-rich-textarea',
  standalone: true,
  imports: [MatIconModule, MatMenuModule, MatCheckboxModule],
  templateUrl: './wwell-rich-textarea.component.html',
  styleUrl: './wwell-rich-textarea.component.scss',
  changeDetection: ChangeDetectionStrategy.OnPush,
})
export class WwellRichTextareaComponent implements AfterViewInit, OnDestroy {
  readonly config = input.required<WwellFieldConfig>();
  readonly control = input.required<FormControl>();
  /** Class applied to the options dropdown's portaled mat-menu panel, so a host page can theme it — mirrors DynamicFieldComponent's own `selectPanelClass`. */
  readonly selectPanelClass = input('');

  protected readonly optionLabel = wwellFieldOptionLabel;
  protected readonly optionValue = wwellFieldOptionValue;
  protected readonly resolvedOptions = computed(() => resolveFieldOptions(this.config()));
  protected readonly hasOptions = computed(() => this.resolvedOptions().length > 0);
  /** The section's own sub-heading (`groupTitle`) already says exactly what this label would — printing both stacks the same word twice. */
  protected readonly repeatsGroupTitle = computed(() => this.config().groupTitle === this.config().label);

  private readonly boxRef = viewChild<ElementRef<HTMLDivElement>>('box');
  private valueChangesSubscription?: Subscription;
  /** Set immediately before this component pushes its own edit into the control, so the
   *  `valueChanges` echo that causes doesn't get rebuilt back into the DOM — rebuilding on every
   *  keystroke would reset the caret to the start of the box on every character typed. */
  private isOwnEdit = false;

  /** Mirrors the control's current plain-text value as a signal purely so `checkedLabels` (and
   *  therefore each checkbox's checked state) stays reactive — kept in sync on every value change
   *  regardless of source (typed, or a checkbox toggle), unlike `isOwnEdit` which only gates
   *  whether the DOM box itself needs rebuilding. */
  private readonly boxText = signal('');
  private readonly checkedLabels = computed(() => new Set(this.linesOf(this.boxText())));
  /** Backs the remove-icon column — one button per current line, same order, always derived from
   *  `boxText` so it (and the checkboxes above, via `checkedLabels`) never drift out of sync with
   *  whatever the box actually contains, regardless of whether a line arrived by checking an
   *  option, typing directly, or a previous remove-icon click. */
  protected readonly currentLines = computed(() => this.linesOf(this.boxText()));

  ngAfterViewInit(): void {
    // Normalizes Enter-key behavior across browsers to one `<div>` per line (Chrome/Edge already
    // default to this; Firefox defaults to `<br>`-separated lines within a single block instead)
    // — renderIntoBox/extractPlainText below both assume "one line == one `<div>`".
    document.execCommand('defaultParagraphSeparator', false, 'div');
    const initial = this.control().value ?? '';
    this.boxText.set(initial);
    this.renderIntoBox(initial);
    this.valueChangesSubscription = this.control().valueChanges.subscribe((value: string) => {
      this.boxText.set(value ?? '');
      if (this.isOwnEdit) {
        this.isOwnEdit = false;
        return;
      }
      this.renderIntoBox(value ?? '');
    });
  }

  ngOnDestroy(): void {
    this.valueChangesSubscription?.unsubscribe();
  }

  /** A dropdown checkbox flipping on appends that option's label as a new line (unless it's
   *  already present); flipping off removes exactly that line. Multiple options can be checked at
   *  once — each becomes its own line, always appended, never inserted mid-text. */
  protected toggleOption(option: string | WwellFieldOption, event: MatCheckboxChange): void {
    const label = this.optionLabel(option);
    const lines = this.linesOf(this.extractPlainText());
    const nextLines = event.checked
      ? (lines.includes(label) ? lines : [...lines, label])
      : lines.filter(line => line !== label);
    this.renderIntoBox(nextLines.join('\n'));
    this.commitFromBox();
  }

  protected isOptionChecked(option: string | WwellFieldOption): boolean {
    return this.checkedLabels().has(this.optionLabel(option));
  }

  /** The remove-icon column's click handler — drops one line by position (not by text, so two
   *  identical lines can be removed independently) and commits, which flows straight back through
   *  `boxText`/`checkedLabels` and unchecks that option's checkbox if the removed line matched one. */
  protected removeLine(index: number): void {
    const lines = this.linesOf(this.boxText());
    lines.splice(index, 1);
    this.renderIntoBox(lines.join('\n'));
    this.commitFromBox();
  }

  protected onBoxInput(): void {
    this.commitFromBox();
  }

  private commitFromBox(): void {
    const text = this.extractPlainText();
    this.isOwnEdit = true;
    this.boxText.set(text);
    this.control().setValue(text);
    this.control().markAsDirty();
    this.control().markAsTouched();
  }

  /** Deliberately does NOT drop blank lines — `currentLines()` (backing the remove-icon column)
   *  needs to stay index-aligned 1:1 with `renderIntoBox`'s one-`<div>`-per-line output, blank
   *  lines included, or a blank line in the middle would leave the icon column one short and every
   *  icon after it would line up with the wrong row. A genuinely empty box still yields `[]` here
   *  (the `text ? ... : []` guard), which is what keeps the icon column hidden when there's really
   *  nothing in the box — see the template's `@if (currentLines().length > 0)`. */
  private linesOf(text: string): string[] {
    return text ? text.split('\n') : [];
  }

  /** `innerText` (not `textContent`) — reflects the box's actual rendered lines as `\n`-joined
   *  text, matching how `renderIntoBox` below wrote them in (one `<div>` per line). */
  private extractPlainText(): string {
    const text = this.boxRef()?.nativeElement.innerText ?? '';
    return text.replace(/\n+$/, '');
  }

  /** Rebuilds the box's content from plain text — one `<div>` per line, an empty line rendered as
   *  `<div><br></div>` so it keeps its line height (a truly empty `<div>` collapses to zero height
   *  in most browsers). Leaves the box with no children at all for an empty value, so the
   *  `:empty::before` placeholder in the stylesheet actually shows. */
  private renderIntoBox(text: string): void {
    const box = this.boxRef()?.nativeElement;
    if (!box) return;
    box.textContent = '';
    if (!text) return;

    text.split('\n').forEach(line => {
      const row = document.createElement('div');
      if (line) row.textContent = line;
      else row.appendChild(document.createElement('br'));
      box.appendChild(row);
    });
  }
}

/** Schema vocabulary for the wwell-planning dynamic form engine. Add a field = edit a schema array, not a template. */

import type { Signal } from '@angular/core';

export type WwellFieldControlType =
  | 'text'
  | 'number'
  | 'textarea'
  | 'select'
  | 'combo'
  | 'rich-textarea'
  | 'toggle-group'
  | 'file';

export interface WwellFieldOption {
  readonly value: string;
  readonly label: string;
  /** Material icon ligature name — used by 'toggle-group' only. */
  readonly icon?: string;
}

export interface WwellFieldValidatorConfig {
  readonly required?: boolean;
  readonly pattern?: RegExp;
  readonly min?: number;
}

/**
 * Field is only shown (and its control enabled) while a sibling control in the same section
 * matches this condition. Exactly one of `equals` | `oneOf` | `notOneOf` | `includesLine` should
 * be set:
 *  - `equals`: sibling value === this value
 *  - `oneOf`: sibling value is one of this set (e.g. a client tier)
 *  - `notOneOf`: sibling value is absent from this set — the fallback for values outside every
 *    named tier (e.g. a client not yet assigned to a tier keeps the default field set)
 *  - `includesLine`: sibling value, split on `\n` and trimmed, includes this exact line — for a
 *    sibling that's a `type: 'rich-textarea'` checkbox list (see WwellRichTextareaComponent),
 *    whose committed value is every checked option newline-joined rather than a single picked
 *    value, so `equals` can only ever match when exactly one box is checked. `openHoleSizeIn`
 *    (wwell-ir-form.schema.ts) is the motivating case: gated on `completion` (a rich-textarea)
 *    having "Open Hole" checked, which must still hold when other options are checked alongside
 *    it — an `equals` gate there hid (and reset — see DynamicSectionComponent.syncControlEnabled)
 *    this field the instant a second option was checked, silently discarding whatever open hole
 *    size had been typed and leaving `resolveOhHalfWidth` (wellbore-depth.helpers.ts) with no real
 *    diameter to scale Open Hole against — it fell back to a fixed wide default, wider than an
 *    actual entered casing OD.
 */
export interface WwellFieldVisibilityCondition {
  readonly field: string;
  readonly equals?: string;
  readonly oneOf?: readonly string[];
  readonly notOneOf?: readonly string[];
  readonly includesLine?: string;
}

  readonly key: string;
  readonly label: string;
  readonly type: WwellFieldControlType;
  readonly placeholder?: string;
  readonly hint?: string;

   *  (e.g. the IR wizard's "Link to Request" picker) rather than being known at schema-definition

  readonly options?: readonly (string | WwellFieldOption)[] | Signal<readonly (string | WwellFieldOption)[]>;

  readonly defaultValue?: string | number;
  /** 'select' only — adds a filter input inside the option panel, narrowing `options` as the user
   *  types. For a picker backed by a long or live-fetched list (e.g. the IR wizard's "Link to
   *  Request", backed by every well eligible for IR initiation) where scrolling an unfiltered

  readonly searchable?: boolean;
  /** 'select' only — when the control already holds a value the option list doesn't contain (a value pre-filled from

  readonly keepUnlistedValue?: boolean;

  readonly readonly?: boolean;

  readonly derived?: WwellDerivedFieldConfig;

   *  Longitude box. Applies in both edit and review mode; the control itself is untouched (still
   *  one value, one key), this only changes how DynamicFieldComponent draws it. */
  readonly splitDisplay?: (value: string) => readonly { readonly label: string; readonly value: string }[];

  readonly groupTitle?: string;

  readonly minRequiredRows?: number;

  readonly showAddButton?: boolean;

  readonly title: string | null;

  readonly fields?: readonly WwellFieldConfig[];

  readonly key: string;
  readonly label: string;
  readonly sections: readonly WwellFormSectionConfig[];

  return typeof option === 'string' ? option : option.label;
}

  return typeof option === 'string' ? option : option.value;

  const options = field.options;

  if (!coordinates) return [{ label: 'Coordinates', value: EMPTY_VALUE, mono: true }];
  return toUtmAxes(coordinates).map(axis => ({
    label: axis.label,
    value: axis.value,
    mono: true,
  }));
}

/** Mirrors the IR wizard's "Request Reference" step (wwell-ir-form.schema.ts) field-for-field,
 *  so the detail page's sections line up 1:1 with the stepper the data was captured through. */
export function buildRequestReferenceRows(
  ir: IntegratedRequirement,
  linkedRequestWellOptions: readonly { readonly value: string; readonly label: string }[] = [],
  /** "Link to Water Well" is only a picker for creating/editing an IR — read-only displays leave it out (the
   *  Water Well Name row below already says which well it is). */
  includeLinkedWell = false,
): readonly KvRow[] {
  const r = ir.requestReference;
  return [
    ...(includeLinkedWell
      ? [{ label: 'Link to Water Well', value: resolveLinkedRequestLabel(r.linkedRequestId, r.wellName, linkedRequestWellOptions) }]
      : []),
    { label: 'Water Well Name', value: fmtStr(r.wellName) },
    { label: 'Letter Number', value: fmtStr(r.letterNumber), mono: true },
    { label: 'Supporting Business', value: fmtStr(r.supportingBusiness) },
    { label: 'Target Aquifer', value: fmtStr(r.targetAquifer), options: true },
    // A brand-new draft is created through the required-number fallback in
    // mapIrFormValueToSubmission (targetDepthFt ?? 0) before the author has entered a real value —
    // treat a literal 0 as "not yet entered" rather than a misleading real reading (see the
    // matching fallback in mapIntegratedRequirementToFormPatch, wwell-ir-edit.mapper.ts).
    { label: 'Target Depth', value: r.targetDepthFt ? fmtNum(r.targetDepthFt, 'ft') : EMPTY_VALUE, mono: true },
    { label: 'Surface Elevation', value: `${fmtNum(r.surfaceElevationFtAsl)} ft ASL`, mono: true },

}

/** Mirrors the schema's "Program Details" section (2.1) — its own rail leaf/edit unit, separate
 *  from Development & Testing (2.2) below. `openHoleSizeIn` only ever means something when the
 *  completion type includes 'Open Hole' (see the field's `visibleWhen` gate in the schema), so
 *  it's only included then rather than showing a dangling "—" the rest of the time.
 *
 *  `completion` is a multi-select — any number of `WWELL_COMPLETION_OPTIONS` can be checked at
 *  once, each becoming its own `\n`-joined line (see wwell-ir-wellbore-diagram.mapper.ts's own
 *  doc comment on this same shape) — so this must check "is 'Open Hole' one of the checked
 *  lines?", the same `includesLine` semantics the schema's `visibleWhen` gate uses
 *  (dynamic-section.component.ts), not a straight equality that only ever matched the
 *  single-selection case and otherwise silently hid an already-entered size. */
export function buildProgramMainRows(ir: IntegratedRequirement): readonly KvRow[] {
  return [...buildProgramCompletionRows(ir), ...buildProgramTreatmentRows(ir)];
}

/** The completion half of `buildProgramMainRows`: type, the sizes that go with it, and the remarks picks (one line). */
export function buildProgramCompletionRows(ir: IntegratedRequirement): readonly KvRow[] {
  const p = ir.programDetails;
  const rows: KvRow[] = [{ label: 'Completion', value: fmtStr(p.completionType) }];

  // Every pick is drawn inside the open hole, so its size is listed whatever the pick.
  if (p.completionType) {
    rows.push({ label: 'Open Hole Size', value: fmtStr(p.openHoleSizeIn), mono: true });
  }
  if (isLinerCompletionType(p.completionType)) {
    rows.push({ label: 'Liner Screen Size', value: fmtStr(p.linerSizeIn), mono: true });
    rows.push({ label: 'Liner Depth (ft)', value: p.linerDepthFt != null ? String(p.linerDepthFt) : 'Target depth', mono: true });
  }
  rows.push({ label: 'Completion Remarks', value: fmtStr(p.completion), options: true, wide: true, icon: 'construction' });
  return rows;
}

/** The acidization / transmission half of `buildProgramMainRows`. */
export function buildProgramTreatmentRows(ir: IntegratedRequirement): readonly KvRow[] {
  const p = ir.programDetails;
  return [

  ];
}

/** Mirrors the schema's "Development & Testing" section (2.2) — see buildProgramMainRows' doc

export function buildProgramDevRows(ir: IntegratedRequirement): readonly KvRow[] {
  const d = ir.programDetails.developmentTesting;
  return [

    { label: 'Sampling', value: fmtStr(d.sampling), options: true, icon: 'colorize' },
    // Both are always listed here; the IR letter / PDF / SAP copy print only the one matching the Hydrology Test Type.
    { label: 'Flow Test', value: fmtStr(d.flowTest), options: true, icon: 'speed', alwaysShow: true },
    { label: 'Development/Testing', value: fmtStr(d.remarks), wide: true, icon: 'engineering', alwaysShow: true },
  ];
}

/** Mirrors the wizard's "Estimated Hydrology" step, including its test-type-conditional
 *  fields (SIWHP for a flow test; water level + pump depth for a pump test) — see the
 *  matching `visibleWhen` gates on `siwhpPsi`/`estimatedWaterLevelFt`/`estimatedPumpDepthFt`
 *  in wwell-ir-form.schema.ts. Any nonzero H2S gets `tone: 'warning'` — a toxic-gas reading is
 *  worth flagging on sight, not just listed at the same visual weight as TDS/rate. */
export function buildHydrogeologyRows(ir: IntegratedRequirement): readonly KvRow[] {
  const isFlowTest = ir.testProgram.testType === 'flow-test';
  const h2sPpm = ir.hydrogeology.h2sPpm;
  const rows: KvRow[] = [
    { label: 'Aquifer', value: fmtStr(ir.hydrogeology.aquifer), options: true },
    { label: 'Test Type', value: isFlowTest ? 'Flow Test' : 'Pump Test', options: true },
    { label: 'Estimated TDS', value: fmtNum(ir.hydrogeology.estimatedTdsPpm, 'ppm'), mono: true },
    { label: 'Estimated Rate', value: fmtNum(ir.testProgram.estimatedRateGpm, 'GPM'), mono: true },
    {
      label: 'H2S',
      value: fmtNum(h2sPpm, 'ppm'),
      tone: h2sPpm ? 'warning' : 'default',
      mono: true,
    },
  ];

  if (isFlowTest) {
    rows.push({ label: 'SIWHP', value: fmtNum(ir.testProgram.siwhpPsi, 'psi'), mono: true });
  } else {
    rows.push(
      { label: 'Estimated Water Level', value: fmtNum(ir.testProgram.estimatedWaterLevelFt, 'ft'), mono: true },
      { label: 'Estimated Pump Depth', value: fmtNum(ir.hydrogeology.estimatedPumpDepthFt, 'ft'), mono: true },
    );
  }
  return rows;
}

export function buildGeologistContactRows(ir: IntegratedRequirement): readonly KvRow[] {
  const c = ir.contacts;
  return [
    { label: 'Primary', value: fmtStr(c.geologistPrimary), icon: 'person' },
    { label: 'Secondary', value: fmtStr(c.geologistSecondary), icon: 'person_outline' },
  ];
}

  private previewRows<TRow>(
    live: readonly TRow[] | null,
    saved: () => readonly TRow[],
    draftRows: readonly TRow[] | null,
    keyOf: (row: TRow) => string,
  ): readonly TRow[] | null {
    if (!draftRows?.length) return live;
    const rows = live ?? saved();
    const keys = new Set(rows.map(keyOf));
    // "Add" refuses a row whose natural key is already taken (it would overwrite that row) — including by
    // another staged draft row — so a colliding draft is not drawn either.
    const additions = draftRows.filter(draft => {
      const key = keyOf(draft);
      if (keys.has(key)) return false;
      keys.add(key);
      return true;
    });
    return additions.length ? [...rows, ...additions] : live;
  }

  /** Watches an offset-well FormArray (the "+" add form, or the section pencil's edit form) and, whenever a row's
   *  "Select Offset Water Well" value changes, fills that row from the picked well. Re-binds each time the form is
   *  opened afresh; the previous subscription is dropped with the previous array. */
  private bindOffsetWellPicker(array: () => FormArray | null): void {
    effect(onCleanup => {
      const live = array();
      if (!live) return;
      const picked = new WeakMap<FormGroup, unknown>();
      const subscription = live.valueChanges.subscribe(() => {
        for (const row of live.controls as FormGroup[]) {
          const value = row.get(OFFSET_WELL_PICKER_KEY)?.value;
          if (picked.get(row) === value) continue;
          picked.set(row, value);
          if (value) this.applyOffsetWellSource(row, value as string);
        }
      });
      onCleanup(() => subscription.unsubscribe());
    });
  }

  /** Fills a staged row from the picked offset water well — every column except distance and location, which are
   *  specific to the new well and stay as the author typed them. The source list comes from the store
   *  (GET /offset-water-wells). */
  private applyOffsetWellSource(row: FormGroup, value: string | null): void {
    const well = this.store.offsetWellSources().find(source => source.wellName === value);
    if (!well) return;

      aquifer: well.aquifer ?? '',
      totalDepthFt: well.totalDepthFt,
      testDurationMin: well.testDurationMin,
      rateGpm: well.rateGpm,
      rpm: well.rpm,
      swlFt: well.swlFt,
      pwlFt: well.pwlFt,
      drawdownFt: well.drawdownFt,
      specificCapacity: well.specificCapacity,
      tdsPpm: well.tdsPpm,
      h2sPpm: well.h2sPpm,
    });
  }

  /** Commits every row staged in a "+" add-form's own FormArray into the section's live FormArray — the
   *  same array inline edits and deletes write to, so Save diffs it like any other change. All-or-nothing:
   *  Save matches rows on their natural key, so if ANY staged row's key is already taken (by an existing
   *  row, or by another row in the same batch) the whole commit is refused rather than silently overwriting
   *  that row, or committing some rows and leaving the colliding one stranded in the add-form. */
  private commitNewRows<TRow>(
    add: SectionEditState<FormArray>,
    section: WwellFormSectionConfig,
    commit: {
      readonly existingKeys: readonly string[];
      readonly keyOf: (row: TRow) => string;
      /** Last tweak to each row as it is committed (e.g. assigning its save key). */
      readonly prepare?: (row: TRow) => TRow;
      readonly liveArray: () => FormArray | null;
      /** Runs once every staged row is in the live array (e.g. to re-sort it). */
      readonly afterCommit?: (array: FormArray) => void;
      readonly duplicate: (key: string) => string;
      readonly added: (count: number) => string;
    },
  ): void {
    const form = add.control();
    if (!form || form.invalid || form.length === 0) return;

    const rows = form.getRawValue() as TRow[];
    const keys = new Set(commit.existingKeys);
    for (const row of rows) {
      const key = commit.keyOf(row);
      if (keys.has(key)) {
        this.notify.warn(commit.duplicate(key));
        return;
      }
      keys.add(key);
    }

    const array = commit.liveArray();
    if (!array) return;
    rows.forEach(staged => {
      const row = commit.prepare ? commit.prepare(staged) : staged;
      addRepeatableRow(array, section.repeatable?.rowFields ?? [], this.fb);
      array.at(array.length - 1).patchValue(row as object);
    });
    commit.afterCommit?.(array);
    add.close();
    this.notify.info(commit.added(rows.length));
  }

  private saveSettledResolvers: ((succeeded: boolean) => void)[] = [];
  private wasSavingForNavigation = false;

  constructor() {
    this.bindOffsetWellPicker(() => this.offsetWellAdd.control());
    this.bindOffsetWellPicker(() => this.offsetWellsEdit.control());
    effect(onCleanup => {
      const ctrl = this.requestReferenceEdit.control();
      if (!ctrl) return;
      const linkedRequestIdControl = ctrl.get('linkedRequestId');
      const subscription = linkedRequestIdControl?.valueChanges.subscribe((linkedRequestId: string) => {
        console.log('[IR:link] "Link to Request" select — valueChanges fired', { linkedRequestId });
        if (linkedRequestId) this.applyLinkedRequestWell(ctrl, linkedRequestId);
      });
      onCleanup(() => subscription?.unsubscribe());
    });

    effect(() => {
      const id = this.ir().id;
      if (this.previousIrId !== null && id !== this.previousIrId) this.closeAllSections();
      this.previousIrId = id;
    });

    effect(() => {
      if (this.ir().irSeq != null) return;
      untracked(() => this.requestReferenceEdit.open());
    });

    effect(() => {
      const saving = this.store.creatingIr();
      const settled = this.wasSavingForNavigation && !saving;
      this.wasSavingForNavigation = saving;
      if (!settled || this.saveSettledResolvers.length === 0) return;

      const succeeded = !this.store.createIrError();
      const resolvers = this.saveSettledResolvers;
      this.saveSettledResolvers = [];
      resolvers.forEach(resolve => resolve(succeeded));
    });
  }

  public discardUnsavedChanges(): void {
    this.closeAllSections();
  }

  public saveUnsavedChangesAndWait(): Promise<boolean> {
    return new Promise(resolve => {
      if (this.store.creatingIr()) {
        this.saveSettledResolvers.push(resolve);
        return;
      }
      if (this.saveAll()) {
        this.saveSettledResolvers.push(resolve);
      } else {
        resolve(false);
      }
    });
  }

  private applyLinkedRequestWell(referenceGroup: FormGroup, linkedRequestId: string): void {
    console.log('[IR:link] applyLinkedRequestWell — start', { linkedRequestId, eligibleWellsLoaded: this.store.linkedRequestWells().length });
    const well = findLinkedWell(linkedRequestId, this.store.linkedRequestWells());
    if (!well) {
      console.log('[IR:link] applyLinkedRequestWell — no matching eligible well, aborting patch', { linkedRequestId });
      return;
    }

    const hasElevation = linkedWellHasElevation(well);
    const hasCoordinates = linkedWellHasCoordinates(well);
    const hasTargetDepth = linkedWellHasTargetDepth(well);
    const hasTargetAquifer = linkedWellHasTargetAquifer(well);
    const hasSupportingBusiness = linkedWellHasSupportingBusiness(well);

    const patch: Record<string, string | number> = { wellName: well.wellName };
    if (hasElevation) patch['surfaceElevationFtAsl'] = well.surfaceElevation as number;
    patch['coordinates'] = hasCoordinates ? formatLinkedWellCoordinates(well) : '';
    if (hasTargetDepth) patch['targetDepthFt'] = well.targetDepth as number;
    patch['targetAquifer'] = hasTargetAquifer ? (well.targetAquifer as string) : '';
    patch['supportingBusiness'] = hasSupportingBusiness ? (well.supportings as string) : '';
    console.log('[IR:link] applyLinkedRequestWell — patching form group', {
      patch,
      hasElevation,
      hasCoordinates,
      hasTargetDepth,
      hasTargetAquifer,
      hasSupportingBusiness,
    });
    referenceGroup.patchValue(patch);

    referenceGroup.get('wellName')?.disable();
    this.setLinkedFieldLock(referenceGroup, 'surfaceElevationFtAsl', hasElevation);
    console.log('[IR:link] applyLinkedRequestWell — done', { rawValue: referenceGroup.getRawValue() });
  }

  private resetFlatSection(

  /** Returns whether a save was actually kicked off — `false` on every early-return branch below
   *  (nothing to save, or a blocked validation). `saveUnsavedChangesAndWait` needs this to settle
   *  its own promise immediately in those cases instead of waiting forever for a `creatingIr`
   *  transition that will now never happen. */
  protected saveAll(): boolean {
    // A record whose stored remarks still carry legacy HTML is saved even with nothing edited — the form already
    // holds the cleaned text, and saving is what writes it back.
    if (this.ir().irSeq !== null && !this.hasUnsavedChanges() && !this.ir().hasLegacyMarkup) {
      this.notify.info('No changes to save.');
      return false;
    }

    let raw = this.currentRawFormValue();

    const referenceCtrl = this.requestReferenceEdit.control();
    if (referenceCtrl) raw = { ...raw, reference: { main: referenceCtrl.getRawValue() } };

    // `letterNumber` carries a `Validators.required` in the schema (wwell-ir-form.schema.ts), but
    // nothing elsewhere in this method ever checks form validity before saving — it only gathers
    // raw values. Without this explicit gate, a blank Letter Number saved silently; the asterisk
    // and mat-error only ever showed once the field itself had been touched.
    if (!raw.reference.main.letterNumber?.trim()) {
      this.requestReferenceEdit.open();
      this.requestReferenceEdit.control()?.get('letterNumber')?.markAsTouched();
      this.expandedTimelineSteps.update(current => new Set(current).add('wellbore'));
      this.focusField('ire-sec-wellbore', 'Letter Number');
      this.notify.error('Letter Number is required before saving.');
      return false;
    }

    const programCtrl = this.programEdit.control();
    if (programCtrl) {
      const flat = programCtrl.getRawValue() as WwellIrRawProgramDetails;
      raw = {
        ...raw,
        programDetails: {
          main: {
            completionType: flat.completionType,
            completion: flat.completion,
            openHoleSizeIn: flat.openHoleSizeIn,
            linerSizeIn: flat.linerSizeIn,
            linerDepthFt: flat.linerDepthFt,
            acidization: flat.acidization,
            transmission: flat.transmission,
          },
          developmentTesting: {
            logging: flat.logging,
            mudLogging: flat.mudLogging,
            sampling: flat.sampling,
            flowTest: flat.flowTest,
            remarks: flat.remarks,
          },
        },
      };
    }

    const geologistCtrl = this.geologistContactEdit.control();
    if (geologistCtrl) raw = { ...raw, contacts: { ...raw.contacts, geologist: geologistCtrl.getRawValue() } };

    const hydrogeologyCtrl = this.hydrogeologyEdit.control();
    if (hydrogeologyCtrl) raw = { ...raw, hydrogeology: { main: hydrogeologyCtrl.getRawValue() } };

    const formationTopsCtrl = this.formationTopsEdit.control();
    if (formationTopsCtrl) raw = { ...raw, subsurface: { ...raw.subsurface, formationTops: formationTopsCtrl.getRawValue() } };

    const topsRemarksCtrl = this.topsRemarksEdit.control();
    if (topsRemarksCtrl) raw = { ...raw, subsurface: { ...raw.subsurface, remarks: topsRemarksCtrl.getRawValue() } };

    const casingCtrl = this.casingEdit.control();
    if (casingCtrl) raw = { ...raw, subsurface: { ...raw.subsurface, casing: casingCtrl.getRawValue() } };

    const offsetWellsCtrl = this.offsetWellsEdit.control();
    if (offsetWellsCtrl) raw = { ...raw, contacts: { ...raw.contacts, offsetWells: offsetWellsCtrl.getRawValue() } };

    const offsetRemarksCtrl = this.offsetRemarksEdit.control();
    if (offsetRemarksCtrl) raw = { ...raw, contacts: { ...raw.contacts, remarks: offsetRemarksCtrl.getRawValue() } };

    this.saveMergedRaw(raw);
    this.closeAllSections();
    return true;
  }

  /** Scrolls a labelled field of an open edit form into view and focuses it — what a blocked Save points at. */
  private focusField(sectionId: string, label: string): void {
    setTimeout(() => {
      const fields = Array.from(this.host.nativeElement.querySelectorAll<HTMLElement>(`#${sectionId} mat-form-field`));
      const field = fields.find(candidate => candidate.querySelector('mat-label')?.textContent?.includes(label));
      const input = field?.querySelector<HTMLInputElement | HTMLTextAreaElement>('input, textarea');
      if (!input) return;
      input.scrollIntoView({ block: 'center', behavior: 'smooth' });
      input.focus({ preventScroll: true });
    }, 50);
  }

    this.flagUnsaved.set(false);
    this.requestReferenceEdit.close();
    this.programEdit.close();
    this.geologistContactEdit.close();
    this.hydrogeologyEdit.close();
    this.formationTopsEdit.close();
    this.topsRemarksEdit.close();
    this.casingEdit.close();
    this.offsetWellsEdit.close();
    this.offsetRemarksEdit.close();
    this.formationTopAdd.close();
    this.casingAdd.close();
    this.offsetWellAdd.close();
  }

  protected submitForReview(): void {
    // The submitted letter/payload is built from the saved record, so unsaved edits would be silently
    // left out — the button is disabled in that state; this guards any other entry point.
    if (this.hasPendingEdits()) return;
    this.store.submitIrForReview(this.ir());
  }

  /** Recalling stops a live SAP workflow — uses the same "confirm, then act" pattern as the other
   *  row-delete actions above, since it's not reversible from here (SAP itself would need a fresh
   *  Submit for Review). */
  protected recallIr(): void {
    const ir = this.ir();
    this.confirmThenAct(
      {
        title: 'Recall this Integrated Requirement?',
        rows: [
          { label: 'IR', value: ir.code },
          { label: 'Well', value: ir.requestReference.wellName || 'Untitled Well' },
        ],
        confirmLabel: 'Recall',
      },
      () => this.store.recallIr(ir),
    );
  }
}

export function buildWwellIrFormSchema(
  linkedRequestOptions: Signal<readonly WwellFieldOption[]>,
  geologistOptions: Signal<readonly string[]>,
  dropdownMenus: Signal<Readonly<Record<string, readonly string[]>>> = signal({}),
  aquiferOptions: Signal<readonly WwellFieldOption[]> = signal([]),
  formationOptions: Signal<readonly WwellFieldOption[]> = signal([]),
  offsetWellAquiferOptions: Signal<readonly WwellFieldOption[]> = signal([]),
): readonly WwellFormStepConfig[] {
  return WWELL_IR_FORM_SCHEMA.map(step => {
    if (step.key === 'subsurface') {
      return {
        ...step,
        sections: step.sections.map(section =>
          section.key === 'formationTops' && section.repeatable
            ? {
                ...section,
                repeatable: {
                  ...section.repeatable,
                  rowFields: section.repeatable.rowFields.map(field => (field.key === 'name' ? { ...field, options: formationOptions } : field)),
                },
              }
            : section.key === 'casing' && section.repeatable
              ? {
                  ...section,
                  repeatable: {
                    ...section.repeatable,
                    rowFields: section.repeatable.rowFields.map(field => {
                      if (field.key === 'type') return { ...field, options: computed(() => dropdownMenus()[WWELL_IR_CASING_TYPE_MENU] ?? []) };
                      if (field.key === 'sizeIn') return { ...field, options: computed(() => dropdownMenus()[WWELL_IR_CASING_SIZE_MENU] ?? []) };
                      return field;
                    }),
                  },
                }
              : section,
        ),
      };
    }
    if (step.key === 'hydrogeology') {
      return {
        ...step,
        sections: step.sections.map(section => ({
          ...section,
          fields: section.fields?.map(field => (field.key === 'aquifer' ? { ...field, options: aquiferOptions } : field)),
        })),
      };
    }
    if (step.key === 'programDetails') {
      return {
        ...step,
        sections: step.sections.map(section => ({
          ...section,
          fields: section.fields?.map(field => withDropdownMenuOptions(field, dropdownMenus)),
        })),
      };
    }
    if (step.key === 'reference') {
      return {
        ...step,
        sections: step.sections.map(section => {
          if (section.key !== 'main' || !section.fields) return section;
          return {
            ...section,
            fields: section.fields.map(field =>
              field.key === 'linkedRequestId' ? { ...field, options: linkedRequestOptions } : field,
            ),
          };
        }),
      };
    }
    if (step.key === 'contacts') {
      return {
        ...step,
        sections: step.sections.map(section => {
          if (section.key === 'offsetWells' && section.repeatable) {
            return {
              ...section,
              repeatable: {
                ...section.repeatable,
                rowFields: section.repeatable.rowFields.map(field => (field.key === 'aquifer' ? { ...field, options: offsetWellAquiferOptions } : field)),
              },
            };
          }
          if (section.key !== 'geologist' || !section.fields) return section;
          return {
            ...section,
            fields: section.fields.map(field =>
              field.key === 'geologistPrimary' || field.key === 'geologistSecondary' ? { ...field, options: geologistOptions } : field,
            ),
          };
        }),
      };
    }
    return step;
  });
}

  const menu = WWELL_IR_DROPDOWN_MENU_BY_FIELD[field.key];
  if (!menu) return field;

import { ChangeDetectionStrategy, Component, computed, input, signal } from '@angular/core';
import { MatIconModule } from '@angular/material/icon';
import { IntegratedRequirement } from '@models/wwell-planning/integrated-requirement.model';
import { WwellCollapseAnimateDirective } from '../../shared/wwell-collapse-animate/wwell-collapse-animate.directive';
import {
  EMPTY_VALUE,
  buildProgramCompletionRows,
  buildProgramDevRows,
  buildProgramTreatmentRows,
} from '../wwell-ir-editor/wwell-ir-detail-rows.util';
import { WwellIrKvValueComponent } from '../wwell-ir-kv-value/wwell-ir-kv-value.component';

/** A KV value as its instructions, one per line — empty when nothing was entered. */
function splitSteps(value: string): readonly string[] {
  return value
    .split('\n')
    .map(line => line.trim())
    .filter(line => line && line !== EMPTY_VALUE);
}

function hasValue(value: string): boolean {
  return !!value.trim() && value.trim() !== EMPTY_VALUE;
}

/** Read-only Program Details, shared by the editor's General step and the IR view's Details page.
 *
 *  Three blocks — Completion (captioned), Treatment and Development & Testing (set apart by a rule); a block with nothing
 *  to show is left out. In each, short values (type, sizes) are a spec strip across the top and long-form ones (picked
 *  instructions, remarks) are rows of one aligned accordion list: label on the left, then the first instruction as a
 *  one-line preview (or, opened, every instruction as a numbered step), then a chevron. Rows with nothing entered are
 *  hidden, except Flow Test and Development/Testing. The first card (Completion Remarks) starts open. The host can fold/unfold everything through
 *  `toggleAll()` / `allOpen()` (the editor's section-head button does). */
@Component({
  selector: 'app-wwell-ir-program-details',
  standalone: true,
  imports: [MatIconModule, WwellCollapseAnimateDirective, WwellIrKvValueComponent],
  templateUrl: './wwell-ir-program-details.component.html',
  styleUrl: './wwell-ir-program-details.component.scss',
  changeDetection: ChangeDetectionStrategy.OnPush,
})
export class WwellIrProgramDetailsComponent {
  /** Optional so a host that reads `allOpen()` earlier in its own template than this element (the editor's section-head
   *  button) can't hit a required-input error before the first binding — until `ir` arrives there is simply nothing to show. */
  readonly ir = input<IntegratedRequirement | null>(null);

  /** Labels of the cards currently expanded. */
  protected readonly openCards = signal<ReadonlySet<string>>(new Set(['Completion Remarks']));

  protected readonly groups = computed(() => {
    const ir = this.ir();
    if (!ir) return [];
    return [
      { title: 'Completion', showTitle: true, rows: buildProgramCompletionRows(ir) },
      { title: 'Treatment', showTitle: false, rows: buildProgramTreatmentRows(ir) },
      { title: 'Development & Testing', showTitle: false, rows: buildProgramDevRows(ir) },
    ]
      .map(({ title, showTitle, rows }) => ({
        title,
        showTitle,
        // Nothing entered means nothing shown — a blank spec cell or an empty row is noise. Only rows flagged `alwaysShow`
        // (Flow Test, Development/Testing) stay listed, as "Not specified".
        specs: rows.filter(row => !row.options && !row.wide && hasValue(row.value)),
        cards: rows
          .filter(row => row.options || row.wide)
          .map(row => ({ row, steps: splitSteps(row.value) }))
          .filter(card => card.steps.length || card.row.alwaysShow),
      }))
      .filter(group => group.specs.length || group.cards.length);
  });

  /** Rows that have something to fold — a row with one instruction (or none) just shows it, so it has no toggle. */
  private readonly expandableCards = computed(() =>
    this.groups().flatMap(group => group.cards.filter(card => card.steps.length > 1).map(card => card.row.label)),
  );

  /** Every foldable card is open. */
  readonly allOpen = computed(() => {
    const labels = this.expandableCards();
    const open = this.openCards();
    return labels.length > 0 && labels.every(label => open.has(label));
  });

  toggleAll(): void {
    this.openCards.set(this.allOpen() ? new Set() : new Set(this.expandableCards()));
  }

  protected toggleCard(label: string): void {
    this.openCards.update(open => {
      const next = new Set(open);
      if (!next.delete(label)) next.add(label);
      return next;
    });
  }
}

import { ChangeDetectionStrategy, Component, computed, inject, input } from '@angular/core';
import { MatButtonModule } from '@angular/material/button';
import { MatIconModule } from '@angular/material/icon';
import { MatProgressSpinnerModule } from '@angular/material/progress-spinner';
import { IntegratedRequirement } from '@models/wwell-planning/integrated-requirement.model';
import { IrAttachment, IrAttachmentSlot } from '@models/wwell-planning/ir-attachment.model';
import { WWELL_IR_ATTACHMENT_SLOTS, irAttachmentsFor } from '../../wwell-ir-attachments.config';
import {
  EMPTY_VALUE,
  KvRow,
  buildGeologistContactRows,
  buildHydrogeologyRows,
  buildRequestReferenceRows,
} from '../../wwell-ir-editor/wwell-ir-detail-rows.util';
import { WwellIrStore } from '../../store/wwell-ir.store';
import { WWELL_IR_SUBMITTABLE_STATUSES, wwellIrHasRequiredSubmitFields } from '../../wwell-ir.config';
import { WwellIrProgramDetailsComponent } from '../../wwell-ir-program-details/wwell-ir-program-details.component';
import { WwellIrViewKvCardComponent } from '../wwell-ir-view-kv-card/wwell-ir-view-kv-card.component';

/** "Details" step of the read-only IR view: General Information, Program Details, Geologist
 *  Contact, Attachments, Area Water Wells, Estimated Hydrology, and — mirroring the reference
 *  GWD/AGWA "Integrated Requirements" screen — an Approval action at the bottom. Every KV card
 *  besides Attachments/Area Water Wells reuses the exact same `KvRow` builders
 *  WwellIrEditorComponent's own read display already uses, so this page never drifts out of sync
 *  with what the inline editor shows for the same record. */
@Component({
  selector: 'app-wwell-ir-view-details',
  standalone: true,
  imports: [MatIconModule, MatButtonModule, MatProgressSpinnerModule, WwellIrViewKvCardComponent, WwellIrProgramDetailsComponent],
  templateUrl: './wwell-ir-view-details.component.html',
  styleUrl: './wwell-ir-view-details.component.scss',
  changeDetection: ChangeDetectionStrategy.OnPush,
})
export class WwellIrViewDetailsComponent {
  readonly ir = input.required<IntegratedRequirement>();

  protected readonly attachmentSlots = WWELL_IR_ATTACHMENT_SLOTS;

  protected attachmentsFor(slot: IrAttachmentSlot): readonly IrAttachment[] {
    return irAttachmentsFor(this.ir().attachments, slot);
  }
  /** Resolves `requestReference.linkedRequestId` to a well name — see
   *  buildRequestReferenceRows' own doc comment. Falls back to the raw id when omitted. */
  readonly linkedRequestWellOptions = input<readonly { readonly value: string; readonly label: string }[]>([]);

  private readonly store = inject(WwellIrStore);
  protected readonly isSubmitting = this.store.updatingIrStatus;

  // Record-level fields (`code`/`updatedDate`) layered onto the request-reference fields —
  // matches the reference screen's "General Information" card, which shows both together.
  protected readonly generalInfoRows = computed<readonly KvRow[]>(() => [
    { label: 'IR #', value: this.ir().code, mono: true },
    ...buildRequestReferenceRows(this.ir(), this.linkedRequestWellOptions()),
    { label: 'Last Updated', value: this.ir().updatedDate ?? EMPTY_VALUE, mono: true },
  ]);
  protected readonly geologistContactRows = computed(() => buildGeologistContactRows(this.ir()));
  protected readonly hydrogeologyRows = computed(() => buildHydrogeologyRows(this.ir()));

  // "Request Approval" only makes sense before the record has already left the author's hands —
  // same gate WwellIrEditorComponent's own "Submit for Review" button uses, `sapWfId` check
  // included (see that gate's own doc comment for why `status` alone can't be trusted here). An
  // unsaved draft has no `ir_seq` to file the SAP workflow under, so it's excluded too, as is a
  // saved-but-incomplete record (missing a field the schema itself marks required).
  protected readonly canRequestApproval = computed(
    () =>
      this.ir().irSeq != null &&
      wwellIrHasRequiredSubmitFields(this.ir()) &&
      WWELL_IR_SUBMITTABLE_STATUSES.includes(this.ir().status) &&
      this.ir().sapWfId == null,
  );

  protected requestApproval(): void {
    this.store.submitIrForReview(this.ir());
  }
}
