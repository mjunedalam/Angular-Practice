```ts
/**
 * WLDA letters — mirrors EXAD.EXAD_AGWA_WLDA_LTR (docs/wlda/exad_agwa_wlda_ltr_schema.md) and the
 * pre-well plan row (EXAD_RCD_PREWAP) every letter belongs to. A letter has no well fields of its
 * own: the well name, number, depth and so on are read from its plan.
 */

/** `draft` = not saved yet; `created` = saved, no SAP workflow result yet; `approved` = SAP APPROVED. */
export type WldaStatus = 'draft' | 'created' | 'approved';

/** One row of `GET /planning/api/v1/wlda-letters` — the letter columns plus its joined plan columns. */
export interface WldaLetter {
  readonly prewapSeq: number;
  readonly ltrId: number;
  readonly svXutmCord: number | null;
  readonly svYutmCord: number | null;
  readonly svUtmZn: number | null;
  readonly ltrRmks: string | null;
  readonly sapWfId: number | null;
  readonly sapWfStatus: string | null;
  readonly sapWfAprvDt: string | null;
  readonly epCrDataDt: string | null;
  readonly epCrDataCd: string | null;
  readonly wPrimHid: string | null;
  readonly wNum: string | null;
  readonly wellType: string | null;
  readonly tdMd: number | null;
  readonly targetForm: string | null;
  readonly gospAreaPlant: string | null;
  readonly currentFlg: string | null;
}

/** The plan columns the create form shows, from `GET /planning/api/v1/prewaps` (EXAD_RCD_PREWAP). */
export interface WldaPrewap {
  readonly prewapSeq: number;
  readonly wPrimHid: string | null;
  readonly wNum: string | null;
  readonly wellType: string | null;
  readonly tdMd: number | null;
  readonly targetForm: string | null;
  readonly gospAreaPlant: string | null;
  readonly svUtmZn: number | null;
}

/** Body of `POST /planning/api/v1/prewaps/{prewapSeq}/wlda-letters`; prewapSeq travels in the path. */
export interface WldaCreatePayload {
  readonly svXutmCord: number | null;
  readonly svYutmCord: number | null;
  readonly svUtmZn: number | null;
  readonly ltrRmks: string | null;
  readonly sapWfId: number | null;
  readonly sapWfStatus: string | null;
  readonly sapWfAprvDt: string | null;
}

/** Body of `PUT .../wlda-letters/{ltrId}` — the update proc writes all four, so unchanged SAP columns are echoed back. */
export interface WldaUpdatePayload {
  readonly ltrRmks: string | null;
  readonly sapWfId: number | null;
  readonly sapWfStatus: string | null;
  readonly sapWfAprvDt: string | null;
}

export interface WldaCreateResult {
  readonly prewapSeq: number;
  readonly ltrId: number;
}

/** What the author types into: the letter's editable columns, the well details, and the files for S3. */
export interface WldaDraft {
  readonly prewapSeq: number | null;
  /** Well details: pre-filled from the chosen plan and editable while creating. They belong to the plan, not the letter. */
  readonly waterWellName: string;
  readonly field: string;
  readonly wellNumber: string;
  readonly reservoir: string;
  readonly wellType: string;
  readonly totalDepth: string;
  /** Cathodic protection required — shown on the letter only; EXAD_AGWA_WLDA_LTR has no column for it. */
  readonly cathodicProtection: boolean;
  /** Distribution list, one entry per line — shown on the letter only; no column for it either. */
  readonly distribution: string;
  /** SV_XUTM_CORD — East (X), NUMBER(8,2). */
  readonly svXutmCord: string;
  /** SV_YUTM_CORD — North (Y), NUMBER(9,2). */
  readonly svYutmCord: string;
  /** SV_UTM_ZN — NUMBER(2). */
  readonly svUtmZn: string;
  /** LTR_RMKS — VARCHAR2(2000). */
  readonly ltrRmks: string;
  /** New files to upload on save. */
  readonly files: readonly File[];
  /** S3 keys of stored files the author removed; deleted from S3 on save (edit only). */
  readonly removedKeys: readonly string[];
}

/** A file on the sheet: stored in S3 (`key` set) or picked and waiting to upload (`pendingIndex` set). */
export interface WldaAttachment {
  readonly key: string | null;
  readonly name: string;
  readonly pendingIndex: number | null;
}

/** One row of the SAP approval workflow table printed on the letter. */
export interface WldaWorkflowRow {
  readonly name: string;
  readonly title: string;
  readonly status: string;
  readonly approvedDate: string;
}

/** Everything the sheet renders, from an unsaved draft, a saved letter, or a letter being edited. */
export interface WldaLetterView {
  readonly code: string;
  /** The typed water well name; empty for a saved letter, whose well is named from its plan. */
  readonly wellName: string;
  /** null = not recorded (saved letters), true/false = ticked Yes/No on the form. */
  readonly cathodic: boolean | null;
  readonly status: WldaStatus;
  readonly date: string | null;
  readonly field: string;
  readonly wellNumber: string;
  readonly wellType: string;
  readonly reservoir: string;
  readonly totalDepth: string;
  readonly gosp: string;
  readonly north: string;
  readonly east: string;
  readonly zone: string;
  readonly remarks: string;
  readonly attachments: readonly WldaAttachment[];
  /** Empty until the SAP workflow is wired in; the sheet and the PDF show a placeholder meanwhile. */
  readonly workflow: readonly WldaWorkflowRow[];
  /** Empty until the distribution list is provided; shown as a placeholder meanwhile. */
  readonly distribution: readonly string[];
}
```

```ts
import { HttpErrorResponse } from '@angular/common/http';
import { computed, inject } from '@angular/core';
import { patchState, signalStore, withComputed, withHooks, withMethods, withState } from '@ngrx/signals';
import { Observable, catchError, forkJoin, last, map, of, switchMap } from 'rxjs';
import { WldaAttachment, WldaDraft, WldaLetter, WldaPrewap } from '@models/wwell-planning/wlda.model';
import { S3Service } from '@services/s3/s3.service';
import { NotificationService } from '@shared/components/notification/notification.service';
import { AuthStore } from 'src/app/features/auth/store/auth.store';
import { WldaService } from '../wlda.service';
import {
  WLDA_EMPTY_DRAFT,
  draftOfLetter,
  isWldaDraftDirty,
  toWldaCreatePayload,
  toWldaUpdatePayload,
  findPlanForWell,
  validateWldaDraft,
  wldaCode,
  wldaKey,
  wldaS3Prefix,
} from '../wlda.util';

/** `empty` = nothing open; `view` = a saved letter, read-only; `create` / `edit` = the draft is live. */
export type WldaMode = 'empty' | 'view' | 'create' | 'edit';

interface WldaState {
  readonly letters: readonly WldaLetter[];
  readonly prewaps: readonly WldaPrewap[];
  /** The letter being authored (create) or changed (edit — then `selectedKey` is also set). */
  readonly draft: WldaDraft | null;
  /** `wldaKey` of the saved letter open on the right, or null. */
  readonly selectedKey: string | null;
  /** Files stored in S3 for the opened letter. */
  readonly storedAttachments: readonly WldaAttachment[];
  readonly loading: boolean;
  readonly saving: boolean;
  readonly deleting: boolean;
  readonly loadError: string | null;
}

const initialState: WldaState = {
  letters: [],
  prewaps: [],
  draft: null,
  selectedKey: null,
  storedAttachments: [],
  loading: false,
  saving: false,
  deleting: false,
  loadError: null,
};

/** Prefers the backend's own `{ message }` over Angular's generic "Http failure response" text. */
function errorMessage(err: unknown, fallback: string): string {
  if (err instanceof HttpErrorResponse) {
    const body: unknown = err.error;
    const message = body && typeof body === 'object' ? (body as { message?: unknown }).message : undefined;
    return err.status !== 0 && typeof message === 'string' && message ? message : fallback;
  }
  return err instanceof Error ? err.message : fallback;
}

function fillFromPlan(draft: WldaDraft, plan: WldaPrewap): WldaDraft {
  const blank = (value: string) => value.trim() === '';
  return {
    ...draft,
    waterWellName: blank(draft.waterWellName) ? `${plan.wPrimHid ?? ''}-${plan.wNum ?? ''}` : draft.waterWellName,
    reservoir: blank(draft.reservoir) ? (plan.targetForm ?? '') : draft.reservoir,
    wellType: blank(draft.wellType) ? (plan.wellType ?? '') : draft.wellType,
    totalDepth: blank(draft.totalDepth) ? (plan.tdMd == null ? '' : String(plan.tdMd)) : draft.totalDepth,
    svUtmZn: blank(draft.svUtmZn) ? (plan.svUtmZn == null ? '' : String(plan.svUtmZn)) : draft.svUtmZn,
  };
}

export const WldaStore = signalStore(
  withState<WldaState>(initialState),

  withComputed(({ letters, prewaps, draft, selectedKey }) => {
    const selectedLetter = computed(() => letters().find(letter => wldaKey(letter) === selectedKey()) ?? null);
    const draftPlan = computed(() => {
      const current = draft();
      if (!current) return null;
      if (selectedLetter()) return prewaps().find(plan => plan.prewapSeq === current.prewapSeq) ?? null;
      return findPlanForWell(prewaps(), current.field, current.wellNumber);
    });
    return {
      selectedLetter,
      draftPlan,
      draftErrors: computed(() => {
        const current = draft();
        return current ? validateWldaDraft(current, selectedLetter() ? 'edit' : 'create', draftPlan()) : [];
      }),
      mode: computed<WldaMode>(() => {
        if (draft()) return selectedLetter() ? 'edit' : 'create';
        return selectedLetter() ? 'view' : 'empty';
      }),
      isDirty: computed(() => {
        const current = draft();
        return current ? isWldaDraftDirty(current, selectedLetter()) : false;
      }),
    };
  }),

  withMethods((store, wldaService = inject(WldaService), s3 = inject(S3Service), authStore = inject(AuthStore), notify = inject(NotificationService)) => {
    const networkId = () => authStore.networkId() ?? '';

    /** Uploads each file under the letter's prefix; resolves to the names that failed (letter stays saved). */
    function uploadFiles(files: readonly File[], prewapSeq: number, ltrId: number): Observable<string[]> {
      if (files.length === 0) return of([]);
      const prefix = wldaS3Prefix(prewapSeq, ltrId);
      return forkJoin(
        files.map(file =>
          s3.store(file, prefix, networkId(), file.name).pipe(
            last(),
            map((): string | null => null),
            catchError(() => of(file.name)),
          ),
        ),
      ).pipe(map(outcomes => outcomes.filter((name): name is string => name !== null)));
    }

    /** Deletes stored objects by key; resolves to the names that failed. */
    function deleteStored(keys: readonly string[], names: ReadonlyMap<string, string>): Observable<string[]> {
      if (keys.length === 0) return of([]);
      return forkJoin(
        keys.map(key =>
          s3.delete(key, networkId()).pipe(
            map((): string | null => null),
            catchError(() => of(names.get(key) ?? key)),
          ),
        ),
      ).pipe(map(outcomes => outcomes.filter((name): name is string => name !== null)));
    }

    function loadStored(letter: WldaLetter, key: string): void {
      s3.search(`${wldaS3Prefix(letter.prewapSeq, letter.ltrId)}/`).subscribe({
        next: res => {
          if (store.selectedKey() !== key) return;
          patchState(store, {
            storedAttachments: res.objects.map(object => ({ key: object.OBJ_KEY, name: object.ORIGINAL_FILENAME, pendingIndex: null })),
          });
        },
        error: () => {
          if (store.selectedKey() === key) notify.error('Could not load the letter attachments');
        },
      });
    }

    /** Re-reads one letter (single-letter GET) and its files, e.g. after a save. */
    function refreshLetter(prewapSeq: number, ltrId: number): void {
      const key = wldaKey({ prewapSeq, ltrId });
      wldaService.getLetter(prewapSeq, ltrId).subscribe({
        next: fresh => {
          patchState(store, { letters: store.letters().map(item => (wldaKey(item) === key ? fresh : item)) });
          loadStored(fresh, key);
        },
        error: (err: unknown) => notify.error(errorMessage(err, 'Failed to refresh the WLDA letter')),
      });
    }

    return {
      loadAll(): void {
        patchState(store, { loading: true, loadError: null });
        forkJoin({ prewaps: wldaService.getPrewaps(), letters: wldaService.getLetters() }).subscribe({
          next: ({ prewaps, letters }) => patchState(store, { prewaps, letters, loading: false }),
          error: (err: unknown) => patchState(store, { loading: false, loadError: errorMessage(err, 'Failed to load WLDA letters') }),
        });
      },

      startDraft(): void {
        patchState(store, { draft: WLDA_EMPTY_DRAFT, selectedKey: null, storedAttachments: [] });
      },

      /** Switches the opened letter from view to edit. */
      startEdit(): void {
        const letter = store.selectedLetter();
        if (letter) patchState(store, { draft: draftOfLetter(letter) });
      },

      patchDraft(patch: Partial<WldaDraft>): void {
        const current = store.draft();
        if (!current) return;
        let next: WldaDraft = { ...current, ...patch };
        // Creating: once Field + Well Number land on a plan, fill the details it already knows (blank fields only).
        if (!store.selectedLetter() && ('field' in patch || 'wellNumber' in patch)) {
          const before = findPlanForWell(store.prewaps(), current.field, current.wellNumber);
          const plan = findPlanForWell(store.prewaps(), next.field, next.wellNumber);
          if (plan && plan !== before) next = fillFromPlan(next, plan);
        }
        patchState(store, { draft: next });
      },

      /** Drops the draft: a create goes back to nothing open, an edit back to viewing the saved letter. */
      discardDraft(): void {
        patchState(store, { draft: null });
      },

      /** Opens a saved letter: the listed row at once, then a fresh single-letter GET and its S3 files. */
      select(key: string | null): void {
        patchState(store, { selectedKey: key, storedAttachments: [], draft: null });
        const letter = store.letters().find(item => wldaKey(item) === key);
        if (letter && key) refreshLetter(letter.prewapSeq, letter.ltrId);
      },

      /** Saves the letter first (it owns the LTR_ID the S3 prefix needs), then uploads each file. */
      createLetter(): void {
        const draft = store.draft();
        const plan = store.draftPlan();
        if (store.mode() !== 'create' || !draft || !plan || store.saving() || validateWldaDraft(draft, 'create', plan).length > 0) return;
        patchState(store, { saving: true });

        wldaService
          .createLetter(plan.prewapSeq, toWldaCreatePayload(draft))
          .pipe(switchMap(result => uploadFiles(draft.files, result.prewapSeq, result.ltrId).pipe(map(failed => ({ result, failed })))))
          .subscribe({
            next: ({ result, failed }) => {
              patchState(store, { saving: false, draft: null, selectedKey: wldaKey(result), storedAttachments: [] });
              notify.info(`${wldaCode(result.ltrId)} created.`);
              if (failed.length > 0) notify.error(`Letter saved, but these files did not upload: ${failed.join(', ')}`);
              wldaService.getLetters().subscribe({
                next: letters => {
                  patchState(store, { letters });
                  refreshLetter(result.prewapSeq, result.ltrId);
                },
              });
            },
            error: (err: unknown) => {
              patchState(store, { saving: false });
              notify.error(errorMessage(err, 'Failed to create the WLDA letter'));
            },
          });
      },

      /** Saves remarks (coordinates are not updatable), removes dropped files from S3, uploads new ones. */
      updateLetter(): void {
        const draft = store.draft();
        const letter = store.selectedLetter();
        if (store.mode() !== 'edit' || !draft || !letter || store.saving() || validateWldaDraft(draft, 'edit').length > 0) return;
        patchState(store, { saving: true });
        const names = new Map(store.storedAttachments().map(file => [file.key ?? '', file.name]));

        wldaService
          .updateLetter(letter.prewapSeq, letter.ltrId, toWldaUpdatePayload(draft, letter))
          .pipe(
            switchMap(() =>
              forkJoin({
                removeFailed: deleteStored(draft.removedKeys, names),
                uploadFailed: uploadFiles(draft.files, letter.prewapSeq, letter.ltrId),
              }),
            ),
          )
          .subscribe({
            next: ({ removeFailed, uploadFailed }) => {
              patchState(store, { saving: false, draft: null });
              notify.info(`${wldaCode(letter.ltrId)} updated.`);
              if (removeFailed.length > 0) notify.error(`Could not remove: ${removeFailed.join(', ')}`);
              if (uploadFailed.length > 0) notify.error(`Letter saved, but these files did not upload: ${uploadFailed.join(', ')}`);
              refreshLetter(letter.prewapSeq, letter.ltrId);
            },
            error: (err: unknown) => {
              patchState(store, { saving: false });
              notify.error(errorMessage(err, 'Failed to update the WLDA letter'));
            },
          });
      },

      /** Deletes the letter, then its S3 files best-effort (the row is already gone, so a failure only leaves orphans). */
      deleteLetter(): void {
        const letter = store.selectedLetter();
        if (!letter || store.deleting()) return;
        patchState(store, { deleting: true });
        const keys = store.storedAttachments().flatMap(file => (file.key ? [file.key] : []));

        wldaService
          .deleteLetter(letter.prewapSeq, letter.ltrId)
          .pipe(switchMap(() => deleteStored(keys, new Map())))
          .subscribe({
            next: orphaned => {
              const key = wldaKey(letter);
              patchState(store, {
                deleting: false,
                draft: null,
                selectedKey: null,
                storedAttachments: [],
                letters: store.letters().filter(item => wldaKey(item) !== key),
              });
              notify.info(`${wldaCode(letter.ltrId)} deleted.`);
              if (orphaned.length > 0) notify.error(`Letter deleted, but ${orphaned.length} stored file(s) could not be removed.`);
            },
            error: (err: unknown) => {
              patchState(store, { deleting: false });
              notify.error(errorMessage(err, 'Failed to delete the WLDA letter'));
            },
          });
      },
    };
  }),

  withHooks({ onInit: store => store.loadAll() }),
);
```

```ts
} from './wlda.config';

const PAGE_WIDTH = 595.28;
const PAGE_HEIGHT = 841.89;
const MARGIN = 34;
const CONTENT_WIDTH = PAGE_WIDTH - MARGIN * 2;
const BOTTOM_LIMIT = 48;

const INK = rgb(0.08, 0.1, 0.13);
const LABEL = rgb(0.36, 0.4, 0.45);
const LINE = rgb(0.17, 0.19, 0.23);
const GREEN = rgb(0.12, 0.48, 0.23);
const TABLE_HEAD_FILL = rgb(0.74, 0.82, 0.91);
const TABLE_LINE = rgb(0.55, 0.6, 0.67);

interface Fonts {
  readonly regular: PDFFont;
  readonly bold: PDFFont;
  readonly mono: PDFFont;
}

/** Top-down drawing cursor: `y` is the distance from the page's top edge, so layout reads like the form. */
class Sheet {
  y = MARGIN;
  page: PDFPage;

  constructor(
    private readonly doc: PDFDocument,
    readonly fonts: Fonts,
  ) {
    this.page = doc.addPage([PAGE_WIDTH, PAGE_HEIGHT]);
  }

  /** Starts a new page when `height` would not fit above the bottom margin. */
  ensureSpace(height: number): void {
    if (this.y + height <= PAGE_HEIGHT - BOTTOM_LIMIT) return;
    this.page = this.doc.addPage([PAGE_WIDTH, PAGE_HEIGHT]);
    this.y = MARGIN;
  }

  /** Text whose baseline sits `baselineFromTop` below the page top. Characters the font can't encode become `?`. */
  text(value: string, x: number, baselineFromTop: number, size: number, font: PDFFont, color: RGB = INK): void {
    this.page.drawText(printable(value, font), { x, y: PAGE_HEIGHT - baselineFromTop, size, font, color });
  }

  /** Text centred on `centerX`. */
  centered(value: string, centerX: number, baselineFromTop: number, size: number, font: PDFFont, color: RGB = INK): void {
    const safe = printable(value, font);
    this.text(safe, centerX - font.widthOfTextAtSize(safe, size) / 2, baselineFromTop, size, font, color);
  }

  hline(x1: number, x2: number, fromTop: number, thickness = 0.8, color: RGB = LINE): void {
    this.page.drawLine({ start: { x: x1, y: PAGE_HEIGHT - fromTop }, end: { x: x2, y: PAGE_HEIGHT - fromTop }, thickness, color });
  }

  vline(x: number, fromTop1: number, fromTop2: number, thickness = 0.8, color: RGB = LINE): void {
    this.page.drawLine({ start: { x, y: PAGE_HEIGHT - fromTop1 }, end: { x, y: PAGE_HEIGHT - fromTop2 }, thickness, color });
  }

  rect(x: number, topFromTop: number, width: number, height: number, options: { fill?: RGB; border?: RGB; borderWidth?: number }): void {
    this.page.drawRectangle({
      x,
      y: PAGE_HEIGHT - topFromTop - height,
      width,
      height,
      color: options.fill,
      borderColor: options.border,
      borderWidth: options.borderWidth ?? 0,
    });
  }
}

/** Standard PDF fonts only cover WinAnsi; anything else (e.g. Arabic) is replaced rather than crashing the download. */
function printable(value: string, font: PDFFont): string {
  const supported = new Set(font.getCharacterSet());
  return Array.from(value.replace(/[\r\n\t]+/g, ' '))
    .map(char => (supported.has(char.codePointAt(0) ?? 0) ? char : '?'))
    .join('');
}

/** Largest size (down to 6) at which `value` still fits `maxWidth`, so long values shrink instead of spilling over the cell border. */
function fitSize(value: string, font: PDFFont, size: number, maxWidth: number): number {
  const safe = printable(value, font);
  let current = size;
  while (current > 6 && font.widthOfTextAtSize(safe, current) > maxWidth) current -= 0.5;
  return current;
}

/** Greedy word wrap to `maxWidth`; a single word wider than the line is hard-split. */
function wrap(value: string, font: PDFFont, size: number, maxWidth: number): string[] {
  const lines: string[] = [];
  for (const paragraph of printable(value, font).split(/ {2,}|\n/)) {
    let line = '';
    for (const word of paragraph.split(' ')) {
      const candidate = line ? `${line} ${word}` : word;
      if (font.widthOfTextAtSize(candidate, size) <= maxWidth) {
        line = candidate;
        continue;
      }
      if (line) lines.push(line);
      line = word;
      while (font.widthOfTextAtSize(line, size) > maxWidth && line.length > 1) {
        let cut = line.length - 1;
        while (cut > 1 && font.widthOfTextAtSize(line.slice(0, cut), size) > maxWidth) cut -= 1;
        lines.push(line.slice(0, cut));
        line = line.slice(cut);
      }
    }
    lines.push(line);
  }
  return lines;
}

function formatLongDate(iso: string | null): string {
  const date = iso ? new Date(iso) : new Date();
  const valid = Number.isNaN(date.getTime()) ? new Date() : date;
  return valid.toLocaleDateString('en-US', { year: 'numeric', month: 'long', day: 'numeric' });
}

const dash = (value: string): string => value.trim() || '—';

/** Column x-edges for a row of cells sized by `weights`, spanning the full content width. */
function columnEdges(weights: readonly number[]): number[] {
  const total = weights.reduce((sum, weight) => sum + weight, 0);
  const edges = [MARGIN];
  for (const weight of weights) edges.push((edges[edges.length - 1] ?? MARGIN) + (CONTENT_WIDTH * weight) / total);
  return edges;
}

  labelLines.forEach((line, index) => sheet.text(line, left + 4, top + 10 + index * 8.5, 7, regular, LABEL));
  const center = (left + right) / 2;
  const startY = top + height - 8 - (values.length - 1) * 12;
  values.forEach((value, index) => {
    const size = fitSize(value, bold, 9, right - left - 8);
    sheet.centered(value, center, startY + index * 12, size, bold);
    if (options.underline) {
      const half = bold.widthOfTextAtSize(printable(value, bold), size) / 2;
      sheet.hline(center - half, center + half, startY + index * 12 + 1.6, 0.6, INK);
    }
  });
}

function drawCoordinates(sheet: Sheet, left: number, top: number, view: WldaLetterView): void {
  const { regular, bold } = sheet.fonts;
  const rows: [string, string][] = [
    ['NORTH', view.north],
    ['EAST', view.east],
    ['ZONE', view.zone],
  ];
  rows.forEach(([label, value], index) => {
    const baseline = top + 30 + index * 13;
    sheet.text(label, left + 18, baseline, 8, regular, LABEL);
    sheet.text(dash(value), left + 56, baseline, 9.5, bold);
  });
}

function drawFieldGrid(sheet: Sheet, view: WldaLetterView): void {
  const { regular } = sheet.fonts;
  const right = MARGIN + CONTENT_WIDTH;

function drawWorkflowTable(sheet: Sheet, view: WldaLetterView): void {
  const { regular } = sheet.fonts;
  const right = MARGIN + CONTENT_WIDTH;
  const edges = columnEdges([25, 33, 17, 25]);
  const headHeight = 18;
  const rowHeight = 17;

  sheet.ensureSpace(headHeight + rowHeight * Math.max(view.workflow.length, 1) + 8);
  const top = sheet.y;
  sheet.rect(MARGIN, top, CONTENT_WIDTH, headHeight, { fill: TABLE_HEAD_FILL, border: TABLE_LINE, borderWidth: 0.6 });
  WLDA_WORKFLOW_COLUMNS.forEach((label, index) => {
    const from = edges[index] ?? MARGIN;
    const to = edges[index + 1] ?? right;
    sheet.centered(label, (from + to) / 2, top + 12.5, 10, sheet.fonts.mono);
  });
  sheet.y = top + headHeight;

  if (view.workflow.length === 0) {
    sheet.rect(MARGIN, sheet.y, CONTENT_WIDTH, rowHeight * 2, { border: TABLE_LINE, borderWidth: 0.6 });
    sheet.centered(WLDA_WORKFLOW_PLACEHOLDER, MARGIN + CONTENT_WIDTH / 2, sheet.y + 21, 8.5, regular, LABEL);
    sheet.y += rowHeight * 2;
    return;
  }
  for (const row of view.workflow) {
    sheet.ensureSpace(rowHeight);
    const rowTop = sheet.y;
    sheet.rect(MARGIN, rowTop, CONTENT_WIDTH, rowHeight, { border: TABLE_LINE, borderWidth: 0.6 });
    [row.name, row.title, row.status, row.approvedDate].forEach((cell, index) => {
      const from = edges[index] ?? MARGIN;
      sheet.text(cell, from + 4, rowTop + 11.5, 8.6, sheet.fonts.mono);
    });
    edges.slice(1, -1).forEach(x => sheet.vline(x, rowTop, rowTop + rowHeight, 0.6, TABLE_LINE));
    sheet.y += rowHeight;
  }
}

function drawDistribution(sheet: Sheet, view: WldaLetterView): void {
  const { regular } = sheet.fonts;
  sheet.ensureSpace(60);
  sheet.y += 14;
  sheet.text('DISTRIBUTION:', MARGIN + 2, sheet.y + 8, 8, regular, LABEL);
  if (view.distribution.length === 0) {
    sheet.text(WLDA_DISTRIBUTION_PLACEHOLDER, MARGIN + 24, sheet.y + 24, 9, regular, LABEL);
    sheet.y += 36;
    return;
  }
  const half = Math.ceil(view.distribution.length / 2);
  const columns = [view.distribution.slice(0, half), view.distribution.slice(half)];
  const columnWidth = CONTENT_WIDTH / 2 - 16;
  let used = 0;
  columns.forEach((entries, columnIndex) => {
    let offset = 0;
    entries.forEach(entry => {
      wrap(entry, regular, 8.2, columnWidth).forEach(line => {
        sheet.text(line, MARGIN + 18 + columnIndex * (CONTENT_WIDTH / 2), sheet.y + 22 + offset, 8.2, regular);
        offset += 10.5;
      });
    });
    used = Math.max(used, offset);
  });
  sheet.y += 26 + used;
}

/** Builds the WLDA letter as a PDF laid out like Aramco Form 6208, followed by the SAP workflow table and distribution list. */
export async function buildWldaPdf(view: WldaLetterView): Promise<Uint8Array> {
  const doc = await PDFDocument.create();
  doc.setTitle(`${view.code} — Well Location & Drilling Approval`);
  const fonts: Fonts = {
    regular: await doc.embedFont(StandardFonts.Helvetica),
    bold: await doc.embedFont(StandardFonts.HelveticaBold),
    mono: await doc.embedFont(StandardFonts.CourierBold),
  };
  const sheet = new Sheet(doc, fonts);
  drawHeader(sheet, view);
  drawToLine(sheet, view.status === 'approved');
  drawFieldGrid(sheet, view);
  drawRelocationAndRemarks(sheet, view);
  drawWorkflowTable(sheet, view);
  drawDistribution(sheet, view);
  return doc.save();
}

/** Browser download of the generated PDF under `fileName`. */
export async function downloadWldaPdf(view: WldaLetterView, fileName: string): Promise<void> {
  const bytes = await buildWldaPdf(view);
  const url = URL.createObjectURL(new Blob([bytes as BlobPart], { type: 'application/pdf' }));
  const link = document.createElement('a');
  link.href = url;
  link.download = fileName;
  document.body.appendChild(link);
  link.click();
  link.remove();
  URL.revokeObjectURL(url);
}
```

```ts
      .get<ApiResponse<WldaLetter>>(`${this.apiUrl}/prewaps/${prewapSeq}/wlda-letters/${ltrId}`)
```

```ts
export const WLDA_MAX_X_UTM = 999_999.99;
export const WLDA_MAX_Y_UTM = 9_999_999.99;
export const WLDA_MIN_UTM_ZONE = 1;
export const WLDA_MAX_UTM_ZONE = 60;
export const WLDA_MAX_REMARKS_BYTES = 2000;

const DECIMAL_2DP = /^\d+(\.\d{1,2})?$/;
const INTEGER = /^\d+$/;

export const WLDA_EMPTY_DRAFT: WldaDraft = {

  field: '',
  wellNumber: '',
  reservoir: '',
  wellType: '',
  totalDepth: '',
  cathodicProtection: false,
  distribution: '',

/** Letter reference shown in the UI — the letter id is the only per-letter identifier the table has. */

}

/** Stable list key: LTR_ID alone is only unique per plan in the table's own design. */

export function wldaWellName(plan: Pick<WldaPrewap, 'wPrimHid' | 'wNum'> | null): string {
  if (!plan) return '';
  return [plan.wPrimHid, plan.wNum].filter(part => !!part?.trim()).join(' ');
}

function utf8Bytes(value: string): number {
  return new Blob([value]).size;
}

const REQUIRED_CREATE_FIELDS: readonly (readonly [keyof WldaDraft, string])[] = [
  ['waterWellName', 'Water Well Name'],
  ['field', 'Field'],
  ['wellNumber', 'Well Number'],
  ['reservoir', 'Reservoir'],
  ['wellType', 'Well Type'],
  ['totalDepth', 'Total Depth'],
  ['svYutmCord', 'North (Y)'],
  ['svXutmCord', 'East (X)'],
];

/**
 * Every problem with the draft, in one list so the author fixes the form in one pass. The starred
 * fields are only required when creating: in edit mode the well and location are locked.
 */
export function validateWldaDraft(draft: WldaDraft, mode: 'create' | 'edit' = 'create', plan: WldaPrewap | null = null): readonly string[] {
  const errors: string[] = [];
  if (mode === 'create') {
    for (const [key, label] of REQUIRED_CREATE_FIELDS) {
      if (String(draft[key]).trim() === '') errors.push(`${label} is required.`);
    }
    if (plan === null && draft.field.trim() !== '' && draft.wellNumber.trim() !== '') {
      errors.push(`No pre-well plan exists for ${draft.field.trim()} ${draft.wellNumber.trim()} — pick a Field and Well Number from the suggestions.`);
    }
  } else if (draft.prewapSeq == null) {
    errors.push('This letter has no pre-well plan.');
  }

  const x = draft.svXutmCord.trim();
  if (x && (!DECIMAL_2DP.test(x) || Number(x) > WLDA_MAX_X_UTM)) {
    errors.push(`East (X) must be a number up to ${WLDA_MAX_X_UTM} with at most 2 decimals.`);
  }
  const y = draft.svYutmCord.trim();
  if (y && (!DECIMAL_2DP.test(y) || Number(y) > WLDA_MAX_Y_UTM)) {
    errors.push(`North (Y) must be a number up to ${WLDA_MAX_Y_UTM} with at most 2 decimals.`);
  }
  const zone = draft.svUtmZn.trim();
  if (zone && (!INTEGER.test(zone) || Number(zone) < WLDA_MIN_UTM_ZONE || Number(zone) > WLDA_MAX_UTM_ZONE)) {
    errors.push(`Zone must be a whole number from ${WLDA_MIN_UTM_ZONE} to ${WLDA_MAX_UTM_ZONE}.`);
  }
  if (utf8Bytes(draft.ltrRmks) > WLDA_MAX_REMARKS_BYTES) {
    errors.push(`Remarks must be at most ${WLDA_MAX_REMARKS_BYTES} bytes.`);
  }
  return errors;
}

function numberOrNull(value: string): number | null {
  const trimmed = value.trim();
  return trimmed ? Number(trimmed) : null;
}

    sapWfAprvDt: null,

}

/** True once the author has changed anything worth confirming before it is thrown away. */
export function isWldaDraftDirty(draft: WldaDraft, letter: WldaLetter | null): boolean {
  if (draft.files.length > 0 || draft.removedKeys.length > 0) return true;
  const baseline = letter ? draftOfLetter(letter) : WLDA_EMPTY_DRAFT;
  const keys = Object.keys(baseline).filter(key => key !== 'files' && key !== 'removedKeys') as (keyof WldaDraft)[];
  return keys.some(key => draft[key] !== baseline[key]);
}

/** The pre-well plan a new letter is filed under: the one whose field code and well number match what was typed. */
export function findPlanForWell(prewaps: readonly WldaPrewap[], field: string, wellNumber: string): WldaPrewap | null {
  const wantedField = field.trim().toLowerCase();
  const wantedNumber = wellNumber.trim().toLowerCase();
  if (!wantedField || !wantedNumber) return null;
  return prewaps.find(plan => textOf(plan.wPrimHid).trim().toLowerCase() === wantedField && textOf(plan.wNum).trim().toLowerCase() === wantedNumber) ?? null;
}

function baseView(
  source: Pick<WldaPrewap, 'wPrimHid' | 'wNum' | 'wellType' | 'targetForm' | 'tdMd' | 'gospAreaPlant'> | null,
): Pick<WldaLetterView, 'field' | 'wellNumber' | 'wellType' | 'reservoir' | 'totalDepth' | 'gosp'> {
  return {
    field: textOf(source?.wPrimHid),
    wellNumber: textOf(source?.wNum),
    wellType: textOf(source?.wellType),
    reservoir: textOf(source?.targetForm),
    totalDepth: textOf(source?.tdMd),
    gosp: textOf(source?.gospAreaPlant),
  };
}

function pendingAttachments(files: readonly File[]): WldaAttachment[] {
  return files.map((file, index) => ({ key: null, name: file.name, pendingIndex: index }));
}

export function viewOfDraft(draft: WldaDraft, plan: WldaPrewap | null): WldaLetterView {
  return {
    code: 'New WLDA',
    status: 'draft',
    date: null,
    wellName: draft.waterWellName,
    cathodic: draft.cathodicProtection,
    field: draft.field,
    wellNumber: draft.wellNumber,
    wellType: draft.wellType,
    reservoir: draft.reservoir,
    totalDepth: draft.totalDepth,
    gosp: textOf(plan?.gospAreaPlant),
    north: draft.svYutmCord,
    east: draft.svXutmCord,
    zone: draft.svUtmZn,
    remarks: draft.ltrRmks,
    attachments: pendingAttachments(draft.files),
    workflow: [],
    distribution: draft.distribution.split('\n').map(line => line.trim()).filter(line => line !== ''),
  };
}

/** A saved letter. While editing (`draft` set) remarks and files come from the draft instead. */
export function viewOfLetter(letter: WldaLetter, stored: readonly WldaAttachment[], draft: WldaDraft | null = null): WldaLetterView {
  const visibleStored = draft ? stored.filter(file => !draft.removedKeys.includes(file.key ?? '')) : stored;
  return {
    ...baseView(letter),
    code: wldaCode(letter.ltrId),
    wellName: '',
    cathodic: null,
    status: statusOfLetter(letter),
    date: letter.sapWfAprvDt ?? letter.epCrDataDt,
    north: textOf(letter.svYutmCord),
    east: textOf(letter.svXutmCord),
    zone: textOf(letter.svUtmZn),
    remarks: draft ? draft.ltrRmks : textOf(letter.ltrRmks),
    attachments: [...visibleStored, ...pendingAttachments(draft?.files ?? [])],
    workflow: [],
    distribution: [],
  };
}
```
