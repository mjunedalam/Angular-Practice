import { inject, Injectable } from '@angular/core';
import { EmailRequest } from 'src/app/shared/models/email/email-request.model';
import { Observable } from 'rxjs';

import { MorningReport } from '@models/morning-report/morning-report.model';

@Injectable({
  providedIn: 'root'
})
export class EmailService {

  private extConfigService = inject(ExternalConfigService);
  private http = inject(HttpClient);
  private apiUrl = `${this.extConfigService.settings.emailServiceUrl}` + "/send-email";


  private formatReportDate(dateStr: string): string {
    const [y, m, d] = dateStr.split('-').map(Number);
    const date = new Date(y, m - 1, d);
    const days = ['Sunday','Monday','Tuesday','Wednesday','Thursday','Friday','Saturday'];
    const months = ['January','February','March','April','May','June','July','August','September','October','November','December'];
    const dd = String(date.getDate()).padStart(2, '0');
    return `${days[date.getDay()]} ${dd} ${months[date.getMonth()]} ${date.getFullYear()}`;
  }

  sendEmail(emailRequest: EmailRequest): Observable<number> {
    const headers = new HttpHeaders({ 'Content-Type': 'application/json' });
    return this.http.post(this.apiUrl, emailRequest, { headers, responseType: 'text', observe: 'response' }).pipe(
      map(res => res.status)
    );
  }

  buildEmailRequest(
    reports: MorningReport[],
    mapImageData: string,
    recipientEmail: string,
    waterWellTestResults?: WaterWellTestEmailResult[],
    reportDate?: string,
  ): EmailRequest {
    return {
      subject: EMAIL_SUBJECT,
      from: recipientEmail,
      to: [recipientEmail],
      cc: [recipientEmail],
      bcc: [recipientEmail],
      replyTo: '',
      templateName: EMAIL_TEMPLATE_NAME,
      generatePdf: false,
      templateData: {
        morningReport: reports,
        mapImageData: mapImageData,
        waterWellTestResults,
        mrReportDate: this.formatReportDate(reportDate ?? new Date().toISOString().slice(0, 10))
      }
    };
  }

  /** The approved-IR email: sent by and to the logged-in user, CC from EXAD_DHUB_LKUP_DATA, the PDF attached.
   *  The backend wraps `content` in the morning report's header/footer shell; `details` become a table in the same style. */
  buildIrApprovedEmailRequest(
    recipientEmail: string,
    cc: string[],
    label: string,
    fileName: string,
    pdfBase64: string,
    details: readonly { readonly label: string; readonly value: string }[] = [],
    subject = `Approved Integrated Requirements - ${label}`,
  ): EmailRequest {
    const escape = (text: string): string => text.replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;');
    const cell = 'border:1px solid #c0c0c0; padding:6px 10px; font-size:12px; text-align:left;';
    const rows = details
      .map(
        row =>
          `<tr><td style="${cell} background-color:#ddeaf7; color:#003d99; font-weight:700; width:35%;">${escape(row.label)}</td>` +
          `<td style="${cell} color:#111827;">${escape(row.value || '—')}</td></tr>`,
      )
      .join('');
    const table = rows
      ? `<table style="border-collapse:collapse; width:100%; margin:12px 0;"><tr><td colspan="2" style="${cell} background-color:#d0cece; color:#8b0000; font-weight:700; font-size:14px; text-align:center;">INTEGRATED REQUIREMENT</td></tr>${rows}</table>`
      : '';
    return {
      subject,
      from: recipientEmail,
      to: [recipientEmail],
      cc,
      replyTo: '',
      templateName: EMAIL_PLAIN_TEMPLATE_NAME,
      generatePdf: false,
      templateData: {
        title: 'Approved Integrated Requirements',
        content:
          `<p style="margin:0 0 8px;">Please find below the approved Integrated Requirements for <b>${escape(label)}</b>. ` +
          `The signed letter is attached as a PDF.</p>${table}`,
        attachments: [{ filename: fileName, contentType: 'application/pdf', b64: pdfBase64 }],
      },
    };
  }
}

import * as THREE from 'three';

import { ANIM, DiagramLayout, WellboreDiagramData } from '@models/well-design/wellbore-diagram.model';
import { ICasingIR } from '@shared/models/wwell/casing-ir.model';
import { IDrlgCsg } from '@shared/models/wwell/drlg-csg.model';
import { computeCasingHalfWidth, inchLabel } from '@shared/utils/wellbore-math.util';
import { LinerEntry, resolveLiners } from '../../renderers/liner.renderer';
import { getCasingTier } from '../../renderers/wellbore-depth.helpers';
import { createCasingMaterial, createLinerMaterial, createScreenGridMaterial } from '../utils/materials.factory';
import { createLabelSprite } from '../utils/label-sprite.util';
import { Depth3DScale, Reveal3DTarget } from '../wellbore-3d.types';

const RADIAL_SEGMENTS = 40;

// Label styling/positioning mirrors the 2D SVG's `.casing-label__primary` line — same "est vs
// act" text content and green-for-actual color, plus a thin leader line matching 2D's
// `.casing-label__line` connector. Labels are collected into their own group (see
// `CasingBuildResult.labelsGroup`) that the caller keeps *out* of the camera's bounding-box fit —
// a single long label (e.g. a screen + pre-perf liner's tags) sitting on a narrow inner pipe can
// be visually wider than every casing combined, and letting that drive the fit zoomed the whole
// scene out until the pipes themselves looked tiny. Framing on pipe geometry alone keeps the
// zoom level predictable; labels can now afford to be large and bold instead of squeezed down to
// avoid blowing up that measurement.
const LABEL_GAP = 0.4;
const LABEL_FONT_SIZE = 48;
const LABEL_SCALE = 1.6;
const LABEL_PLANNED_COLOR = '#0f172a';
const LABEL_ACTUAL_COLOR = '#16a34a';
const LEADER_LINE_MATERIAL = new THREE.LineBasicMaterial({ color: 0x94a3b8, transparent: true, opacity: 0.85 });

// Mirrors 2D's `.casing-size-inner` text — the OD size painted directly on the pipe itself
// (centered on its axis, no leader line, no background pill), always black regardless of
// est/act, unlike the callout label above which keeps the green/black distinction.
const INNER_LABEL_FONT_SIZE = 40;
const INNER_LABEL_SCALE = 1.1;
const INNER_LABEL_COLOR = '#000000';

function attachPipeLabel(labelsGroup: THREE.Group, text: string, radius: number, y: number, hasActual: boolean): void {
  const sprite = createLabelSprite({
    text,
    color: hasActual ? LABEL_ACTUAL_COLOR : LABEL_PLANNED_COLOR,
    fontSize: LABEL_FONT_SIZE,
    fontWeight: '900',
    scale: LABEL_SCALE,
  });
  const leaderEndX = radius + LABEL_GAP;
  sprite.position.set(leaderEndX + sprite.scale.x / 2, y, 0);
  labelsGroup.add(sprite);

  // Thin leader line from the pipe wall to the label — same role as 2D's `.casing-label__line`.
  const leader = new THREE.Line(
    new THREE.BufferGeometry().setFromPoints([
      new THREE.Vector3(radius, y, 0),
      new THREE.Vector3(leaderEndX, y, 0),
    ]),
    LEADER_LINE_MATERIAL,
  );
  labelsGroup.add(leader);
}

/** Mirrors `renderCasingLabels`'/`renderLinerLabels`' `.casing-size-inner` text — just the OD
 *  size, centered on the pipe's own axis at the given depth, painted straight onto the pipe. */
function attachInnerSizeLabel(labelsGroup: THREE.Group, size: string, y: number): void {
  const sprite = createLabelSprite({
    text: `${inchLabel(size)}`,
    color: INNER_LABEL_COLOR,
    background: 'rgba(255,255,255,0)',
    fontSize: INNER_LABEL_FONT_SIZE,
    fontWeight: '900',
    scale: INNER_LABEL_SCALE,
  });
  sprite.position.set(0, y, 0);
  labelsGroup.add(sprite);
}

// Mirrors 2D's `.liner-hanger` — a solid black marker at the point each liner is hung/anchored
// from the bottom of the tubular above it (`buildScreenHanger`, drawn as two flat wedges flaring
// inward from the liner's own OD wall). A cylindrical liner has no single flat "wall" to flare
// two wedges from, so the 3D equivalent is a real 360° collar instead: a short black cone frustum,
// wide (`HANGER_FLARE_PX` beyond the liner's own half-width, matching 2D's `size` flare) at the
// shallow end where it meets the casing above, tapering back down to the liner's own OD over a
// short vertical span (`HANGER_HEIGHT_UNITS`).
const HANGER_COLOR = 0x000000;
const HANGER_FLARE_PX = 26;
const HANGER_HEIGHT_UNITS = 0.3;

function buildLinerHanger(depthScale: Depth3DScale, halfWidth: number, hangerDepth: number): THREE.Mesh {
  const innerRadius = depthScale.toRadius(halfWidth);
  const outerRadius = depthScale.toRadius(halfWidth + HANGER_FLARE_PX);
  const topY = depthScale.toY(hangerDepth);
  const geometry = new THREE.CylinderGeometry(outerRadius, innerRadius, HANGER_HEIGHT_UNITS, RADIAL_SEGMENTS);
  const material = new THREE.MeshStandardMaterial({ color: HANGER_COLOR, roughness: 0.5, metalness: 0.3 });
  const mesh = new THREE.Mesh(geometry, material);
  mesh.name = 'liner-hanger';
  mesh.position.y = topY - HANGER_HEIGHT_UNITS / 2;
  return mesh;
}

/** Mirrors `renderCasingLabels`' primary label text in casing.renderer.ts (size/type/depth,
 *  trailing " *" only for as-drilled/actual entries). */
function casingLabelText(entry: CasingEntry): string {
  const suffix = entry.hasActual ? ' *' : '';
  return `${inchLabel(entry.size)} ${entry.csgType} @ ${Math.round(entry.bottomDepth).toLocaleString()} ft${suffix}`;
}

/** Mirrors `renderLinerLabels`' primary label text in liner.renderer.ts (OD/screen/pre-perf tags
 *  and trailing " *" for as-drilled/actual entries). */
function linerLabelText(liner: LinerEntry): string {
  const screenTag = liner.isScreen ? ' (Screen)' : '';
  const prePerfTag = liner.isPrePerfLiner ? ' (Pre Perf Liner)' : '';
  const suffix = liner.hasActual ? ' *' : '';
  return `${inchLabel(liner.odSz)} Liner @ ${Math.round(liner.botDepth).toLocaleString()} ft${screenTag}${prePerfTag}${suffix}`;
}

export interface CasingBuildResult {
  /** Pipe meshes only — pass this (and only this) to the camera's bounding-box fit. */
  readonly group: THREE.Group;
  /** Text label sprites — render these, but keep them out of the camera-fit measurement. */
  readonly labelsGroup: THREE.Group;
  readonly reveals: Reveal3DTarget[];
  /** Bottom-most Y of the deepest structural casing shoe — completion builder anchors off this. */
  readonly lastCasingBottomY: number;
}

interface CasingEntry {
  readonly size: string;
  readonly csgType: string;
  readonly topDepth: number;
  readonly bottomDepth: number;
  readonly tier: number;
  readonly hasActual: boolean;
}

function uniqueByOd(drlgCasings: IDrlgCsg[]): IDrlgCsg[] {
  const map = new Map<string, IDrlgCsg>();
  for (const d of drlgCasings) {
    if (d.wCsgBotDpth <= 0) continue;
    const existing = map.get(d.wCsgOdSz);
    if (!existing || d.wCsgBotDpth > existing.wCsgBotDpth) map.set(d.wCsgOdSz, d);
  }
  return [...map.values()].sort((a, b) => b.wCsgBotDpth - a.wCsgBotDpth);
}

function resolveCasingEntries(casings: ICasingIR[], drlgCasings: IDrlgCsg[]): CasingEntry[] {
  if (casings.length === 0) {
    return uniqueByOd(drlgCasings).map((actual, i) => ({
      size: actual.wCsgOdSz,
      csgType: 'Casing',
      topDepth: 0,
      bottomDepth: actual.wCsgBotDpth,
      tier: i,
      hasActual: true,
    }));
  }

  return casings
    .filter(c => c.csgType !== 'Liner' && c.csgType !== 'Gravel Pack')
    .map(csg => {
      const tier = getCasingTier(csg, casings);
      const actual = drlgCasings.find(d => d.wCsgOdSz === csg.csgSize) ?? null;
      const hasActual = actual !== null && actual.wCsgBotDpth > 0;
      return {
        size: hasActual ? actual!.wCsgOdSz : csg.csgSize,
        csgType: csg.csgType,
        topDepth: 0,
        bottomDepth: hasActual ? actual!.wCsgBotDpth : csg.csgDepth,
        tier,
        hasActual,
      };
    });
}

/** Plain, constant-radius open pipe wall — a straight tube like a real drilling casing, with no
 *  decorative shoe taper. `openEnded` (no top/bottom caps) so the transparent walls read as a
 *  hollow tube, letting inner casings/liners show through. */
export function buildPipeMesh(radius: number, topY: number, bottomY: number, material: THREE.Material): THREE.Mesh {
  const height = Math.max(0.05, topY - bottomY);
  const geometry = new THREE.CylinderGeometry(radius, radius, height, RADIAL_SEGMENTS, 1, true);
  const mesh = new THREE.Mesh(geometry, material);
  mesh.position.y = (topY + bottomY) / 2;
  return mesh;
}

export interface CasingTiming {
  /** Seconds — mirrors `t_casingsStart` in the 2D component's sequential timeline. */
  readonly casingsStart: number;
  /** Seconds — mirrors `t_linersStart` (computed via `computeOverlayDelay`, kept identical to 2D). */
  readonly linersStart: number;
}

export function buildCasings3D(
  data: WellboreDiagramData,
  layout: DiagramLayout,
  depthScale: Depth3DScale,
  { casingsStart, linersStart }: CasingTiming,
): CasingBuildResult {
  const group = new THREE.Group();
  group.name = 'casings';
  const labelsGroup = new THREE.Group();
  labelsGroup.name = 'casing-labels';
  const reveals: Reveal3DTarget[] = [];
  const entries = resolveCasingEntries(data.casings, data.drlgCasings);
  let lastCasingBottomY = 0;
  let lastCasingBottomDepth = 0;

  // Mirrors `renderCasingLabels`' `sortedShoeDepths`/rank lookup exactly: the inner size label
  // for a given casing sits at the midpoint of the section unique to it — from the next-shallower
  // casing's shoe (or ground level for the shallowest) down to its own shoe — not simply its own
  // top/bottom, since `entry.topDepth` is always 0 here (every structural casing starts at
  // ground) and would otherwise put every inner label at the same nonsensical near-surface spot.
  const sortedShoeDepths = entries.map((e) => e.bottomDepth).sort((a, b) => a - b);

  entries.forEach((entry, i) => {
    const hw = computeCasingHalfWidth(entry.tier, layout.baseHalfWidth, layout.halfWidthIncrement);
    const radius = depthScale.toRadius(hw);
    const topY = depthScale.toY(entry.topDepth);
    const bottomY = depthScale.toY(entry.bottomDepth);
    const isConductor = entry.csgType.toLowerCase() === 'conductor';
    const material = createCasingMaterial(entry.hasActual, isConductor);
    const mesh = buildPipeMesh(radius, topY, bottomY, material);
    mesh.name = `casing-${entry.size}`;
    group.add(mesh);
    attachPipeLabel(labelsGroup, casingLabelText(entry), radius, bottomY, entry.hasActual);

    const rank = sortedShoeDepths.indexOf(entry.bottomDepth);
    const innerTopDepth = rank > 0 ? sortedShoeDepths[rank - 1] : 0;
    const innerMidY = depthScale.toY((innerTopDepth + entry.bottomDepth) / 2);
    attachInnerSizeLabel(labelsGroup, entry.size, innerMidY);

    const plane = new THREE.Plane(new THREE.Vector3(0, 1, 0), -topY);
    material.clippingPlanes = [plane];
    const animOrder = entries.length - 1 - i;
    reveals.push({
      mesh,
      clipPlane: plane,
      topY,
      bottomY,
      delay: casingsStart + animOrder * (ANIM.CASING_STAGGER / 1000),
      duration: ANIM.CASING_DURATION / 1000,
    });

    lastCasingBottomY = Math.min(lastCasingBottomY, bottomY);
    lastCasingBottomDepth = Math.max(lastCasingBottomDepth, entry.bottomDepth);
  });

  const liners = resolveLiners(data.casings, data.drlgCasings, layout, data.wellDesign);

  liners.forEach((liner, i) => {
    const radius = depthScale.toRadius(liner.halfWidth);
    const topY = depthScale.toY(liner.topDepth);
    const bottomY = depthScale.toY(liner.botDepth);
    const material = liner.isScreen ? createScreenGridMaterial() : createLinerMaterial(liner.hasActual, liner.isScreen);
    const mesh = buildPipeMesh(radius, topY, bottomY, material);
    mesh.name = `liner-${liner.odSz}`;
    group.add(mesh);

    // Mirrors 2D's `hangerY`: anchored where *this* liner exits the tubular above it — the
    // deepest structural casing shoe for the first liner, or the previous liner's own shoe for
    // every liner after that — not this liner's own `topDepth`, which as-drilled data can report
    // as sitting well above that overlap point.
    const hangerDepth = i === 0 ? lastCasingBottomDepth : liners[i - 1].botDepth;
    group.add(buildLinerHanger(depthScale, liner.halfWidth, hangerDepth));

    attachPipeLabel(labelsGroup, linerLabelText(liner), radius, bottomY, liner.hasActual);
    const linerMidY = depthScale.toY((liner.topDepth + liner.botDepth) / 2);
    attachInnerSizeLabel(labelsGroup, liner.odSz, linerMidY);

    const plane = new THREE.Plane(new THREE.Vector3(0, 1, 0), -topY);
    material.clippingPlanes = [plane];
    const delay = (liners.length - 1 - i) * (ANIM.CASING_STAGGER / 1000);
    reveals.push({
      mesh,
      clipPlane: plane,
      topY,
      bottomY,
      delay: linersStart + delay,
      duration: ANIM.OH_DURATION / 1000,
    });

    lastCasingBottomY = Math.min(lastCasingBottomY, bottomY);
  });

  return { group, labelsGroup, reveals, lastCasingBottomY };
}

export function renderCasingElements(options: RenderCasingElementsOptions): void {
    const { casings, drlgCasings, layout, wellDesign, rootG, defsEl, centerX, scale, linersStart, labelsStart, animation } = options;
    const liners = resolveLiners(casings, drlgCasings, layout, wellDesign);
    const lastCasingBotDepth = resolveLastCasingShoe(casings, drlgCasings);

    renderCasings(options);
    renderLiners({ rootG, defsEl, layout, liners, centerX, scale, linersStart, labelsStart, animation, lastCasingBotDepth });

function renderCasings({
    rootG,
    defsEl,
    layout,
    casings,
    drlgCasings,
    centerX,
    scale,
    casingsStart,
    gravelStart,
    openHoleHalfWidthPx,
}: RenderCasingElementsOptions): void {
    if (casings.length === 0) {
        renderCasingsFromActual({ rootG, defsEl, layout, drlgCasings, centerX, scale, casingsStart });
        return;
    }

    const { baseHalfWidth, halfWidthIncrement, shoeCurveOffset } = layout;
    const completionTopPx = scale(resolveCompletionTopDepth(casings, drlgCasings));

    casings.forEach((csg, i) => {
        if (csg.csgType === 'Liner') return;
        const tier = getCasingTier(csg, casings);
        const hw = computeCasingHalfWidth(tier, baseHalfWidth, halfWidthIncrement);
        const isConductor = csg.csgType.toLowerCase() === 'conductor';
        const isGP = csg.csgType === 'Gravel Pack';
        const actualCsg = drlgCasings.find(d => d.wCsgOdSz === csg.csgSize) ?? null;
        const hasActual = actualCsg !== null && actualCsg.wCsgBotDpth > 0;
        const displayDepth = hasActual ? actualCsg!.wCsgBotDpth : csg.csgDepth;
        const bottomPx = scale(displayDepth);
        const clipId = `dyn-casing-clip-${i}`;
        const animOrder = casings.length - 1 - i;
        const delay = animOrder * ANIM.CASING_STAGGER;
        const gpInnerHW = computeCasingHalfWidth(0, layout.baseHalfWidth, layout.halfWidthIncrement);
        const gpLsHW = Math.min(gpInnerHW, openHoleHalfWidthPx) - layout.linerScreenInset;
        const gpMaxOuter = openHoleHalfWidthPx - layout.gpScreenGap;
        const gpAnnulusWidth = Math.max(0, (gpMaxOuter - gpLsHW) * layout.gpFillRatio);
        const gpOuterEdge = gpLsHW + gpAnnulusWidth;
        const gpAnnulusCenter = gpLsHW + gpAnnulusWidth / 2;
        const clipRadius = isGP ? gpOuterEdge : openHoleHalfWidth(hw) + 20;

        const clipRect = defsEl
            .append('clipPath')
            .attr('class', 'dyn-clip')
            .attr('id', clipId)
            .append('rect')
            .attr('x', centerX - clipRadius - (isGP ? 0 : 15))
            .attr('y', isGP ? completionTopPx : -5)
            .attr('width', clipRadius * 2 + (isGP ? 0 : 30))
            .attr('height', 0);

        if (isGP) {
            const gravelBottomPx = scale(csg.csgDepth);
            rootG.append('path')
                .attr('class', 'gravelHole')
                .attr('d', buildGravelPackUPath(centerX, gpAnnulusCenter, completionTopPx, gravelBottomPx))
                .attr('stroke', 'url(#gravelpattern)')
                .attr('stroke-width', String(gpAnnulusWidth))
                .style('fill', 'none')
                .attr('clip-path', `url(#${clipId})`);
            clipRect.transition()
                .delay(gravelStart)
                .duration(ANIM.GRAVEL_DURATION)
                .ease(easeCubicInOut)
                .attr('height', gravelBottomPx - completionTopPx);
            return;
        }

        const casingD = buildCasingPath(centerX, hw, 0, bottomPx, shoeCurveOffset);

        if (hasActual) {
            rootG.append('path')
                .attr('class', 'casing')
                .attr('d', casingD)
                .attr('clip-path', `url(#${clipId})`)
                .attr('fill', 'url(#actualCasingGradient)')
                .attr('stroke', '#000000')
                .attr('stroke-width', '5')
                .style('opacity', 0.78);
        } else {
            const plannedFillGrad = isConductor
                ? 'url(#mainGradient)'
                : `url(#${casingGradientId(csg.csgType)})`;
            rootG.append('path')
                .attr('class', 'casing')
                .attr('d', casingD)
                .attr('clip-path', `url(#${clipId})`)
                .attr('fill', plannedFillGrad)
                .attr('stroke', '#000000')
                .attr('stroke-width', '5')
                .style('opacity', isConductor ? 0.75 : 0.40);
        }

        clipRect.transition()
            .delay(casingsStart + delay)
            .duration(ANIM.CASING_DURATION)
            .ease(easeCubicInOut)
            .attr('height', bottomPx + shoeCurveOffset + 20);
    });
}

function uniqueCasingsByOd(drlgCasings: IDrlgCsg[]): IDrlgCsg[] {
    const map = new Map<string, IDrlgCsg>();
    for (const d of drlgCasings) {
        if (d.wCsgBotDpth <= 0) continue;
        const existing = map.get(d.wCsgOdSz);
        if (!existing || d.wCsgBotDpth > existing.wCsgBotDpth) {
            map.set(d.wCsgOdSz, d);
        }
    }
    return [...map.values()].sort((a, b) => b.wCsgBotDpth - a.wCsgBotDpth);
}

function renderCasingsFromActual({

function renderCasingLabels({
    rootG,
    layout,
    casings,
    drlgCasings,
    centerX,
    scale,
    labelsStart,
    openHoleHalfWidthPx,
    animation,
}: RenderCasingElementsOptions): void {
    if (casings.length === 0) {
        renderCasingLabelsFromActual({ rootG, layout, drlgCasings, centerX, scale, labelsStart, animation });
        return;
    }

    const { baseHalfWidth, halfWidthIncrement, linerScreenInset, gpScreenGap, gpFillRatio } = layout;
    const gpLabelInnerHW = computeCasingHalfWidth(0, baseHalfWidth, halfWidthIncrement);
    const gpLabelLsHW = Math.min(gpLabelInnerHW, openHoleHalfWidthPx) - linerScreenInset;
    const gpLabelMaxOuter = openHoleHalfWidthPx - gpScreenGap;
    const gpLabelAnnulus = Math.max(0, (gpLabelMaxOuter - gpLabelLsHW) * gpFillRatio);
    const gpLabelOuterEdge = gpLabelLsHW + gpLabelAnnulus;

    const sortedShoeDepths = casings
        .filter(c => c.csgType !== 'Liner' && c.csgType !== 'Gravel Pack')
        .map(c => {
            const act = drlgCasings.find(d => d.wCsgOdSz === c.csgSize) ?? null;
            return (act !== null && act.wCsgBotDpth > 0) ? act.wCsgBotDpth : c.csgDepth;
        })
        .sort((a, b) => a - b);

    casings.forEach((csg) => {
        if (csg.csgType === 'Liner') return;
        const tier = getCasingTier(csg, casings);
        const hw = computeCasingHalfWidth(tier, baseHalfWidth, halfWidthIncrement);
        const actualCsg = drlgCasings.find(d => d.wCsgOdSz === csg.csgSize) ?? null;
        const hasActual = actualCsg !== null && actualCsg.wCsgBotDpth > 0;
        const displayDepth = hasActual ? actualCsg!.wCsgBotDpth : csg.csgDepth;
        const displaySize = hasActual ? actualCsg!.wCsgOdSz : csg.csgSize;
        const shoePx = scale(displayDepth);
        const labelYPx = csg.csgType === 'Gravel Pack' ? scale(csg.csgDepth) : shoePx;
        const rEdge = csg.csgType === 'Gravel Pack'
            ? centerX + gpLabelOuterEdge
            : centerX + hw;
        const lEnd = rEdge + 26;
        const labelX = lEnd + 8;
        const lg = rootG.append('g').attr('class', 'casing-label');

        lg.append('line')
            .attr('class', 'casing-label__line')
            .attr('x1', rEdge).attr('x2', lEnd)
            .attr('y1', labelYPx).attr('y2', labelYPx);
        lg.append('path')
            .attr('class', 'casing-label__arrow')
            .attr('d', buildArrowHeadRight(lEnd, labelYPx, 6));

        if (csg.csgType === 'Gravel Pack') {
            lg.append('text')
                .attr('class', 'casing-label__primary')
                .attr('x', labelX).attr('y', labelYPx + 4)
                .text(csg.csgRemarks || 'Gravel Pack');
        } else {
            const primaryEl = lg.append('text')
                .attr('class', 'casing-label__primary')
                .attr('x', labelX).attr('y', shoePx + 4);
            if (hasActual) {
                primaryEl
                    .attr('fill', '#22c55e')
                    .text(`${inchLabel(displaySize)} ${csg.csgType} @ ${displayDepth.toLocaleString()} *`);
            } else {
                primaryEl.text(`${inchLabel(csg.csgSize)} ${csg.csgType} @ ${csg.csgDepth.toLocaleString()}`);
                if (csg.csgRemarks) {
                    lg.append('text')
                        .attr('class', 'casing-label__secondary')
                        .attr('x', labelX).attr('y', shoePx + 24)
                        .text(`(${csg.csgRemarks})`);
                }
            }

            const innerDelay = labelsStart + (animation.lineStyle === 'draw' ? animation.lineDur : 0);
            const rank = sortedShoeDepths.indexOf(displayDepth);
            const topDepth = rank > 0 ? sortedShoeDepths[rank - 1] : 0;
            const innerMidY = scale((topDepth + displayDepth) / 2);
            rootG.append('text')
                .attr('class', 'casing-size-inner')
                .attr('x', centerX).attr('y', innerMidY)
                .attr('text-anchor', 'middle')
                .attr('dominant-baseline', 'central')
                .style('opacity', 0)
                .attr('fill', hasActual ? '#0f2d1a' : '')
                .style('font-weight', hasActual ? '800' : 'normal')
                .text(`${inchLabel(displaySize)}`)
                .transition().delay(innerDelay).duration(animation.fadeDur).ease(easeCubicInOut)
                .style('opacity', 1);
        }

        animateLabel(lg, labelsStart, animation);
    });
}

function buildCasingTooltipController(
    tooltipLayer: GSel,
    layout: DiagramLayout,
    centerX: number,
) {
    interface Row { label: string; value: string; hilite?: boolean; }

    let isVisible = false;
    let lastSide: 'left' | 'right' | null = null;

    const hide = (): void => {
        isVisible = false;
        lastSide = null;
        tooltipLayer.interrupt().transition().duration(120).style('opacity', 0);
    };

    const renderCard = (
        title: string,
        badge: string,
        accent: string,
        rows: readonly Row[],
        hw: number,
        mouseX: number,
        mouseY: number,
    ): void => {
        const ROW_H = 28;
        const tooltipH = 60 + rows.length * ROW_H + 16;
        const tooltipW = CSG_TOOLTIP_W;
        const onRight = mouseX > centerX;
        const side: 'left' | 'right' = onRight ? 'right' : 'left';
        const casingEdgeX = onRight ? centerX + hw : centerX - hw;
        const mutedAccent = accent === '#16a34a' ? 'rgba(22,163,74,0.14)' : 'rgba(37,99,235,0.14)';

        const rawTooltipX = onRight
            ? casingEdgeX + CSG_TOOLTIP_GAP
            : casingEdgeX - CSG_TOOLTIP_GAP - tooltipW;
        const tooltipX = Math.max(8, Math.min(rawTooltipX, layout.wellboreViewWidth - tooltipW - 8));
        const tooltipY = Math.max(4, Math.min(mouseY - tooltipH / 2, layout.drawingHeight - tooltipH - 4));
        const connY = Math.min(Math.max(mouseY, tooltipY + 18), tooltipY + tooltipH - 18);

        tooltipLayer.selectAll('*').remove();
        tooltipLayer.raise();

        // Connector: elbow from casing edge to tooltip connector-side edge
        const connTipX = onRight ? tooltipX : tooltipX + tooltipW;
        const connMidX = onRight ? tooltipX - 8 : tooltipX + tooltipW + 8;
        tooltipLayer.append('path')
            .attr('d', `M${casingEdgeX},${mouseY} H${connMidX} Q${connTipX},${mouseY} ${connTipX},${connY} H${connTipX}`)
            .attr('fill', 'none').attr('stroke', accent)
            .attr('stroke-width', 1.4).attr('stroke-dasharray', '4,3')
            .attr('stroke-linecap', 'round').attr('opacity', 0.65);

        // Anchor dot on casing edge
        tooltipLayer.append('circle')
            .attr('cx', casingEdgeX).attr('cy', mouseY)
            .attr('r', 3.2).attr('fill', '#fff')
            .attr('stroke', accent).attr('stroke-width', 1.6);

        // Card background
        tooltipLayer.append('rect')
            .attr('x', tooltipX).attr('y', tooltipY)
            .attr('width', tooltipW).attr('height', tooltipH)
            .attr('rx', 8).attr('fill', '#ffffff')
            .attr('stroke', '#e2e8f0').attr('stroke-width', 1.2)
            .attr('filter', 'drop-shadow(0 8px 16px rgba(15,23,42,0.18))');

        // Accent rail on the connector side of the card
        const railX = onRight ? tooltipX : tooltipX + tooltipW - 5;
        tooltipLayer.append('rect')
            .attr('x', railX).attr('y', tooltipY)
            .attr('width', 5).attr('height', tooltipH)
            .attr('rx', 2).attr('fill', accent);

        // Badge
        const badgeW = badge.length * 8.5 + 16;
        const badgeX = tooltipX + 12;
        tooltipLayer.append('rect')
            .attr('x', badgeX).attr('y', tooltipY + 11)
            .attr('width', badgeW).attr('height', 22)
            .attr('rx', 11).attr('fill', mutedAccent);
        tooltipLayer.append('text')
            .attr('x', badgeX + badgeW / 2).attr('y', tooltipY + 27)
            .attr('text-anchor', 'middle')
            .attr('font-size', '14').attr('font-family', CSG_TOOLTIP_FONT)
            .attr('font-weight', '900').attr('letter-spacing', '0.5')
            .attr('fill', accent).text(badge);

        // Title
        tooltipLayer.append('text')
            .attr('x', tooltipX + tooltipW - 12).attr('y', tooltipY + 27)
            .attr('text-anchor', 'end')
            .attr('font-size', '17').attr('font-family', CSG_TOOLTIP_FONT)
            .attr('font-weight', '800').attr('fill', '#0f172a').text(title);

        // Divider
        tooltipLayer.append('line')
            .attr('x1', tooltipX + 12).attr('x2', tooltipX + tooltipW - 12)
            .attr('y1', tooltipY + 44).attr('y2', tooltipY + 44)
            .attr('stroke', '#e2e8f0').attr('stroke-width', 1);

        // Rows
        rows.forEach((row, i) => {
            const rowY = tooltipY + 64 + i * ROW_H;
            tooltipLayer.append('text')
                .attr('x', tooltipX + 14).attr('y', rowY)
                .attr('font-size', '15').attr('font-family', CSG_TOOLTIP_FONT)
                .attr('font-weight', '700').attr('fill', '#64748b').text(row.label);
            tooltipLayer.append('text')
                .attr('x', tooltipX + tooltipW - 14).attr('y', rowY)
                .attr('text-anchor', 'end')
                .attr('font-size', '17').attr('font-family', CSG_TOOLTIP_FONT)
                .attr('font-weight', '800')
                .attr('fill', row.hilite ? accent : '#0f172a').text(row.value);
        });

        // Fade in only on first show or when side changes; instant reposition otherwise
        const sideChanged = side !== lastSide;
        lastSide = side;
        if (!isVisible || sideChanged) {
            tooltipLayer.interrupt().style('opacity', 0).transition().duration(150).style('opacity', 1);
            isVisible = true;
        } else {
            tooltipLayer.style('opacity', 1);
        }
    };

    const showForCasing = (
        csg: ICasingIR | null,
        actual: IDrlgCsg | null,
        hw: number,
        mouseX: number,
        mouseY: number,
        accent: string,
    ): void => {
        const hasActual = actual !== null && (actual.wCsgBotDpth ?? 0) > 0;
        const size = csg?.csgSize ?? actual?.wCsgOdSz ?? '?';
        const typeName = csg?.csgType ?? 'Casing';
        const typeLabel = typeName.toLowerCase() === 'conductor' ? 'CONDUCTOR' : typeName.toUpperCase();
        const badge = hasActual ? `${typeLabel} ✓` : typeLabel;

        const rows: Row[] = [];
        if (hasActual) {
            rows.push({ label: 'actual bottom depth', value: `${actual!.wCsgBotDpth.toLocaleString()} ft`, hilite: true });
            if (csg) rows.push({ label: 'est bottom depth', value: `${csg.csgDepth.toLocaleString()} ft` });
            if (csg?.csgRemarks) rows.push({ label: 'Remark', value: csg.csgRemarks });
        } else if (csg) {
            rows.push({ label: 'est bottom depth', value: `${csg.csgDepth.toLocaleString()} ft` });
            if (csg.csgRemarks) rows.push({ label: 'Remark', value: csg.csgRemarks });
        } else if (actual) {
            rows.push({ label: 'actual bottom depth', value: `${actual.wCsgBotDpth.toLocaleString()} ft`, hilite: true });
        }

        renderCard(`${inchLabel(size)} ${typeName}`, badge, accent, rows, hw, mouseX, mouseY);
    };

    const showForLiner = (liner: LinerEntry, mouseX: number, mouseY: number): void => {
        const length = liner.botDepth - liner.topDepth;
        const typeLabel = liner.isScreen ? 'SCREEN' : 'LINER';
        const accent = liner.hasActual ? '#16a34a' : '#2563eb';
        const badge = liner.hasActual ? `${typeLabel} ✓` : typeLabel;
        const typeName = liner.isPrePerfLiner ? 'Pre Perf Liner' : liner.isScreen ? 'Screen Liner' : 'Liner';
        const rows: Row[] = [];
        // Only show a genuine top-of-liner value (as-drilled data, or an explicit IR TOL) —
        // never the computed last-casing/liner-shoe attach point (see LinerEntry.hasActualTop).
        if (liner.hasActualTop) {
            rows.push({ label: 'Top (TOL)', value: `${liner.topDepth.toLocaleString()} ft` });
        }
        rows.push(
            { label: 'Bottom Depth', value: `${liner.botDepth.toLocaleString()} ft`, hilite: liner.hasActual },
            { label: 'Length', value: `${length.toLocaleString()} ft` },
        );
        renderCard(`${inchLabel(liner.odSz)} ${typeName}`, badge, accent, rows, liner.halfWidth, mouseX, mouseY);
    };

    return { showForCasing, showForLiner, hide };
}

            .attr('x', centerX - liner.halfWidth).attr('y', topPx)

            .attr('fill', 'transparent')
            .style('pointer-events', 'all').style('cursor', 'crosshair')
            .on('mousemove', (event: MouseEvent) => {
                const [mx, my] = pointer(event, node);
                ctrl.showForLiner(liner, mx, my);
            })
            .on('mouseleave', ctrl.hide);
    });

    hoverLayer.raise();
    tooltipLayer.raise();
}

function resolveTargetLinerIndex(liners: readonly RawLinerEntry[]): number | null {
    const idx = liners.findIndex(is7InchLiner);
    return idx === -1 ? null : idx;
}

export type LinerRuleKind = 'none' | 'single' | 'single-fallback' | 'multi-target' | 'multi-fallback';

export interface LinerRule {
    readonly kind: LinerRuleKind;
    readonly startDepth: number;
    /** Fixed structural bottom for 'single' and 'multi-target' (the target liner's own
     *  shoe). Null for 'none', 'single-fallback', and 'multi-fallback' — those are bounded
     *  by total/current/target depth, data this liner-only function doesn't have, so the
     *  caller resolves them. */
    readonly structuralEndDepth: number | null;
    /** The is7InchLiner match completion elements are drawn around — gets the screen
     *  texture when lsFlg='Y'. Null when no liner qualifies (kind is 'none',
     *  'single-fallback', or 'multi-fallback' — those get the generic Liner Screen shape
     *  instead, see completion.renderer's renderOpenHoleAndScreen). */
    readonly targetLiner: RawLinerEntry | null;
    /** The liner immediately above the interval start, used only to blend Open Hole's
     *  on-screen width between it and the outer casing (see resolveOhHalfWidth). */
    readonly boundaryLiner: RawLinerEntry | null;
}

/**
 * Implements docs/plan/logic.md Rules 1-3 over a depth-only liner list: which liner (if
 * any) is the 7" completion target, and the depth interval Open Hole / Gravel Pack /
 * Perforated Interval share as a result.
 */
export function resolveLinerRule(liners: readonly RawLinerEntry[], lastCasingShoe: number): LinerRule {
    // Rule 1: No Liner
    if (liners.length === 0) {
        return { kind: 'none', startDepth: lastCasingShoe, structuralEndDepth: null, targetLiner: null, boundaryLiner: null };
    }

    // Rule 2: Single Liner
    if (liners.length === 1) {
        const only = liners[0];
        if (!is7InchLiner(only)) {
            // Not the 7" screen liner: still an ordinary liner with open hole/completion
            // below it — draw from its own shoe down to total/current/target depth, same
            // as a no-match multi-liner well (docs/plan/logic.md leaves this case
            // unaddressed; this mirrors Rule 3's fallback rather than drawing nothing).
            return { kind: 'single-fallback', startDepth: only.botDepth, structuralEndDepth: null, targetLiner: null, boundaryLiner: only };
        }
        return { kind: 'single', startDepth: lastCasingShoe, structuralEndDepth: only.botDepth, targetLiner: only, boundaryLiner: only };
    }

    // Rule 3: Multiple Liners
    const targetIndex = resolveTargetLinerIndex(liners);
    if (targetIndex !== null) {
        const target = liners[targetIndex];
        const startDepth = targetIndex === 0 ? lastCasingShoe : liners[targetIndex - 1].botDepth;
        return {
            kind: 'multi-target',
            startDepth,
            structuralEndDepth: target.botDepth,
            targetLiner: target,
            boundaryLiner: targetIndex === 0 ? null : liners[targetIndex - 1],
        };
    }

    // Fallback: multiple liners, none match 7"
    const deepest = liners[liners.length - 1];
    return { kind: 'multi-fallback', startDepth: deepest.botDepth, structuralEndDepth: null, targetLiner: null, boundaryLiner: deepest };
}

function assignHalfWidths(baseHalfWidth: number, count: number): number[] {
    return Array.from({ length: count }, (_, i) =>
        Math.max(22, baseHalfWidth - (i + 1) * LINER_HW_DECREMENT)
    );
}

/**
 * Determines whether a liner qualifies as a 7" pre-perforated liner.
 *
 * Both conditions must pass:
 *   1. The liner's own OD size (DRLG_CSG.wLnrOdSz, preferred over the planned size) is 7".
 *   2. EXAD_GWD_WELL_DESIGN.perforatedFlg is 'Y' (case-insensitive).
 *
 * perforatedRemarks is free text and only ever used for label copy — it must never gate
 * visibility, otherwise perforatedFlg='Y' with a blank/missing remarks field would silently
 * hide a feature the well design explicitly opted into.
 */
function isPrePerforatedLiner(
    odSz: string | null | undefined,
    wellDesign: IWellDesign | null | undefined,
): boolean {
    if (parseInchSize(odSz) !== SCREEN_LINER_SIZE_IN) return false;
    return (wellDesign?.perforatedFlg ?? '').trim().toUpperCase() === 'Y';
}

/** Attaches halfWidth + isScreen/isPrePerfLiner render flags to a depth-only liner list. */
function attachRenderFlags(
    raw: RawLinerEntry[],
    layout: DiagramLayout,
    wellDesign?: IWellDesign | null,
): LinerEntry[] {
    const widths = assignHalfWidths(layout.baseHalfWidth, raw.length);
    const targetIndex = resolveTargetLinerIndex(raw);
    const lsFlgYes = (wellDesign?.lsFlg ?? '').trim().toUpperCase() === 'Y';

    return raw.map((liner, i) => ({
        ...liner,
        halfWidth: widths[i],
        isScreen: i === targetIndex && lsFlgYes,
        isPrePerfLiner: isPrePerforatedLiner(liner.odSz, wellDesign),
    }));
}

/**
 * Extracts liner entries from drlgCasings when no ICasingIR data is available.
 * Deduplicates by wLnrOdSz, sorts shallowest-first.
 */
function rawLinersFromDrlg(drlgCasings: IDrlgCsg[]): RawLinerEntry[] {
    const dedupMap = new Map<string, RawLinerEntry>();
    for (const d of drlgCasings) {
        if (!d.wLnrOdSz || (d.wLnrBotDepth ?? 0) <= 0) continue;
        const existing = dedupMap.get(d.wLnrOdSz);
        if (!existing || d.wLnrBotDepth! > existing.botDepth) {
            dedupMap.set(d.wLnrOdSz, {
                odSz: d.wLnrOdSz,
                topDepth: d.wLnrTopDepth ?? d.wCsgBotDpth,
                botDepth: d.wLnrBotDepth!,
                hasActual: true,
                hasActualTop: (d.wLnrTopDepth ?? 0) > 0,
            });
        }
    }

    return [...dedupMap.values()].sort((a, b) => a.botDepth - b.botDepth);
}

/**
 * Fallback top-of-liner anchor when no matching drlgCasing liner row exists: the shoe of
 * the last (deepest) non-liner casing. Per casing, prefer its actual (as-drilled)
 * DRLG_CSG shoe over its IR (planned) depth; only use the IR depth when no actual record
 * exists for that casing.
 *
 * When the IR is liner-only (no non-liner casing row at all — e.g. a standalone-liner
 * well), there's no IR casing to read a shoe from, so fall back to the deepest actual
 * non-liner casing record instead of defaulting straight to 0.
 */
export function resolveLastCasingShoe(casings: ICasingIR[], drlgCasings: IDrlgCsg[]): number {
    const nonLinerCasings = casings.filter(c => c.csgType !== 'Liner');

    if (nonLinerCasings.length > 0) {
        return nonLinerCasings.reduce((max, c) => {
            const actualShoe = drlgCasings.find(d => d.wCsgOdSz === c.csgSize)?.wCsgBotDpth ?? 0;
            return Math.max(max, actualShoe > 0 ? actualShoe : c.csgDepth);
        }, 0);
    }

    // Note: a row's wLnrOdSz (a liner hung off THIS casing) doesn't disqualify its own
    // wCsgBotDpth as a real casing shoe — IDrlgCsg legitimately bundles both on one row.
    const actualCasingOnly = drlgCasings.filter(d => d.wCsgBotDpth > 0);
    if (actualCasingOnly.length > 0) {
        return Math.max(...actualCasingOnly.map(d => d.wCsgBotDpth));
    }
    return Math.max(0, ...drlgCasings.map(d => d.wCsgBotDpth).filter(v => v > 0));
}

/**
 * Extracts liner entries from ICasingIR + drlgCasings for actual depth override.
 *
 * Source priority per liner:
 *   - DRLG_CSG (actual/as-drilled) overrides depth when a matching wLnrOdSz exists.
 *   - EXAD_GWD_IR_CASING (planned) is the fallback for depth and the primary OD source.
 *
 * Extra DRLG_CSG liners with no corresponding IR entry are appended at the end.
 */
function rawLinersFromIR(casings: ICasingIR[], drlgCasings: IDrlgCsg[]): RawLinerEntry[] {
    // Sorted shallow-to-deep so the chaining below (each un-anchored liner hangs from the
    // one immediately above it) walks the string in physical order, matching
    // buildLinerTopAnchors' identical convention on the IR wizard side.
    const irLiners = [...casings.filter(c => c.csgType === 'Liner')].sort((a, b) => a.csgDepth - b.csgDepth);

    const lastCasingShoe = resolveLastCasingShoe(casings, drlgCasings);

    const matchedOdSizes = new Set<string>();

    // Only the shallowest liner without its own actual top falls back to lastCasingShoe.
    // Every liner after it hangs from the liner immediately above — carried via chainTop,
    // which always holds that previous liner's own resolved (actual-preferred, IR-fallback)
    // botDepth — otherwise two+ stacked liners with no actual data would all render starting
    // at the same depth (the last casing's shoe) instead of hanging from one another.
    let chainTop = lastCasingShoe;
    const irEntries: RawLinerEntry[] = irLiners.map(csg => {
        // Source priority: use DRLG_CSG depth when available; fall back to IR depth.
        const actual = drlgCasings.find(d => d.wLnrOdSz === csg.csgSize) ?? null;
        const hasActualBot = (actual?.wLnrBotDepth ?? 0) > 0;
        const hasActualTop = (actual?.wLnrTopDepth ?? 0) > 0;
        if (actual?.wLnrOdSz) matchedOdSizes.add(actual.wLnrOdSz);
        const botDepth = hasActualBot ? actual!.wLnrBotDepth! : csg.csgDepth;
        // `actual.wCsgBotDpth` is only a meaningful fallback (a real parent-casing shoe) when
        // it's genuinely positive — the wwell-ir wizard's synthetic anchors always hardcode it
        // to 0 (see buildLinerTopAnchors), so this correctly falls through to chainTop for them
        // instead of collapsing every un-anchored liner to depth 0.
        const fallbackTopDepth = (actual?.wCsgBotDpth ?? 0) > 0 ? actual!.wCsgBotDpth : chainTop;
        const topDepth = hasActualTop ? actual!.wLnrTopDepth! : fallbackTopDepth;
        chainTop = botDepth;
        return {
            odSz: actual?.wLnrOdSz ?? csg.csgSize,
            topDepth,
            botDepth,
            hasActual: hasActualBot,
            hasActualTop,
        };
    });

    const extraEntries: RawLinerEntry[] = drlgCasings
        .filter(d => d.wLnrOdSz != null && (d.wLnrBotDepth ?? 0) > 0 && !matchedOdSizes.has(d.wLnrOdSz!))
        .map(d => ({
            odSz: d.wLnrOdSz!,
            topDepth: d.wLnrTopDepth ?? d.wCsgBotDpth,
            botDepth: d.wLnrBotDepth!,
            hasActual: true,
            hasActualTop: (d.wLnrTopDepth ?? 0) > 0,
        }));

    return [...irEntries, ...extraEntries].sort((a, b) => a.botDepth - b.botDepth);
}

function renderLinerShapes({
    rootG,
    defsEl,
    layout,
    liners,
    centerX,
    scale,
    linersStart,
    lastCasingBotDepth,
}: RenderLinersOptions): void {
    const { shoeCurveOffset } = layout;

    liners.forEach((liner, i) => {
        const topPx = scale(liner.topDepth);
        const botPx = scale(liner.botDepth);
        const clipId = `dyn-liner-clip-${i}`;

        rootG.append('path')
            .attr('class', 'liner-hanger')

            .attr('stroke', '#000000')

            .attr('stroke-linejoin', 'round')
            .style('opacity', 0)

            .delay(thisLinerStart)
            .duration(Math.round(ANIM.CASING_DURATION / 3))
            .ease(easeCubicInOut)
            .style('opacity', 1);

        const clipRect = defsEl.append('clipPath')
            .attr('class', 'dyn-clip')
            .attr('id', clipId)
            .append('rect')
            .attr('x', centerX - clipRadius - 15)
            .attr('y', topPx - 5)
            .attr('width', clipRadius * 2 + 30)
            .attr('height', 0);

        clipRect.transition()
            .delay(thisLinerStart)
            .duration(ANIM.CASING_DURATION)
            .ease(easeCubicInOut)
            .attr('height', botPx - topPx + shoeCurveOffset + 20);

        rootG.append('path')
            .attr('class', liner.isScreen ? 'liner liner--screen' : 'liner')
            .attr('d', casingPath)
            .attr('clip-path', `url(#${clipId})`)
            .attr('fill', fill)
            .attr('stroke', stroke)
            .attr('stroke-width', '2')
            .style('opacity', opacity);

        rootG.append('ellipse')
            .attr('class', 'liner-top-ellipse')
            .attr('cx', centerX)
            .attr('cy', topPx)
            .attr('rx', liner.halfWidth)
            .attr('ry', 4)
            .attr('clip-path', `url(#${clipId})`)
            .attr('fill', capFill)
            .attr('stroke', stroke)
            .attr('stroke-width', '2')
            .style('opacity', opacity);

        // Darken the part of the liner that is inside the casing to simulate interior shadowing

                .attr('y', topPx)

                .attr('height', exitPx - topPx)
                .attr('fill', '#0f172a')
                .style('opacity', 0.15)
                .attr('clip-path', `url(#${clipId})`);

            // This perfectly simulates the front wall of the casing acting as a glass layer.
            rootG.append('rect')
                .attr('class', 'liner-glass-reflection')
                .attr('x', centerX - liner.halfWidth)
                .attr('y', topPx)
                .attr('width', liner.halfWidth * 2)
                .attr('height', exitPx - topPx)
                .attr('fill', 'url(#mainGradient)')
                .style('opacity', 0.45)
                .attr('clip-path', `url(#${clipId})`);

            // Shadow cast by the casing shoe onto the liner as it exits the hole
            rootG.append('ellipse')
                .attr('cx', centerX)
                .attr('cy', exitPx)
                .attr('rx', liner.halfWidth + 8)
                .attr('ry', 4)
                .attr('fill', 'none')
                .attr('stroke', '#020617')
                .attr('stroke-width', '5')
                .style('opacity', 0.6)
                .attr('clip-path', `url(#${clipId})`);

                .attr('class', 'liner liner--screen-grid')

                .attr('clip-path', `url(#${clipId})`)
                .attr('fill', 'url(#gridPattern)')
                .attr('fill-opacity', 0.92)
                .attr('stroke', 'none');
        }
    });
}

                .attr('x', tolEnd - 8).attr('y', topPx + 4)
                .text(`TOL @ ${liner.topDepth.toLocaleString()}`);
            animateLabel(tolLg, labelsStart, animation);
        }

    });
}

function blobToBase64(blob: Blob): Promise<string> {
  return new Promise((resolve, reject) => {

      );
    } catch {
      this.notify.error('Failed to send the approved IR copy.');
    } finally {

    }
  }

  protected async downloadPdf(): Promise<void> {
    const blob = await this.buildPdfBlob();
    if (!blob) return;
    this.pdfStage.set('ready');
    await new Promise<void>(resolve => setTimeout(resolve, PDF_READY_HOLD_MS));
    this.downloadingPdf.set(false);
    this.pdfStage.set(null);
    const url = URL.createObjectURL(blob);
    const link = document.createElement('a');
    link.href = url;
    link.download = this.pdfFileName();
    link.click();
    URL.revokeObjectURL(url);
  }

  /** "GWD-49-2026_SHYB_1031.pdf" — the letter number and well, falling back to the IR code. */
  private pdfFileName(): string {
    const { letterNumber, wellName } = this.ir().requestReference;
    const parts = [letterNumber ? `GWD-${letterNumber}` : this.ir().code, wellName].filter((part): part is string => !!part);
    return `${parts.join('_').replace(/[\\/:*?"<>|\s]+/g, '-')}.pdf`;
  }

  /** Captures the diagram, has the backend render the letter and merges the attachments; `null` (after
   *  a toast) when the backend refuses. Drives `downloadingPdf`/`pdfStage` for the download. */
  private async buildPdfBlob(): Promise<Blob | null> {
    const irSeq = this.ir().irSeq;
    if (irSeq === null || irSeq === undefined || this.downloadingPdf()) return null;

    this.downloadingPdf.set(true);
    try {
      // Always print SAP's live workflow (status, stamp and approval table), never what was cached
      // when the record was opened — it may have been approved/rejected/resubmitted since.
      this.pdfStage.set('status');
      const ir = await this.irStore.syncApprovalStatus(this.ir());
      this.pdfStage.set('diagram');
      const wellboreImageBase64 = await this.letterView()?.captureWellboreImage();
      this.pdfStage.set('letter');
      const letterBlob = await firstValueFrom(
        this.wwellIrService.downloadLetterPdf(
          irSeq,
          wellboreImageBase64,
          resolveLetterStampStatus(ir),
          resolveLetterWorkflow(ir) ?? undefined,
        ),
      );
      this.pdfStage.set('attachments');
      return await this.letterPdf.mergeAttachments(letterBlob, ir);
    } catch (err: unknown) {
      this.downloadingPdf.set(false);
      this.pdfStage.set(null);
      const message = await this.extractDownloadErrorMessage(err);
      this.notify.error(message ?? 'Failed to download the integrated requirement letter PDF.');
      return null;
    }
  }

  /** `downloadLetterPdf` requests `responseType: 'blob'` so a successful PDF response can be
   *  turned straight into a download link — but Angular's HttpClient applies that same
   *  responseType to error bodies too, so even agwa-planning's JSON error envelope
   *  (`{ message: "Cannot generate IR letter: missing required field(s): ..." }`) arrives as an
   *  unparsed Blob instead of a plain object. Without this, the backend's specific reason (e.g.
   *  a missing Letter Number) was silently discarded in favor of a generic failure message.
   *  Reads it via `FileReader` rather than `Blob.text()` — the latter isn't implemented by
   *  jsdom (this repo's test environment), while `FileReader.readAsText` is. */
  private extractDownloadErrorMessage(err: unknown): Promise<string | null> {
    if (!(err instanceof HttpErrorResponse) || !(err.error instanceof Blob)) return Promise.resolve(null);
    return new Promise(resolve => {
      const reader = new FileReader();
      reader.onload = () => resolve(this.parseErrorMessage(reader.result));
      reader.onerror = () => resolve(null);
      reader.readAsText(err.error as Blob);
    });
  }

  private parseErrorMessage(text: FileReader['result']): string | null {
    if (typeof text !== 'string') return null;
    try {
      const parsed: unknown = JSON.parse(text);
      const message = parsed && typeof parsed === 'object' ? (parsed as { message?: unknown }).message : undefined;
      return typeof message === 'string' ? message : null;
    } catch {
      return null;
    }
  }

  /** Header stat: how many of the editor's 7 rail sections (Wellbore, General, Horizons, Casing,
   *  Hydrogeology, Area Water Wells, Attachments) hold meaningful data, per the same
   *  wwellIrTimelineStepStatus completeness rule the editor's Timeline view already uses — so this
   *  view's "N/7 sections" reading always agrees with what Timeline mode shows for the same record. */
  protected readonly sectionProgress = computed(() => {
    const ir = this.ir();
    const total = WWELL_IR_SECTION_RAIL.length;
    const completed = WWELL_IR_SECTION_RAIL.filter(group => wwellIrTimelineStepStatus(ir, group.key) === 'completed').length;
    return { completed, total, percent: Math.round((completed / total) * 100) };
  });

  protected setTab(tab: WwellIrViewTab): void {
    this.activeTab.set(tab);
  }
}

/**
 * Parses a fractional inch size string to a decimal number.
 * Handles formats like "8 1/2", "12 1/4", "9 5/8", "8.5", "6".
 * Returns null if the value cannot be parsed.
 */
export function parseInchSize(value: string | null | undefined): number | null {
  if (!value) return null;
  const m = value.match(/(\d+(?:\.\d+)?)\s*(?:(\d+)\/(\d+))?/);
  if (!m) return null;
  const whole = parseFloat(m[1]);
  if (m[2] && m[3]) return whole + parseInt(m[2]) / parseInt(m[3]);
  return whole;
}

/**
 * Detects whether free-text remarks reference a given inch size anywhere in the string.
 *
 * Remarks fields are free-form (e.g. `+/- 500 ft of 7" screen`), so a naive
 * parseInchSize() would return the first number token (500) and miss the real size.
 * This scans every whole/fractional size token in the text and matches on value.
 */
export function remarksReferenceSize(remarks: string | null | undefined, targetInches: number): boolean {
  if (!remarks) return false;
  const tokenRe = /(\d+(?:\.\d+)?)(?:\s+(\d+)\/(\d+))?/g;
  for (const m of remarks.matchAll(tokenRe)) {
    const whole = parseFloat(m[1]);
    const value = m[2] && m[3] ? whole + parseInt(m[2], 10) / parseInt(m[3], 10) : whole;
    if (value === targetInches) return true;
  }
  return false;
}
