## src/app/features/wwell-planning/wwell-ir/wwell-ir-form/wwell-ir-casing-warnings.util.ts

```ts
import { parseInchSize } from '@shared/utils/wellbore-math.util';
import { WwellIrRawCasingRow } from './wwell-ir-form.types';

/**
 * `mapCasings` (wwell-ir-wellbore-diagram.mapper.ts) now orders casings by entered SIZE, so
 * the diagram's width genuinely tracks the inch value a user typed rather than depth order —
 * see that function's doc comment. But `casing.renderer.ts` (presentation/well-bore-view —
 * off-limits to edit) always draws a casing as a full tube from ground level down to its own
 * depth, and paints later array entries — now the numerically widest — on top. A structurally
 * inverted pair (a deeper casing entered wider than some shallower casing) is therefore no
 * longer mislabeled, but the wider, deeper one now paints over the narrower, shallower one
 * wherever their depth ranges overlap, which can fully hide the shallower casing's shape. Two
 * casings can't really nest that way in a physical well either way, so this flags the
 * combination as a data-entry warning rather than trying to silently pick a "correct" look for
 * something that can't actually happen in reality.
 *
 * Liner rows are excluded — `casing.renderer.ts` skips them entirely for tier/width purposes
 * (liners hang from casing shoes via a separate anchor system, see
 * wwell-ir-wellbore-diagram.mapper.ts's `buildLinerTopAnchors`), so they're never part of the
 * telescoping width comparison this warns about.
 */
const NON_TIERED_CASING_TYPE = 'Liner';

export interface CasingSizeWarning {
  /** Index into the original `rows` array — matches the row's on-screen "#N" (1-based via +1). */
  readonly rowIndex: number;
  readonly size: string;
  readonly conflictsWithRowIndex: number;
  readonly conflictsWithSize: string;
}

interface StructuralCasingEntry {
  readonly index: number;
  readonly size: string;
  readonly depthFtBgl: number;
  readonly parsedSize: number;
}

function toStructuralEntries(rows: readonly WwellIrRawCasingRow[]): StructuralCasingEntry[] {
  const entries: StructuralCasingEntry[] = [];
  rows.forEach((row, index) => {
    if (row.type === NON_TIERED_CASING_TYPE || row.depthFtBgl === null) return;
    const parsedSize = parseInchSize(row.sizeIn);
    if (parsedSize === null) return;
    entries.push({ index, size: row.sizeIn, depthFtBgl: row.depthFtBgl, parsedSize });
  });
  return entries;
}

export function findInvertedCasingSizeRows(rows: readonly WwellIrRawCasingRow[]): readonly CasingSizeWarning[] {
  const structural = toStructuralEntries(rows);
  const warnings: CasingSizeWarning[] = [];

  for (const entry of structural) {
    const shallowerAndNarrower = structural.find(
      other =>
        other.index !== entry.index &&
        other.depthFtBgl < entry.depthFtBgl &&
        other.parsedSize < entry.parsedSize,
    );

    if (shallowerAndNarrower) {
      warnings.push({
        rowIndex: entry.index,
        size: entry.size,
        conflictsWithRowIndex: shallowerAndNarrower.index,
        conflictsWithSize: shallowerAndNarrower.size,
      });
    }
  }

  return warnings;
}
```

```ts
 * a user makes any single selection (e.g. just picking "Liner" from the Type dropdown before
 * touching Size or Depth). Mapping every row unconditionally meant a half-filled row already
 * rendered a shape/label on the diagram (a "Liner @ 0" annotation with no real size or depth
 * behind it) as soon as it existed, rather than once it actually described something real. These
 * gates require every field the diagram actually draws from before a row counts — reused by both
 * the Subsurface & Construction step's live preview and the Well Design View step (both call
 * `buildWellboreDiagramData`).
 */
function isRenderableCasingRow(row: WwellIrRawCasingRow): boolean {
  return !!row.type && !!row.sizeIn && row.depthFtBgl !== null && row.depthFtBgl !== undefined;
}

function isRenderableFormationTop(row: WwellIrRawFormationTop): boolean {
  return !!row.name && row.depthFt !== null && row.depthFt !== undefined;
}

const COMPLETION_OPEN_HOLE = 'Open Hole';
const COMPLETION_SCREENED = 'Screened';
const COMPLETION_SCREENED_GRAVEL_PACK = 'Screened + Gravel Pack';
const COMPLETION_LINER_PERFORATED = 'Liner + Perforated';

/** Mirrors wwell-ir-design-preview.renderer.ts's MIN_SCALE_DEPTH_FT: anchors the depth
 * scale to a stable minimum so a shallow/partial entry doesn't stretch across the full
 * drawing height, and so the scale's domain is never degenerate ([0, 0]) before the
 * (required, but non-linear-stepper-skippable) target depth field is filled in. */
const MIN_SCALE_DEPTH_FT = 500;

/**
 * Never used to match a real casing — `casing.renderer.ts` / `wellbore-depth.helpers.ts`
 * match `drlgCasings` rows to casings by `wCsgOdSz === csg.csgSize`, so this sentinel must be
 * a value no user-entered casing size could ever equal. Only `buildLinerTopAnchors`' rows use
 * this; they're read exclusively through `wLnrOdSz`, never `wCsgOdSz`.
 */
const LINER_ANCHOR_SENTINEL = '__wwell_ir_liner_anchor__';

/**
 * `casing.renderer.ts`'s `getCasingTier` derives each casing's on-screen half-width purely
 * from its position in this array via `indexOf` (tier 0 = narrowest, each later tier wider) —
 * it never reads `csgSize` for width, only for the text label. It also paints casings in this
 * same array order, later entries on top, so whichever entry ends up widest is also drawn last
 * — the standard "wide, shallow casing painted over a narrow, deep one" telescoping look.
 *
 * Sorting by SIZE (ascending — smallest first, so the largest entered size lands last/widest)
 * is what makes width actually track the number the user typed, rather than depth order. For
 * well-formed data (size decreasing with depth, real telescoping construction) this produces
 * the exact same order as depth-descending would — size and depth are inversely correlated
 * there, so there's no behavior change for the common case. It only diverges for physically
 * inverted rows (a deeper casing entered wider than a shallower one, flagged separately by
 * `findInvertedCasingSizeRows` in wwell-ir-casing-warnings.util.ts) — there, the wider-but-
 * deeper entry now paints last and can fully cover the narrower-but-shallower one in their
 * overlapping depth range, since two casings can't actually nest that way in reality either.
 * That's an unavoidable consequence of a genuinely invalid combination, not a rendering bug —
 * see `casing.renderer.ts` (off-limits to edit) for why: `buildCasingPath` always draws a
 * casing as a full tube from ground level (0) to its own depth, so the only way to reflect its
 * real entered size is via this array's order.
 *
 * Unparseable sizes sort as 0in (narrowest/first) — the safest fallback, since
 * `isRenderableCasingRow` already guarantees `sizeIn` is non-empty text, just not necessarily
 * a `parseInchSize`-recognizable number.
 */
function mapCasings(rows: readonly WwellIrRawCasingRow[]): ICasingIR[] {
  const casings: ICasingIR[] = rows.map(row => ({
    csgSize: row.sizeIn,
    csgType: row.type,
    csgDepth: row.depthFtBgl ?? 0,
    csgRemarks: row.remarks || null,
  }));
  return [...casings].sort((a, b) => (parseInchSize(a.csgSize) ?? 0) - (parseInchSize(b.csgSize) ?? 0));
}

/**
 * `liner.renderer.ts`'s `rawLinersFromIR` can only learn a liner's TOP depth from a matching
 * `drlgCasings` (as-drilled) row — with none supplied, every planned liner falls back to the
 * SAME top (the deepest structural casing's shoe), so two or more stacked liners all render
 * starting at the same point instead of hanging from one another (visible as overlapping
 * "TOL @ ..." labels).
 *
 * This builds synthetic `IDrlgCsg` "anchor" entries carrying ONLY the correct chained
 * `wLnrTopDepth` per liner — never `wLnrBotDepth` or a real `wCsgOdSz` — because
 * `rawLinersFromIR` gates `hasActual` (and therefore the green-vs-silver color choice)
 * strictly on `wLnrBotDepth`, and gates the casing-actual lookup strictly on `wCsgOdSz`.
 * Leaving both unset fixes the top-depth chain while keeping every casing and liner silver.
 *
 * Chain rule (matches the physical assumption already baked into liner.renderer.ts, e.g.
 * `renderLinerShapes`'s `hangerY`): the shallowest liner hangs from the deepest structural
 * casing's shoe; each deeper liner hangs from the liner immediately above it. Casing rows
 * whose type is Conductor, Surface, Intermediate, or Production are all "structural" here —
 * anything that isn't `'Liner'`, exactly matching how casing.renderer.ts itself distinguishes
 * casing from liner (`csgType !== 'Liner'`).
 *
 * A row's own `tolFtBgl` (the wizard's "Top Of Liner (TOL)" field, shown only for Liner rows)
 * overrides this auto-chained value when the user has filled it in — the chain computation
 * above becomes the fallback for rows left blank. Explicit TOLs don't otherwise perturb the
 * chain: the next liner's fallback top is still this row's own `depthFtBgl` (its shoe), not
 * its `tolFtBgl`, since a liner physically hangs from the casing/liner above it regardless of
 * how far down its own top was entered.
 */
function buildLinerTopAnchors(rows: readonly WwellIrRawCasingRow[]): IDrlgCsg[] {
  const structuralDepths = rows.filter(row => row.type !== LINER_TYPE).map(row => row.depthFtBgl ?? 0);
  const lastCasingShoe = structuralDepths.length ? Math.max(...structuralDepths) : 0;

  const linersShallowToDeep = rows
    .filter(row => row.type === LINER_TYPE)
    .slice()
    .sort((a, b) => (a.depthFtBgl ?? 0) - (b.depthFtBgl ?? 0));

  let anchorTop = lastCasingShoe;
  return linersShallowToDeep.map(row => {
    const anchor: IDrlgCsg = {
      wCsgOdSz: LINER_ANCHOR_SENTINEL,
      wCsgBotDpth: 0,
      wLnrOdSz: row.sizeIn,
      wLnrTopDepth: row.tolFtBgl ?? anchorTop,
      wLnrBotDepth: null,
    };
    anchorTop = row.depthFtBgl ?? anchorTop;
    return anchor;
  });
}

function mapGeologicTops(
  rows: readonly WwellIrRawFormationTop[],
  surfaceElevationFtAsl: number | null,
): ITopsIR[] {
  return rows.map(row => {
    const planTvdDepth = row.depthFt ?? 0;
    return {
      stLongCd: row.name,
      planTvdDepth,
      planSsDepth: surfaceElevationFtAsl !== null ? surfaceElevationFtAsl - planTvdDepth : 0,
    };
  });
}

/**
 * Derives completion-element flags/remarks from the "Completion" dropdown (Program
 * Details step) plus, for Open Hole, the conditionally-shown "Open Hole Size" field.
 * `well-bore-view`'s renderers gate Open Hole strictly on a parseable inch size in
 * `ohRemarks`, and Gravel Pack / Liner Screen strictly on non-empty remarks text — the
 * constant fallback labels below exist only to satisfy those gates when the dropdown
 * implies the element but no free-text remarks field exists for it in this form.
 *
 * "Liner + Perforated" only actually renders once a Casing-table Liner row is exactly 7"
 * OD — that 7" convention is hardcoded inside liner.renderer.ts (PRE_PERF_LINER_SIZE_IN)
 * and is intentionally left untouched here.
 */
function buildWellDesignFromForm(programMain: WwellIrRawProgramMain): IWellDesign {
  const completion = programMain.completion;
  const isOpenHole = completion === COMPLETION_OPEN_HOLE;
  const isGravelPack = completion === COMPLETION_SCREENED_GRAVEL_PACK;
  const isScreened = completion === COMPLETION_SCREENED || isGravelPack;
  const isLinerPerforated = completion === COMPLETION_LINER_PERFORATED;

  return {
    name: null,
    prewapSeq: null,
    staWaterLvl: null,
    swLvlTxt: null,
    pumpLvl: null,
    ohFlg: isOpenHole ? 'Y' : 'N',
    ohRemarks: isOpenHole ? (programMain.openHoleSizeIn || null) : null,
    gpFlg: isGravelPack ? 'Y' : 'N',
    gpRemarks: isGravelPack ? 'Gravel Pack' : null,
    perforatedFlg: isLinerPerforated ? 'Y' : 'N',
    perforatedRemarks: isLinerPerforated ? '7" Pre-Perforated Liner' : null,
    lsFlg: isScreened ? 'Y' : 'N',
    lsRemarks: isScreened ? 'Screen' : null,
  };
}

function computeDisplayDepthFt(raw: WwellIrFormRawValue): number {
  const depths = [
    raw.reference.main.targetDepthFt ?? 0,
    ...raw.subsurface.casing.map(row => row.depthFtBgl ?? 0),
    ...raw.subsurface.formationTops.map(row => row.depthFt ?? 0),
  ];
  return Math.max(...depths, MIN_SCALE_DEPTH_FT);
}

/**
 * Gates the *entire* diagram — depth scale, ground level line, and total-depth label included,
 * not just individual casing/formation-top shapes (see `isRenderableCasingRow`/
 * `isRenderableFormationTop` above). Before any row is complete, `well-bore-view` still draws
 * that scaffold around nothing: a bare 0–500ft ruler, a "GROUND LEVEL" line, and a
 * "TOTAL DEPTH: 0 FT" label with no actual construction data behind any of it. Callers should
 * check this first and render an empty state instead of the diagram until it's true.
 */
export function hasWellboreDiagramData(raw: WwellIrFormRawValue): boolean {
  return (
    raw.subsurface.casing.some(isRenderableCasingRow) ||
    raw.subsurface.formationTops.some(isRenderableFormationTop)
  );
}

export function buildWellboreDiagramData(raw: WwellIrFormRawValue): WellboreDiagramData {
  const targetDepthFt = raw.reference.main.targetDepthFt ?? 0;
  const renderableCasings = raw.subsurface.casing.filter(isRenderableCasingRow);
  const renderableFormationTops = raw.subsurface.formationTops.filter(isRenderableFormationTop);

  return {
    wellName: raw.reference.main.wellName,
    totalDepth: targetDepthFt,
    displayDepth: computeDisplayDepthFt(raw),
    casings: mapCasings(renderableCasings),
    drlgCasings: buildLinerTopAnchors(renderableCasings),
    geologicTops: mapGeologicTops(renderableFormationTops, raw.reference.main.surfaceElevationFtAsl),
    hydrogeology: null,
    prewap: null,
    rigActivity: null,
    currentDepth: 0,
    targetDepth: targetDepthFt,
    mudCirculation: [],
    wellDesign: buildWellDesignFromForm(raw.programDetails.main),
  };
}
```

```ts
  // Review, and the dedicated Well Design View step) redraws as casing/formation/completion
  // data changes anywhere earlier in the wizard.
  private readonly rawFormValue: Signal<WwellIrFormRawValue> = toSignal(this.form.valueChanges, {
    initialValue: this.form.getRawValue() as WwellIrFormRawValue,
  });

  protected readonly activeStepIndex = signal(0);

  // Position-based fill for the custom step rail (see wwell-ir-new-page.component.scss's
  // `.nip__rail`) — width is a percentage of the *whole* track div, but the track itself is
  // inset by half a step's flex-width on each side (so it starts/ends at the first/last
  // chip's center, not the track container's edge). Dividing by `schema.length` rather than
  // `schema.length - 1` bakes that inset directly into the percentage: for n evenly-flexed
  // steps the chip centers sit at (i + 0.5) / n, and the track's own left inset is 0.5 / n,
  // so distance-from-left-inset-to-chip-i's-center reduces exactly to i / n.
  protected readonly railProgressPercent = computed(
    () => (this.activeStepIndex() / this.schema.length) * 100,
  );

  /*
   * Feeds every `app-well-bore-view` instance in this wizard — same real diagram
   * everywhere, just adapted from the raw form value into the richer WellboreDiagramData
   * shape.
   *
   * `well-bore-view`'s own redraw effect never reads its `animTrigger` input (confirmed by
   * reading well-bore-view.component.ts — only `diagramData`/`diagramLayout`/`isFullscreen`
   * are tracked), so it only ever redraws when `diagramData` gets a NEW object reference.
   * Angular Material's horizontal stepper renders every step's content upfront and keeps it
   * mounted (hidden inactive steps get `visibility: hidden` + an off-screen `transform`, not
   * `display: none`/destroyed) — so the very first redraw (with its clip-path "grow" reveal
   * transition, see casing.renderer.ts's `renderCasings`) runs while this step is still
   * hidden and transformed off-screen, and has already finished — collapsed — by the time
   * the user actually navigates here. Tracking `activeStepIndex` below forces a brand-new
   * `WellboreDiagramData` object (and therefore a fresh redraw + reveal animation) every time
   * the active step changes, so it's guaranteed to run once while genuinely visible. This is
   * also a wwell-ir-local fix, per the standing rule not to modify anything under
   * `presentation/well-bore-view/`.
   */
  protected readonly wellboreDiagramData: Signal<WellboreDiagramData> = computed(() => {
    this.activeStepIndex();
    return buildWellboreDiagramData(this.rawFormValue());
  });

  // Gates the diagram card's scale/ground-level/total-depth chrome, not just individual
  // casing/formation-top shapes — see `hasWellboreDiagramData`'s own doc comment.
  protected readonly hasWellboreData: Signal<boolean> = computed(() =>
    hasWellboreDiagramData(this.rawFormValue()),
  );

  // Surfaces cases where a deeper casing's entered size is larger than a shallower casing's —
  // see `findInvertedCasingSizeRows`'s doc comment for why the diagram can never draw that
  // combination correctly.
  protected readonly casingSizeWarnings: Signal<readonly CasingSizeWarning[]> = computed(() =>
    findInvertedCasingSizeRows(this.rawFormValue().subsurface.casing),
  );

  // `wellboreDiagramData().totalDepth` falls back to 0 when Target Depth (Request Reference,
  // a required field) hasn't been filled in yet — reachable via the stepper's free (non-linear)
  // navigation. `well-bore-view` always draws its "TOTAL DEPTH: 0 FT" label regardless (see the
  // standing rule against editing anything under `presentation/well-bore-view/`), so this instead
  // drives a page-local CSS hook (`.nip__wellbore-diagram-row--no-depth` in
  // wwell-ir-new-page.component.scss) that hides that one D3-rendered label group.
  protected readonly hasTargetDepth: Signal<boolean> = computed(
    () => (this.rawFormValue().reference.main.targetDepthFt ?? 0) > 0,
  );
  @ViewChildren('stepContent') private readonly stepContents?: QueryList<ElementRef<HTMLElement>>;
  @ViewChildren('reviewCard', { read: ElementRef }) private readonly reviewCards?: QueryList<ElementRef<HTMLElement>>;

  private previousStepIndex = 0;
  private wasSaving = false;
  private reviewRevealed = false;

  constructor() {
    effect(() => {
      const index = this.activeStepIndex();
      const host = this.stepContents?.get(index);
      if (!host) return;

      const direction: WwellStepDirection = index >= this.previousStepIndex ? 'forward' : 'backward';
      playIfMotionAllowed(buildStepTransitionTimeline(host.nativeElement, direction));
      this.previousStepIndex = index;
    });

    effect(() => {
      // Keyed on the 'attachments' step itself (not "last step") — Well Design View is now
      // appended after it, so the review UI is no longer on the final step.
      const isReviewStep = WWELL_IR_FORM_SCHEMA[this.activeStepIndex()]?.key === 'attachments';
      if (!isReviewStep || this.reviewRevealed) return;

      const cards = this.reviewCards?.map(card => card.nativeElement) ?? [];
      if (!cards.length) return;

      playIfMotionAllowed(buildReviewRevealTimeline(cards));
      this.reviewRevealed = true;
    });

    effect(() => {
      const saving = this.isSaving();
      const error = this.store.createIntegratedRequirementError();

      if (error) {
        this.snackBar.open(error, 'Dismiss', { duration: SNACKBAR_DURATION_MS });
      } else if (this.wasSaving && !saving) {
        this.router.navigate(['..'], { relativeTo: this.route });
      }
      this.wasSaving = saving;
    });
  }

  protected stepGroup(stepKey: string): FormGroup {
    return this.form.get(stepKey) as FormGroup;
  }

  protected sectionControl(stepKey: string, sectionKey: string): FormGroup | FormArray {
    return this.stepGroup(stepKey).get(sectionKey) as FormGroup | FormArray;
  }

  protected isStepValid(stepKey: string): boolean {
    return this.stepGroup(stepKey).valid;
  }

  protected reviewCompletionSummary(): { done: number; total: number } {
    const total = this.reviewSteps.length;
    const done = this.reviewSteps.filter(step => this.isStepValid(step.key)).length;
    return { done, total };
  }

  protected onReviewCardOpened(index: number): void {
    this.reviewCards?.get(index)?.nativeElement.scrollIntoView({ behavior: 'smooth', block: 'nearest' });
  }

  protected onStepSelectionChange(index: number): void {
    this.activeStepIndex.set(index);
  }

  protected cancel(): void {
    this.router.navigate(['..'], { relativeTo: this.route });
  }

  protected saveAsDraft(): void {
    this.submitAs('draft');
  }

  protected submit(stepper: MatStepper): void {
    if (this.form.invalid) {
      this.form.markAllAsTouched();
      const invalidIndex = WWELL_IR_FORM_SCHEMA.findIndex(step => this.stepGroup(step.key).invalid);
      if (invalidIndex >= 0) stepper.selectedIndex = invalidIndex;
      return;
    }
    this.submitAs('in-progress');
  }

  private submitAs(status: 'draft' | 'in-progress'): void {
    const raw = this.form.getRawValue() as WwellIrFormRawValue;
    const { payload, files } = mapIrFormValueToSubmission(raw, status);
    this.store.createIntegratedRequirement(payload, files);
  }
}
```
