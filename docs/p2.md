```ts
import * as THREE from 'three';

import { OrbitCameraController } from '../utils/orbit-camera.controller';
import { frameTicker } from '../utils/frame-ticker.util';
import { SCENE_HEIGHT_UNITS } from '../utils/depth-3d.util';

/**
 * Owns the Three.js renderer/scene/camera/lights and the render loop for one wellbore-3d-view
 * instance. Rendering is driven by the shared `frameTicker` (one `requestAnimationFrame` loop), so the
 * render pass and the component's zoom sync run in a fixed order each frame.
 */
export class WellboreSceneEngine {
  readonly scene = new THREE.Scene();
  readonly camera: THREE.PerspectiveCamera;
  readonly renderer: THREE.WebGLRenderer;
  readonly rigGroup = new THREE.Group();
  readonly orbit: OrbitCameraController;

  private readonly resizeObserver: ResizeObserver;
  private readonly tickFn: (time: number, deltaTime: number) => void;
  private lastWidth = 0;
  private lastHeight = 0;
  private disposed = false;

  constructor(private readonly host: HTMLElement) {
    this.renderer = new THREE.WebGLRenderer({ antialias: true, alpha: true });
    this.renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));
    this.renderer.localClippingEnabled = true;
    this.renderer.shadowMap.enabled = false;

    // This canvas is inserted via plain DOM API, not Angular's template — so it never receives
    // the component's `_ngcontent-*` encapsulation attribute, and the component SCSS rule that
    // stretches `canvas` to 100% width/height silently never matches it. Combined with
    // `setSize(..., false)` below (which intentionally skips setting CSS size), the canvas would
    // otherwise render at its raw attribute size (CSS pixels = display size × device pixel ratio)
    // instead of the container's actual size — i.e. up to 2x too big, clipped by the parent's
    // `overflow: hidden` down to just its top-left quadrant, with anything centered in the real
    // scene appearing to sit in the bottom-right corner of that crop. Setting these inline
    // guarantees correct sizing regardless of Angular's style scoping.
    const canvasEl = this.renderer.domElement;
    canvasEl.style.display = 'block';
    canvasEl.style.width = '100%';
    canvasEl.style.height = '100%';
    canvasEl.style.touchAction = 'none';
    canvasEl.style.cursor = 'grab';
    host.appendChild(canvasEl);

    // Without explicit tone mapping, bright specular highlights on the fairly metallic casing
    // material simply clip to pure white instead of rolling off — which is what was reading as
    // "washed out white" instead of the intended green/silver metal colors.
    this.renderer.outputColorSpace = THREE.SRGBColorSpace;
    this.renderer.toneMapping = THREE.ACESFilmicToneMapping;
    this.renderer.toneMappingExposure = 1.1;

    this.camera = new THREE.PerspectiveCamera(45, 1, 0.1, 500);
    this.scene.add(this.rigGroup);
    // Plain white, matching the 2D SVG diagram's card background.
    this.scene.background = new THREE.Color(0xffffff);
    this.scene.fog = new THREE.Fog(0xffffff, 90, 220);

    this.setupLights();

    this.orbit = new OrbitCameraController(this.camera, this.renderer.domElement, {
      initialRadius: SCENE_HEIGHT_UNITS * 1.1,
      targetY: -SCENE_HEIGHT_UNITS * 0.4,
      minRadius: SCENE_HEIGHT_UNITS * 0.35,
      maxRadius: SCENE_HEIGHT_UNITS * 2.4,
    });

    this.resizeObserver = new ResizeObserver(() => this.handleResize());
    this.resizeObserver.observe(host);
    this.handleResize();

    this.tickFn = (_time, deltaTime) => {
      this.orbit.update(deltaTime / 1000);
      this.renderer.render(this.scene, this.camera);
    };
    frameTicker.add(this.tickFn);
  }

  private setupLights(): void {
    const hemi = new THREE.HemisphereLight(0xffffff, 0x3a4657, 0.65);
    this.scene.add(hemi);

    const key = new THREE.DirectionalLight(0xffffff, 0.9);
    key.position.set(18, 30, 22);
    this.scene.add(key);

    const rim = new THREE.DirectionalLight(0x8fd3ff, 0.3);
    rim.position.set(-20, 10, -18);
    this.scene.add(rim);

    const fill = new THREE.PointLight(0xffe8b8, 0.25, 60);
    fill.position.set(0, 4, 12);
    this.scene.add(fill);
  }

  private handleResize(): void {
    const width = Math.max(1, this.host.clientWidth);
    const height = Math.max(1, this.host.clientHeight);
    if (width === this.lastWidth && height === this.lastHeight) return;
    this.lastWidth = width;
    this.lastHeight = height;
    this.camera.aspect = width / height;
    this.camera.updateProjectionMatrix();
    this.renderer.setSize(width, height, false);
  }

  dispose(): void {
    if (this.disposed) return;
    this.disposed = true;
    frameTicker.remove(this.tickFn);
    this.resizeObserver.disconnect();
    this.orbit.dispose();
    this.renderer.dispose();
    this.renderer.domElement.remove();
  }
}
```

```ts
/** Per-frame callback: `deltaTime` is the milliseconds since the previous frame. */
export type FrameCallback = (time: number, deltaTime: number) => void;

/** A gap longer than this (tab was hidden, thread stalled) is treated as one normal frame, so the camera doesn't jump. */
const LAG_THRESHOLD_MS = 500;
const LAG_ADJUSTED_MS = 33;

const callbacks = new Set<FrameCallback>();
let frameHandle: number | null = null;
let lastTime = 0;

function tick(now: number): void {
  const gap = now - lastTime;
  const deltaTime = gap > LAG_THRESHOLD_MS ? LAG_ADJUSTED_MS : gap;
  lastTime = now;

  // Copied so a callback can remove itself (or another) mid-frame without skipping its neighbour.
  [...callbacks].forEach(callback => {
    if (callbacks.has(callback)) callback(now, deltaTime);
  });

  frameHandle = callbacks.size > 0 ? requestAnimationFrame(tick) : null;
}

/** One shared `requestAnimationFrame` loop; callbacks run in the order they were added. */
export const frameTicker = {
  add(callback: FrameCallback): void {
    callbacks.add(callback);
    if (frameHandle !== null) return;
    lastTime = performance.now();
    frameHandle = requestAnimationFrame(tick);
  },

  remove(callback: FrameCallback): void {
    callbacks.delete(callback);
    if (callbacks.size === 0 && frameHandle !== null) {
      cancelAnimationFrame(frameHandle);
      frameHandle = null;
    }
  },
};
```

```ts
import * as THREE from 'three';

const MIN_POLAR = 0.25;
const MAX_POLAR = Math.PI - 0.25;
const DAMPING = 0.08;
const AUTO_ROTATE_SPEED = 0.06; // radians / second

/**
 * Minimal spherical-coordinates orbit/zoom/pan controller.
 *
 * A hand-rolled replacement for `three/addons/controls/OrbitControls` — the addons build
 * ships ESM-only and isn't transformable under the project's current Jest config, so this
 * keeps the 3D view fully self-contained and test-safe without touching jest.config.js.
 */
/** Default azimuth/polar angle the view opens on. Casing/liner/conductor pipes are built with no
 *  top or bottom caps (see `buildPipeMesh`), so whichever end faces the camera reads as a hollow
 *  opening rather than a solid surface. A level, phi=90° camera sits at the vertical *midpoint* of
 *  the well (the target is the framed box's center) — that put the camera *below* the shallower
 *  casings/conductor near ground level, looking up into their open tops from underneath, which
 *  read as the tube flaring outward/upward like a megaphone bell instead of a shaft going down
 *  into the ground. Tilting the camera down (phi < 90°) puts it above that midpoint looking
 *  downward instead, so every open top — including the conductor's, at the very top — is seen
 *  from above like looking down a shaft, matching the "going into the ground" read the 2D
 *  elevation diagram implies. `resetOrientation()`/the "recenter" button return to this same angle. */
const FRONT_THETA = 0;
const FRONT_PHI = 1.05;

export class OrbitCameraController {
  /** Starts static (like the 2D diagram) — the user opts into auto-rotate via the toolbar. */
  autoRotate = false;

  private smoothedRadius: number;
  private targetRadius: number;
  private smoothedTheta: number;      // azimuth
  private targetTheta: number;
  private smoothedPhi: number;        // polar
  private targetPhi: number;
  private readonly target = new THREE.Vector3();
  private isDragging = false;
  private lastX = 0;
  private lastY = 0;
  private minRadius: number;
  private maxRadius: number;

  private readonly onPointerDown = (e: PointerEvent): void => {
    this.isDragging = true;
    this.autoRotate = false;
    this.lastX = e.clientX;
    this.lastY = e.clientY;
    this.dom.setPointerCapture(e.pointerId);
  };

  private readonly onPointerMove = (e: PointerEvent): void => {
    if (!this.isDragging) return;
    const dx = e.clientX - this.lastX;
    const dy = e.clientY - this.lastY;
    this.lastX = e.clientX;
    this.lastY = e.clientY;
    this.targetTheta -= dx * 0.006;
    this.targetPhi = Math.min(MAX_POLAR, Math.max(MIN_POLAR, this.targetPhi - dy * 0.006));
  };

  private readonly onPointerUp = (e: PointerEvent): void => {
    this.isDragging = false;
    this.dom.releasePointerCapture(e.pointerId);
  };

  private readonly onWheel = (e: WheelEvent): void => {
    e.preventDefault();
    const factor = Math.exp(e.deltaY * 0.001);
    this.targetRadius = Math.min(this.maxRadius, Math.max(this.minRadius, this.targetRadius * factor));
  };

  constructor(
    private readonly camera: THREE.PerspectiveCamera,
    private readonly dom: HTMLElement,
    opts: { initialRadius: number; targetY: number; minRadius: number; maxRadius: number },
  ) {
    this.smoothedRadius = this.targetRadius = opts.initialRadius;
    this.smoothedTheta = this.targetTheta = FRONT_THETA;
    this.smoothedPhi = this.targetPhi = FRONT_PHI;
    this.minRadius = opts.minRadius;
    this.maxRadius = opts.maxRadius;
    this.target.set(0, opts.targetY, 0);

    this.dom.addEventListener('pointerdown', this.onPointerDown);
    this.dom.addEventListener('pointermove', this.onPointerMove);
    this.dom.addEventListener('pointerup', this.onPointerUp);
    this.dom.addEventListener('pointerleave', this.onPointerUp);
    this.dom.addEventListener('wheel', this.onWheel, { passive: false });
  }

  setTargetY(y: number): void {
    this.target.y = y;
  }

  /** Tweenable orbit radius — lets the reveal timeline drive a cinematic dolly with its
   *  own easing curve, on top of the controller's own per-frame damping. */
  get radius(): number {
    return this.targetRadius;
  }

  set radius(value: number) {
    this.targetRadius = Math.min(this.maxRadius, Math.max(this.minRadius, value));
  }

  /** The *damped* radius actually applied to the camera this frame — unlike `radius` (the
   *  target the user just scrolled/tweened to), this is what the camera position is built from
   *  in `update()`. Anything that needs to stay pixel-perfect in sync with the rendered camera
   *  (e.g. the depth ruler re-anchoring itself to the current view every frame) must read this,
   *  not `radius`, or it will visibly snap ahead of the camera's own eased motion. */
  get currentRadius(): number {
    return this.smoothedRadius;
  }

  /** Read-only snapshot of the current look-at target, in case it's ever off the X=0 centerline. */
  get targetPosition(): THREE.Vector3 {
    return this.target.clone();
  }

  /** Tweenable azimuth (radians) — used for the establishing-shot swing-around. */
  get azimuth(): number {
    return this.targetTheta;
  }

  set azimuth(value: number) {
    this.targetTheta = value;
  }

  /** Snaps back to the flat, front-elevation angle the view opens on — used by the "recenter" button. */
  resetOrientation(): void {
    this.targetTheta = FRONT_THETA;
    this.targetPhi = FRONT_PHI;
  }

  /**
   * Fits a world-space bounding box into the camera's current FOV/aspect and updates the orbit
   * target + zoom bounds to match. Returns the "rest" distance that comfortably frames it.
   *
   * Deliberately fits the box's actual height/width per axis rather than wrapping it in a
   * bounding *sphere* — a well is extremely tall and thin (tens of units tall, a handful of
   * units wide), so a sphere large enough to enclose the full height is mostly empty space
   * around a sliver-thin object. Backing the camera off far enough to fit that sphere in every
   * direction left the well looking tiny in the middle of the frame; fitting height-to-vertical-
   * FOV and width-to-horizontal-FOV separately lets the camera sit much closer.
   *
   * The two axes intentionally use *different* margins. `marginHeight` defaults to ~1.14 so the
   * vertical fill ratio (1/marginHeight ≈ 88%) matches the 2D SVG's fixed
   * `drawingHeight / svgHeight` ratio (740/840) — the caller passes in a box whose Y range is
   * pinned to the nominal [0, displayDepth] depth range (see `wellbore-3d-view.component.ts`'s
   * `frameCamera`), so this keeps "ground level" (depth 0) sitting at the same proportional
   * screen height as the sibling `<app-depth-scale>` ruler's "0" tick, instead of drifting per
   * well based on how deep that well's actual casings happen to reach. `marginWidth` stays more
   * generous (1.4) since the ruler has no opinion on width — that only needs to guarantee wells
   * with many (wide) casings never clip outside the viewbox.
   *
   * Zoom bounds are pinned tightly to this same fit, on purpose: `minRadius` is set to the exact
   * *unmargined* distance for whichever axis is dominant — the distance at which that axis fills
   * 100% of the frame edge-to-edge — so scrolling in can never push the geometry outside the
   * viewbox. `maxRadius` stays within a modest multiple of the fit distance so scrolling out
   * can't shrink the content down to an illegibly tiny speck either.
   */
  frameBox(box: THREE.Box3, marginHeight = 1.14, marginWidth = 1.4): number {
    const size = box.getSize(new THREE.Vector3());
    const center = box.getCenter(new THREE.Vector3());
    this.target.copy(center);

    const halfHeight = Math.max(size.y / 2, 0.05);
    const halfWidth = Math.max(Math.max(size.x, size.z) / 2, 0.05);

    // `camera.aspect` can be a stale/garbage value for the very first frame or two before the
    // host element's ResizeObserver has fired — falling back to 1 keeps this fully finite
    // instead of ever letting a NaN/zero aspect poison targetRadius (and blank the whole scene).
    const safeAspect = Number.isFinite(this.camera.aspect) && this.camera.aspect > 0 ? this.camera.aspect : 1;
    const halfV = THREE.MathUtils.degToRad(this.camera.fov) / 2;
    const halfH = Math.atan(Math.tan(halfV) * safeAspect);
    const heightFit = (halfHeight / Math.tan(halfV)) * marginHeight;
    const widthFit = (halfWidth / Math.tan(halfH)) * marginWidth;
    const fitDistance = Math.max(heightFit, widthFit);

    // Whichever axis is dominant, dividing its own margin back out gives the exact distance at
    // which it fills the frame edge-to-edge (still inside the viewbox, just with zero breathing
    // room) — anything closer than that would clip. Using the dominant axis's own margin (rather
    // than always the smaller one) keeps this precise instead of overly conservative.
    const noMarginFit = heightFit >= widthFit ? heightFit / marginHeight : widthFit / marginWidth;
    this.minRadius = noMarginFit;
    this.maxRadius = fitDistance * 1.5;

    return fitDistance;
  }

  /** Advances damping/auto-rotate and writes the resulting position onto the camera. Call once per frame. */
  update(deltaSeconds: number): void {
    if (this.autoRotate && !this.isDragging) {
      this.targetTheta += AUTO_ROTATE_SPEED * deltaSeconds;
    }

    this.smoothedRadius += (this.targetRadius - this.smoothedRadius) * DAMPING;
    this.smoothedTheta += (this.targetTheta - this.smoothedTheta) * DAMPING;
    this.smoothedPhi += (this.targetPhi - this.smoothedPhi) * DAMPING;

    const sinPhi = Math.sin(this.smoothedPhi);
    this.camera.position.set(
      this.target.x + this.smoothedRadius * sinPhi * Math.sin(this.smoothedTheta),
      this.target.y + this.smoothedRadius * Math.cos(this.smoothedPhi),
      this.target.z + this.smoothedRadius * sinPhi * Math.cos(this.smoothedTheta),
    );
    this.camera.lookAt(this.target);
  }

  dispose(): void {
    this.dom.removeEventListener('pointerdown', this.onPointerDown);
    this.dom.removeEventListener('pointermove', this.onPointerMove);
    this.dom.removeEventListener('pointerup', this.onPointerUp);
    this.dom.removeEventListener('pointerleave', this.onPointerUp);
    this.dom.removeEventListener('wheel', this.onWheel);
  }
}
```

```ts
  Component,
  DestroyRef,
  ElementRef,
  effect,
  inject,
  input,
  viewChild,
} from '@angular/core';
import { frameTicker } from './utils/frame-ticker.util';
import * as THREE from 'three';

import { FormationInfoViewModel } from '@models/active-wwell/active-wwell-view.model';
import {
  DEFAULT_ANIM_CONFIG,
  DIAGRAM_LAYOUT,
  DiagramLayout,
  WellboreAnimConfig,
  WellboreDiagramData,
} from '@models/well-design/wellbore-diagram.model';
import { pickTickInterval } from '@shared/utils/wellbore-math.util';
import { buildCasings3D } from './builders/casing-3d.builder';
import { buildDepthScale3D } from './builders/depth-scale-3d.builder';
import { WellboreSceneEngine } from './engine/scene-engine';
import { createDepth3DScale, SCENE_HEIGHT_UNITS } from './utils/depth-3d.util';
import { disposeObject3D } from './utils/dispose3d.util';
import { Depth3DScale } from './wellbore-3d.types';

/** Only used when the built content is empty (e.g. no casing data yet) — real framing is
 *  always computed from the nominal depth range + actual geometry width, see `frameCamera()` below. */
const FALLBACK_REST_RADIUS = SCENE_HEIGHT_UNITS * 1.5;

/** Extra world-space buffer beyond the widest depth-scale label so text never touches the exact
 *  edge of the visible frustum. */
const DEPTH_SCALE_SAFETY_MARGIN = 0.5;
/** Floor on how close to the pipe the ruler is ever allowed to sit, however tight the frustum
 *  gets at the closest permitted zoom — keeps the reach calculation from ever going negative. */
const DEPTH_SCALE_MIN_REACH = 0.5;
const WORLD_UP = new THREE.Vector3(0, 1, 0);

/**
 * TEMP scope reduction, per request: this view currently draws ONLY the conductor/casing/liner
 * pipes, fully visible and static (no reveal animation, no cinematic camera). Open hole, gravel
 * pack, water level, drill string, pump marker and geologic-top rings — plus the reveal
 * timeline that animated all of it — are on hold until the core casing/liner geometry itself is
 * confirmed to render correctly. See the `real-3d/builders/` and `real-3d/animation/` folders
 * for that code; nothing was deleted, this component just doesn't call into it right now.
 */
@Component({
  selector: 'app-wellbore-3d-view',
  standalone: true,
  changeDetection: ChangeDetectionStrategy.OnPush,
  templateUrl: './wellbore-3d-view.component.html',
  styleUrl: './wellbore-3d-view.component.scss',
})
export class Wellbore3DViewComponent {
  readonly diagramData = input.required<WellboreDiagramData | null>();
  readonly animTrigger = input.required<number>();
  readonly animConfig = input<WellboreAnimConfig>(DEFAULT_ANIM_CONFIG);
  readonly formationTops = input<FormationInfoViewModel[]>([]);
  readonly diagramLayout = input<DiagramLayout>(DIAGRAM_LAYOUT);
  readonly isFullscreen = input<boolean>(false);

  private readonly hostRef = viewChild<ElementRef<HTMLDivElement>>('canvasHost');
  private readonly destroyRef = inject(DestroyRef);

  private engine: WellboreSceneEngine | null = null;
  private activeLayout: DiagramLayout | null = null;
  /** Pipe meshes only — the *only* group ever measured for camera framing. */
  private contentGroup: THREE.Group | null = null;
  /** Casing/liner text label sprites + leader lines — cosmetic, excluded from framing. */
  private labelsGroup: THREE.Group | null = null;
  private restRadius = FALLBACK_REST_RADIUS;
  private pendingFrameId: number | null = null;

  // ── Depth-scale ruler state — built once per `rebuildScene()` call (same tick logic/interval as
  // the 2D ruler), then only *repositioned* every frame as the camera zooms/orbits (see
  // `syncDepthScaleToZoom`). Deliberately never added under `contentGroup`, so the ruler can never
  // feed back into the pipes' own framing. ──
  private depthScale: Depth3DScale | null = null;
  private displayDepthValue = 0;
  private depthScaleAxisGroup: THREE.Group | null = null;
  private depthScaleLabelsGroup: THREE.Group | null = null;
  private depthScaleLeftReach = 0;

  /** Runs every render frame for the lifetime of the component (not just while data is loaded) —
   *  reads `this.engine` fresh each time so it never goes stale across teardown/rebuild cycles,
   *  and no-ops cheaply whenever there's nothing to sync yet. */
  private readonly zoomSyncFn = (): void => {
    if (!this.engine) return;
    this.syncDepthScaleToZoom(this.engine);
  };

  constructor() {
    effect(() => {
      const hostEl = this.hostRef();
      const data = this.diagramData();
      const layout = this.diagramLayout();
      this.animTrigger();  // track — parent bumps this when it wants a fresh redraw
      this.isFullscreen(); // track — layout identity changes alongside this, see below

      if (!hostEl?.nativeElement || !data) return;

      // Re-bootstrap the renderer when layout identity changes (normal ↔ fullscreen), mirroring
      // WellBoreViewComponent's `_activeLayout` guard for the 2D SVG root.
      if (this.activeLayout !== layout) {
        this.activeLayout = layout;
        this.teardownEngine();
      }

      if (!this.engine) {
        this.engine = new WellboreSceneEngine(hostEl.nativeElement);
      }

      this.rebuildScene(data, layout);
    });

    frameTicker.add(this.zoomSyncFn);
    this.destroyRef.onDestroy(() => {
      frameTicker.remove(this.zoomSyncFn);
      this.teardownEngine();
    });
  }

  protected resetCamera(): void {
    if (!this.engine) return;
    this.engine.orbit.radius = this.restRadius;
    this.engine.orbit.resetOrientation();
  }

  protected toggleAutoRotate(): void {
    if (!this.engine) return;
    this.engine.orbit.autoRotate = !this.engine.orbit.autoRotate;
  }

  private rebuildScene(data: WellboreDiagramData, layout: DiagramLayout): void {
    const engine = this.engine;
    if (!engine) return;

    if (this.contentGroup) {
      engine.rigGroup.remove(this.contentGroup);
      disposeObject3D(this.contentGroup);
    }
    if (this.labelsGroup) {
      engine.rigGroup.remove(this.labelsGroup);
      disposeObject3D(this.labelsGroup);
    }
    if (this.depthScaleAxisGroup) {
      engine.rigGroup.remove(this.depthScaleAxisGroup);
      disposeObject3D(this.depthScaleAxisGroup);
      this.depthScaleAxisGroup = null;
    }
    if (this.depthScaleLabelsGroup) {
      engine.rigGroup.remove(this.depthScaleLabelsGroup);
      disposeObject3D(this.depthScaleLabelsGroup);
      this.depthScaleLabelsGroup = null;
    }

    const group = new THREE.Group();
    group.name = 'wellbore-3d-content';
    engine.rigGroup.add(group);
    this.contentGroup = group;

    const labelsGroup = new THREE.Group();
    labelsGroup.name = 'wellbore-3d-labels';
    engine.rigGroup.add(labelsGroup);
    this.labelsGroup = labelsGroup;

    const displayDepth = Math.max(data.totalDepth, data.displayDepth);
    const depthScale = createDepth3DScale(displayDepth);
    this.depthScale = depthScale;
    this.displayDepthValue = displayDepth;

    const casings = buildCasings3D(data, layout, depthScale, { casingsStart: 0, linersStart: 0 });
    group.add(casings.group);
    labelsGroup.add(casings.labelsGroup);

    // No reveal animation for now — snap every clip plane straight to "fully shown" so the
    // conductor/casing/liner pipes are immediately visible instead of waiting on a reveal timeline.
    casings.reveals.forEach(({ clipPlane, bottomY }) => {
      clipPlane.constant = -bottomY;
    });

    // Exactly mirrors `DepthScaleComponent.drawScale`'s interval logic — same 100/200/500ft
    // buckets, doubled in fullscreen — so tick spacing reads identically to the 2D ruler. Built
    // once here (like the pipes), never rebuilt reactively; `syncDepthScaleToZoom` only
    // repositions the already-built groups every frame.
    const baseInterval = pickTickInterval(displayDepth);
    const interval = this.isFullscreen() ? baseInterval * 2 : baseInterval;
    const builtScale = buildDepthScale3D({ totalDepth: displayDepth, depthScale, interval });
    engine.rigGroup.add(builtScale.axisGroup);
    engine.rigGroup.add(builtScale.labelsGroup);
    this.depthScaleAxisGroup = builtScale.axisGroup;
    this.depthScaleLabelsGroup = builtScale.labelsGroup;
    this.depthScaleLeftReach = builtScale.leftReach;

    this.frameCamera(engine, group, depthScale, displayDepth);
  }

  /** Fits the orbit camera to the pipe geometry just built (never the depth-scale ruler — see the
   *  field doc comments above). Runs once synchronously and once more on the next animation
   *  frame — the host element's ResizeObserver may not have reported its real size yet on the
   *  very first pass (e.g. right after this view is inserted behind the `@defer` block), which
   *  would frame against a stale 1:1 aspect ratio and leave the pipes off-center in the canvas. */
  private frameCamera(engine: WellboreSceneEngine, group: THREE.Group, depthScale: Depth3DScale, displayDepth: number): void {
    if (this.pendingFrameId !== null) {
      cancelAnimationFrame(this.pendingFrameId);
      this.pendingFrameId = null;
    }

    const apply = (): number => {
      // `Box3.setFromObject` reads each mesh's `matrixWorld`, which is only populated during a
      // render pass — measuring it synchronously, before the scene has ever actually rendered,
      // reads stale (identity) transforms and produces a bounding box that has nothing to do
      // with where the pipes really are. Force the update explicitly instead of hoping a render
      // happened to run first.
      group.updateMatrixWorld(true);
      const box = new THREE.Box3().setFromObject(group);
      if (box.isEmpty()) return FALLBACK_REST_RADIUS;

      // Pin the vertical frame to the *nominal* [0, displayDepth] range instead of the casing
      // geometry's actual extent — mirroring the 2D SVG diagram's `createDepthScale(displayDepth,
      // drawingHeight)`, which always maps this exact same fixed range across its full height.
      // Most wells' casings/liners stop well short of displayDepth (it's totalDepth plus a fixed
      // padding — see `selectDisplayDepth`), so framing on pipe height alone would zoom in
      // tighter than "ground level" and the deepest nominal depth actually correspond to.
      box.min.y = depthScale.toY(displayDepth);
      box.max.y = depthScale.toY(0);

      return engine.orbit.frameBox(box);
    };

    this.restRadius = apply();
    engine.orbit.radius = this.restRadius;

    this.pendingFrameId = requestAnimationFrame(() => {
      this.pendingFrameId = null;
      this.restRadius = apply();
      engine.orbit.radius = this.restRadius;
    });
  }

  /**
   * Keeps the 3D depth ruler acting like a real depth-wheel gauge: every frame it re-anchors
   * itself to the extreme left edge of whatever's *currently* in view (using the camera's
   * damped, actually-rendered radius — see `OrbitCameraController.currentRadius` — so it never
   * visibly snaps ahead of the camera's own eased motion). Tick geometry itself is built once per
   * `rebuildScene()` call (same fixed interval as the 2D ruler) and never rebuilt here — this only
   * repositions the existing groups, so zooming/orbiting can't trigger the flashing/duplication
   * that came from rebuilding tick geometry reactively.
   *
   * Position is computed from the camera's *actual* current forward/right vectors, not a
   * world-X assumption — the earlier version placed the ruler along the fixed world X axis,
   * which only happens to line up with "screen left" at the default front-elevation angle
   * (theta=0). As soon as the user orbits left/right, world-X stops meaning "left of camera": the
   * ruler would swing through world space independently of where the camera actually pointed,
   * at some angles cutting straight through the pipes. Deriving "left" from the camera's own
   * forward vector each frame keeps the ruler at a constant apparent position to the camera's
   * left — sitting still on screen while the camera orbits around the well, not swinging.
   */
  private syncDepthScaleToZoom(engine: WellboreSceneEngine): void {
    if (!this.depthScale || !this.depthScaleAxisGroup || !this.depthScaleLabelsGroup) return;

    const safeAspect = Number.isFinite(engine.camera.aspect) && engine.camera.aspect > 0 ? engine.camera.aspect : 1;
    const halfV = THREE.MathUtils.degToRad(engine.camera.fov) / 2;
    const halfH = Math.atan(Math.tan(halfV) * safeAspect);

    // Re-anchor to the extreme left of the current view every frame — "rolls" with the camera
    // instead of sitting at a fixed world position that would drift toward (or past) the pipes
    // as the user zooms in, or shrink into the middle of a lot of empty space zooming out.
    const radius = engine.orbit.currentRadius;
    const frustumHalfWidth = radius * Math.tan(halfH);

    // Camera's actual horizontal forward direction, whatever the current orbit angle — not an
    // assumed fixed axis. `getWorldDirection` recomputes from the camera's own current
    // position/quaternion, so this is correct at any theta, not just the front-elevation default.
    const forward = new THREE.Vector3();
    engine.camera.getWorldDirection(forward);
    forward.y = 0;
    if (forward.lengthSq() < 1e-6) forward.set(0, 0, -1); // camera looking straight down/up (polar-clamped, shouldn't happen)
    forward.normalize();
    const right = new THREE.Vector3().crossVectors(forward, WORLD_UP).normalize();

    const target = engine.orbit.targetPosition;
    const reach = Math.max(DEPTH_SCALE_MIN_REACH, frustumHalfWidth - this.depthScaleLeftReach - DEPTH_SCALE_SAFETY_MARGIN);
    const rulerX = target.x - right.x * reach;
    const rulerZ = target.z - right.z * reach;

    this.depthScaleAxisGroup.position.set(rulerX, 0, rulerZ);
    this.depthScaleLabelsGroup.position.set(rulerX, 0, rulerZ);
  }

  private teardownEngine(): void {
    if (this.pendingFrameId !== null) {
      cancelAnimationFrame(this.pendingFrameId);
      this.pendingFrameId = null;
    }
    if (this.contentGroup) {
      disposeObject3D(this.contentGroup);
      this.contentGroup = null;
    }
    if (this.labelsGroup) {
      disposeObject3D(this.labelsGroup);
      this.labelsGroup = null;
    }
    if (this.depthScaleAxisGroup) {
      disposeObject3D(this.depthScaleAxisGroup);
      this.depthScaleAxisGroup = null;
    }
    if (this.depthScaleLabelsGroup) {
      disposeObject3D(this.depthScaleLabelsGroup);
      this.depthScaleLabelsGroup = null;
    }
    this.depthScale = null;
    this.engine?.dispose();
    this.engine = null;
  }
}
```

```ts
  ChangeDetectionStrategy,

  ElementRef,
  OnDestroy,
  QueryList,
  ViewChildren,
  computed,
  input,
} from '@angular/core';
import { AbstractControl, FormArray, FormControl, FormGroup } from '@angular/forms';
import { Subscription } from 'rxjs';
import { DynamicFieldComponent, WwellDynamicFieldMode } from '../dynamic-field/dynamic-field.component';
import { playIfMotionAllowed } from '../motion.util';
import { RepeatableGroupComponent } from '../repeatable-group/repeatable-group.component';
import { defaultValueFor } from '../wwell-dynamic-form.builder';
import { WwellFieldConfig, WwellFormSectionConfig } from '../wwell-dynamic-form.types';
import {
  buildConditionalFieldEnterTimeline,
  buildConditionalFieldExitTimeline,
  setConditionalFieldVisibility,
} from './conditional-field.animation';

@Component({
  selector: 'app-dynamic-section',
  standalone: true,
  imports: [DynamicFieldComponent, RepeatableGroupComponent],
  templateUrl: './dynamic-section.component.html',
  styleUrl: './dynamic-section.component.scss',
  changeDetection: ChangeDetectionStrategy.OnPush,
})
export class DynamicSectionComponent implements AfterViewInit, OnDestroy {
  readonly section = input.required<WwellFormSectionConfig>();
  readonly control = input.required<AbstractControl>();
  readonly mode = input<WwellDynamicFieldMode>('edit');
  readonly selectPanelClass = input('');

  @ViewChildren('fieldHost') private readonly fieldHosts?: QueryList<ElementRef<HTMLElement>>;

  /** Only fields with `visibleWhen` grow a `#fieldHost` wrapper — this stays index-aligned with that query list. */
  protected readonly conditionalFields = computed(() => (this.section().fields ?? []).filter(field => field.visibleWhen));

  /** Tracks last-applied visibility per field key so we only animate on an actual flip, not every keystroke. */
  private readonly lastVisibility = new Map<string, boolean>();
  private valueChangesSubscription?: Subscription;

  ngAfterViewInit(): void {
    this.syncConditionalFields();
    this.valueChangesSubscription = this.control().valueChanges.subscribe(() => this.syncConditionalFields());
  }

  ngOnDestroy(): void {
    this.valueChangesSubscription?.unsubscribe();
  }

  protected fieldControl(key: string): FormControl {
    return (this.control() as FormGroup).get(key) as FormControl;
  }

  protected asFormArray(): FormArray {
    return this.control() as FormArray;
  }

  private syncConditionalFields(): void {
    this.conditionalFields().forEach((field, index) => {
      const host = this.fieldHosts?.get(index)?.nativeElement;
      if (host) this.applyFieldVisibility(field, host);
    });
  }

  private applyFieldVisibility(field: WwellFieldConfig, host: HTMLElement): void {
    const isVisible = this.isFieldVisible(field);
    const seenBefore = this.lastVisibility.has(field.key);
    const wasVisible = this.lastVisibility.get(field.key);
    this.lastVisibility.set(field.key, isVisible);

    if (!seenBefore) {
      setConditionalFieldVisibility(host, isVisible);
      this.syncControlEnabled(field, isVisible);
      return;
    }
    if (isVisible === wasVisible) return;

    if (isVisible) {
      this.syncControlEnabled(field, true);
      playIfMotionAllowed(buildConditionalFieldEnterTimeline(host));
    } else {
      playIfMotionAllowed(buildConditionalFieldExitTimeline(host, () => this.syncControlEnabled(field, false)));
    }
  }

  private isFieldVisible(field: WwellFieldConfig): boolean {
    const condition = field.visibleWhen;
    if (!condition) return true;

    const sibling = (this.control() as FormGroup).get(condition.field);
    const value = sibling?.value;
    if (condition.equals !== undefined) return value === condition.equals;
    if (condition.oneOf) return condition.oneOf.includes(value);
    if (condition.notOneOf) return !condition.notOneOf.includes(value);
    if (condition.includesLine !== undefined) {
      return typeof value === 'string' && value.split('\n').some(line => line.trim() === condition.includesLine);
    }
    return true;
  }

  /** Hidden fields are disabled (excluded from validity/`.value`) and reset so a stale value from the other mode never submits. */
  private syncControlEnabled(field: WwellFieldConfig, isVisible: boolean): void {
    const control = this.fieldControl(field.key);
    if (isVisible) {
      if (control.disabled) control.enable({ emitEvent: false });
      return;
    }
    if (control.enabled) {
      control.reset(defaultValueFor(field), { emitEvent: false });
      control.disable({ emitEvent: false });
    }
  }
}
```

```ts
/**
 * Tiny Web Animations API wrapper for the form/panel motion in this feature — opacity, translate,
 * scale and height tweens with eases, staggers and an on-complete hook. Replaces what GSAP was used
 * for here; builders return a `MotionTimeline` and callers hand it to `playIfMotionAllowed`.
 */

export type MotionEase = 'power1.out' | 'power1.in' | 'power2.out' | 'power2.in' | 'power3.out' | 'back.out';

// CSS equivalents of the matching GSAP eases (quad / cubic / quart / back).
const EASE_CSS: Readonly<Record<MotionEase, string>> = {
  'power1.out': 'cubic-bezier(0.5, 1, 0.89, 1)',
  'power1.in': 'cubic-bezier(0.11, 0, 0.5, 0)',
  'power2.out': 'cubic-bezier(0.33, 1, 0.68, 1)',
  'power2.in': 'cubic-bezier(0.32, 0, 0.67, 1)',
  'power3.out': 'cubic-bezier(0.25, 1, 0.5, 1)',
  'back.out': 'cubic-bezier(0.34, 1.8, 0.64, 1)',
};

const MS_PER_S = 1000;

export interface MotionProps {
  readonly opacity?: number;
  readonly x?: number;
  readonly y?: number;
  readonly scale?: number;
  /** Pixels, or `'auto'` to animate to/from the element's natural height. */
  readonly height?: number | 'auto';
  readonly transformOrigin?: string;
}

export interface MotionTween {
  readonly targets: readonly HTMLElement[];
  /** Omit for a "to" tween that starts from the element's current state. */
  readonly from?: MotionProps;
  readonly to: MotionProps;
  readonly duration: number;
  readonly ease: MotionEase;
  /** Seconds between each target's start. */
  readonly stagger?: number;
  /** Seconds from the timeline's start. */
  readonly at?: number;
}

export interface MotionTimeline {
  play(): void;
  /** Jumps every target to its end state and runs `onComplete`. */
  complete(): void;
}

export function prefersReducedMotion(): boolean {
  if (typeof window === 'undefined' || typeof window.matchMedia !== 'function') return false;
  return window.matchMedia('(prefers-reduced-motion: reduce)').matches;
}

/** Plays the timeline, or jumps straight to its end state when the user prefers reduced motion. */
export function playIfMotionAllowed(timeline: MotionTimeline): MotionTimeline {
  if (prefersReducedMotion()) {
    timeline.complete();
    return timeline;
  }
  timeline.play();
  return timeline;
}

/** Instantly pins an element's opacity/height (no tween). */
export function setMotionState(host: HTMLElement, props: Pick<MotionProps, 'opacity' | 'height'>): void {
  applyFinalStyles(host, props);
}

export function createMotionTimeline(tweens: readonly MotionTween[], onComplete?: () => void): MotionTimeline {
  let finished = false;
  const running = new Set<Animation>();

  const finish = (): void => {
    if (finished) return;
    finished = true;
    onComplete?.();
  };

  return {
    play(): void {
      const jobs = tweens.flatMap(tween => tween.targets.map((el, index) => ({ tween, el, index })));
      if (jobs.length === 0 || typeof jobs[0].el.animate !== 'function') {
        this.complete();
        return;
      }

      let remaining = jobs.length;
      jobs.forEach(({ tween, el, index }) => {
        const animation = el.animate(buildKeyframes(el, tween), {
          duration: tween.duration * MS_PER_S,
          delay: ((tween.at ?? 0) + index * (tween.stagger ?? 0)) * MS_PER_S,
          easing: EASE_CSS[tween.ease],
          fill: 'both',
        });
        running.add(animation);

        let settled = false;
        const settle = (): void => {
          if (settled) return;
          settled = true;
          running.delete(animation);
          // Commit the end state to inline styles first, then drop the animation's own fill.
          applyFinalStyles(el, tween.to);
          animation.cancel();
          remaining -= 1;
          if (remaining === 0) finish();
        };
        animation.onfinish = settle;
        animation.oncancel = settle;
      });
    },

    complete(): void {
      running.forEach(animation => {
        animation.oncancel = null;
        animation.onfinish = null;
        animation.cancel();
      });
      running.clear();
      tweens.forEach(tween => tween.targets.forEach(el => applyFinalStyles(el, tween.to)));
      finish();
    },
  };
}

function toTransform(props: MotionProps): string | null {
  if (props.x === undefined && props.y === undefined && props.scale === undefined) return null;
  return `translate(${props.x ?? 0}px, ${props.y ?? 0}px) scale(${props.scale ?? 1})`;
}

function measureAutoHeight(el: HTMLElement): number {
  const previous = el.style.height;
  el.style.height = 'auto';
  const height = el.offsetHeight;
  el.style.height = previous;
  return height;
}

function resolveHeight(el: HTMLElement, height: number | 'auto'): string {
  return `${height === 'auto' ? measureAutoHeight(el) : height}px`;
}

function toKeyframe(el: HTMLElement, props: MotionProps): Keyframe {
  const frame: Keyframe = {};
  if (props.opacity !== undefined) frame['opacity'] = props.opacity;
  const transform = toTransform(props);
  if (transform !== null) frame['transform'] = transform;
  if (props.height !== undefined) frame['height'] = resolveHeight(el, props.height);
  return frame;
}

function buildKeyframes(el: HTMLElement, tween: MotionTween): Keyframe[] {
  if (tween.from?.transformOrigin) el.style.transformOrigin = tween.from.transformOrigin;
  const to = toKeyframe(el, tween.to);
  // "to" tween with a height target: start from the current rendered height, since WAAPI can't interpolate from `auto`.
  const from = tween.from
    ? toKeyframe(el, tween.from)
    : tween.to.height !== undefined
      ? { height: `${el.offsetHeight}px` }
      : null;
  return from ? [from, to] : [to];
}

function applyFinalStyles(el: HTMLElement, props: MotionProps): void {
  if (props.opacity !== undefined) el.style.opacity = String(props.opacity);
  const transform = toTransform(props);
  if (transform !== null) el.style.transform = transform;
  if (props.height !== undefined) el.style.height = props.height === 'auto' ? 'auto' : `${props.height}px`;
}
```

```ts
import { MatIconModule } from '@angular/material/icon';
import { Subscription } from 'rxjs';
import {

  buildConditionalFieldExitTimeline,
  setConditionalFieldVisibility,

import { DynamicFieldComponent, WwellDynamicFieldMode } from '../dynamic-field/dynamic-field.component';
import { playIfMotionAllowed } from '../motion.util';
import { addRepeatableRow, defaultValueFor } from '../wwell-dynamic-form.builder';
import { WwellFieldConfig, WwellRepeatableRowConfig } from '../wwell-dynamic-form.types';
import { buildRowEnterTimeline, buildRowExitTimeline } from './repeatable-row.animation';

@Component({
  selector: 'app-repeatable-group',
  standalone: true,
  imports: [DynamicFieldComponent, MatButtonModule, MatIconModule],
  templateUrl: './repeatable-group.component.html',
  styleUrl: './repeatable-group.component.scss',
  changeDetection: ChangeDetectionStrategy.OnPush,

export class RepeatableGroupComponent implements AfterViewInit, OnDestroy {
  private readonly fb = inject(FormBuilder);
  private readonly cdr = inject(ChangeDetectorRef);

  readonly config = input.required<WwellRepeatableRowConfig>();
  readonly formArray = input.required<FormArray>();
  readonly mode = input<WwellDynamicFieldMode>('edit');
  readonly selectPanelClass = input('');

  @ViewChildren('rowHost') private readonly rowHosts?: QueryList<ElementRef<HTMLElement>>;

  /**
   * Row-level `visibleWhen` support — `app-dynamic-section` implements this for flat
   * `section().fields`, but never for `section().repeatable` rows, so it's ported here
   * rather than duplicated per-feature. Each row gets its own independent visibility
   * state, resolved against that row's own `FormGroup` (not the whole array), because a
   * repeatable row's condition (e.g. "this row's Type is Liner") only ever depends on a
   * sibling field within the *same* row.
   */
  @ViewChildren('rowFieldHost') private readonly rowFieldHosts?: QueryList<ElementRef<HTMLElement>>;

  /** Only fields with `visibleWhen` grow a `#rowFieldHost` wrapper — the template renders
   * exactly this many per row, in this order, so `row index * length + field index` maps
   * a (row, field) pair onto the flat, document-order `rowFieldHosts` query list. */
  protected readonly conditionalRowFields = computed(() => this.config().rowFields.filter(field => field.visibleWhen));

  /** Bumped only on user-triggered `addRow()` so the effect below can play an entrance animation for it. */
  private readonly addedRowTick = signal(0);

  /** Tracks last-applied visibility per (row, field key) so we only animate on an actual flip. Keyed by row
   * object identity (not index) so it survives rows being added/removed/reordered without misattribution. */
  private readonly lastVisibility = new Map<FormGroup, Map<string, boolean>>();
  private valueChangesSubscription?: Subscription;
  private rowFieldHostsSubscription?: Subscription;

  constructor() {
    effect(() => {
      const tick = this.addedRowTick();
      if (tick === 0) return;
      const last = this.rowHosts?.last;
      if (last) playIfMotionAllowed(buildRowEnterTimeline(last.nativeElement));
    });
  }

  ngAfterViewInit(): void {
    this.syncDerivedFields();
    this.syncConditionalRowFields();
    this.valueChangesSubscription = this.formArray().valueChanges.subscribe(() => this.onFormArrayChanged());
    // Re-sync whenever the query list's elements change (row added/removed) — a freshly
    // added row's #rowFieldHost isn't queryable until after this fires.
    this.rowFieldHostsSubscription = this.rowFieldHosts?.changes.subscribe(() => this.syncConditionalRowFields());
    // See `onFormArrayChanged`'s doc comment — the very first render can be just as stale as
    // any later one, for the same reason, so it needs the same forced re-check.
    this.cdr.detectChanges();
  }

  ngOnDestroy(): void {
    this.valueChangesSubscription?.unsubscribe();
    this.rowFieldHostsSubscription?.unsubscribe();
  }

  protected rows(): FormGroup[] {
    return this.formArray().controls as FormGroup[];
  }

  /**
   * Pushing/removing a row (or editing any field in one) mutates the `formArray` input's own
   * `FormArray` object in place — the input's own signal reference never changes, so Angular's
   * standard OnPush triggers (changed input, event sourced from this component's own template,
   * a *directly template-read* signal changing) never fire for a `RepeatableGroupComponent`
   * instance that didn't cause the mutation itself. That's invisible for the normal "edit"
   * instance (its own "+ Add"/remove buttons are template-sourced click events, so zone.js
   * re-checks it for free) but left every *other* instance of the same section — concretely,
   * the read-only copy the Attachments & Review step renders for the same FormArray — stuck
   * forever showing whatever row count existed when it first mounted. Confirmed live: filling
   * in an Offset Water Well and opening the review step showed a completely empty section,
   * with the underlying FormArray provably holding the right row the whole time. Forcing a
   * synchronous `detectChanges()` here — rather than the usual `markForCheck()` — is deliberate:
   * this component sits inside a `mat-expansion-panel`, and Material detaches a collapsed
   * panel's content from the ambient change-detection walk, so merely flagging "needs a check"
   * isn't enough when nothing ever walks back down to it on its own.
   */
  private onFormArrayChanged(): void {
    this.syncDerivedFields();
    this.syncConditionalRowFields();
    this.cdr.detectChanges();
  }

  protected fieldControl(row: FormGroup, key: string): FormControl {
    return row.get(key) as FormControl;
  }

  protected addRow(): void {
    addRepeatableRow(this.formArray(), this.config().rowFields, this.fb);
    this.addedRowTick.update(tick => tick + 1);
  }

  protected removeRow(index: number): void {
    const row = this.rows()[index];
    const host = this.rowHosts?.get(index);
    if (!host) {
      this.formArray().removeAt(index);
      this.lastVisibility.delete(row);
      return;
    }

    playIfMotionAllowed(buildRowExitTimeline(host.nativeElement, () => {
      this.formArray().removeAt(index);
      this.lastVisibility.delete(row);
    }));
  }

  /** Writes each `derived` field's computed value into its control. Only a changed value is written, so the
   *  valueChanges this triggers settles on the next pass instead of looping. */
  private syncDerivedFields(): void {
    const derivedFields = this.config().rowFields.filter(field => field.derived);
    if (!derivedFields.length) return;

    for (const row of this.rows()) {
      for (const field of derivedFields) {
        const control = this.fieldControl(row, field.key);
        const value = field.derived?.compute(row.getRawValue()) ?? null;
        if (control.value !== value) control.setValue(value);
      }
    }
  }

  private syncConditionalRowFields(): void {
    const conditionalFields = this.conditionalRowFields();
    if (!conditionalFields.length) return;

    this.rows().forEach((row, rowIndex) => {
      conditionalFields.forEach((field, fieldIndex) => {
        const host = this.rowFieldHosts?.get(rowIndex * conditionalFields.length + fieldIndex)?.nativeElement;
        if (host) this.applyFieldVisibility(row, field, host);
      });
    });
  }

  private applyFieldVisibility(row: FormGroup, field: WwellFieldConfig, host: HTMLElement): void {
    const rowVisibility = this.lastVisibility.get(row) ?? new Map<string, boolean>();
    this.lastVisibility.set(row, rowVisibility);

    const isVisible = this.isFieldVisible(row, field);
    const seenBefore = rowVisibility.has(field.key);
    const wasVisible = rowVisibility.get(field.key);
    rowVisibility.set(field.key, isVisible);

    if (!seenBefore) {
      setConditionalFieldVisibility(host, isVisible);
      this.syncControlEnabled(row, field, isVisible);
      return;
    }
    if (isVisible === wasVisible) return;

    if (isVisible) {
      this.syncControlEnabled(row, field, true);
      playIfMotionAllowed(buildConditionalFieldEnterTimeline(host));
    } else {
      playIfMotionAllowed(buildConditionalFieldExitTimeline(host, () => this.syncControlEnabled(row, field, false)));
    }
  }

  private isFieldVisible(row: FormGroup, field: WwellFieldConfig): boolean {
    const condition = field.visibleWhen;
    if (!condition) return true;

    const sibling = row.get(condition.field);
    const value = sibling?.value;
    if (condition.equals !== undefined) return value === condition.equals;
    if (condition.oneOf) return condition.oneOf.includes(value);
    if (condition.notOneOf) return !condition.notOneOf.includes(value);
    if (condition.includesLine !== undefined) {
      return typeof value === 'string' && value.split('\n').some(line => line.trim() === condition.includesLine);
    }
    return true;
  }

  /** Hidden fields are disabled (excluded from validity/`.value`) and reset so a stale value from another row/mode never submits. */
  private syncControlEnabled(row: FormGroup, field: WwellFieldConfig, isVisible: boolean): void {
    const control = this.fieldControl(row, field.key);
    if (isVisible) {
      if (control.disabled) control.enable({ emitEvent: false });
      return;
    }
    if (control.enabled) {
      control.reset(defaultValueFor(field), { emitEvent: false });
      control.disable({ emitEvent: false });
    }
  }
}
```

```ts
const COLLAPSE_SLIDE_PX = 6;

 * can hold several mat-form-fields or a multi-row table, and tweening `height` (even to/from a
 * measured 'auto') forces a synchronous layout recalculation on every single frame, which is what
 * made this read as stuttering/heavy instead of smooth. `transform`/`opacity` are compositor-only —
 * the browser can run them entirely on the GPU with zero layout work (see the project's own
 * performance rules: prefer transform/opacity, avoid animating height/width/margin/padding).
 *
 * The tradeoff: surrounding content still shifts in the same frame the element mounts/unmounts
 * (no smooth height "push") rather than over the animation's duration — invisible in practice
 * since it lines up with the very start of the fade, and far lighter than a stuttering alternative.
 */
export function buildCollapseEnterTimeline(host: HTMLElement, onComplete: () => void): MotionTimeline {
  return createMotionTimeline(
    [
      {
        targets: [host],
        from: { opacity: 0, y: -COLLAPSE_SLIDE_PX },
        to: { opacity: 1, y: 0 },
        duration: COLLAPSE_ENTER_DURATION_S,
        ease: 'power1.out',
      },
    ],
    onComplete,
  );
}

/** `onComplete` resolves `animate.leave`'s own `animationComplete()` — Angular only detaches the
 *  node once that runs, so the exit tween is guaranteed to finish on screen first regardless of
 *  what triggered the removal (a click, a signal flip elsewhere, whatever). */
export function buildCollapseExitTimeline(host: HTMLElement, onComplete: () => void): MotionTimeline {
  return createMotionTimeline(
    [{ targets: [host], to: { opacity: 0, y: -COLLAPSE_SLIDE_PX }, duration: COLLAPSE_EXIT_DURATION_S, ease: 'power1.in' }],
    onComplete,
  );
}
```

```ts
import { Directive, ElementRef, inject } from '@angular/core';
import type { AnimationCallbackEvent } from '@angular/core';
import { playIfMotionAllowed } from '../dynamic-form/motion.util';
import { buildCollapseEnterTimeline, buildCollapseExitTimeline } from './wwell-collapse-animate.animation';

/**
 * Drop-in smooth open/close for any block Angular's control flow (`@if`/`@for`) adds or removes
 * from the DOM — a section swapping from its read display to an edit form, a timeline step
 * expanding, a filter row appearing. Without this, `@if` just snaps the DOM in and out instantly,
 * which is the "tabs" in this editor (section edit toggles, timeline step accordions) reading as
 * abrupt rather than smooth.
 *
 * Hooks Angular's native `animate.enter`/`animate.leave` (fired right before insertion / right
 * before removal — see @angular/core's `AnimationCallbackEvent`) to run a lightweight, compositor-
 * only (opacity + transform, never height — see wwell-collapse-animate.animation.ts) animation
 * instead of nothing. `animate.leave`'s `animationComplete()` is what actually lets Angular
 * detach the node, so the exit tween is guaranteed to finish on screen before the element
 * disappears, no matter what triggered the removal. Respects prefers-reduced-motion via the
 * shared `playIfMotionAllowed` (jumps straight to the end state, still resolving
 * `animationComplete()` either way, so a reduced-motion user is never left with an undetached
 * node stuck mid-removal).
 *
 * Usage: add the `appWwellCollapseAnimate` attribute directly to whichever element is the root of
 * the `@if`/`@else` branch that should transition — e.g. both the read-display root and the
 * edit-form root of a section, so switching between them cross-fades instead of swapping instantly.
 */
@Directive({
  selector: '[appWwellCollapseAnimate]',
  standalone: true,
  host: {
    '(animate.enter)': 'onEnter($event)',
    '(animate.leave)': 'onLeave($event)',
  },
})
export class WwellCollapseAnimateDirective {
  private readonly hostEl = inject<ElementRef<HTMLElement>>(ElementRef).nativeElement;

  protected onEnter(event: AnimationCallbackEvent): void {
    playIfMotionAllowed(buildCollapseEnterTimeline(this.hostEl, () => event.animationComplete()));
  }

  protected onLeave(event: AnimationCallbackEvent): void {
    playIfMotionAllowed(buildCollapseExitTimeline(this.hostEl, () => event.animationComplete()));
  }
}
```

```ts
  private previewRows<TRow>(
    live: readonly TRow[] | null,
    saved: () => readonly TRow[],
    draftRows: readonly TRow[] | null,
    keyOf: (row: TRow) => string,
  ): readonly TRow[] | null {
    if (!draftRows?.length) return live;
    const rows = live ?? saved();
    const keys = new Set(rows.map(keyOf));
    // "Add" refuses a row whose natural key is already taken (it would overwrite that row) — including by
    // another staged draft row — so a colliding draft is not drawn either.
    const additions = draftRows.filter(draft => {
      const key = keyOf(draft);
      if (keys.has(key)) return false;
      keys.add(key);
      return true;
    });
    return additions.length ? [...rows, ...additions] : live;
  }

  /** Commits every row staged in a "+" add-form's own FormArray into the section's live FormArray — the
   *  same array inline edits and deletes write to, so Save diffs it like any other change. All-or-nothing:
   *  Save matches rows on their natural key, so if ANY staged row's key is already taken (by an existing
   *  row, or by another row in the same batch) the whole commit is refused rather than silently overwriting
   *  that row, or committing some rows and leaving the colliding one stranded in the add-form. */
  private commitNewRows<TRow>(
    add: SectionEditState<FormArray>,
    section: WwellFormSectionConfig,
    commit: {
      readonly existingKeys: readonly string[];
      readonly keyOf: (row: TRow) => string;
      /** Last tweak to each row as it is committed (e.g. assigning its save key). */
      readonly prepare?: (row: TRow) => TRow;
      readonly liveArray: () => FormArray | null;
      /** Runs once every staged row is in the live array (e.g. to re-sort it). */
      readonly afterCommit?: (array: FormArray) => void;
      readonly duplicate: (key: string) => string;
      readonly added: (count: number) => string;
    },
  ): void {
    const form = add.control();
    if (!form || form.invalid || form.length === 0) return;

    const rows = form.getRawValue() as TRow[];
    const keys = new Set(commit.existingKeys);
    for (const row of rows) {
      const key = commit.keyOf(row);
      if (keys.has(key)) {
        this.notify.warn(commit.duplicate(key));
        return;
      }
      keys.add(key);
    }

    const array = commit.liveArray();
    if (!array) return;
    rows.forEach(staged => {
      const row = commit.prepare ? commit.prepare(staged) : staged;
      addRepeatableRow(array, section.repeatable?.rowFields ?? [], this.fb);
      array.at(array.length - 1).patchValue(row as object);
    });
    commit.afterCommit?.(array);
    add.close();
    this.notify.info(commit.added(rows.length));
  }

  constructor() {
    this.bindOffsetWellPicker(() => this.offsetWellAdd.control());
    this.bindOffsetWellPicker(() => this.offsetWellsEdit.control());
    effect(onCleanup => {
      const ctrl = this.requestReferenceEdit.control();
      if (!ctrl) return;
      const linkedRequestIdControl = ctrl.get('linkedRequestId');
      const subscription = linkedRequestIdControl?.valueChanges.subscribe((linkedRequestId: string) => {
        console.log('[IR:link] "Link to Request" select — valueChanges fired', { linkedRequestId });
        if (linkedRequestId) this.applyLinkedRequestWell(ctrl, linkedRequestId);

      onCleanup(() => subscription?.unsubscribe());

    effect(() => {
      const id = this.ir().id;
      if (this.previousIrId !== null && id !== this.previousIrId) this.closeAllSections();
      this.previousIrId = id;
    });

    effect(() => {
      if (this.ir().irSeq != null) return;
      untracked(() => this.requestReferenceEdit.open());
    });

    effect(() => {
      const saving = this.store.creatingIr();
      const settled = this.wasSavingForNavigation && !saving;
      this.wasSavingForNavigation = saving;
      if (!settled || this.saveSettledResolvers.length === 0) return;

      const succeeded = !this.store.createIrError();
      const resolvers = this.saveSettledResolvers;
      this.saveSettledResolvers = [];
      resolvers.forEach(resolve => resolve(succeeded));
    });

  private applyLinkedRequestWell(referenceGroup: FormGroup, linkedRequestId: string): void {
    console.log('[IR:link] applyLinkedRequestWell — start', { linkedRequestId, eligibleWellsLoaded: this.store.linkedRequestWells().length });
    const well = findLinkedWell(linkedRequestId, this.store.linkedRequestWells());
    if (!well) {
      console.log('[IR:link] applyLinkedRequestWell — no matching eligible well, aborting patch', { linkedRequestId });
      return;
    }

    const hasElevation = linkedWellHasElevation(well);
    const hasCoordinates = linkedWellHasCoordinates(well);
    const hasTargetDepth = linkedWellHasTargetDepth(well);
    const hasTargetAquifer = linkedWellHasTargetAquifer(well);
    const hasSupportingBusiness = linkedWellHasSupportingBusiness(well);

    const patch: Record<string, string | number> = { wellName: well.wellName };
    if (hasElevation) patch['surfaceElevationFtAsl'] = well.surfaceElevation as number;
    patch['coordinates'] = hasCoordinates ? formatLinkedWellCoordinates(well) : '';
    if (hasTargetDepth) patch['targetDepthFt'] = well.targetDepth as number;
    patch['targetAquifer'] = hasTargetAquifer ? (well.targetAquifer as string) : '';
    patch['supportingBusiness'] = hasSupportingBusiness ? (well.supportings as string) : '';
    console.log('[IR:link] applyLinkedRequestWell — patching form group', {
      patch,
      hasElevation,
      hasCoordinates,
      hasTargetDepth,
      hasTargetAquifer,
      hasSupportingBusiness,
    });
    referenceGroup.patchValue(patch);

    referenceGroup.get('wellName')?.disable();
    this.setLinkedFieldLock(referenceGroup, 'surfaceElevationFtAsl', hasElevation);
    console.log('[IR:link] applyLinkedRequestWell — done', { rawValue: referenceGroup.getRawValue() });

  private resetFlatSection(
    edit: SectionEditState<FormGroup, unknown>,
    section: WwellFormSectionConfig,

    key: 'program' | 'geologist',

    edit.open();

    if (this.hasSavedIrSeq()) {

  private saveMergedRaw(raw: WwellIrFormRawValue): void {
    const current = this.ir();
    const status: Extract<IntegratedRequirementStatus, 'draft' | 'in-progress'> =
      current.status === 'draft' || current.status === 'in-progress' ? current.status : 'in-progress';

  /** Returns whether a save was actually kicked off — `false` on every early-return branch below
   *  (nothing to save, or a blocked validation). `saveUnsavedChangesAndWait` needs this to settle
   *  its own promise immediately in those cases instead of waiting forever for a `creatingIr`
   *  transition that will now never happen. */
  protected saveAll(): boolean {
    // A record whose stored remarks still carry legacy HTML is saved even with nothing edited — the form already
    // holds the cleaned text, and saving is what writes it back.
    if (this.ir().irSeq !== null && !this.hasUnsavedChanges() && !this.ir().hasLegacyMarkup) {
      this.notify.info('No changes to save.');
      return false;
    }

    let raw = this.currentRawFormValue();

    const referenceCtrl = this.requestReferenceEdit.control();
    if (referenceCtrl) raw = { ...raw, reference: { main: referenceCtrl.getRawValue() } };

    // `letterNumber` carries a `Validators.required` in the schema (wwell-ir-form.schema.ts), but
    // nothing elsewhere in this method ever checks form validity before saving — it only gathers
    // raw values. Without this explicit gate, a blank Letter Number saved silently; the asterisk
    // and mat-error only ever showed once the field itself had been touched.
    if (!raw.reference.main.letterNumber?.trim()) {
      this.requestReferenceEdit.open();
      this.requestReferenceEdit.control()?.get('letterNumber')?.markAsTouched();
      this.expandedTimelineSteps.update(current => new Set(current).add('wellbore'));
      this.focusField('ire-sec-wellbore', 'Letter Number');
      this.notify.error('Letter Number is required before saving.');
      return false;
    }

    const programCtrl = this.programEdit.control();
    if (programCtrl) {
      const flat = programCtrl.getRawValue() as WwellIrRawProgramDetails;
      raw = {
        ...raw,
        programDetails: {
          main: {
            completionType: flat.completionType,
            completion: flat.completion,
            openHoleSizeIn: flat.openHoleSizeIn,
            linerSizeIn: flat.linerSizeIn,
            linerDepthFt: flat.linerDepthFt,
            acidization: flat.acidization,
            transmission: flat.transmission,
          },
          developmentTesting: {
            logging: flat.logging,
            mudLogging: flat.mudLogging,
            sampling: flat.sampling,
            flowTest: flat.flowTest,
            remarks: flat.remarks,
          },
        },
      };
    }

    const geologistCtrl = this.geologistContactEdit.control();
    if (geologistCtrl) raw = { ...raw, contacts: { ...raw.contacts, geologist: geologistCtrl.getRawValue() } };

    const hydrogeologyCtrl = this.hydrogeologyEdit.control();
    if (hydrogeologyCtrl) raw = { ...raw, hydrogeology: { main: hydrogeologyCtrl.getRawValue() } };

    const formationTopsCtrl = this.formationTopsEdit.control();
    if (formationTopsCtrl) raw = { ...raw, subsurface: { ...raw.subsurface, formationTops: formationTopsCtrl.getRawValue() } };

    const topsRemarksCtrl = this.topsRemarksEdit.control();
    if (topsRemarksCtrl) raw = { ...raw, subsurface: { ...raw.subsurface, remarks: topsRemarksCtrl.getRawValue() } };

    const casingCtrl = this.casingEdit.control();
    if (casingCtrl) raw = { ...raw, subsurface: { ...raw.subsurface, casing: casingCtrl.getRawValue() } };

    const offsetWellsCtrl = this.offsetWellsEdit.control();
    if (offsetWellsCtrl) raw = { ...raw, contacts: { ...raw.contacts, offsetWells: offsetWellsCtrl.getRawValue() } };

    const offsetRemarksCtrl = this.offsetRemarksEdit.control();
    if (offsetRemarksCtrl) raw = { ...raw, contacts: { ...raw.contacts, remarks: offsetRemarksCtrl.getRawValue() } };

    this.saveMergedRaw(raw);
    this.closeAllSections();
    return true;
  }

      input.scrollIntoView({ block: 'center', behavior: 'smooth' });
```

```ts
  constructor() {
    effect(() => {

    // Reports the real outcome of a save once it settles — the editor deliberately shows nothing
    // at click time (see saveMergedRaw), since the request is still in flight there (its own Save

        // own follow-up GET resolves a sapWfId — no separate call needed here.

        // its attachment slots can be fetched (and the first moment uploads become possible).

        // now — send it against the id the save just assigned.

    // workflow id → SAP /trigger → stamp sap_wf_id/sap_wf_status on the header). Same settled-edge

    // because the request is still in flight — the editor's own button shows a spinner until then.
    //

    // visible trace was the Submit for Review button disappearing (the status leaves
    // WWELL_IR_SUBMITTABLE_STATUSES, so canSubmitForReview drops it), which reads as the click
    // having done nothing. A failure is already toasted by the sapTriggerError effect above, so

      const settled = this.wasRecallingIr && !recalling;
      this.wasRecallingIr = recalling;

      const recalledCode = this.store.irs().find(ir => ir.irSeq === this.selectedIrSeq())?.code;
      this.notify.info(`Workflow for ${recalledCode ?? 'this integrated requirement'} recalled successfully.`);

      if (message) this.notify.error(message);

      const message = this.store.irApprovalTimelineError();
      if (message) this.notify.error(message);

      const message = this.store.deleteIrError();
      if (message) this.notify.error(message);

    // while in flight (see wwell-ir-editor.component.html's isDeleting()), so nothing needs
    // announcing at click time. A failure is already toasted by the effect above.
    effect(() => {
      const deleting = this.store.deletingIr();
      const settled = this.wasDeletingIr && !deleting;
      this.wasDeletingIr = deleting;
      if (!settled || this.store.deleteIrError()) return;

      this.notify.info('Integrated requirement deleted.');

      // The deleted record is already gone from store.irs() by this point (WwellIrStore.deleteIr
      // removes it on success) — if it was the one open, fall back to null so the "default to
      // first visible record" effect below picks whatever the list now shows first (or leaves the
      // panel empty if nothing remains), the same path a fresh session start already uses.
      if (this.selectedIrSeq() !== null && !this.store.irs().some(ir => ir.irSeq === this.selectedIrSeq())) {
        this.selectedIrSeq.set(null);
        this.previousSelectedIrSeq = null;
        this.deselected.set(false);
      }
    });

    // Default to the first *visible* record once the list loads — mirrors the reference app,
    // which always has something selected in the detail pane rather than starting on an empty
    // state. Reads `rows()` (lifecycle/status/search-scoped), not the raw store list, so the
    // default selection is never a row the Active/Inactive toggle is currently hiding. Skipped
    // while a pendingDraftIr is open — selectedIrSeq is intentionally left null throughout
    // drafting (the draft isn't in `rows()`), and this effect would otherwise immediately fetch
    // and select an unrelated real record's detail out from under the still-open draft.
    effect(() => {
      if (this.pendingDraftIr() || this.deselected()) return;
      const firstIrSeq = this.rows()[0]?.irSeq;
      if (firstIrSeq != null && this.selectedIrSeq() === null) {
        this.selectedIrSeq.set(firstIrSeq);
      }
    });

    // default-selection effect opens (and fetches detail for) its first row; the list GET that lands
    // moments later then replaces every row with a bare summary, wiping that merged detail while the
    // selection stays put — an empty-looking record. Re-fetch the open record's detail once the list settles.
    effect(() => {

      this.wasLoadingIrs = loading;
      if (!settled || this.store.irsError()) return;

        this.store.loadIrDetail(irSeq);

    // (casing/tops/hydrogeology) by ir_seq, its primary key.

      // exad_gwd_ir_* columns, so they can't ride along in the detail blocks above.
      this.store.loadIrAttachments(irSeq);

      // them surface on whichever record was just selected. (This effect deliberately no-ops on the
      // save transition, see the comment above, so it can't discard files mid-upload.)
      this.store.clearPendingAttachments();

      // "Link to Request" dropdown and its live auto-populate-on-pick (see
      // wwell-ir-editor.component.ts's applyLinkedRequestWell / wwell-ir-linked-well.util.ts),
      // used when the user picks a NEW link on the opened record. Eligibility can change
      // server-side between selections in a long-lived session (another user initiates an IR for
      // a well, a well newly becomes eligible...) — no cached/one-time snapshot, always the

    const draft = buildIntegratedRequirementFromPayload(payload, { id: 'draft', code: 'New IR', irSeq: null });
    // Start the new draft with empty slots — a previous draft's held files must not carry over.
    this.store.clearPendingAttachments();

    this.pendingDraftIr.set(draft);
    this.selectedIrSeq.set(null);
    this.previousSelectedIrSeq = null;
    this.deselected.set(false);
    // A brand-new draft has nothing committed to view read-only yet — always open it for editing,
    // regardless of whatever mode the previously-open record was left in.
    this.detailMode.set('edit');

    const host = this.detailHost?.nativeElement;
    if (host) playIfMotionAllowed(buildDetailRevealTimeline(host));

   *  record with no `sapWfId` (never submitted, so there is nothing in SAP to ask about) or no

  protected refreshApprovalStatus(ir: IntegratedRequirement): void {

    this.store.refreshIrApprovalStatus(ir.irSeq, ir.sapWfId);

  // ── Unsaved-changes guard — backs three separate "leaving the currently-open record" paths with
  // one prompt: switching the selected row / opening "New IR" / clicking the view (eye) icon (all
  // via guardNavigation below), and a real Angular route change away from this page entirely (via
  // canDeactivate(), wired up as the 'integrated-requirements' route's CanDeactivate guard — see
  // app.routes.ts and unsaved-changes.guard.ts). Browser tab close/refresh is covered separately by
  // the beforeunload listener below, which can't run this same async dialog flow.

  /** Router entry point (ComponentCanDeactivate) — resolves `true` once it's safe to actually leave
   *  the route: immediately if nothing is unsaved, after a confirmed Discard, or after a confirmed
   *  Save has settled. `false` means stay (Cancel, or a save that failed). */
  canDeactivate(): Observable<boolean> {
    const editor = this.editorRef;
    if (!editor?.hasPendingEdits()) return of(true);

    // Draw every pending edit in dark green behind the prompt, so it is clear what Save / Discard acts on.
    editor.showUnsavedHighlight();
    return this.confirmDialog
      .openWithResult({
        title: 'Unsaved changes',
        subtitle: 'This Integrated Requirement has section edits that haven’t been saved yet. Save them before leaving?',
        confirmLabel: 'Save & Leave',
        discardLabel: 'Discard Changes',
        cancelLabel: 'Stay Here',
        severity: 'warning',
      })
      .pipe(
        switchMap(result => {
          if (result === 'cancel') {
            editor.clearUnsavedHighlight();
            return of(false);
          }
          if (result === 'discard') {
            editor.discardUnsavedChanges();
            return of(true);
          }
          return from(editor.saveUnsavedChangesAndWait()).pipe(tap(saved => !saved && editor.clearUnsavedHighlight()));
        }),
      );
  }

  /** In-page equivalent of canDeactivate() — same prompt/save/discard flow, but for switching what
   *  this page itself shows (selecting another row, opening a blank "New IR", or the view-icon's
   *  mode switch) rather than leaving the route. `action` only runs once it's actually safe to
   *  proceed. */
  private guardNavigation(action: () => void): void {
    this.canDeactivate()
      .pipe(take(1))
      .subscribe(canLeave => {
        if (canLeave) action();
      });
  }

  /** Native browser navigation (tab close, refresh, typed URL) bypasses the Router entirely, so
   *  canDeactivate() never runs for it — this is the one place that has to use the browser's own
   *  confirmation prompt instead (no custom dialog is possible here; per spec, `returnValue` must be
   *  set and the browser supplies its own generic copy). */
  @HostListener('window:beforeunload', ['$event'])
  protected onBeforeUnload(event: BeforeUnloadEvent): void {
    if (!this.editorRef?.hasPendingEdits()) return;
    event.preventDefault();
    event.returnValue = '';
  }
}
```
