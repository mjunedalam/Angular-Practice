```typescript
import { easeCubicInOut } from 'd3-ease';
import 'd3-transition';

import { ANIM, DiagramLayout, WellboreDiagramData } from '@models/well-design/wellbore-diagram.model';
import {
  buildArrowHeadLeft,
  buildArrowHeadRight,
  buildGravelPackUPath,
  buildOpenHolePath,
  buildScreenHanger,
  computeCasingHalfWidth,
} from '@shared/utils/wellbore-math.util';
import { animateLabel } from './animation.helpers';
import {
  resolveCompletionBottomDepth,
  resolveCompletionBottomPx,
  resolveCompletionTopDepth,
  resolveCompletionTopPx,
  resolveOhHalfWidth,
  resolvePrePerforatedLinerDepth,
  shouldRenderGravelPack,
  shouldRenderLinerScreen,
  shouldRenderOpenHole,
} from './wellbore-depth.helpers';
import { DefsSel, DepthScale, GSel, ResolvedAnimation } from './wellbore-renderer.types';

export interface RenderCompletionElementsOptions {
  readonly rootG: GSel;
  readonly defsEl: DefsSel;
  readonly layout: DiagramLayout;
  readonly data: WellboreDiagramData;
  readonly centerX: number;
  readonly scale: DepthScale;
  readonly ohStart: number;
  readonly ohLabelStart: number;
  readonly hangerStart: number;
  readonly animation: ResolvedAnimation;
}

export function renderCompletionElements(options: RenderCompletionElementsOptions): void {
  renderOpenHoleAndScreen(options);
  renderGravelPackDesign(options);
  renderPrePerforatedLiner(options);
}

function renderOpenHoleAndScreen({
  rootG,
  defsEl,
  layout,
  data,
  centerX,
  scale,
  ohStart,
  ohLabelStart,
  animation,
}: RenderCompletionElementsOptions): void {
  if (!data.casings.length) return;

  const { wellDesign } = data;
  const shouldDrawOH = shouldRenderOpenHole(wellDesign);
  // When DRLG_CSG liners are present, liner.renderer owns all liner rendering (shapes,
  // hangers, and the 7" screen via isScreen). The legacy completion-path screen is only
  // drawn when there are no as-drilled liners and the well-design flags opt in.
  const hasDrlgLiners = data.drlgCasings.some(d => d.wLnrOdSz && (d.wLnrBotDepth ?? 0) > 0);
  const shouldDrawLS = !hasDrlgLiners && shouldRenderLinerScreen(wellDesign);
  if (!shouldDrawOH && !shouldDrawLS) return;

  const { baseHalfWidth, halfWidthIncrement, linerScreenInset, gpScreenGap } = layout;
  const innerHW = computeCasingHalfWidth(0, baseHalfWidth, halfWidthIncrement);
  const ohHW = resolveOhHalfWidth(data, layout);
  const screenHW = Math.min(innerHW, ohHW) - linerScreenInset;
  const liner = data.casings.find(c => c.csgType === 'Liner');
  const gpCasing = data.casings.find(c => c.csgType === 'Gravel Pack');
  const shoePx = resolveCompletionTopPx(data.casings, data.drlgCasings, scale, wellDesign);
  const completionBottomDepth = resolveCompletionBottomDepth(data);
  const completionBottomPx = scale(completionBottomDepth);
  const hasGP = shouldRenderGravelPack(wellDesign) || !!gpCasing;
  const gpBottomPx = gpCasing ? Math.min(scale(gpCasing.csgDepth), completionBottomPx) : completionBottomPx;
  const screenBottomPx = wellDesign
    ? (shouldDrawLS && hasGP ? gpBottomPx - gpScreenGap : completionBottomPx)
    : (liner ? scale(liner.csgDepth) : completionBottomPx - 15);
  const clipHW = shouldDrawOH ? ohHW : screenHW;
  const clipId = 'dyn-oh-clip';
  const clipRect = defsEl.append('clipPath').attr('class', 'dyn-clip').attr('id', clipId).append('rect')
    .attr('x', centerX - clipHW - 15).attr('y', shoePx - 5).attr('width', clipHW * 2 + 30).attr('height', 0);

  if (shouldDrawOH) {
    rootG.append('path').attr('class', 'open-hole')
      .attr('d', buildOpenHolePath(centerX, ohHW, shoePx, completionBottomPx))
      .attr('clip-path', `url(#${clipId})`);
  }

  if (shouldDrawLS) {
    rootG.append('path').attr('class', 'liner-screen')
      .attr('d', buildOpenHolePath(centerX, screenHW, shoePx, screenBottomPx))
      .attr('clip-path', `url(#${clipId})`);
  }

  clipRect.transition().delay(ohStart).duration(ANIM.OH_DURATION).ease(easeCubicInOut).attr('height', completionBottomPx - shoePx + 5);

  if (shouldDrawOH) {
    const ohLabel = wellDesign?.ohRemarks ?? '8 1/2" Open Hole';
    const ohLY = shoePx + (completionBottomPx - shoePx) * 0.2;
    const ohLG = rootG.append('g').attr('class', 'open-hole-label');
    const ohLineX1 = centerX - ohHW;
    const ohLineX2 = centerX - ohHW - 33;

    ohLG.append('line')
      .attr('class', 'oh-label-line')
      .attr('x1', ohLineX1).attr('x2', ohLineX2).attr('y1', ohLY).attr('y2', ohLY);
    ohLG.append('path')
      .attr('class', 'oh-label-arrow')
      .attr('d', buildArrowHeadLeft(ohLineX2, ohLY, 6));
    ohLG.append('text')
      .attr('class', 'open-hole-text')
      .attr('x', ohLineX2 - 6).attr('y', ohLY + 4).attr('text-anchor', 'end').text(ohLabel);

    animateLabel(ohLG, ohLabelStart, animation);
  }

  if (shouldDrawLS) {
    const shoeDepth = resolveCompletionTopDepth(data.casings, data.drlgCasings, wellDesign);
    const fallbackFt = liner ? liner.csgDepth - shoeDepth : completionBottomDepth - shoeDepth;
    const screenLabel = wellDesign?.lsRemarks ?? (liner?.csgRemarks ?? `+/- ${fallbackFt.toLocaleString()} ft of Screen`);
    const scrDelay = ohLabelStart + ANIM.OVERLAY_FADE + ANIM.SEQ_GAP;
    const scrLG = rootG.append('g').attr('class', 'screen-label');
    const scrX1 = centerX + screenHW;
    const scrX2 = centerX + screenHW + 33;

    scrLG.append('line')
      .attr('class', 'screen-label-line')
      .attr('x1', scrX1).attr('x2', scrX2).attr('y1', screenBottomPx).attr('y2', screenBottomPx);
    scrLG.append('path')
      .attr('d', buildArrowHeadRight(scrX2, screenBottomPx, 6));
    scrLG.append('text')
      .attr('class', 'screen-label-text')
      .attr('x', scrX2 + 6).attr('y', screenBottomPx + 4).text(screenLabel);

    animateLabel(scrLG, scrDelay, animation);
  }
}

function renderGravelPackDesign({
  rootG,
  defsEl,
  layout,
  data,
  centerX,
  scale,
  ohStart,
  ohLabelStart,
  animation,
}: RenderCompletionElementsOptions): void {
  const wellDesign = data.wellDesign;
  if (!wellDesign || !shouldRenderGravelPack(wellDesign) || !data.casings.length) return;

  const { baseHalfWidth, halfWidthIncrement, linerScreenInset, gpScreenGap, gpFillRatio } = layout;
  const innerHW = computeCasingHalfWidth(0, baseHalfWidth, halfWidthIncrement);
  const ohHW = resolveOhHalfWidth(data, layout);
  const lsHW = Math.min(innerHW, ohHW) - linerScreenInset;
  const gpMaxOuter = ohHW - gpScreenGap;
  const annulusWidth = Math.max(0, (gpMaxOuter - lsHW) * gpFillRatio);
  const gpOuterEdge = lsHW + annulusWidth;
  const annulusCenter = lsHW + annulusWidth / 2;
  const startDepthPx = resolveCompletionTopPx(data.casings, data.drlgCasings, scale, wellDesign);
  const completionBottomPx = resolveCompletionBottomPx(data, scale);

  const clipId = 'dyn-gp-design-clip';
  const clipRect = defsEl.append('clipPath').attr('class', 'dyn-clip').attr('id', clipId).append('rect')
    .attr('x', centerX - gpOuterEdge).attr('y', startDepthPx)
    .attr('width', gpOuterEdge * 2).attr('height', 0);

  rootG.append('path')
    .attr('class', 'gravelHole')
    .attr('d', buildGravelPackUPath(centerX, annulusCenter, startDepthPx, completionBottomPx))
    .attr('stroke', 'url(#gravelpattern)')
    .attr('stroke-width', String(annulusWidth))
    .style('fill', 'none')
    .attr('clip-path', `url(#${clipId})`);

  clipRect.transition()
    .delay(ohStart)
    .duration(ANIM.GRAVEL_DURATION)
    .ease(easeCubicInOut)
    .attr('height', completionBottomPx - startDepthPx);

  const labelY = startDepthPx + (completionBottomPx - startDepthPx) * 0.5;
  const rEdge = centerX + gpOuterEdge;
  const lEnd = rEdge + 26;
  const lg = rootG.append('g').attr('class', 'gp-design-label');

  lg.append('line')
    .attr('class', 'screen-label-line')
    .attr('x1', rEdge).attr('x2', lEnd).attr('y1', labelY).attr('y2', labelY);
  lg.append('path').attr('d', buildArrowHeadRight(lEnd, labelY, 6));
  lg.append('text')
    .attr('class', 'screen-label-text')
    .attr('x', lEnd + 6).attr('y', labelY + 4).text(wellDesign.gpRemarks);

  animateLabel(lg, ohLabelStart, animation);
}

// Matches the `+4` clearance liner.renderer uses between a liner's hanger bracket and its
// own top depth, so the perforated overlay starts flush with — not below — the hanger.
const PRE_PERF_HANGER_CLEARANCE_PX = 4;

function renderPrePerforatedLiner({
  rootG,
  defsEl,
  layout,
  data,
  centerX,
  scale,
  ohStart,
}: RenderCompletionElementsOptions): void {
  const wellDesign = data.wellDesign;
  const prePerfDepth = resolvePrePerforatedLinerDepth(wellDesign, data.drlgCasings, data.casings);
  if (!wellDesign || !prePerfDepth) return;

  const { openHoleHwMargin } = layout;
  const ohHW = resolveOhHalfWidth(data, layout);
  const perfHwInset = Math.max(12, Math.min(20, openHoleHwMargin * 0.65));
  const prePerfHW = Math.max(10, ohHW - perfHwInset);
  const topPx = scale(prePerfDepth.topDepth);
  // Never draw past the liner's own bottom depth, and still respect the current display cap.
  const botDepth = Math.min(prePerfDepth.botDepth, resolveCompletionBottomDepth(data));
  const completionBottomPx = scale(botDepth);
  // Start just after the liner hanger and run all the way to the liner's bottom — do not
  // shorten either end.
  const perfTopPx = Math.min(topPx + PRE_PERF_HANGER_CLEARANCE_PX, completionBottomPx);
  const perfBottomPx = completionBottomPx;

  const clipId = 'dyn-perf-clip';
  const clipRect = defsEl.append('clipPath').attr('class', 'dyn-clip').attr('id', clipId).append('rect')
    .attr('x', centerX - prePerfHW - 6).attr('y', perfTopPx - 5)
    .attr('width', (prePerfHW + 6) * 2).attr('height', 0);

  const perfGroup = rootG.append('g')
    .attr('class', 'perf-wall-group')
    .attr('clip-path', `url(#${clipId})`);

  // Same outline as open hole (left wall, bottom shoe, right wall in one continuous path) —
  // just a bigger dash gap and a thicker, darker stroke so it reads as its own element.
  perfGroup.append('path')
    .attr('class', 'pre-perf-liner')
    .attr('d', buildOpenHolePath(centerX, prePerfHW, perfTopPx, perfBottomPx));
  // odSz comes straight from DRLG_CSG and may already carry a trailing inch mark
  // (e.g. `7"`) — strip it so the tooltip never doubles up as `7"" Pre Perf Liner`.
  const prePerfOdSz = prePerfDepth.odSz.replace(/["']/g, '').trim();
  perfGroup.append('title').text(`${prePerfOdSz}" Pre Perf Liner`);

  clipRect.transition()
    .delay(ohStart)
    .duration(ANIM.OH_DURATION)
    .ease(easeCubicInOut)
    .attr('height', perfBottomPx - perfTopPx + 10);
}

export function renderScreenHanger({
  rootG,
  layout,
  data,
  centerX,
  scale,
  hangerStart,
}: RenderCompletionElementsOptions): void {
  if (!data.casings.length) return;
  // Liner hangers are drawn per-liner by liner.renderer (for both DRLG and IR sources).
  // This function no longer draws a hanger for the liner itself — that produced a
  // duplicate OUTER hanger set. It only draws the well-design liner-screen hanger
  // (lsFlg='Y') for a screen that has no corresponding liner geometry.
  if (!shouldRenderLinerScreen(data.wellDesign)) return;
  const hasDrlgLiners = data.drlgCasings.some(d => d.wLnrOdSz && (d.wLnrBotDepth ?? 0) > 0);
  if (hasDrlgLiners) return;

  const { baseHalfWidth, halfWidthIncrement, linerScreenInset } = layout;
  const innerHW = computeCasingHalfWidth(0, baseHalfWidth, halfWidthIncrement);
  const ohHW = resolveOhHalfWidth(data, layout);
  const screenHW = Math.min(innerHW, ohHW) - linerScreenInset;
  const hangerG = rootG.append('g').attr('class', 'screen-hangers').style('opacity', 0);

  const screenY = scale(resolveCompletionTopDepth(data.casings, data.drlgCasings, data.wellDesign));
  const { left, right } = buildScreenHanger(centerX, screenY, screenHW);
  hangerG.append('path').attr('class', 'screen-hanger').attr('stroke', '#000000').attr('stroke-width', '1.5').attr('fill', '#000000').attr('d', left);
  hangerG.append('path').attr('class', 'screen-hanger').attr('stroke', '#000000').attr('stroke-width', '1.5').attr('fill', '#000000').attr('d', right);

  hangerG.transition().delay(hangerStart).duration(ANIM.HANGER_DURATION).ease(easeCubicInOut).style('opacity', 1);
}
```

```typescript
import { DiagramLayout, WellboreDiagramData } from '@models/well-design/wellbore-diagram.model';
import { ICasingIR } from '@shared/models/wwell/casing-ir.model';
import { IDrlgCsg } from '@shared/models/wwell/drlg-csg.model';
import { IWellDesign } from '@shared/models/wwell/well-design.model';
import { computeCasingHalfWidth, parseInchSize, remarksReferenceSize } from '@shared/utils/wellbore-math.util';
import { isLastLinerScreen, resolveAnchorLiner, resolveLinerDepths, resolveLiners } from './liner.renderer';
import { DepthScale } from './wellbore-renderer.types';

const OVERRUN_COMPLETION_LENGTH_FT = 1200;

export function resolveShoeDepth(casings: ICasingIR[], drlgCasings: IDrlgCsg[]): number {
  const irNonLiner = casings.filter(c => c.csgType !== 'Liner' && c.csgType !== 'Gravel Pack');
  const deepest = irNonLiner[0];
  if (!deepest) return drlgCasings[0]?.wCsgBotDpth ?? 0;

  const actual = drlgCasings.find(d => d.wCsgOdSz === deepest.csgSize) ?? null;
  return (actual !== null && actual.wCsgBotDpth > 0 ? actual.wCsgBotDpth : null) ?? deepest.csgDepth;
}

/**
 * Depth Open Hole / Gravel Pack / Pre-Perf start from:
 *   - 2+ liners: below the anchor (second-deepest) liner's bottom — the deepest liner is
 *     the screen and sits inside that zone on its own.
 *   - 1 liner, not flagged as the screen: below its bottom (same rule).
 *   - 1 liner flagged as the screen: there's no separate pipe liner above it — it's run
 *     into the open hole for its own full length, overlapping up into the casing shoe for
 *     its hanger seal. So completion elements must start at the CASING SHOE (not the
 *     liner's own top, which sits inside that overlap/seal zone, and not its bottom, which
 *     would leave it un-surrounded) so Open Hole/Gravel Pack visually contain the screen.
 *   - No liners: the casing shoe.
 */
export function resolveCompletionTopDepth(
  casings: ICasingIR[],
  drlgCasings: IDrlgCsg[],
  wellDesign?: IWellDesign | null,
): number {
  const liners = resolveLinerDepths(casings, drlgCasings);
  if (liners.length === 0) return resolveShoeDepth(casings, drlgCasings);

  if (liners.length === 1) {
    const only = liners[0];
    return isLastLinerScreen(only.odSz, wellDesign)
      ? resolveShoeDepth(casings, drlgCasings)
      : only.botDepth;
  }

  return liners[liners.length - 2].botDepth;
}

export function resolveCompletionTopPx(
  casings: ICasingIR[],
  drlgCasings: IDrlgCsg[],
  scale: DepthScale,
  wellDesign?: IWellDesign | null,
): number {
  return scale(resolveCompletionTopDepth(casings, drlgCasings, wellDesign));
}

export function resolveCompletionBottomDepth(data: WellboreDiagramData): number {
  const completionTopDepth = resolveCompletionTopDepth(data.casings, data.drlgCasings, data.wellDesign);
  const isCurrentDepthBeyondTarget = data.targetDepth > 0 && data.currentDepth > data.targetDepth;

  const naturalBottom = isCurrentDepthBeyondTarget
    ? Math.max(completionTopDepth, Math.min(data.totalDepth, completionTopDepth + OVERRUN_COMPLETION_LENGTH_FT))
    : data.totalDepth;

  return Math.min(naturalBottom, data.displayDepth);
}

export function resolveCompletionBottomPx(data: WellboreDiagramData, scale: DepthScale): number {
  return scale(resolveCompletionBottomDepth(data));
}

function isFlagYes(flg: string | null | undefined): boolean {
  return (flg ?? '').trim().toUpperCase() === 'Y';
}

function hasText(value: string | null | undefined): boolean {
  return !!value?.trim();
}

export function shouldRenderOpenHole(wellDesign: IWellDesign | null | undefined): boolean {
  if (!isFlagYes(wellDesign?.ohFlg)) return false;
  return parseInchSize(wellDesign?.ohRemarks) !== null;
}

export function shouldRenderGravelPack(wellDesign: IWellDesign | null | undefined): boolean {
  if (!isFlagYes(wellDesign?.gpFlg)) return false;
  return hasText(wellDesign?.gpRemarks);
}

export function shouldRenderLinerScreen(wellDesign: IWellDesign | null | undefined): boolean {
  if (!isFlagYes(wellDesign?.lsFlg)) return false;
  return hasText(wellDesign?.lsRemarks);
}

const PRE_PERF_LINER_SIZE_IN = 7;

export interface PrePerforatedLinerDepth {
  readonly odSz: string;
  readonly topDepth: number;
  readonly botDepth: number;
}

/**
 * Resolves the pre-perforated liner overlay depth range.
 *
 * All three conditions must pass:
 *   1. EXAD_GWD_WELL_DESIGN.perforatedFlg is 'Y' (case-insensitive).
 *   2. EXAD_GWD_WELL_DESIGN.perforatedRemarks references a 7" size somewhere in its free text.
 *   3. The anchor liner (see resolveAnchorLiner) has a 7" OD.
 *
 * The returned depths come from that anchor liner's own top/bottom so the overlay is
 * drawn only across its interval — and, with 2+ liners, from the same liner Open Hole
 * and Gravel Pack are drawn below, not the deepest (screen) liner.
 */
export function resolvePrePerforatedLinerDepth(
  wellDesign: IWellDesign | null | undefined,
  drlgCasings: IDrlgCsg[],
  casings: ICasingIR[] = [],
): PrePerforatedLinerDepth | null {
  if (!isFlagYes(wellDesign?.perforatedFlg)) return null;
  if (!remarksReferenceSize(wellDesign?.perforatedRemarks, PRE_PERF_LINER_SIZE_IN)) return null;

  const anchor = resolveAnchorLiner(casings, drlgCasings);
  if (!anchor || parseInchSize(anchor.odSz) !== PRE_PERF_LINER_SIZE_IN) return null;

  return { odSz: anchor.odSz, topDepth: anchor.topDepth, botDepth: anchor.botDepth };
}

export function shouldRenderPrePerforatedLiner(
  wellDesign: IWellDesign | null | undefined,
  drlgCasings: IDrlgCsg[] = [],
  casings: ICasingIR[] = [],
): boolean {
  return resolvePrePerforatedLinerDepth(wellDesign, drlgCasings, casings) !== null;
}

export function resolveOhHalfWidth(data: WellboreDiagramData, layout: DiagramLayout): number {
  const { baseHalfWidth, halfWidthIncrement, openHoleHwMargin } = layout;
  const innerHW = computeCasingHalfWidth(0, baseHalfWidth, halfWidthIncrement);

  if (!isFlagYes(data.wellDesign?.ohFlg)) return innerHW + openHoleHwMargin;

  const ohDiameter = parseInchSize(data.wellDesign?.ohRemarks);
  if (ohDiameter === null) return innerHW + openHoleHwMargin;

  // innerHW is the on-screen half-width of the deepest structural casing (tier 0) — the
  // casing the anchor liner hangs from — so its real OD is the outer bound OH must stay inside.
  const structuralCasings = data.casings.filter(
    c => c.csgType !== 'Liner' && c.csgType !== 'Gravel Pack',
  );
  const outerOD = parseInchSize(structuralCasings[0]?.csgSize ?? null);

  // When liners exist, OH must render strictly outside the anchor liner it surrounds and
  // inside the casing it hangs from (e.g. a 12 1/4" OH must sit outside a 7" liner and
  // inside a 13 3/8" casing). Per-liner on-screen widths (anchorLiner.halfWidth, innerHW)
  // are a cosmetic decrement scheme with no fixed relationship to real inches, so scaling
  // one of them by a raw diameter ratio can overshoot the casing or undershoot the liner.
  // Instead, interpolate between the two known on-screen widths using the real diameter
  // ratio as the blend factor — this guarantees OH always lands between them.
  const liners = resolveLiners(data.casings, data.drlgCasings, layout, data.wellDesign);
  const anchorLiner = liners.length > 1 ? liners[liners.length - 2] : (liners[0] ?? null);

  if (anchorLiner !== null) {
    const linerOD = parseInchSize(anchorLiner.odSz);
    if (linerOD !== null && linerOD > 0) {
      if (outerOD !== null && outerOD > linerOD) {
        const blend = Math.min(1, Math.max(0, (ohDiameter - linerOD) / (outerOD - linerOD)));
        return anchorLiner.halfWidth + blend * (innerHW - anchorLiner.halfWidth);
      }
      // No known casing OD to blend against: fall back to a direct ratio, clamped so OH
      // never renders outside the innermost casing's on-screen width.
      return Math.min(anchorLiner.halfWidth * (ohDiameter / linerOD), innerHW);
    }
  }

  // No liners: scale relative to innermost structural casing
  if (outerOD !== null && outerOD > 0 && ohDiameter < outerOD) {
    return innerHW * (ohDiameter / outerOD);
  }

  return innerHW + openHoleHwMargin;
}

export function getCasingTier(csg: ICasingIR, casings: ICasingIR[]): number {
  const structuralCasings = casings.filter(item => item.csgType !== 'Gravel Pack');
  const idx = structuralCasings.indexOf(csg);
  return idx === -1 ? 0 : idx;
}
```

## src/app/features/wwell-planning/wwell-ir/wwell-ir-design-preview/wwell-ir-design-preview.renderer.ts

```typescript
import { Selection, select } from 'd3-selection';
import { easeCubicOut } from 'd3-ease';
import 'd3-transition';
import {
  IntegratedRequirementCasingRow,
  IntegratedRequirementFormationTop,
} from '@models/wwell-planning/wwell-planning.model';
import {
  buildCasingPath,
  buildDepthTicks,
  buildOpenHolePath,
  buildScreenHanger,
  computeCasingHalfWidth,
  createDepthScale,
  formatDepth,
  openHoleHalfWidth,
  pickTickInterval,
} from '@shared/utils/wellbore-math.util';

export interface WwellIrDesignPreviewRenderData {
  readonly formationTops: readonly IntegratedRequirementFormationTop[];
  readonly casing: readonly IntegratedRequirementCasingRow[];
  readonly surfaceElevationFtAsl: number | null;
  readonly targetDepthFt: number;
}

/** A casing/liner row placed on the depth scale, with the half-width its shape is drawn at. */
interface PlacedRow {
  readonly row: IntegratedRequirementCasingRow;
  readonly halfWidth: number;
  readonly topPx: number;
  readonly bottomPx: number;
}

const DRAWING_HEIGHT = 380;
const TOP_MARGIN = 54;
const BOTTOM_MARGIN = 50;
const RULER_X = 46;
const GEO_LINE_X = 132;
const CENTER_X = 300;
const ROW_SPAN = 520;
const BASE_HALF_WIDTH = 10;
const HALF_WIDTH_INCREMENT = 15;
const LINER_HALF_WIDTH_DECREMENT = 13;
const MIN_LINER_HALF_WIDTH = 6;
const SHOE_CURVE_OFFSET = 9;
const LABEL_LINE_GAP = 22;
const LABEL_TEXT_GAP = 6;
const HANGER_LABEL_GAP = 46;
const OPEN_HOLE_MARGIN = 12;
const MIN_OPEN_HOLE_FT = 1;
// Mirrors `presentation/well-bore-view`'s selectTotalDepth/selectDisplayDepth pattern: the scale
// is anchored to a stable minimum rather than whatever partial data happens to be entered, so a
// single shallow value doesn't stretch across the entire drawing height.
const MIN_SCALE_DEPTH_FT = 500;
// The <g> root is measured with getBBox() after every draw and the SVG's viewBox is refit to it
// (plus this padding) — the diagram can never render outside its own viewBox, regardless of how
// many casing rows, how deep the well is, or how long a formation/callout label happens to be.
const VIEWBOX_PADDING = 14;
const FADE_DURATION_MS = 220;

type GSel = Selection<SVGGElement, unknown, null, undefined>;
type PathSel = Selection<SVGPathElement, unknown, null, undefined>;

function isLinerType(type: string): boolean {
  return type.trim().toLowerCase() === 'liner';
}

function computeFormationPenetration(
  depthFtBgl: number,
  formationTops: readonly IntegratedRequirementFormationTop[],
): { readonly name: string; readonly penetrationFt: number } | null {
  const penetrated = [...formationTops].filter(top => top.depthFt <= depthFtBgl).sort((a, b) => b.depthFt - a.depthFt);
  const nearest = penetrated[0];
  if (!nearest) return null;
  const penetrationFt = depthFtBgl - nearest.depthFt;
  return penetrationFt > 0 ? { name: nearest.name, penetrationFt } : null;
}

function buildCasingPrimaryLabel(row: IntegratedRequirementCasingRow): string {
  const kind = row.type.trim().toLowerCase();
  const suffix = kind === 'conductor' || kind === 'liner' ? '' : ' casing';
  return `${row.sizeIn} ${row.type}${suffix} set at ${formatDepth(row.depthFtBgl)}`;
}

function computeMaxDepthFt(data: WwellIrDesignPreviewRenderData): number {
  const depths = [data.targetDepthFt, ...data.casing.map(row => row.depthFtBgl), ...data.formationTops.map(top => top.depthFt)].filter(
    (value): value is number => typeof value === 'number' && value > 0,
  );
  const deepest = depths.length ? Math.max(...depths) : 0;
  return Math.max(deepest, MIN_SCALE_DEPTH_FT);
}

// Deepest-first among the full casing strings only — tier 0 = deepest = narrowest. Liner rows are
// placed separately since they don't participate in this tiering (they hang inside whichever
// string is deepest here, rather than running to the surface themselves).
function placeCasings(casing: readonly IntegratedRequirementCasingRow[], scale: (depth: number) => number): readonly PlacedRow[] {
  return [...casing]
    .filter(row => !isLinerType(row.type))
    .sort((a, b) => b.depthFtBgl - a.depthFtBgl)
    .map((row, tier) => ({
      row,
      halfWidth: computeCasingHalfWidth(tier, BASE_HALF_WIDTH, HALF_WIDTH_INCREMENT),
      topPx: TOP_MARGIN,
      bottomPx: TOP_MARGIN + scale(row.depthFtBgl),
    }));
}

// Each liner hangs from the bottom of the previous string (the deepest casing, or the liner above
// it) — shallowest liner first so the chain can walk outward-in.
function placeLiners(
  casing: readonly IntegratedRequirementCasingRow[],
  placedCasings: readonly PlacedRow[],
  scale: (depth: number) => number,
): readonly PlacedRow[] {
  let hangFromDepth = placedCasings[0]?.row.depthFtBgl ?? 0;
  let hangFromHalfWidth = placedCasings[0]?.halfWidth ?? BASE_HALF_WIDTH + HALF_WIDTH_INCREMENT;

  return [...casing]
    .filter(row => isLinerType(row.type))
    .sort((a, b) => a.depthFtBgl - b.depthFtBgl)
    .map(row => {
      const halfWidth = Math.max(hangFromHalfWidth - LINER_HALF_WIDTH_DECREMENT, MIN_LINER_HALF_WIDTH);
      const placed: PlacedRow = {
        row,
        halfWidth,
        topPx: TOP_MARGIN + scale(hangFromDepth),
        bottomPx: TOP_MARGIN + scale(row.depthFtBgl),
      };
      hangFromDepth = row.depthFtBgl;
      hangFromHalfWidth = halfWidth;
      return placed;
    });
}

function fadeIn(selection: PathSel): void {
  selection.style('opacity', '0').transition().duration(FADE_DURATION_MS).ease(easeCubicOut).style('opacity', '1');
}

/**
 * Imperatively builds the well-bore schematic inside `svgEl` using D3 selections — mirrors
 * `presentation/well-bore-view`'s renderer pattern (full DOM rebuild per draw, driven by an
 * Angular `effect()` in the host component) rather than Angular template bindings. Because
 * Angular's emulated view-encapsulation attribute is never applied to elements D3 appends here,
 * every class this function sets must be styled via `:host ::ng-deep` in the component SCSS
 * (same reason `wellbore-view.component.scss` does it).
 *
 * After every element is appended, the root `<g>` is measured with `getBBox()` and the SVG's
 * `viewBox` is refit to exactly that measured extent (plus a small padding) — this is what
 * guarantees the diagram can never render outside its own viewBox no matter how much casing/
 * formation data is entered or how long a label's text turns out to be.
 */
export function renderWwellIrDesignPreview(svgEl: SVGSVGElement, data: WwellIrDesignPreviewRenderData): void {
  const svg = select(svgEl);
  svg.selectAll('*').remove();
  svg.attr('width', '100%').attr('height', '100%').attr('preserveAspectRatio', 'xMidYMid meet');

  const root: GSel = svg.append('g').attr('class', 'preview__root');

  const maxDepthFt = computeMaxDepthFt(data);
  const scale = createDepthScale(maxDepthFt, DRAWING_HEIGHT);
  const surfaceY = TOP_MARGIN;

  const placedCasings = placeCasings(data.casing, scale);
  const placedLiners = placeLiners(data.casing, placedCasings, scale);
  const innermostRow = placedLiners.length ? placedLiners[placedLiners.length - 1] : (placedCasings[0] ?? null);

  // ── Column headers ─────────────────────────────────────────────────────
  root
    .append('text')
    .attr('class', 'preview__col-header')
    .attr('x', RULER_X)
    .attr('y', 20)
    .text('Depth Below');
  root
    .append('text')
    .attr('class', 'preview__col-header')
    .attr('x', RULER_X)
    .attr('y', 32)
    .text('Ground (Ft)');
  root
    .append('text')
    .attr('class', 'preview__col-header')
    .attr('x', GEO_LINE_X)
    .attr('y', 20)
    .text('Geologic');
  root
    .append('text')
    .attr('class', 'preview__col-header')
    .attr('x', GEO_LINE_X)
    .attr('y', 32)
    .text('Horizon Tops');

  // ── Depth ruler ─────────────────────────────────────────────────────────
  root
    .append('line')
    .attr('class', 'preview__ruler-line')
    .attr('x1', RULER_X)
    .attr('y1', surfaceY)
    .attr('x2', RULER_X)
    .attr('y2', surfaceY + DRAWING_HEIGHT);

  const depthTicks = buildDepthTicks(maxDepthFt, pickTickInterval(maxDepthFt)).map(depth => ({ depth, y: TOP_MARGIN + scale(depth) }));
  depthTicks.forEach(tick => {
    root
      .append('line')
      .attr('class', 'preview__ruler-tick')
      .attr('x1', RULER_X - 5)
      .attr('y1', tick.y)
      .attr('x2', RULER_X)
      .attr('y2', tick.y);
    root
      .append('text')
      .attr('class', 'preview__ruler-label')
      .attr('x', RULER_X - 9)
      .attr('y', tick.y + 3)
      .text(tick.depth.toLocaleString());
  });

  // ── Surface line ────────────────────────────────────────────────────────
  root.append('line').attr('class', 'preview__surface-line').attr('x1', 0).attr('y1', surfaceY).attr('x2', ROW_SPAN).attr('y2', surfaceY);
  if (data.surfaceElevationFtAsl !== null) {
    root
      .append('text')
      .attr('class', 'preview__surface-label')
      .attr('x', CENTER_X - 90)
      .attr('y', surfaceY - 8)
      .text(`Surface — ${data.surfaceElevationFtAsl} ft ASL`);
  }

  // ── Geologic horizon tops ───────────────────────────────────────────────
  const formationLines = [...data.formationTops]
    .sort((a, b) => a.depthFt - b.depthFt)
    .map(top => ({ name: top.name, depthFt: top.depthFt, y: TOP_MARGIN + scale(top.depthFt) }));
  formationLines.forEach(band => {
    root.append('line').attr('class', 'preview__geo-line').attr('x1', 0).attr('y1', band.y).attr('x2', ROW_SPAN).attr('y2', band.y);
    root
      .append('line')
      .attr('class', 'preview__geo-tick')
      .attr('x1', GEO_LINE_X - 14)
      .attr('y1', band.y)
      .attr('x2', GEO_LINE_X + 14)
      .attr('y2', band.y);
    root
      .append('text')
      .attr('class', 'preview__geo-depth')
      .attr('x', GEO_LINE_X - 20)
      .attr('y', band.y + 3)
      .text(band.depthFt.toLocaleString());
    root
      .append('text')
      .attr('class', 'preview__geo-name')
      .attr('x', GEO_LINE_X + 18)
      .attr('y', band.y + 3)
      .text(band.name);
  });

  // ── Casing / liner strings ─────────────────────────────────────────────
  // Paint shallow/wide casings first, then deep/narrow on top, so telescoping strings read
  // correctly; liners are floating segments so they simply paint after (no z-fight to resolve).
  const casingShapes = [...placedCasings]
    .reverse()
    .map(placed => buildCasingPath(CENTER_X, placed.halfWidth, placed.topPx, placed.bottomPx, SHOE_CURVE_OFFSET))
    .concat(placedLiners.map(placed => buildCasingPath(CENTER_X, placed.halfWidth, placed.topPx, placed.bottomPx, SHOE_CURVE_OFFSET)));
  casingShapes.forEach(path => {
    fadeIn(root.append('path').attr('class', 'preview__casing').attr('d', path));
  });

  // Size text centered inside each string's own visible telescoping band — a casing's visible
  // band runs from the shoe of the string above it (or the surface, for the outermost string)
  // down to its own shoe; a liner's band is simply its own top-to-bottom segment.
  const shallowToDeep = [...placedCasings].reverse();
  shallowToDeep.forEach((placed, i) => {
    const bandTop = i === 0 ? TOP_MARGIN : shallowToDeep[i - 1].bottomPx;
    const y = (bandTop + placed.bottomPx) / 2;
    root.append('text').attr('class', 'preview__casing-size').attr('x', CENTER_X).attr('y', y + 3).text(placed.row.sizeIn);
  });
  placedLiners.forEach(placed => {
    const y = (placed.topPx + placed.bottomPx) / 2;
    root.append('text').attr('class', 'preview__casing-size').attr('x', CENTER_X).attr('y', y + 3).text(placed.row.sizeIn);
  });

  // ── Liner hanger ─────────────────────────────────────────────────────────
  const firstLiner = placedLiners[0];
  if (firstLiner) {
    const { left, right } = buildScreenHanger(CENTER_X, firstLiner.topPx, firstLiner.halfWidth, 8, 18);
    root.append('path').attr('class', 'preview__hanger').attr('d', `${left} ${right}`);
    const labelEndX = CENTER_X - firstLiner.halfWidth - HANGER_LABEL_GAP;
    root
      .append('line')
      .attr('class', 'preview__hanger-line')
      .attr('x1', CENTER_X - firstLiner.halfWidth)
      .attr('y1', firstLiner.topPx)
      .attr('x2', labelEndX)
      .attr('y2', firstLiner.topPx);
    root
      .append('text')
      .attr('class', 'preview__hanger-label')
      .attr('x', labelEndX - LABEL_TEXT_GAP)
      .attr('y', firstLiner.topPx + 3)
      .attr('text-anchor', 'end')
      .text('Liner Hanger');
  }

  // ── Open hole ────────────────────────────────────────────────────────────
  if (innermostRow && data.targetDepthFt > innermostRow.row.depthFtBgl + MIN_OPEN_HOLE_FT) {
    const halfWidth = openHoleHalfWidth(innermostRow.halfWidth, OPEN_HOLE_MARGIN);
    const bottomY = TOP_MARGIN + scale(data.targetDepthFt);
    const lineEndX = CENTER_X + halfWidth + LABEL_LINE_GAP;
    const labelX = lineEndX + LABEL_TEXT_GAP;
    const showBullPlug = placedLiners.length > 0;

    root
      .append('path')
      .attr('class', 'preview__openhole')
      .attr('d', buildOpenHolePath(CENTER_X, halfWidth, innermostRow.bottomPx, bottomY));
    root.append('text').attr('class', 'preview__openhole-label').attr('x', labelX).attr('y', bottomY - 6).text('Open Hole');

    if (showBullPlug) {
      root
        .append('line')
        .attr('class', 'preview__bullplug')
        .attr('x1', CENTER_X - halfWidth)
        .attr('y1', bottomY)
        .attr('x2', CENTER_X + halfWidth)
        .attr('y2', bottomY);
      root.append('text').attr('class', 'preview__bullplug-label').attr('x', labelX).attr('y', bottomY + 12).text('Bull Plug');
    }
  }

  // ── Right-side casing callouts ─────────────────────────────────────────
  const casingCallouts = [...placedCasings, ...placedLiners]
    .sort((a, b) => a.row.depthFtBgl - b.row.depthFtBgl)
    .map(placed => {
      const rEdgeX = CENTER_X + placed.halfWidth;
      const lineEndX = rEdgeX + LABEL_LINE_GAP;
      const penetration = computeFormationPenetration(placed.row.depthFtBgl, data.formationTops);
      const secondary = penetration ? `${formatDepth(penetration.penetrationFt)} into ${penetration.name}` : placed.row.remarks || undefined;
      return {
        y: placed.bottomPx,
        rEdgeX,
        lineEndX,
        labelX: lineEndX + LABEL_TEXT_GAP,
        primary: buildCasingPrimaryLabel(placed.row),
        secondary,
      };
    });
  casingCallouts.forEach(callout => {
    root
      .append('line')
      .attr('class', 'preview__callout-line')
      .attr('x1', callout.rEdgeX)
      .attr('y1', callout.y)
      .attr('x2', callout.lineEndX)
      .attr('y2', callout.y);
    root.append('text').attr('class', 'preview__callout-primary').attr('x', callout.labelX).attr('y', callout.y - 2).text(callout.primary);
    if (callout.secondary) {
      root.append('text').attr('class', 'preview__callout-secondary').attr('x', callout.labelX).attr('y', callout.y + 11).text(callout.secondary);
    }
  });

  // ── Total depth ──────────────────────────────────────────────────────────
  if (data.targetDepthFt > 0) {
    const totalDepthY = TOP_MARGIN + scale(data.targetDepthFt);
    root
      .append('line')
      .attr('class', 'preview__td-line')
      .attr('x1', CENTER_X - 60)
      .attr('y1', totalDepthY)
      .attr('x2', CENTER_X + 60)
      .attr('y2', totalDepthY);
    root
      .append('text')
      .attr('class', 'preview__td-label')
      .attr('x', CENTER_X)
      .attr('y', totalDepthY + 30)
      .attr('text-anchor', 'middle')
      .text(`TOTAL DEPTH = ${data.targetDepthFt.toLocaleString()} ft`);
  }

  // ── Auto-fit viewBox: measure the real rendered extent (real glyph metrics included) and
  // refit the viewBox to it, so nothing — long labels, many casing tiers, deep scales — can
  // ever fall outside the visible box. getBBox() is unimplemented in JSDOM (Jest), so this falls
  // back to the fixed provisional box there instead of throwing. ──────────────────────────────
  const rootNode = root.node();
  const bbox = rootNode && typeof rootNode.getBBox === 'function' ? rootNode.getBBox() : null;
  if (bbox && bbox.width > 0 && bbox.height > 0) {
    const x = bbox.x - VIEWBOX_PADDING;
    const y = bbox.y - VIEWBOX_PADDING;
    const width = bbox.width + VIEWBOX_PADDING * 2;
    const height = bbox.height + VIEWBOX_PADDING * 2;
    svg.attr('viewBox', `${x} ${y} ${width} ${height}`);
  } else {
    svg.attr('viewBox', `0 0 ${ROW_SPAN} ${DRAWING_HEIGHT + TOP_MARGIN + BOTTOM_MARGIN}`);
  }
}
```

```typescript
import {
  ChangeDetectionStrategy,
  Component,
  computed,

import { MatSnackBar } from '@angular/material/snack-bar';

  IntegratedRequirementCasingRow,
  IntegratedRequirementFormationTop,
} from '@models/wwell-planning/wwell-planning.model';
import { DynamicSectionComponent } from '../../shared/dynamic-form/dynamic-section/dynamic-section.component';
import { playIfMotionAllowed } from '../../shared/dynamic-form/gsap-motion.util';
import { buildReviewRevealTimeline } from '../../shared/dynamic-form/review-reveal.animation';
import { buildStepTransitionTimeline, WwellStepDirection } from '../../shared/dynamic-form/step-transition.animation';
import { buildWwellDynamicForm } from '../../shared/dynamic-form/wwell-dynamic-form.builder';
import { WwellApprovalMiniChainComponent } from '../../shared/wwell-approvals/wwell-approval-timeline/wwell-approval-mini-chain.component';
import { WwellPlanningStore } from '../../store/wwell-planning.store';
import { WwellIrDesignPreviewComponent } from '../wwell-ir-design-preview/wwell-ir-design-preview.component';
import { WWELL_IR_CHAIN_PREVIEW } from '../wwell-ir-form/wwell-ir-form.config';
import { mapIrFormValueToSubmission } from '../wwell-ir-form/wwell-ir-form.mapper';
import { WWELL_IR_FORM_SCHEMA } from '../wwell-ir-form/wwell-ir-form.schema';
import { WwellIrFormRawValue } from '../wwell-ir-form/wwell-ir-form.types';

const SNACKBAR_DURATION_MS = 3000;
const SELECT_PANEL_CLASS = 'nip-select-panel';

@Component({
  selector: 'app-wwell-ir-new-page',
  standalone: true,
  imports: [
    ReactiveFormsModule,
    MatButtonModule,
    MatIconModule,
    MatStepperModule,
    MatExpansionModule,
    DynamicSectionComponent,
    WwellApprovalMiniChainComponent,
    WwellIrDesignPreviewComponent,
  ],
  templateUrl: './wwell-ir-new-page.component.html',
  styleUrl: './wwell-ir-new-page.component.scss',
  changeDetection: ChangeDetectionStrategy.OnPush,
  // Required for the scoped M3 theme (and the portaled select/autocomplete panel it targets) to
  // actually reach those elements — see wwell-ir-new-page.component.scss for the full rationale.
  encapsulation: ViewEncapsulation.None,
  providers: [
    // Default indicator logic shows a pencil (not a checkmark) for any completed-but-editable,
    // non-active step — and *every* step here is editable (we never set [editable]="false", since
    // that flag also gates CdkStepper's backward-navigation, which we still need). This "guideline"
    // mode instead shows: done (green check) whenever completed and not the active step, regardless
    // of editable — exactly what we want, without touching navigability.
    { provide: STEPPER_GLOBAL_OPTIONS, useValue: { displayDefaultIndicatorType: false } },
  ],
})
export class WwellIrNewPageComponent {
  private readonly fb = inject(FormBuilder);
  private readonly store = inject(WwellPlanningStore);
  private readonly router = inject(Router);
  private readonly route = inject(ActivatedRoute);
  private readonly snackBar = inject(MatSnackBar);

  protected readonly schema = WWELL_IR_FORM_SCHEMA;
  protected readonly reviewSteps = WWELL_IR_FORM_SCHEMA.slice(0, -1);
  protected readonly chainPreview = WWELL_IR_CHAIN_PREVIEW;
  protected readonly selectPanelClass = SELECT_PANEL_CLASS;

  protected readonly form = buildWwellDynamicForm(WWELL_IR_FORM_SCHEMA, this.fb);
  protected readonly isSaving = this.store.creatingIntegratedRequirement;

  // Live-mirrors the form so the review step's well-bore preview redraws as casing/formation
  // rows are added, edited, or removed back in the "Subsurface & Construction" step.
  private readonly rawFormValue: Signal<WwellIrFormRawValue> = toSignal(this.form.valueChanges, {
    initialValue: this.form.getRawValue() as WwellIrFormRawValue,
  });

  protected readonly wellboreCasing = computed<readonly IntegratedRequirementCasingRow[]>(() =>
    this.rawFormValue().subsurface.casing.map(row => ({
      type: row.type,
      sizeIn: row.sizeIn,
      depthFtBgl: row.depthFtBgl ?? 0,
      remarks: row.remarks || null,
    })),
  );

  protected readonly wellboreFormationTops = computed<readonly IntegratedRequirementFormationTop[]>(() =>
    this.rawFormValue().subsurface.formationTops.map(row => ({ name: row.name, depthFt: row.depthFt ?? 0 })),
  );

  protected readonly wellboreTargetDepthFt = computed(() => this.rawFormValue().reference.main.targetDepthFt ?? 0);
  protected readonly wellboreSurfaceElevationFtAsl = computed(() => this.rawFormValue().reference.main.surfaceElevationFtAsl);
  protected readonly wellboreWellName = computed(() => this.rawFormValue().reference.main.wellName || null);

  protected readonly activeStepIndex = signal(0);
  @ViewChildren('stepContent') private readonly stepContents?: QueryList<ElementRef<HTMLElement>>;
  @ViewChildren('reviewCard', { read: ElementRef }) private readonly reviewCards?: QueryList<ElementRef<HTMLElement>>;

  private previousStepIndex = 0;
  private wasSaving = false;
  private reviewRevealed = false;

  constructor() {
    effect(() => {
      const index = this.activeStepIndex();
      const host = this.stepContents?.get(index);
      if (!host) return;

      const direction: WwellStepDirection = index >= this.previousStepIndex ? 'forward' : 'backward';
      playIfMotionAllowed(buildStepTransitionTimeline(host.nativeElement, direction));
      this.previousStepIndex = index;
    });

    effect(() => {
      const isReviewStep = this.activeStepIndex() === WWELL_IR_FORM_SCHEMA.length - 1;
      if (!isReviewStep || this.reviewRevealed) return;

      const cards = this.reviewCards?.map(card => card.nativeElement) ?? [];
      if (!cards.length) return;

      playIfMotionAllowed(buildReviewRevealTimeline(cards));
      this.reviewRevealed = true;
    });

    effect(() => {
      const saving = this.isSaving();
      const error = this.store.createIntegratedRequirementError();

      if (error) {
        this.snackBar.open(error, 'Dismiss', { duration: SNACKBAR_DURATION_MS });
      } else if (this.wasSaving && !saving) {
        this.router.navigate(['..'], { relativeTo: this.route });
      }
      this.wasSaving = saving;
    });
  }

  protected stepGroup(stepKey: string): FormGroup {
    return this.form.get(stepKey) as FormGroup;
  }

  protected sectionControl(stepKey: string, sectionKey: string): FormGroup | FormArray {
    return this.stepGroup(stepKey).get(sectionKey) as FormGroup | FormArray;
  }

  protected isStepValid(stepKey: string): boolean {
    return this.stepGroup(stepKey).valid;
  }

  protected reviewCompletionSummary(): { done: number; total: number } {
    const total = this.reviewSteps.length;
    const done = this.reviewSteps.filter(step => this.isStepValid(step.key)).length;
    return { done, total };
  }

  protected onReviewCardOpened(index: number): void {
    this.reviewCards?.get(index)?.nativeElement.scrollIntoView({ behavior: 'smooth', block: 'nearest' });
  }

  protected onStepSelectionChange(index: number): void {
    this.activeStepIndex.set(index);
  }

  protected cancel(): void {
    this.router.navigate(['..'], { relativeTo: this.route });
  }

  protected saveAsDraft(): void {
    this.submitAs('draft');
  }

  protected submit(stepper: MatStepper): void {
    if (this.form.invalid) {
      this.form.markAllAsTouched();
      const invalidIndex = WWELL_IR_FORM_SCHEMA.findIndex(step => this.stepGroup(step.key).invalid);
      if (invalidIndex >= 0) stepper.selectedIndex = invalidIndex;
      return;
    }
    this.submitAs('in-progress');
  }

  private submitAs(status: 'draft' | 'in-progress'): void {
    const raw = this.form.getRawValue() as WwellIrFormRawValue;
    const { payload, files } = mapIrFormValueToSubmission(raw, status);
    this.store.createIntegratedRequirement(payload, files);
  }
}
```
