import { computed, inject } from '@angular/core';
import { patchState, signalStore, withComputed, withMethods, withState } from '@ngrx/signals';
import { rxMethod } from '@ngrx/signals/rxjs-interop';
import { tapResponse } from '@ngrx/operators';
import { pipe, switchMap, tap } from 'rxjs';
import { CreateIntegratedRequirementPayload, IntegratedRequirement } from '@models/wwell-planning/integrated-requirement.model';
import { mergeIrDetailBlocksIntoIntegratedRequirement } from '../wwell-ir-form/wwell-ir-detail-response.mapper';
import { WwellIrBulkSavePayload } from '../wwell-ir-form/wwell-ir-payload.model';
import { WwellIrService } from '../wwell-ir.service';
import { WwellIrStoreState, initialWwellIrState } from './wwell-ir.state';

function errorMessage(err: unknown, fallback: string): string {
  return err instanceof Error ? err.message : fallback;
}

/** A repeat "Save as Draft" on the same IR reuses its ir_seq (see wwell-ir-payload.mapper.ts), so
 *  the response describes the *same* record, not a new one — replace it in place instead of
 *  prepending a duplicate. Falls back to prepend for a genuinely new IR (irSeq not seen yet). */
function upsertByIrSeq(list: readonly IntegratedRequirement[], created: IntegratedRequirement): IntegratedRequirement[] {
  const existingIndex = created.irSeq != null ? list.findIndex(item => item.irSeq === created.irSeq) : -1;
  if (existingIndex === -1) return [created, ...list];
  // `created` is rebuilt client-side from what was just submitted (see
  // WwellIrService.buildIrFromPayload) — it has no way to know the record's real approval chain
  // (that's a separate, still-mocked endpoint; see loadIrApprovalTimeline), so every resave mints
  // a brand new "just initiated" chain. Harmless for a first-time INSERT (nothing to preserve
  // yet), but a repeat save on an already-in-progress/approved record — including every single
  // inline section edit on the detail page — would otherwise wipe out its real chain state on
  // every save. Keep whichever chain this record already had.
  return list.map((item, index) => (index === existingIndex ? { ...created, chain: item.chain } : item));
}

export const WwellIrStore = signalStore(
  withState<WwellIrStoreState>(initialWwellIrState),

  withComputed(({ irsLoading, linkedRequestWells, linkedRequestWellsLoading, irsError, linkedRequestWellsError, createIrError, updateIrStatusError, irDetailError, irApprovalTimelineError }) => ({
    isLoading: computed(() => irsLoading() || linkedRequestWellsLoading()),
    /** Trimmed {value,label} view of linkedRequestWells for the "Link to Request" picker itself —
     *  see wwell-ir-new-page.component.ts's `applyLinkedRequestWell` for where the full record
     *  (looked up from linkedRequestWells by this same value) drives the auto-populate. */
    linkedRequestWellOptions: computed(() => linkedRequestWells().map(well => ({ value: String(well.prewapSeq), label: well.wellName }))),
    activeErrors: computed(() =>
      [irsError(), linkedRequestWellsError(), createIrError(), updateIrStatusError(), irDetailError(), irApprovalTimelineError()].filter(
        (message): message is string => message !== null,
      ),
    ),
    hasError: computed(
      () =>
        irsError() !== null ||
        linkedRequestWellsError() !== null ||
        createIrError() !== null ||
        updateIrStatusError() !== null ||
        irDetailError() !== null ||
        irApprovalTimelineError() !== null,
    ),
  })),

  withMethods((store, svc = inject(WwellIrService)) => {
    const loadIrs = rxMethod<void>(
      pipe(
        tap(() => patchState(store, { irsLoading: true, irsError: null })),
        switchMap(() =>
          svc.getIrs().pipe(
            tapResponse({
              next: irs => patchState(store, { irs, irsLoading: false }),
              error: (err: unknown) => patchState(store, { irsLoading: false, irsError: errorMessage(err, 'Failed to load integrated requirements') }),
            }),
          ),
        ),
      ),
    );

    const loadLinkedRequestWells = rxMethod<void>(
      pipe(
        tap(() => patchState(store, { linkedRequestWellsLoading: true, linkedRequestWellsError: null })),
        switchMap(() =>
          svc.getEligibleWellsForIr().pipe(
            tapResponse({
              next: linkedRequestWells => patchState(store, { linkedRequestWells, linkedRequestWellsLoading: false }),
              error: (err: unknown) =>
                patchState(store, {
                  linkedRequestWellsLoading: false,
                  linkedRequestWellsError: errorMessage(err, 'Failed to load wells eligible for IR initiation'),
                }),
            }),
          ),
        ),
      ),
    );

    // Keyed by ir_seq (the real primary key once an IR is saved) — switchMap cancels a still-in-
    // flight fetch for a previously-selected row if the user picks another one before it resolves,
    // so a slow first response can never land after (and clobber) a faster later selection.
    const loadIrDetail = rxMethod<number>(
      pipe(
        tap(() => patchState(store, { irDetailLoading: true, irDetailError: null })),
        switchMap(irSeq =>
          svc.getIrDetailBlocks(irSeq).pipe(
            tapResponse({
              next: blocks => {
                const existing = store.irs().find(item => item.irSeq === irSeq);
                // Nothing to merge onto if the summary row isn't loaded (shouldn't happen from the
                // normal select-a-row flow) — just clear the loading flag and leave state as-is.
                if (!existing) {
                  patchState(store, { irDetailLoading: false });
                  return;
                }
                const merged = mergeIrDetailBlocksIntoIntegratedRequirement(existing, blocks);
                patchState(store, { irs: upsertByIrSeq(store.irs(), merged), irDetailLoading: false });
              },
              error: (err: unknown) =>
                patchState(store, { irDetailLoading: false, irDetailError: errorMessage(err, 'Failed to load integrated requirement detail') }),
            }),
          ),
        ),
      ),
    );

    // Separate call from loadIrDetail above — mirrors the future real API shape (approval process
    // gets its own endpoint), mocked client-side in the meantime. Same switchMap-cancels-a-stale-
    // fetch reasoning applies when the selection changes quickly.
    const loadIrApprovalTimeline = rxMethod<number>(
      pipe(
        tap(() => patchState(store, { irApprovalTimelineLoading: true, irApprovalTimelineError: null })),
        switchMap(irSeq =>
          svc.getIrApprovalTimeline(irSeq).pipe(
            tapResponse({
              next: chain => {
                patchState(store, {
                  irs: store.irs().map(item => (item.irSeq === irSeq ? { ...item, chain } : item)),
                  irApprovalTimelineLoading: false,
                });
              },
              error: (err: unknown) =>
                patchState(store, { irApprovalTimelineLoading: false, irApprovalTimelineError: errorMessage(err, 'Failed to load approval timeline') }),
            }),
          ),
        ),
      ),
    );

    return {
      loadIrs(): void {
        loadIrs();
      },

      loadIrDetail(irSeq: number): void {
        loadIrDetail(irSeq);
      },

      loadIrApprovalTimeline(irSeq: number): void {
        loadIrApprovalTimeline(irSeq);
      },

      loadLinkedRequestWells(): void {
        loadLinkedRequestWells();
      },

      createIr(
        payload: CreateIntegratedRequirementPayload,
        irBulkPayload: WwellIrBulkSavePayload,
        attachmentNames: readonly string[] = [],
      ): void {
        patchState(store, { creatingIr: true, createIrError: null });
        svc.createIr(payload, irBulkPayload, attachmentNames).subscribe({
          next: created =>
            patchState(store, {
              irs: upsertByIrSeq(store.irs(), created),
              // Falls back to whatever was already known so an UPDATE response missing irSeq never clears it.
              draftIrSeq: created.irSeq ?? store.draftIrSeq(),
              creatingIr: false,
            }),
          error: (err: unknown) => patchState(store, { creatingIr: false, createIrError: errorMessage(err, 'Failed to create integrated requirement') }),
        });
      },

      /** Call when a fresh "new IR" session starts, so it never inherits a previous draft's ir_seq. */
      resetIrDraft(): void {
        patchState(store, { draftIrSeq: null });
      },

      /** Call when the wizard opens in edit mode for an existing IR — makes every Save/Submit on
       *  that visit issue an UPDATE against this ir_seq instead of an INSERT. */
      startEditingIr(irSeq: number): void {
        patchState(store, { draftIrSeq: irSeq });
      },

      submitIrForReview(ir: IntegratedRequirement): void {
        patchState(store, { updatingIrStatus: true, updateIrStatusError: null });
        svc.submitIrForReview(ir).subscribe({
          next: updated => patchState(store, { irs: store.irs().map(item => (item.id === updated.id ? updated : item)), updatingIrStatus: false }),
          error: (err: unknown) =>
            patchState(store, { updatingIrStatus: false, updateIrStatusError: errorMessage(err, 'Failed to submit integrated requirement for review') }),
        });
      },
    };
  }),
);

    this.saveMergedRaw(this.currentRawFormValue(), finalNames);
    this.closeAttachmentsEdit();
  }
}

import { IntegratedRequirement, IntegratedRequirementTestType } from '@models/wwell-planning/integrated-requirement.model';
import {
  IrCasingRow,
  IrHeaderRow,
  IrHydrogeologyRow,
  IrTopsRow,
  WwellIrDetailTableBlock,
  WwellIrTableName,
} from './wwell-ir-payload.model';

/** Case-insensitive on `candidate.table`: the GET /{irSeq} detail endpoint echoes back
 *  UPPERCASE table names (EXAD_GWD_IR_HEADER...) per exad_agwa_queries_pkg.get_ir_data_prc,
 *  while WwellIrTableName (and the bulk-save write path) uses lowercase — a strict `===` here
 *  never matched, so every block silently resolved to empty and wiped casing/formationTops on
 *  every detail fetch. */
function rowsFor<TRow>(blocks: readonly WwellIrDetailTableBlock[], table: WwellIrTableName): readonly TRow[] {
  const block = blocks.find(candidate => candidate.table.toLowerCase() === table);
  return (block?.data as readonly TRow[] | undefined) ?? [];
}

/** Inverse of the `h2s: '${ppm} ppm'` formatting in wwell-ir-payload.mapper.ts. */
function parseH2sPpm(h2s: string | null | undefined): number | null {
  if (!h2s) return null;
  const value = parseFloat(h2s);
  return Number.isFinite(value) ? value : null;
}

/**
 * Enriches an already-known IntegratedRequirement (e.g. from the GET-all summary list) with the
 * real casing/tops/hydrogeology/program data from a GET /{irSeq} detail fetch. Merges onto
 * `existing` rather than building from scratch because the detail response carries none of the
 * domain-only fields (client, wellName, contacts, status, chain, updatedDate...) that table has no
 * column for — see wwell-ir-summary.mapper.ts for the same limitation on the list side. Field
 * mappings mirror buildWwellIrBulkSavePayload in wwell-ir-payload.mapper.ts exactly (this is that
 * function's inverse) — keep the two in sync if either changes.
 */
export function mergeIrDetailBlocksIntoIntegratedRequirement(
  existing: IntegratedRequirement,
  blocks: readonly WwellIrDetailTableBlock[],
): IntegratedRequirement {
  const header = rowsFor<IrHeaderRow>(blocks, 'exad_gwd_ir_header')[0] as IrHeaderRow | undefined;
  const casingRows = rowsFor<IrCasingRow>(blocks, 'exad_gwd_ir_casing');
  const topsRows = rowsFor<IrTopsRow>(blocks, 'exad_gwd_ir_tops');
  const hydroRow = rowsFor<IrHydrogeologyRow>(blocks, 'exad_gwd_ir_hydrogeology')[0] as IrHydrogeologyRow | undefined;

  return {
    ...existing,
    formationTops: topsRows.map(row => ({
      name: row.st_long_cd,
      depthFt: row.plan_tvd_dpth,
      subSeaFtAsl: row.plan_ss_dpth ?? null,
    })),
    // tolFtBgl has no column yet (buildWwellIrBulkSavePayload never writes it either) — stays null.
    casing: casingRows.map(row => ({
      type: row.csg_type,
      sizeIn: row.csg_size,
      depthFtBgl: row.csg_depth,
      tolFtBgl: null,
      remarks: row.csg_remarks ?? null,
    })),
    programDetails: {
      ...existing.programDetails,
      completion: header?.completion ?? header?.cmpl_type ?? existing.programDetails.completion,
    },
    hydrogeology: hydroRow
      ? {
          aquifer: hydroRow.st_long_cd,
          estimatedTdsPpm: hydroRow.est_water_quality,
          h2sPpm: parseH2sPpm(hydroRow.h2s),
          // Not persisted anywhere in the wire contract yet — carry forward whatever was known.
          estimatedPumpDepthFt: existing.hydrogeology.estimatedPumpDepthFt,
        }
      : existing.hydrogeology,
    testProgram: {
      ...existing.testProgram,
      testType: (header?.test_type as IntegratedRequirementTestType | null | undefined) ?? existing.testProgram.testType,
      estimatedRateGpm: hydroRow?.est_productivity ?? existing.testProgram.estimatedRateGpm,
      estimatedWaterLevelFt: hydroRow?.est_sta_water_lvl ?? existing.testProgram.estimatedWaterLevelFt,
    },
  };
}

/**
 * Carries the previously-assigned ir_seq (if any) so a repeat save UPDATEs instead of
 * re-INSERTing, plus (when known) the record as it was last loaded/saved — used to diff each
 * repeatable row's OWN action instead of inheriting one blanket action for the whole payload.
 * `existing` is intentionally distinguished from "omitted": omitting it (or passing `null`) falls
 * back to the old whole-payload action (every row shares `resolveRowAction(irSeq)`) — the only
 * safe choice when the caller has no snapshot to diff against, since guessing would otherwise
 * risk re-labeling an already-persisted row as a fresh INSERT (duplicate) or a still-current row
 * as DELETE (data loss).
 */
export interface WwellIrSaveContext {
  readonly irSeq: number | null;
  readonly existing?: Pick<IntegratedRequirement, 'casing' | 'formationTops' | 'hydrogeology' | 'testProgram'> | null;
}

const NEW_IR_CONTEXT: WwellIrSaveContext = { irSeq: null, existing: null };

function resolveRowAction(irSeq: number | null): IrRowAction {
  return irSeq === null ? 'INSERT' : 'UPDATE';
}

const isBlankCasingRow = (row: WwellIrRawCasingRow): boolean =>
  !row.type && !row.sizeIn && !row.remarks && row.depthFtBgl == null;

const isBlankFormationTop = (row: WwellIrRawFormationTop): boolean =>
  !row.name && !row.description && row.depthFt == null && row.subSeaFtAsl == null;

/** Natural keys the backend's UPDATE/DELETE branches match repeatable rows by (see
 *  EXAD_AGWA_IUD_PKG.save_casing/save_tops in 05-agwa-iud-package.sql) — a row whose key isn't
 *  already present in `existing` is genuinely new and must be sent as INSERT, not UPDATE, or the
 *  backend's `WHERE <key match>` finds nothing and silently drops it. */
const casingKey = (row: { readonly type: string; readonly sizeIn: string }): string => `${row.type} ${row.sizeIn}`;
const topKey = (row: { readonly name: string }): string => row.name;

/** Only the columns save_casing's/save_tops' UPDATE branch actually writes (type/size are the
 *  match key itself, not a writable column) — comparing anything else could flag a row "changed"
 *  when nothing the backend would touch actually differs. */
const casingRowUnchanged = (existingRow: IntegratedRequirementCasingRow, row: WwellIrRawCasingRow): boolean =>
  existingRow.depthFtBgl === (row.depthFtBgl ?? 0) && (existingRow.remarks ?? null) === (row.remarks || null);

const topRowUnchanged = (existingRow: IntegratedRequirementFormationTop, row: WwellIrRawFormationTop): boolean =>
  existingRow.depthFt === (row.depthFt ?? 0) && (existingRow.subSeaFtAsl ?? null) === (row.subSeaFtAsl ?? null);

function buildCasingRows(
  raw: readonly WwellIrRawCasingRow[],
  irSeq: number | null,
  existing: readonly IntegratedRequirementCasingRow[] | null | undefined,
): IrCasingRow[] {
  const currentRows = raw.filter(row => !isBlankCasingRow(row));

  if (existing == null) {
    const action = resolveRowAction(irSeq);
    return currentRows.map(row => ({
      action,
      ir_seq: irSeq,
      csg_size: row.sizeIn,
      csg_type: row.type,
      csg_depth: row.depthFtBgl ?? 0,
      csg_remarks: row.remarks || null,
    }));
  }

  const existingByKey = new Map(existing.map(row => [casingKey(row), row]));
  const keptKeys = new Set<string>();

  // A row this save didn't actually change is sent as '' (no-op), not a redundant UPDATE — see
  // IrRowAction's doc comment: the backend's per-row action switch then skips the DML entirely
  // instead of re-writing a row with the exact values it already has on every single save.
  const upserts: IrCasingRow[] = currentRows.map(row => {
    const key = casingKey(row);
    const existingRow = existingByKey.get(key);
    if (existingRow) keptKeys.add(key);
    const action: IrRowAction = existingRow ? (casingRowUnchanged(existingRow, row) ? '' : 'UPDATE') : 'INSERT';
    return {
      action,
      ir_seq: existingRow ? irSeq : null,
      csg_size: row.sizeIn,
      csg_type: row.type,
      csg_depth: row.depthFtBgl ?? 0,
      csg_remarks: row.remarks || null,
    };
  });

  const deletes: IrCasingRow[] = existing
    .filter(row => !keptKeys.has(casingKey(row)))
    .map(row => ({
      action: 'DELETE',
      ir_seq: irSeq,
      csg_size: row.sizeIn,
      csg_type: row.type,
      csg_depth: row.depthFtBgl,
      csg_remarks: row.remarks,
    }));

  return [...upserts, ...deletes];
}

function buildTopsRows(
  raw: readonly WwellIrRawFormationTop[],
  irSeq: number | null,
  existing: readonly IntegratedRequirementFormationTop[] | null | undefined,

  const currentRows = raw.filter(row => !isBlankFormationTop(row));

  if (existing == null) {
    const action = resolveRowAction(irSeq);

  const upserts: IrTopsRow[] = currentRows.map(row => {

/** True when nothing save_hydrogeology's UPDATE branch would actually write has changed —
 *  est_sta_water_lvl/est_productivity come from the form's (and domain model's) testProgram
 *  section, not hydrogeology, so both are needed to tell "unchanged" from "changed". */
const hydrogeologyRowUnchanged = (
  existingHydro: IntegratedRequirementHydrogeology,
  existingTestProgram: IntegratedRequirementTestProgram,
  raw: WwellIrRawHydrology,
): boolean =>
  existingTestProgram.estimatedWaterLevelFt === raw.estimatedWaterLevelFt &&
  existingHydro.estimatedTdsPpm === raw.estimatedTdsPpm &&
  existingTestProgram.estimatedRateGpm === raw.estimatedRateGpm &&
  existingHydro.h2sPpm === raw.h2sPpm;

/** Only ever 0 or 1 row (the form has a single hydrogeology section, not a repeatable list), keyed
 *  by aquifer (st_long_cd) the same way the backend matches it — so switching to a different
 *  aquifer must DELETE the old row, not silently orphan it under a key nothing will ever match again. */
function buildHydrogeologyRows(
  raw: WwellIrRawHydrology,
  irSeq: number | null,
  existing: Pick<IntegratedRequirement, 'hydrogeology' | 'testProgram'> | null | undefined,
): IrHydrogeologyRow[] {
  const currentAquifer = raw.aquifer || null;

  if (existing == null) {
    if (!currentAquifer) return [];
    const action = resolveRowAction(irSeq);
    return [
      {
        action,
        ir_seq: irSeq,
        st_long_cd: currentAquifer,
        est_sta_water_lvl: raw.estimatedWaterLevelFt,
        est_water_quality: raw.estimatedTdsPpm,
        est_productivity: raw.estimatedRateGpm,
        flowing: null,
        h2s: raw.h2sPpm != null ? `${raw.h2sPpm} ppm` : null,
      },
    ];
  }

  const existingAquifer = existing.hydrogeology.aquifer || null;
  const rows: IrHydrogeologyRow[] = [];

  if (existingAquifer && existingAquifer !== currentAquifer) {
    rows.push({
      action: 'DELETE',
      ir_seq: irSeq,
      st_long_cd: existingAquifer,
      est_sta_water_lvl: null,
      est_water_quality: null,
      est_productivity: null,
      flowing: null,
      h2s: null,
    });
  }

  if (currentAquifer) {
    const isExisting = existingAquifer === currentAquifer;
    const action: IrRowAction = isExisting
      ? hydrogeologyRowUnchanged(existing.hydrogeology, existing.testProgram, raw)
        ? ''
        : 'UPDATE'
      : 'INSERT';
    rows.push({
      action,
      ir_seq: isExisting ? irSeq : null,
      st_long_cd: currentAquifer,
      est_sta_water_lvl: raw.estimatedWaterLevelFt,
      est_water_quality: raw.estimatedTdsPpm,
      est_productivity: raw.estimatedRateGpm,
      flowing: null,
      h2s: raw.h2sPpm != null ? `${raw.h2sPpm} ppm` : null,
    });
  }

  return rows;
}

  private syncRailOffsets(): void {

  private onSaveSucceeded(): void {

  private applyLinkedRequestWell(linkedRequestId: string): void {
    const well = this.store.linkedRequestWells().find(candidate => String(candidate.prewapSeq) === linkedRequestId);
    if (!well) return;

  private populateFormForEdit(ir: IntegratedRequirement): void {

    this.stepGroup('programDetails').patchValue(patch.programDetails);

    this.populateRepeatableSection('subsurface', 'casing', ir.casing, casingRowPatch);

  // TEMPORARY: validation gate disabled per explicit request — Submit saves whatever is filled
  // in across all steps, same as Save as Draft, instead of blocking on invalid/incomplete fields.
  protected submit(): void {
    this.submitAs('in-progress');
  }

  private submitAs(status: 'draft' | 'in-progress'): void {
    const raw = this.form.getRawValue() as WwellIrFormRawValue;
    const { payload, files } = mapIrFormValueToSubmission(raw, status);
    const knownIrSeq = this.store.draftIrSeq();
    // Reusing the remembered ir_seq turns a repeat "Save as Draft" into an UPDATE instead of a
    // duplicate INSERT. `existing` is what this record looked like as of the last save/load — the
    // payload mapper diffs the current form against it so each casing/tops/hydrogeology row gets
    // its own real INSERT/UPDATE/DELETE (a row added on this pass is genuinely new even though the
    // IR itself already has an ir_seq) instead of one blanket action for the whole submission.
    const existing = knownIrSeq === null ? null : (this.store.irs().find(ir => ir.irSeq === knownIrSeq) ?? null);
    const irBulkPayload = buildWwellIrBulkSavePayload(raw, { irSeq: knownIrSeq, existing });
    this.lastSubmitStatus = status;
    this.lastSubmitWasUpdate = knownIrSeq !== null;
    this.store.createIr(payload, irBulkPayload, files.map(file => file.name));
  }
}

import { HttpClient } from '@angular/common/http';

import { mapIrHeaderSummariesToIntegratedRequirements } from './wwell-ir-form/wwell-ir-summary.mapper';

@Injectable({ providedIn: 'root' })
export class WwellIrService {
  private readonly http = inject(HttpClient);
  private readonly extConfigService = inject(ExternalConfigService);

  private get apiUrl(): string {
    const serverUrl = this.extConfigService.settings.planningServiceUrl ?? '';
    return `${serverUrl}/planning/api/v1`;
  }

  getIrs(): Observable<IntegratedRequirement[]> {
    return this.http
      .get<ApiResponse<IrHeaderSummary>>(`${this.apiUrl}/integrated-requirements`)
      .pipe(map(res => mapIrHeaderSummariesToIntegratedRequirements(res.data)));

  /**
   * Wells still eligible to have an IR initiated (backed by exad_rcd_prewap via
   * exad_agwa_queries_pkg.get_ir_wells_prc) — feeds the "New IR" wizard's Request Reference
   * "Link to Request" picker. Returns the full record (not just {value,label}) so selecting one

   */
  getEligibleWellsForIr(): Observable<IrEligibleWell[]> {
    return this.http.get<ApiResponse<IrEligibleWell>>(`${this.apiUrl}/wells/eligible-for-ir`).pipe(map(res => res.data));
  }

  /**
   * ir_seq is the real primary key once an IR is saved — selecting a row in the list passes it
   * here to fetch the full { table, data } detail (casing/tops/hydrogeology) for that one record.
   * No offline-mock fallback here (unlike the other reads above): faking detail for a specific,
   * already-known-real record would misrepresent it rather than degrade gracefully, so a failure
   * surfaces as a real error instead — see WwellIrStore.loadIrDetail.
   */
  getIrDetailBlocks(irSeq: number): Observable<readonly WwellIrDetailTableBlock[]> {
    return this.http
      .get<ApiResponse<WwellIrDetailTableBlock>>(`${this.apiUrl}/integrated-requirements/${irSeq}`)
      .pipe(map(res => res.data));
  }

  /**
   * The approval-process API is a separate call from the detail fetch above and doesn't exist yet
   * — selecting a row fires both, this one just has nothing real to hit. Mocked client-side (with
   * an artificial delay so it reads as its own network round-trip) until a real
   * `GET .../integrated-requirements/{irSeq}/approval-timeline` endpoint replaces it.
   */
  getIrApprovalTimeline(irSeq: number): Observable<readonly ApprovalChainStep[]> {
    return of(mockApprovalChainForIrSeq(irSeq)).pipe(delay(300));
  }

  createIr(
    payload: CreateIntegratedRequirementPayload,
    irBulkPayload: WwellIrBulkSavePayload,
    attachmentNames: readonly string[] = [],
  ): Observable<IntegratedRequirement> {
    // The backend only accepts the raw exad_gwd_ir_* bulk-save array as the request body (see
    // wwell-ir-payload.model.ts / IntegratedRequirementResource.java's @Consumes(APPLICATION_JSON))
    // — it has no multipart handling and no columns for the richer domain fields below (client,
    // wellName, contacts, attachments...), so those stay client-side only until exad_agwa_iud_pkg's
    // header table grows to carry them. Sending them today just 415s (wrong Content-Type/body shape).
    return this.http
      .post<ApiResponse<WwellIrSaveResult>>(`${this.apiUrl}/integrated-requirements`, irBulkPayload)
      .pipe(
        map(res => this.buildIrFromPayload(payload, res.data[0]?.irSeq ?? null, attachmentNames)),
        // TODO: remove fallback once every environment has a reachable integrated-requirements endpoint.
        catchError(() => of(this.buildIrFromPayload(payload, null, attachmentNames))),
      );
  }

  submitIrForReview(ir: IntegratedRequirement): Observable<IntegratedRequirement> {
    return this.http
      .post<ApiResponse<IntegratedRequirement>>(`${this.apiUrl}/integrated-requirements/${ir.id}/submit-for-review`, {})
      .pipe(
        map(res => res.data[0]),
        // TODO: remove fallback once the submit-for-review endpoint is live.
        catchError(() => of({ ...ir, status: 'submit-for-review' as const, updatedDate: todayIsoDate() })),
      );
  }

  /** Builds the domain object locally from what was submitted — the backend only ever echoes
   *  back an ir_seq (see WwellIrSaveResult), never the full IntegratedRequirement. `irSeq: null`
   *  (offline fallback) mints a placeholder one, same as a first-time INSERT would eventually get. */
  private buildIrFromPayload(
    payload: CreateIntegratedRequirementPayload,
    irSeq: number | null,
    attachmentNames: readonly string[],
  ): IntegratedRequirement {
    const resolvedIrSeq = irSeq ?? Math.floor(Math.random() * 9000 + 1000);
    // ir_seq-derived, matching mapIrHeaderSummaryToIntegratedRequirement's format — this same
    // record must display the same code whether it was just created or reloaded from the list.
    const code = `IR-${resolvedIrSeq}`;
    return {
      id: code,
      code,
      irSeq: resolvedIrSeq,
      updatedDate: todayIsoDate(),
      status: payload.status,
      requestReference: payload.requestReference,
      formationTops: payload.formationTops,
      casing: payload.casing,
      programDetails: payload.programDetails,
      contacts: payload.contacts,
      offsetWells: payload.offsetWells,
      offsetWellsRemarks: payload.offsetWellsRemarks,
      hydrogeology: payload.hydrogeology,
      testProgram: payload.testProgram,
      attachmentNames,
      chain: buildInitialApprovalChain(WWELL_IR_CHAIN_ROLES, 'You'),
    };
  }
}
