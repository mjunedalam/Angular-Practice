  protected readonly ringPercent = computed(() => approvalRingPercent(this.chain()));

  protected readonly popoverOpen = signal(false);
  /** True for the length of the exit animation: the overlay stays mounted so the card can fade/scale out. */
  protected readonly popoverClosing = signal(false);
  private closeTimer: ReturnType<typeof setTimeout> | null = null;

  protected readonly positions: ConnectedPosition[] = [
    { originX: 'end', originY: 'bottom', overlayX: 'end', overlayY: 'top', offsetY: 8 },
    { originX: 'end', originY: 'top', overlayX: 'end', overlayY: 'bottom', offsetY: -8 },
  ];

  protected toggle(): void {
    if (this.popoverOpen() && !this.popoverClosing()) {
      this.close();
      return;
    }
    this.cancelClose();
    this.popoverOpen.set(true);
  }

  protected close(): void {
    if (!this.popoverOpen() || this.popoverClosing()) return;
    if (prefersReducedMotion()) {
      this.popoverOpen.set(false);
      return;
    }
    this.popoverClosing.set(true);
    this.closeTimer = setTimeout(() => this.cancelClose(true), POPOVER_EXIT_MS);
  }

  private cancelClose(finish = false): void {
    if (this.closeTimer !== null) clearTimeout(this.closeTimer);
    this.closeTimer = null;
    this.popoverClosing.set(false);
    if (finish) this.popoverOpen.set(false);
  }
}

/** Prefers the backend's own `{ message }` over Angular's generic "Http failure response" text. */
function errorMessage(err: unknown, fallback: string): string {
  if (err instanceof HttpErrorResponse) {
    const body: unknown = err.error;
    const message = body && typeof body === 'object' ? (body as { message?: unknown }).message : undefined;
    return err.status !== 0 && typeof message === 'string' && message ? message : fallback;
  }
  return err instanceof Error ? err.message : fallback;
}

/** Prefers SAP's own 422 `detail[].msg` list over the generic `{ message }` envelope. */
function sapErrorMessage(err: unknown, fallback: string): string {
  if (err instanceof HttpErrorResponse) {
    const body: unknown = err.error;
    const detail = body && typeof body === 'object' ? (body as { detail?: unknown }).detail : undefined;
    if (Array.isArray(detail)) {
      const messages = detail
        .map(item => (item && typeof item === 'object' ? (item as { msg?: unknown }).msg : undefined))
        .filter((msg): msg is string => typeof msg === 'string' && msg.length > 0);
      if (messages.length > 0) return messages.join('; ');
    }
  }
  return errorMessage(err, fallback);
}

/** Raw base64 (no `data:` prefix) of bytes — the shape SAP's `attachment_files[].b64` expects. */
function bytesToBase64(bytes: Uint8Array): Promise<string> {
  return new Promise((resolve, reject) => {
    const reader = new FileReader();
    reader.onload = () => resolve(String(reader.result).split(',')[1] ?? '');
    reader.onerror = () => reject(reader.error);
    reader.readAsDataURL(new Blob([bytes as BlobPart]));
  });
}

/** Copies the plan's details into the draft; `overwrite` (an explicit well link) replaces typed values, otherwise only blanks are filled. */
function fillFromPlan(draft: WldaDraft, plan: WldaPrewap, overwrite = false): WldaDraft {
  const blank = (value: string) => overwrite || value.trim() === '';
  return {
    ...draft,
    prewapSeq: plan.prewapSeq,
    field: overwrite ? (plan.wPrimHid ?? '') : draft.field,
    wellNumber: overwrite ? (plan.wNum ?? '') : draft.wellNumber,
    waterWellName: blank(draft.waterWellName) ? `${plan.wPrimHid ?? ''}-${plan.wNum ?? ''}` : draft.waterWellName,
    reservoir: blank(draft.reservoir) ? (plan.targetForm ?? '') : draft.reservoir,
    wellType: blank(draft.wellType) ? (plan.wellType ?? '') : draft.wellType,
    totalDepth: blank(draft.totalDepth) ? (plan.tdMd == null ? '' : String(plan.tdMd)) : draft.totalDepth,
    svYutmCord: blank(draft.svYutmCord) ? (plan.surfaceNorth == null ? '' : String(plan.surfaceNorth)) : draft.svYutmCord,
    svXutmCord: blank(draft.svXutmCord) ? (plan.surfaceEast == null ? '' : String(plan.surfaceEast)) : draft.svXutmCord,
    svUtmZn: blank(draft.svUtmZn) ? (plan.svUtmZn == null ? '' : String(plan.svUtmZn)) : draft.svUtmZn,
  };
}

    function uploadFiles(files: readonly File[], prewapSeq: number, ltrId: number): Observable<string[]> {
      if (files.length === 0) return of([]);
      const prefix = wldaS3Prefix(prewapSeq, ltrId);

        files.map(file =>
          s3.store(file, prefix, networkId(), file.name).pipe(
            last(),
            map((): string | null => null),
            catchError(() => of(file.name)),
          ),
        ),
      ).pipe(map(outcomes => outcomes.filter((name): name is string => name !== null)));
    }

    /** Deletes stored objects by key; resolves to the names that failed. */
    function deleteStored(keys: readonly string[], names: ReadonlyMap<string, string>): Observable<string[]> {
      if (keys.length === 0) return of([]);
      return forkJoin(
        keys.map(key =>
          s3.delete(key, networkId()).pipe(
            map((): string | null => null),
            catchError(() => of(names.get(key) ?? key)),
          ),
        ),
      ).pipe(map(outcomes => outcomes.filter((name): name is string => name !== null)));
    }

      s3.search(`${wldaS3Prefix(letter.prewapSeq, letter.ltrId)}/`).subscribe({
        next: res => {
          if (store.selectedKey() !== key) return;
          patchState(store, {
            storedAttachments: res.objects.map(object => ({ key: object.OBJ_KEY, name: object.ORIGINAL_FILENAME, pendingIndex: null })),
          });
        },
        error: () => {
          if (store.selectedKey() === key) notify.error('Could not load the letter attachments');
        },
      });
    }

    /**
     * "Submit for Approval", same three steps as the IR: mint a workflow id, push the workflow to SAP (the
     * letter PDF plus every stored file as `attachment_files`), then stamp the id and "In Progress" on the
     * letter. A refused trigger leaves the letter exactly as it was (Draft).
     */
    function submitToSap(letter: WldaLetter): Observable<void> {
      const prefix = `${wldaS3Prefix(letter.prewapSeq, letter.ltrId)}/`;
      return irService.getNextSapWorkflowSeq().pipe(
        switchMap(seq => {
          const submitted: WldaLetter = { ...letter, sapWfId: seq, sapWfStatus: WLDA_SAP_WF_STATUS_IN_PROGRESS };
          return s3.search(prefix).pipe(
            switchMap(res => {
              const objects = res.objects;
              const stored = objects.map(object => ({ key: object.OBJ_KEY, name: object.ORIGINAL_FILENAME, pendingIndex: null }));
              const code = wldaCode(letter.ltrId);
              const letter$ = from(buildWldaPdf(viewOfLetter(submitted, stored))).pipe(
                switchMap(bytes => bytesToBase64(bytes)),
                map((b64): SapAttachmentFile => ({ file_name: `${code}-letter.pdf`, b64 })),
              );
              const files$ = objects.map(object =>
                s3.fetch(object.OBJ_KEY).pipe(map((response): SapAttachmentFile => ({ file_name: object.ORIGINAL_FILENAME, b64: response.content }))),
              );
              return forkJoin([letter$, ...files$]);
            }),
            switchMap(attachmentFiles =>
              sap
                .trigger({
                  workflow_title: buildWldaSapWorkflowTitle(wldaCode(letter.ltrId), wldaWellName(letter) || `Plan #${letter.prewapSeq}`, environment.production),
                  initiator_network_id: networkId(),
                  requester_network_id: networkId(),
                  process_id: WLDA_SAP_PROCESS_ID,
                  scenario_id: WLDA_SAP_SCENARIO_ID,
                  workflow_id: String(seq),
                  attachment_files: attachmentFiles,
                  body: [],
                })
                .pipe(catchError((err: unknown) => throwError(() => new Error(sapErrorMessage(err, 'Failed to trigger the SAP approval workflow'))))),
            ),
            switchMap(() =>
              wldaService
                .updateLetter(letter.prewapSeq, letter.ltrId, { ltrRmks: letter.ltrRmks, sapWfId: seq, sapWfStatus: WLDA_SAP_WF_STATUS_IN_PROGRESS, sapWfAprvDt: letter.sapWfAprvDt })
                .pipe(catchError((err: unknown) => throwError(() => new Error(errorMessage(err, 'SAP workflow triggered, but failed to record it on the letter'))))),
            ),
            map((): void => undefined),
          );
        }),
      );
    }

    /** Stamps SAP's live status text onto the listed letter (local only), so the badge follows SAP like the IR's does. */
    function withStatusText(prewapSeq: number, ltrId: number, statusText: string): readonly WldaLetter[] {
      const key = wldaKey({ prewapSeq, ltrId });
      return store.letters().map(item => (wldaKey(item) === key ? { ...item, sapWfStatus: statusText } : item));
    }

    /** Reads the letter's real approval chain and status from SAP; a never-submitted letter has nothing to ask about. */
    function loadApproval(letter: WldaLetter): void {
      const key = wldaKey(letter);
      if (letter.sapWfId == null) {
        patchState(store, { approvalChain: [], approvalLoading: false });
        return;
      }
      patchState(store, { approvalLoading: true });
      sap.getWorkflowStatus(WLDA_SAP_PROCESS_ID, letter.sapWfId).subscribe({
        next: status => {
          if (store.selectedKey() !== key) return;
          patchState(store, {
            approvalChain: mapSapWorkflowStatusToApprovalChain(status),
            approvalLoading: false,
            letters: withStatusText(letter.prewapSeq, letter.ltrId, status.header.wf_status_text),
          });
        },
        error: (err: unknown) => {
          if (store.selectedKey() !== key) return;
          patchState(store, { approvalLoading: false });
          notify.error(sapErrorMessage(err, 'Failed to load the SAP approval workflow status'));
        },
      });
    }

    /** Reads the saved letter back and submits it; resolves to null on success or the failure text (never errors). */
    function submitSaved(prewapSeq: number, ltrId: number): Observable<string | null> {
      return wldaService.getLetter(prewapSeq, ltrId).pipe(
        switchMap(letter => submitToSap(letter)),
        map((): string | null => null),
        catchError((err: unknown) => of(err instanceof Error ? err.message : 'Unexpected error')),
      );
    }

    /** Re-reads one letter (single-letter GET) and its files, e.g. after a save. */
    /** Re-reads one letter (single-letter GET) and its files, e.g. after a save. `reopenEdit` re-bases the open draft on the fresh row. */
    function refreshLetter(prewapSeq: number, ltrId: number, reopenEdit = false): void {
      const key = wldaKey({ prewapSeq, ltrId });
      wldaService.getLetter(prewapSeq, ltrId).subscribe({
        next: fresh => {
          patchState(store, {
            letters: store.letters().map(item => (wldaKey(item) === key ? fresh : item)),
            ...(reopenEdit && store.selectedKey() === key ? { draft: draftOfLetter(fresh) } : {}),
          });
          loadStored(fresh, key);
          if (store.selectedKey() === key) loadApproval(fresh);
        },
        error: (err: unknown) => notify.error(errorMessage(err, 'Failed to refresh the WLDA letter')),
      });
    }

    return {

      uploadLetterFile(file: File): void {

          next: ({ percent, response }) => {
            if (!response) {

              return;
            }
            patchState(store, {
              uploadingFile: false,
              uploadPercent: null,
              storedAttachments: [...store.storedAttachments(), { key: response.key, name: file.name, pendingIndex: null }],
            });
            notify.info(`${file.name} uploaded.`);
          },
          error: (err: unknown) =>
            patchState(store, { uploadingFile: false, uploadPercent: null, uploadError: errorMessage(err, `Failed to upload ${file.name}`) }),
        });
      },

      /** Clears a previous upload failure before the next file's dialog opens. */
      resetUpload(): void {
        patchState(store, { uploadError: null, uploadPercent: null });
      },

      loadAll(): void {
        patchState(store, { loading: true, loadError: null });
        forkJoin({ prewaps: wldaService.getPrewaps(), letters: wldaService.getLetters() }).subscribe({
          next: ({ prewaps, letters }) => patchState(store, { prewaps, letters, loading: false }),
          error: (err: unknown) => patchState(store, { loading: false, loadError: errorMessage(err, 'Failed to load WLDA letters') }),
        });
      },

      startDraft(): void {
        patchState(store, { draft: WLDA_EMPTY_DRAFT, selectedKey: null, storedAttachments: [] });
      },

      /** Switches the opened letter from view to edit. */
      startEdit(): void {
        const letter = store.selectedLetter();
        if (letter) patchState(store, { draft: draftOfLetter(letter) });
      },

      patchDraft(patch: Partial<WldaDraft>): void {
        const current = store.draft();
        if (!current) return;
        let next: WldaDraft = { ...current, ...patch };
        // Typing a different Field / Well Number un-links the chosen well; the typed values decide again.
        if (!store.selectedLetter() && ('field' in patch || 'wellNumber' in patch)) next = { ...next, prewapSeq: null };
        // Creating: once Field + Well Number land on a plan, fill the details it already knows (blank fields only).
        if (!store.selectedLetter() && ('field' in patch || 'wellNumber' in patch)) {
          const before = findPlanForWell(store.prewaps(), current.field, current.wellNumber);
          const plan = findPlanForWell(store.prewaps(), next.field, next.wellNumber);
          if (plan && plan !== before) next = fillFromPlan(next, plan);
        }
        patchState(store, { draft: next });
      },

      linkWell(prewapSeq: number | null): void {
        const current = store.draft();
        if (!current || store.selectedLetter()) return;
        const plan = store.prewaps().find(item => item.prewapSeq === prewapSeq);
        patchState(store, { draft: plan ? fillFromPlan(current, plan, true) : { ...current, prewapSeq: null } });
      },

      /** Drops the draft: a create goes back to nothing open, an edit back to viewing the saved letter. */
      discardDraft(): void {
        patchState(store, { draft: null });
      },

      /** Opens a saved letter: the listed row at once, then a fresh single-letter GET and its S3 files. */
      select(key: string | null): void {
        patchState(store, { selectedKey: key, storedAttachments: [], draft: null, approvalChain: [], approvalLoading: false });
        const letter = store.letters().find(item => wldaKey(item) === key);
        if (letter && key) refreshLetter(letter.prewapSeq, letter.ltrId);
      },

      /** Saves the letter first (it owns the LTR_ID the S3 prefix needs), then uploads each file. */
      createLetter(): void {
        const draft = store.draft();
        const plan = store.draftPlan();
        if (store.mode() !== 'create' || !draft || !plan || store.saving() || validateWldaDraft(draft, 'create', plan).length > 0) return;
        patchState(store, { saving: true });

        wldaService
          .createLetter(plan.prewapSeq, toWldaCreatePayload(draft))
          .pipe(
            switchMap(result =>
              uploadFiles(draft.files, result.prewapSeq, result.ltrId).pipe(
                // Inserted as Draft; only now is the SAP workflow started. Whatever goes wrong from here on
                // leaves the saved Draft in place, so the author can fix it and submit again from the editor.
                switchMap(failed => submitSaved(result.prewapSeq, result.ltrId).pipe(map(sapError => ({ result, failed, sapError })))),
              ),
            ),
          )
          .subscribe({
            next: ({ result, failed, sapError }) => {
              // The draft stays up (now keyed to the saved letter) until the fresh row lands, so the page stays in the editor.
              patchState(store, { saving: false, selectedKey: wldaKey(result), storedAttachments: [], draft: { ...draft, prewapSeq: result.prewapSeq, files: [], removedKeys: [] } });
              if (sapError) notify.error(`${wldaCode(result.ltrId)} saved as Draft, but submitting it for approval failed: ${sapError}`);
              else notify.info(`${wldaCode(result.ltrId)} submitted for approval.`);
              if (failed.length > 0) notify.error(`Letter saved, but these files did not upload: ${failed.join(', ')}`);
              wldaService.getLetters().subscribe({
                next: letters => {
                  patchState(store, { letters });
                  refreshLetter(result.prewapSeq, result.ltrId, true);
                },
                // Saved, but the list could not reload: fall back to viewing so a second Create can't duplicate it.
                error: () => patchState(store, { draft: null }),
              });
            },
            error: (err: unknown) => {
              patchState(store, { saving: false });
              notify.error(errorMessage(err, 'Failed to save the WLDA letter'));
            },
          });
      },

      /** Saves remarks (coordinates are not updatable), removes dropped files from S3, uploads new ones. */
      updateLetter(): void {
        const draft = store.draft();
        const letter = store.selectedLetter();
        if (store.mode() !== 'edit' || !draft || !letter || store.saving() || validateWldaDraft(draft, 'edit').length > 0) return;
        patchState(store, { saving: true });
        const names = new Map(store.storedAttachments().map(file => [file.key ?? '', file.name]));

        wldaService
          .updateLetter(letter.prewapSeq, letter.ltrId, toWldaUpdatePayload(draft, letter))
          .pipe(
            switchMap(() =>
              forkJoin({
                removeFailed: deleteStored(draft.removedKeys, names),
                uploadFailed: uploadFiles(draft.files, letter.prewapSeq, letter.ltrId),
              }),
            ),
          )
          .subscribe({
            next: ({ removeFailed, uploadFailed }) => {
              patchState(store, { saving: false, draft: { ...draft, files: [], removedKeys: [] } });
              notify.info(`${wldaCode(letter.ltrId)} updated.`);
              if (removeFailed.length > 0) notify.error(`Could not remove: ${removeFailed.join(', ')}`);
              if (uploadFailed.length > 0) notify.error(`Letter saved, but these files did not upload: ${uploadFailed.join(', ')}`);
              refreshLetter(letter.prewapSeq, letter.ltrId, true);
            },
            error: (err: unknown) => {
              patchState(store, { saving: false });
              notify.error(errorMessage(err, 'Failed to update the WLDA letter'));
            },
          });
      },

      /** Submits an already-saved Draft (or a rejected/recalled letter) for approval. */
      submitLetter(): void {
        const letter = store.selectedLetter();
        if (!letter || store.saving() || store.isDirty()) return;
        patchState(store, { saving: true });
        submitSaved(letter.prewapSeq, letter.ltrId).subscribe(sapError => {
          patchState(store, { saving: false });
          if (sapError) notify.error(`Submitting ${wldaCode(letter.ltrId)} for approval failed: ${sapError}`);
          else notify.info(`${wldaCode(letter.ltrId)} submitted for approval.`);
          refreshLetter(letter.prewapSeq, letter.ltrId, store.mode() === 'edit');
        });
      },

      /** The approval card's refresh button: re-reads the chain from SAP. */
      refreshApproval(): void {
        const letter = store.selectedLetter();
        if (letter) loadApproval(letter);
      },

      /**
       * Recall: DELETE /recall against the workflow's existing id, then poll GET /display (SAP applies it
       * asynchronously) until it reports Recalled. Nothing is persisted: the live status is read back everywhere.
       */
      recallLetter(): void {
        const letter = store.selectedLetter();
        if (!letter || letter.sapWfId == null || store.recalling()) return;
        const sapWfId = letter.sapWfId;
        const key = wldaKey(letter);
        patchState(store, { recalling: true });
        sap.recall(WLDA_SAP_PROCESS_ID, sapWfId, networkId()).subscribe({
          next: () => {
            let sawStatus = false;
            timer(WLDA_SAP_RECALL_POLL_MS, WLDA_SAP_RECALL_POLL_MS)
              .pipe(
                take(WLDA_SAP_RECALL_POLL_MAX_ATTEMPTS),
                concatMap(() => sap.getWorkflowStatus(WLDA_SAP_PROCESS_ID, sapWfId).pipe(catchError(() => of(null)))),
                tap(status => {
                  if (!status) return;
                  sawStatus = true;
                  patchState(store, {
                    letters: withStatusText(letter.prewapSeq, letter.ltrId, status.header.wf_status_text),
                    ...(store.selectedKey() === key ? { approvalChain: mapSapWorkflowStatusToApprovalChain(status) } : {}),
                  });
                }),
                takeWhile(status => !status || !isSapWorkflowRecalled(status.header.wf_status_text), true),
              )
              .subscribe({
                complete: () => {
                  patchState(store, { recalling: false });
                  if (sawStatus) notify.info(`${wldaCode(letter.ltrId)} recalled.`);
                  else notify.error('Workflow recalled, but failed to refresh its latest status');
                },
              });
          },
          error: (err: unknown) => {
            patchState(store, { recalling: false });
            notify.error(sapErrorMessage(err, 'Failed to recall the SAP approval workflow'));
          },
        });
      },

      /** Deletes the letter, then its S3 files best-effort (the row is already gone, so a failure only leaves orphans). */
      deleteLetter(): void {
        const letter = store.selectedLetter();
        if (!letter || store.deleting()) return;
        patchState(store, { deleting: true });
        const keys = store.storedAttachments().flatMap(file => (file.key ? [file.key] : []));

        wldaService
          .deleteLetter(letter.prewapSeq, letter.ltrId)
          .pipe(switchMap(() => deleteStored(keys, new Map())))
          .subscribe({
            next: orphaned => {
              const key = wldaKey(letter);
              patchState(store, {
                deleting: false,
                draft: null,
                selectedKey: null,
                storedAttachments: [],
                letters: store.letters().filter(item => wldaKey(item) !== key),
              });
              notify.info(`${wldaCode(letter.ltrId)} deleted.`);
              if (orphaned.length > 0) notify.error(`Letter deleted, but ${orphaned.length} stored file(s) could not be removed.`);
            },
            error: (err: unknown) => {
              patchState(store, { deleting: false });
              notify.error(errorMessage(err, 'Failed to delete the WLDA letter'));
            },
          });
      },
    };
  }),

  withHooks({ onInit: store => store.loadAll() }),
);

/** Top-down drawing cursor: `y` is the distance from the page's top edge, so layout reads like the form. */
class Sheet {
  y = MARGIN;
  page: PDFPage;

  constructor(
    private readonly doc: PDFDocument,

  ) {

/** Standard PDF fonts only cover WinAnsi; anything else (e.g. Arabic) is replaced rather than crashing the download. */
function printable(value: string, font: PDFFont): string {
  const supported = new Set(font.getCharacterSet());
  return Array.from(value.replace(/[\r\n\t]+/g, ' '))
    .map(char => (supported.has(char.codePointAt(0) ?? 0) ? char : '?'))
    .join('');
}

/** Largest size (down to 6) at which `value` still fits `maxWidth`, so long values shrink instead of spilling over the cell border. */
function fitSize(value: string, font: PDFFont, size: number, maxWidth: number): number {
  const safe = printable(value, font);
  let current = size;
  while (current > 6 && font.widthOfTextAtSize(safe, current) > maxWidth) current -= 0.5;
  return current;
}

/** Greedy word wrap to `maxWidth`; a single word wider than the line is hard-split. */
function wrap(value: string, font: PDFFont, size: number, maxWidth: number): string[] {
  const lines: string[] = [];
  for (const paragraph of printable(value, font).split(/ {2,}|\n/)) {
    let line = '';
    for (const word of paragraph.split(' ')) {
      const candidate = line ? `${line} ${word}` : word;
      if (font.widthOfTextAtSize(candidate, size) <= maxWidth) {
        line = candidate;
        continue;
      }
      if (line) lines.push(line);
      line = word;
      while (font.widthOfTextAtSize(line, size) > maxWidth && line.length > 1) {
        let cut = line.length - 1;
        while (cut > 1 && font.widthOfTextAtSize(line.slice(0, cut), size) > maxWidth) cut -= 1;
        lines.push(line.slice(0, cut));
        line = line.slice(cut);
      }
    }
    lines.push(line);
  }
  return lines;
}

function formatLongDate(iso: string | null): string {
  const date = iso ? new Date(iso) : new Date();
  const valid = Number.isNaN(date.getTime()) ? new Date() : date;
  return valid.toLocaleDateString('en-US', { year: 'numeric', month: 'long', day: 'numeric' });
}

const dash = (value: string): string => value.trim() || '—';

/** Column x-edges for a row of cells sized by `weights`, spanning the full content width. */
function columnEdges(weights: readonly number[]): number[] {
  const total = weights.reduce((sum, weight) => sum + weight, 0);
  const edges = [MARGIN];
  for (const weight of weights) edges.push((edges[edges.length - 1] ?? MARGIN) + (CONTENT_WIDTH * weight) / total);
  return edges;
}

/** A cell: small grey label at the top-left, then value lines centred in the lower part of the cell. */
function drawCell(
  sheet: Sheet,
  left: number,
  right: number,
  top: number,
  height: number,
  label: string,
  values: readonly string[] = [],
  options: { underline?: boolean } = {},
): void {
  const { regular, bold } = sheet.fonts;
  const labelLines = wrap(label, regular, 7, right - left - 8);
  labelLines.forEach((line, index) => sheet.text(line, left + 4, top + 10 + index * 8.5, 7, regular, LABEL));
  const center = (left + right) / 2;
  const startY = top + height - 8 - (values.length - 1) * 12;
  values.forEach((value, index) => {
    const size = fitSize(value, bold, 9, right - left - 8);
    sheet.centered(value, center, startY + index * 12, size, bold);
    if (options.underline) {
      const half = bold.widthOfTextAtSize(printable(value, bold), size) / 2;
      sheet.hline(center - half, center + half, startY + index * 12 + 1.6, 0.6, INK);
    }
  });
}

}

/** Builds the WLDA letter as a PDF laid out like Aramco Form 6208, followed by the SAP workflow table and distribution list. */
export async function buildWldaPdf(view: WldaLetterView): Promise<Uint8Array> {
  const doc = await PDFDocument.create();
  doc.setTitle(`${view.code} — Well Location & Drilling Approval`);
  const fonts: Fonts = {
    regular: await doc.embedFont(StandardFonts.Helvetica),
    bold: await doc.embedFont(StandardFonts.HelveticaBold),
    mono: await doc.embedFont(StandardFonts.CourierBold),
  };
  const sheet = new Sheet(doc, fonts);
  drawHeader(sheet, view);
  drawToLine(sheet, view.status === 'approved');
  drawFieldGrid(sheet, view);
  drawRelocationAndRemarks(sheet, view);
  drawWorkflowTable(sheet, view);
  drawDistribution(sheet, view);
  return doc.save();
}

/** Browser download of the generated PDF under `fileName`. */
export async function downloadWldaPdf(view: WldaLetterView, fileName: string): Promise<void> {
  const bytes = await buildWldaPdf(view);
  const url = URL.createObjectURL(new Blob([bytes as BlobPart], { type: 'application/pdf' }));
  const link = document.createElement('a');
  link.href = url;
  link.download = fileName;
  document.body.appendChild(link);
  link.click();
  link.remove();
  URL.revokeObjectURL(url);
}

import { ChangeDetectionStrategy, Component, computed, effect, inject, signal, untracked } from '@angular/core';
import { FormsModule } from '@angular/forms';
import { MatDialog } from '@angular/material/dialog';
import { MatButtonModule } from '@angular/material/button';
import { MatIconModule } from '@angular/material/icon';
import { MatTooltipModule } from '@angular/material/tooltip';
import { WldaLetter, WldaLetterView } from '@models/wwell-planning/wlda.model';
import { NotificationService } from '@shared/components/notification/notification.service';
import { ActionConfirmDialogService } from '@shared/components/action-confirm-dialog/action-confirm-dialog.service';
import { WwellApprovalPillComponent } from '../shared/wwell-approvals/wwell-approval-pill/wwell-approval-pill.component';
import { WwellApprovalWorkflowCardComponent } from '../shared/wwell-approvals/wwell-approval-workflow-card/wwell-approval-workflow-card.component';
import { isSapWorkflowTerminal } from '../wwell-ir/wwell-ir.config';
import { WldaStore } from './store/wlda.store';
import { WldaUploadDialogComponent } from './wlda-upload-dialog/wlda-upload-dialog.component';
import { WldaFormComponent } from './wlda-form/wlda-form.component';
import { WldaSheetComponent } from './wlda-sheet/wlda-sheet.component';
import { WLDA_STATUS_META } from './wlda.config';
import { downloadWldaPdf } from './wlda-pdf.util';
import { WLDA_EMPTY_DRAFT, draftOfLetter, statusOfLetter, viewOfDraft, viewOfLetter, wldaCode, wldaKey, wldaWellName } from './wlda.util';

@Component({
  selector: 'app-wlda',
  standalone: true,
  imports: [FormsModule, MatButtonModule, MatIconModule, MatTooltipModule, WldaFormComponent, WldaSheetComponent, WwellApprovalWorkflowCardComponent, WwellApprovalPillComponent],
  templateUrl: './wlda.component.html',
  styleUrl: './wlda.component.scss',
  changeDetection: ChangeDetectionStrategy.OnPush,
})
export class WldaComponent {
  protected readonly store = inject(WldaStore);
  private readonly confirmDialog = inject(ActionConfirmDialogService);
  private readonly notify = inject(NotificationService);
  private readonly dialog = inject(MatDialog);

  protected readonly searchTerm = signal('');
  protected readonly downloading = signal(false);
  protected readonly statusMeta = WLDA_STATUS_META;

  /** Sticky like the IR page: rows open in 'edit' until a row's eye icon (or the View toggle) switches to read-only. */
  protected readonly detailMode = signal<'view' | 'edit'>('edit');
  protected readonly viewTab = signal<'details' | 'letter'>('details');

  protected readonly rows = computed(() => {
    const term = this.searchTerm().trim().toLowerCase();
    const letters = this.store.letters();
    if (!term) return letters;
    return letters.filter(letter => [wldaCode(letter.ltrId), wldaWellName(letter)].some(value => value.toLowerCase().includes(term)));
  });

  /** What the sheet shows: the draft while creating, the saved letter (with draft overrides) otherwise. */
  protected readonly view = computed<WldaLetterView | null>(() => {
    const draft = this.store.draft();
    const letter = this.store.selectedLetter();
    if (letter) return viewOfLetter(letter, this.store.storedAttachments(), draft, this.store.approvalChain());
    return draft ? viewOfDraft(draft, this.store.draftPlan()) : null;
  });

  /** What "changed" is measured against: the saved letter while editing, a blank draft while creating. */
  protected readonly baseline = computed(() => {
    const letter = this.store.selectedLetter();
    return letter ? draftOfLetter(letter) : WLDA_EMPTY_DRAFT;
  });

  /** Options of the "Link to Water Well" dropdown: every pre-well plan, labelled like the IR picker. */
  protected readonly linkOptions = computed(() =>
    this.store.prewaps().map(plan => ({ prewapSeq: plan.prewapSeq, label: wldaWellName(plan) || `Plan #${plan.prewapSeq}` })),
  );

  /** Known plans' field codes / well numbers, offered as suggestions so a typo can't leave Create disabled. */
  protected readonly fieldOptions = computed(() => [...new Set(this.store.prewaps().map(plan => (plan.wPrimHid ?? '').trim()).filter(Boolean))]);
  protected readonly wellNumberOptions = computed(() => {
    const field = this.store.draft()?.field.trim().toLowerCase() ?? '';
    return this.store
      .prewaps()
      .filter(plan => !field || (plan.wPrimHid ?? '').trim().toLowerCase() === field)
      .map(plan => (plan.wNum ?? '').trim())
      .filter(Boolean);
  });

  /** Set by the Back arrow so the auto-select below doesn't immediately reopen what was just closed. */
  private readonly deselected = signal(false);

  constructor() {
    // Like the IR page, opening WLDA lands on the first letter instead of an empty detail pane.
    effect(() => {
      if (this.deselected() || this.store.draft() || this.store.selectedKey() !== null) return;
      const first = this.rows()[0];
      if (first) untracked(() => this.open(wldaKey(first)));
    });
  }

  protected readonly canSave = computed(() => this.store.draftErrors().length === 0 && !this.store.saving());

  protected code(letter: WldaLetter): string {
    return wldaCode(letter.ltrId);
  }

  protected detailRows(view: WldaLetterView): readonly { label: string; value: string }[] {
    return [
      { label: 'Letter', value: view.code },
      { label: 'Status', value: this.statusMeta[view.status].label },
      { label: 'Field', value: view.field },
      { label: 'Well Number', value: view.wellNumber },
      { label: 'Well Type', value: view.wellType },
      { label: 'Reservoir', value: view.reservoir },
      { label: 'Total Depth', value: view.totalDepth },
      { label: 'GOSP / Area Plant', value: view.gosp },
      { label: 'Northing', value: view.north },
      { label: 'Easting', value: view.east },
      { label: 'UTM Zone', value: view.zone },
      { label: 'Remarks', value: view.remarks },
      { label: 'Attachments', value: view.attachments.map(file => file.name).join(', ') },
    ];
  }

  /** Title of the detail header: the typed well name while creating, the plan's well name for a saved letter. */
  protected readonly headline = computed(() => {
    const letter = this.store.selectedLetter();
    const typed = this.store.draft()?.waterWellName.trim() ?? '';
    return (letter ? wldaWellName(letter) : typed) || 'Untitled well';
  });

  protected wellLabel(letter: WldaLetter): string {
    return wldaWellName(letter) || 'Unnamed well';
  }

  protected status(letter: WldaLetter) {
    return this.statusMeta[statusOfLetter(letter)];
  }

  protected isActive(letter: WldaLetter): boolean {
    return wldaKey(letter) === this.store.selectedKey();
  }

  protected select(letter: WldaLetter): void {
    this.guardUnsaved(() => {
      this.deselected.set(false);
      this.open(wldaKey(letter));
    });
  }

  /** The row's eye icon: opens the letter read-only (Details / WLDA Letter tabs) and stays read-only for the next rows. */
  protected viewLetter(letter: WldaLetter): void {
    this.guardUnsaved(() => {
      this.deselected.set(false);
      this.detailMode.set('view');
      this.store.select(wldaKey(letter));
    });
  }

  /** The toolbar's View toggle: drop the draft and show the saved letter read-only. */
  protected showView(): void {
    this.guardUnsaved(() => {
      this.detailMode.set('view');
      this.store.discardDraft();
    });
  }

  protected showEdit(): void {
    this.detailMode.set('edit');
    this.store.startEdit();
  }

  private open(key: string): void {
    this.store.select(key);
    if (this.detailMode() === 'edit') this.store.startEdit();
  }

  protected newDraft(): void {
    this.guardUnsaved(() => {
      this.deselected.set(false);
      this.store.startDraft();
    });
  }

  /** Discard (create): leaves the draft, confirming first if it changed. */
  protected leaveDraft(): void {
    this.guardUnsaved(() => this.store.discardDraft());
  }

  /** Only a Draft or a rejected/recalled letter can be (re)submitted, and only once its edits are saved. */
  protected readonly canSubmit = computed(() => {
    const letter = this.store.selectedLetter();
    if (!letter || this.store.saving() || this.store.isDirty()) return false;
    const status = statusOfLetter(letter);
    return status === 'draft' || status === 'rejected';
  });

  /** Recall replaces Submit once SAP has the workflow, but only while it is still live (nothing left to pull back otherwise). */
  protected readonly canRecall = computed(() => {
    const letter = this.store.selectedLetter();
    return !!letter && letter.sapWfId != null && statusOfLetter(letter) === 'submitted' && !isSapWorkflowTerminal(letter.sapWfStatus);
  });

  /** What the shared Approval Workflow card renders: the live chain plus the workflow id (null = never submitted). */
  protected readonly approval = computed(() => ({ chain: this.store.approvalChain(), sapWfId: this.store.selectedLetter()?.sapWfId ?? null }));

  protected readonly hasWorkflow = computed(() => this.approval().sapWfId !== null);

  protected refreshApproval(): void {
    this.store.refreshApproval();
  }

  protected recall(): void {
    const letter = this.store.selectedLetter();
    if (!letter) return;
    this.confirmDialog
      .open({
        title: 'Recall this WLDA letter?',
        rows: [
          { label: 'Letter', value: wldaCode(letter.ltrId) },
          { label: 'Well', value: wldaWellName(letter) || '—' },
        ],
        confirmLabel: 'Recall',
        severity: 'warning',
        hint: 'This pulls the letter back from the SAP approval workflow. You can submit it again afterwards.',
      })
      .subscribe(confirmed => {
        if (confirmed) this.store.recallLetter();
      });
  }

  protected submit(): void {
    this.store.submitLetter();
  }

  protected save(): void {
    if (this.store.mode() === 'create') this.store.createLetter();
    else this.store.updateLetter();
  }

  /** Downloads the letter exactly as the sheet shows it (an unsaved draft downloads as a draft). */
  protected async download(view: WldaLetterView): Promise<void> {
    if (this.downloading()) return;
    this.downloading.set(true);
    try {
      await downloadWldaPdf(view, `${view.status === 'draft' ? 'WLDA-draft' : view.code}.pdf`);
    } catch {
      this.notify.error('Could not generate the WLDA PDF.');
    } finally {
      this.downloading.set(false);
    }
  }

  /** Each picked file gets its own upload dialog, one after another: a saved letter uploads now, a new one stages the file for Create. */
  protected uploadPicked(files: readonly File[]): void {
    const [file, ...rest] = files;
    if (!file) return;
    const canUpload = this.store.mode() === 'edit';
    this.store.resetUpload();
    this.dialog
      .open<WldaUploadDialogComponent, unknown, boolean>(WldaUploadDialogComponent, {
        data: { store: this.store, file, canUpload },
        panelClass: 'ir-upload-panel',
        autoFocus: false,
      })
      .afterClosed()
      .subscribe(accepted => {
        if (accepted && !canUpload) {
          const draft = this.store.draft();
          if (draft) this.store.patchDraft({ files: [...draft.files, file] });
        }
        this.uploadPicked(rest);
      });
  }

  protected confirmDelete(): void {
    const letter = this.store.selectedLetter();
    if (!letter) return;
    this.confirmDialog
      .open({
        title: 'Delete this WLDA letter?',
        rows: [
          { label: 'Letter', value: wldaCode(letter.ltrId) },
          { label: 'Well', value: wldaWellName(letter) || '—' },
        ],
        confirmLabel: 'Delete',
        severity: 'danger',
      })
      .subscribe(confirmed => {
        if (confirmed) this.store.deleteLetter();
      });
  }

  /** Runs `proceed` straight away unless an open draft has unsaved changes, which need a Discard first. */
  private guardUnsaved(proceed: () => void): void {
    if (!this.store.draft() || !this.store.isDirty()) {
      proceed();
      return;
    }
    this.confirmDialog
      .open({
        title: 'Discard unsaved changes?',
        confirmLabel: 'Discard',
        severity: 'warning',
        hint: 'Your changes to this letter have not been saved.',
      })
      .subscribe(confirmed => {
        if (confirmed) proceed();
      });
  }

function utf8Bytes(value: string): number {

const REQUIRED_CREATE_FIELDS: readonly (readonly [keyof WldaDraft, string])[] = [
  ['waterWellName', 'Water Well Name'],
  ['field', 'Field'],
  ['wellNumber', 'Well Number'],
  ['reservoir', 'Reservoir'],
  ['wellType', 'Well Type'],
  ['totalDepth', 'Total Depth'],
  ['svYutmCord', 'North (Y)'],
  ['svXutmCord', 'East (X)'],
];

/** Field a validation message belongs under; the form shows it inline beneath that field. */
export type WldaErrorKey = keyof WldaDraft;
export type WldaFieldErrors = Readonly<Partial<Record<WldaErrorKey, string>>>;

/**
 * Problems with the draft keyed by the field they belong to (first problem per field). The starred
 * fields are only required when creating: in edit mode the well and location are locked.
 */
export function wldaFieldErrors(draft: WldaDraft, mode: 'create' | 'edit' = 'create', plan: WldaPrewap | null = null): WldaFieldErrors {
  const errors: Partial<Record<WldaErrorKey, string>> = {};
  const add = (key: WldaErrorKey, message: string): void => {
    errors[key] ??= message;
  };
  if (mode === 'create') {
    for (const [key, label] of REQUIRED_CREATE_FIELDS) {
      if (String(draft[key]).trim() === '') add(key, `${label} is required.`);
    }
    if (plan === null && draft.field.trim() !== '' && draft.wellNumber.trim() !== '') {
      add('wellNumber', `No pre-well plan exists for ${draft.field.trim()} ${draft.wellNumber.trim()} — link a water well above.`);
    }
  } else if (draft.prewapSeq == null) {
    add('prewapSeq', 'This letter has no pre-well plan.');
  }

  const x = draft.svXutmCord.trim();
  if (x && (!DECIMAL_2DP.test(x) || Number(x) > WLDA_MAX_X_UTM)) {
    add('svXutmCord', `East (X) must be a number up to ${WLDA_MAX_X_UTM} with at most 2 decimals.`);
  }
  const y = draft.svYutmCord.trim();
  if (y && (!DECIMAL_2DP.test(y) || Number(y) > WLDA_MAX_Y_UTM)) {
    add('svYutmCord', `North (Y) must be a number up to ${WLDA_MAX_Y_UTM} with at most 2 decimals.`);
  }
  const zone = draft.svUtmZn.trim();
  if (zone && (!INTEGER.test(zone) || Number(zone) < WLDA_MIN_UTM_ZONE || Number(zone) > WLDA_MAX_UTM_ZONE)) {
    add('svUtmZn', `Zone must be a whole number from ${WLDA_MIN_UTM_ZONE} to ${WLDA_MAX_UTM_ZONE}.`);
  }
  if (utf8Bytes(draft.ltrRmks) > WLDA_MAX_REMARKS_BYTES) {
    add('ltrRmks', `Remarks must be at most ${WLDA_MAX_REMARKS_BYTES} bytes.`);
  }
  return errors;
}

  return Object.values(wldaFieldErrors(draft, mode, plan)).filter((message): message is string => !!message);
}

}

 *  rather than the click handler and a `minRows: 1` seed both trying to supply that first row. */
function addRowsSection(section: WwellFormSectionConfig, requiredKeys: readonly string[], addLabel: string): WwellFormSectionConfig {
  const rowFields = section.repeatable?.rowFields ?? [];
  const key = `${section.key}NewRows`;
  return {
    key,
    title: null,
    repeatable: {
      key,
      addLabel,
      showAddButton: false,
      rowFields: rowFields.map(field =>
        requiredKeys.includes(field.key) ? { ...field, validators: { ...field.validators, required: true } } : field,
      ),
    },
  };
}

function asNumber(value: unknown): number | null {
  return typeof value === 'number' && Number.isFinite(value) ? value : null;
}

/** Turns the Formation Tops "Sub-sea" field into a read-only, calculated one: surface elevation minus the row's depth,
 *  with an (i) tooltip that spells the formula out with the current numbers. */
function withDerivedSubSea(section: WwellFormSectionConfig, elevation: () => number | null): WwellFormSectionConfig {
  const repeatable = section.repeatable;
  if (!repeatable) return section;
  const subSea: Partial<WwellFieldConfig> = {
    readonly: true,
    placeholder: 'auto',
    derived: {
      compute: row => computeSubSeaFt(elevation(), asNumber(row['depthFt'])),
      explain: row => explainSubSea(elevation(), asNumber(row['depthFt'])),
    },
  };
  return {
    ...section,
    repeatable: {
      ...repeatable,
      rowFields: repeatable.rowFields.map(field => (field.key === 'subSeaFtAsl' ? { ...field, ...subSea } : field)),
    },
  };
}

const OFFSET_WELL_PICKER_KEY = 'sourceWell';

/** Prepends the optional "Select Offset Water Well" picker to a section's row fields: it pre-fills the row from a well's
 *  latest record, after which every field can still be edited — or the row typed by hand without picking at all. */
function withOffsetWellPicker(
  section: WwellFormSectionConfig,
  options: Signal<readonly WwellFieldOption[]>,
): WwellFormSectionConfig {
  const repeatable = section.repeatable;
  if (!repeatable) return section;
  const picker: WwellFieldConfig = {
    key: OFFSET_WELL_PICKER_KEY,
    label: 'Select Offset Water Well',
    type: 'select',
    options,
    placeholder: 'Select a well — fills the fields below',
    searchable: true,
    colSpan: 2,
  };
  return { ...section, repeatable: { ...repeatable, rowFields: [picker, ...repeatable.rowFields] } };
}

  private previewRows<TRow>(
    live: readonly TRow[] | null,
    saved: () => readonly TRow[],
    draftRows: readonly TRow[] | null,
    keyOf: (row: TRow) => string,
  ): readonly TRow[] | null {
    if (!draftRows?.length) return live;
    const rows = live ?? saved();
    const keys = new Set(rows.map(keyOf));
    // "Add" refuses a row whose natural key is already taken (it would overwrite that row) — including by
    // another staged draft row — so a colliding draft is not drawn either.
    const additions = draftRows.filter(draft => {
      const key = keyOf(draft);
      if (keys.has(key)) return false;
      keys.add(key);
      return true;
    });
    return additions.length ? [...rows, ...additions] : live;
  }

  /** Watches an offset-well FormArray (the "+" add form, or the section pencil's edit form) and, whenever a row's
   *  "Select Offset Water Well" value changes, fills that row from the picked well. Re-binds each time the form is
   *  opened afresh; the previous subscription is dropped with the previous array. */
  private bindOffsetWellPicker(array: () => FormArray | null): void {
    effect(onCleanup => {
      const live = array();
      if (!live) return;
      const picked = new WeakMap<FormGroup, unknown>();
      const subscription = live.valueChanges.subscribe(() => {
        for (const row of live.controls as FormGroup[]) {
          const value = row.get(OFFSET_WELL_PICKER_KEY)?.value;
          if (picked.get(row) === value) continue;
          picked.set(row, value);
          if (value) this.applyOffsetWellSource(row, value as string);
        }
      });
      onCleanup(() => subscription.unsubscribe());
    });
  }

  /** Commits every row staged in a "+" add-form's own FormArray into the section's live FormArray — the
   *  same array inline edits and deletes write to, so Save diffs it like any other change. All-or-nothing:
   *  Save matches rows on their natural key, so if ANY staged row's key is already taken (by an existing
   *  row, or by another row in the same batch) the whole commit is refused rather than silently overwriting
   *  that row, or committing some rows and leaving the colliding one stranded in the add-form. */
  private commitNewRows<TRow>(
    add: SectionEditState<FormArray>,
    section: WwellFormSectionConfig,
    commit: {
      readonly existingKeys: readonly string[];
      readonly keyOf: (row: TRow) => string;
      /** Last tweak to each row as it is committed (e.g. assigning its save key). */
      readonly prepare?: (row: TRow) => TRow;
      readonly liveArray: () => FormArray | null;
      /** Runs once every staged row is in the live array (e.g. to re-sort it). */
      readonly afterCommit?: (array: FormArray) => void;
      readonly duplicate: (key: string) => string;
      readonly added: (count: number) => string;
    },
  ): void {
    const form = add.control();
    if (!form || form.invalid || form.length === 0) return;

    const rows = form.getRawValue() as TRow[];
    const keys = new Set(commit.existingKeys);
    for (const row of rows) {
      const key = commit.keyOf(row);
      if (keys.has(key)) {
        this.notify.warn(commit.duplicate(key));
        return;
      }
      keys.add(key);
    }

    const array = commit.liveArray();
    if (!array) return;
    rows.forEach(staged => {
      const row = commit.prepare ? commit.prepare(staged) : staged;
      addRepeatableRow(array, section.repeatable?.rowFields ?? [], this.fb);
      array.at(array.length - 1).patchValue(row as object);
    });
    commit.afterCommit?.(array);
    add.close();
    this.notify.info(commit.added(rows.length));
  }

  protected onSlotDragOver(slot: IrAttachmentSlot, event: DragEvent): void {
    if (!event.dataTransfer?.types.includes('Files')) return;
    event.preventDefault();
    event.dataTransfer.dropEffect = 'copy';
    if (this.dragOverSlot() !== slot) this.dragOverSlot.set(slot);
  }

  protected onSlotDragLeave(slot: IrAttachmentSlot, event: DragEvent): void {
    const next = event.relatedTarget as Node | null;
    if (next && (event.currentTarget as HTMLElement).contains(next)) return;
    if (this.dragOverSlot() === slot) this.dragOverSlot.set(null);
  }

  protected onSlotDrop(slot: IrAttachmentSlot, event: DragEvent): void {
    event.preventDefault();
    this.dragOverSlot.set(null);
    const file = event.dataTransfer?.files?.[0];
    if (!file) return;
    if (!hasAcceptedIrAttachmentExtension(file.name)) {
      this.notify.error(`Only ${toExtensionLabel()} files can be attached.`);
      return;
    }
    this.stageAttachment(slot, file);
  }

  protected onAttachmentPicked(slot: IrAttachmentSlot, event: Event): void {
    const input = event.target as HTMLInputElement | null;
    const file = input?.files?.[0];
    if (input) input.value = '';
    if (!file) return;
    this.stageAttachment(slot, file);
  }

  private stageAttachment(slot: IrAttachmentSlot, file: File): void {
    if (!hasAcceptedIrAttachmentExtension(file.name)) {
      this.notify.error(`Unsupported file type. Attach ${toExtensionLabel()}.`);
      return;
    }
    if (file.size > WWELL_IR_ATTACHMENT_MAX_BYTES) {
      this.notify.error('Attachment exceeds the 25 MB limit.');
      return;
    }

    const pendingId = this.store.addPendingAttachment(slot, file);
    // Straight into the upload dialog; dismissing it leaves the file staged on its row.
    this.dialog.open(WwellIrAttachmentUploadDialogComponent, {
      data: { store: this.store, slot, slotLabel: this.slotLabel(slot), slotIcon: this.slotIcon(slot), pendingId, irSeq: this.editorIrSeq() },
      panelClass: 'ir-upload-panel',
      autoFocus: false,
    });
  }

  private saveSettledResolvers: ((succeeded: boolean) => void)[] = [];
  private wasSavingForNavigation = false;

  constructor() {
    this.bindOffsetWellPicker(() => this.offsetWellAdd.control());
    this.bindOffsetWellPicker(() => this.offsetWellsEdit.control());
    effect(onCleanup => {
      const ctrl = this.requestReferenceEdit.control();
      if (!ctrl) return;
      const linkedRequestIdControl = ctrl.get('linkedRequestId');
      const subscription = linkedRequestIdControl?.valueChanges.subscribe((linkedRequestId: string) => {
        console.log('[IR:link] "Link to Request" select — valueChanges fired', { linkedRequestId });
        if (linkedRequestId) this.applyLinkedRequestWell(ctrl, linkedRequestId);
      });
      onCleanup(() => subscription?.unsubscribe());
    });

    effect(() => {
      const id = this.ir().id;
      if (this.previousIrId !== null && id !== this.previousIrId) this.closeAllSections();
      this.previousIrId = id;
    });

    effect(() => {
      if (this.ir().irSeq != null) return;
      untracked(() => this.requestReferenceEdit.open());
    });

    effect(() => {
      const saving = this.store.creatingIr();
      const settled = this.wasSavingForNavigation && !saving;
      this.wasSavingForNavigation = saving;
      if (!settled || this.saveSettledResolvers.length === 0) return;

      const succeeded = !this.store.createIrError();
      const resolvers = this.saveSettledResolvers;
      this.saveSettledResolvers = [];
      resolvers.forEach(resolve => resolve(succeeded));
    });
  }

  public discardUnsavedChanges(): void {
    this.closeAllSections();
  }

  public saveUnsavedChangesAndWait(): Promise<boolean> {
    return new Promise(resolve => {
      if (this.store.creatingIr()) {
        this.saveSettledResolvers.push(resolve);
        return;
      }
      if (this.saveAll()) {
        this.saveSettledResolvers.push(resolve);
      } else {
        resolve(false);
      }
    });
  }

  private applyLinkedRequestWell(referenceGroup: FormGroup, linkedRequestId: string): void {
    console.log('[IR:link] applyLinkedRequestWell — start', { linkedRequestId, eligibleWellsLoaded: this.store.linkedRequestWells().length });
    const well = findLinkedWell(linkedRequestId, this.store.linkedRequestWells());
    if (!well) {
      console.log('[IR:link] applyLinkedRequestWell — no matching eligible well, aborting patch', { linkedRequestId });
      return;
    }

    const hasElevation = linkedWellHasElevation(well);
    const hasCoordinates = linkedWellHasCoordinates(well);
    const hasTargetDepth = linkedWellHasTargetDepth(well);
    const hasTargetAquifer = linkedWellHasTargetAquifer(well);
    const hasSupportingBusiness = linkedWellHasSupportingBusiness(well);

    const patch: Record<string, string | number> = { wellName: well.wellName };
    if (hasElevation) patch['surfaceElevationFtAsl'] = well.surfaceElevation as number;
    patch['coordinates'] = hasCoordinates ? formatLinkedWellCoordinates(well) : '';
    if (hasTargetDepth) patch['targetDepthFt'] = well.targetDepth as number;
    patch['targetAquifer'] = hasTargetAquifer ? (well.targetAquifer as string) : '';
    patch['supportingBusiness'] = hasSupportingBusiness ? (well.supportings as string) : '';
    console.log('[IR:link] applyLinkedRequestWell — patching form group', {
      patch,
      hasElevation,
      hasCoordinates,
      hasTargetDepth,
      hasTargetAquifer,
      hasSupportingBusiness,
    });
    referenceGroup.patchValue(patch);

    referenceGroup.get('wellName')?.disable();
    this.setLinkedFieldLock(referenceGroup, 'surfaceElevationFtAsl', hasElevation);
    console.log('[IR:link] applyLinkedRequestWell — done', { rawValue: referenceGroup.getRawValue() });
  }

  /** `blank` empties every field outright — including those with a schema default (Liner Screen Size) —
   *  instead of restoring that default. Program Details uses it so the whole General step's program
   *  data is deleted; Geologist Contact is a separate section and is never touched by that reset. */
  private resetFlatSection(
    edit: SectionEditState<FormGroup, unknown>,
    section: WwellFormSectionConfig,
    label: string,
    key: 'program' | 'geologist',
    blank = false,
  ): void {
    edit.open();
    const ctrl = edit.control();
    if (!ctrl) return;
    const defaults: Record<string, unknown> = {};
    (section.fields ?? []).forEach(field => {
      defaults[field.key] = blank ? blankValueFor(field) : defaultValueFor(field);
    });
    ctrl.patchValue(defaults);
    // A saved record is deleted for real, right now (the button is a delete icon, not a draft edit). A
    // never-saved draft has nothing stored to delete, so there the form is just emptied.
    if (this.hasSavedIrSeq()) {
      this.deletingSection.set(key);
      // A save that does not start (nothing to save, a validation stop) never raises `isSaving`, so the marker is dropped here.
      if (!this.saveAll()) this.deletingSection.set(null);
      return;
    }
    this.notify.info(`${label} cleared.`);
  }

  private confirmThenAct(
    config: { title: string; rows?: { label: string; value: string }[]; confirmLabel: string },
    onConfirmed: () => void,
  ): void {
    this.confirmDialog
      .open({ title: config.title, rows: config.rows ?? [], confirmLabel: config.confirmLabel, severity: 'danger' })
      .pipe(takeUntilDestroyed(this.destroyRef))
      .subscribe(confirmed => {
        if (confirmed) onConfirmed();
      });
  }

  private confirmThenDelete(title: string, rows: { label: string; value: string }[], onConfirmed: () => void): void {
    this.confirmThenAct({ title, rows, confirmLabel: 'Delete' }, onConfirmed);
  }

  private currentRawFormValue(): WwellIrFormRawValue {
    return mapIntegratedRequirementToRawFormValue(this.ir());
  }

  private saveMergedRaw(raw: WwellIrFormRawValue): void {
    const current = this.ir();
    const status: Extract<IntegratedRequirementStatus, 'draft' | 'in-progress'> =
      current.status === 'draft' || current.status === 'in-progress' ? current.status : 'in-progress';
    const { payload } = mapIrFormValueToSubmission(raw, status);
    const irBulkPayload = buildWwellIrBulkSavePayload(raw, {
      irSeq: current.irSeq ?? null,
      existing: current,
      geologistEmpNumByName: this.store.geologistEmpNumByName(),
    });
    this.store.createIr(payload, irBulkPayload);
  }

  /** Returns whether a save was actually kicked off — `false` on every early-return branch below
   *  (nothing to save, or a blocked validation). `saveUnsavedChangesAndWait` needs this to settle
   *  its own promise immediately in those cases instead of waiting forever for a `creatingIr`
   *  transition that will now never happen. */
  protected saveAll(): boolean {
    // A record whose stored remarks still carry legacy HTML is saved even with nothing edited — the form already
    // holds the cleaned text, and saving is what writes it back.
    if (this.ir().irSeq !== null && !this.hasUnsavedChanges() && !this.ir().hasLegacyMarkup) {
      this.notify.info('No changes to save.');
      return false;
    }

    let raw = this.currentRawFormValue();

    const referenceCtrl = this.requestReferenceEdit.control();
    if (referenceCtrl) raw = { ...raw, reference: { main: referenceCtrl.getRawValue() } };

    // `letterNumber` carries a `Validators.required` in the schema (wwell-ir-form.schema.ts), but
    // nothing elsewhere in this method ever checks form validity before saving — it only gathers
    // raw values. Without this explicit gate, a blank Letter Number saved silently; the asterisk
    // and mat-error only ever showed once the field itself had been touched.
    if (!raw.reference.main.letterNumber?.trim()) {
      this.requestReferenceEdit.open();
      this.requestReferenceEdit.control()?.get('letterNumber')?.markAsTouched();
      this.expandedTimelineSteps.update(current => new Set(current).add('wellbore'));
      this.focusField('ire-sec-wellbore', 'Letter Number');
      this.notify.error('Letter Number is required before saving.');
      return false;
    }

    const programCtrl = this.programEdit.control();
    if (programCtrl) {
      const flat = programCtrl.getRawValue() as WwellIrRawProgramDetails;
      raw = {
        ...raw,
        programDetails: {
          main: {
            completionType: flat.completionType,
            completion: flat.completion,
            openHoleSizeIn: flat.openHoleSizeIn,
            linerSizeIn: flat.linerSizeIn,
            linerDepthFt: flat.linerDepthFt,
            acidization: flat.acidization,
            transmission: flat.transmission,
          },
          developmentTesting: {
            logging: flat.logging,
            mudLogging: flat.mudLogging,
            sampling: flat.sampling,
            flowTest: flat.flowTest,
            remarks: flat.remarks,
          },
        },
      };
    }

    const geologistCtrl = this.geologistContactEdit.control();
    if (geologistCtrl) raw = { ...raw, contacts: { ...raw.contacts, geologist: geologistCtrl.getRawValue() } };

    const hydrogeologyCtrl = this.hydrogeologyEdit.control();
    if (hydrogeologyCtrl) raw = { ...raw, hydrogeology: { main: hydrogeologyCtrl.getRawValue() } };

    const formationTopsCtrl = this.formationTopsEdit.control();
    if (formationTopsCtrl) raw = { ...raw, subsurface: { ...raw.subsurface, formationTops: formationTopsCtrl.getRawValue() } };

    const topsRemarksCtrl = this.topsRemarksEdit.control();
    if (topsRemarksCtrl) raw = { ...raw, subsurface: { ...raw.subsurface, remarks: topsRemarksCtrl.getRawValue() } };

    const casingCtrl = this.casingEdit.control();
    if (casingCtrl) raw = { ...raw, subsurface: { ...raw.subsurface, casing: casingCtrl.getRawValue() } };

    const offsetWellsCtrl = this.offsetWellsEdit.control();
    if (offsetWellsCtrl) raw = { ...raw, contacts: { ...raw.contacts, offsetWells: offsetWellsCtrl.getRawValue() } };

    const offsetRemarksCtrl = this.offsetRemarksEdit.control();
    if (offsetRemarksCtrl) raw = { ...raw, contacts: { ...raw.contacts, remarks: offsetRemarksCtrl.getRawValue() } };

    this.saveMergedRaw(raw);
    this.closeAllSections();
    return true;
  }

  /** Scrolls a labelled field of an open edit form into view and focuses it — what a blocked Save points at. */
  private focusField(sectionId: string, label: string): void {
    setTimeout(() => {
      const fields = Array.from(this.host.nativeElement.querySelectorAll<HTMLElement>(`#${sectionId} mat-form-field`));
      const field = fields.find(candidate => candidate.querySelector('mat-label')?.textContent?.includes(label));
      const input = field?.querySelector<HTMLInputElement | HTMLTextAreaElement>('input, textarea');
      if (!input) return;
      input.scrollIntoView({ block: 'center', behavior: 'smooth' });
      input.focus({ preventScroll: true });
    }, 50);
  }

  private closeAllSections(): void {
    this.flagUnsaved.set(false);
    this.requestReferenceEdit.close();
    this.programEdit.close();
    this.geologistContactEdit.close();
    this.hydrogeologyEdit.close();
    this.formationTopsEdit.close();
    this.topsRemarksEdit.close();
    this.casingEdit.close();
    this.offsetWellsEdit.close();
    this.offsetRemarksEdit.close();
    this.formationTopAdd.close();
    this.casingAdd.close();
    this.offsetWellAdd.close();
  }

  protected submitForReview(): void {
    // The submitted letter/payload is built from the saved record, so unsaved edits would be silently
    // left out — the button is disabled in that state; this guards any other entry point.
    if (this.hasPendingEdits()) return;
    this.store.submitIrForReview(this.ir());
  }

  /** Recalling stops a live SAP workflow — uses the same "confirm, then act" pattern as the other
   *  row-delete actions above, since it's not reversible from here (SAP itself would need a fresh
   *  Submit for Review). */
  protected recallIr(): void {
    const ir = this.ir();
    this.confirmThenAct(
      {
        title: 'Recall this Integrated Requirement?',
        rows: [
          { label: 'IR', value: ir.code },
          { label: 'Well', value: ir.requestReference.wellName || 'Untitled Well' },
        ],
        confirmLabel: 'Recall',
      },
      () => this.store.recallIr(ir),
    );
  }
}
