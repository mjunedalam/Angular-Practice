import { HttpClient, HttpContext, HttpParams } from '@angular/common/http';
import { inject, Injectable } from '@angular/core';
import { Observable } from 'rxjs';
import { SapTriggerRequest, SapTriggerResponse } from '@models/sap/sap-trigger.model';
import { SapWorkflowStatusResponse } from '@models/sap/sap-workflow-status.model';
import { SKIP_AUTH } from '@interceptors/auth.interceptor';
import { SKIP_AUTO_LOGOUT, SUPPRESS_GLOBAL_ERROR } from '@interceptors/error.interceptor';
import { ExternalConfigService } from 'src/app/shared/services/external-config.service';

@Injectable({ providedIn: 'root' })
export class SapService {
  private readonly http = inject(HttpClient);
  private readonly extConfigService = inject(ExternalConfigService);

  private get baseUrl(): string {
    return this.extConfigService.settings.sapServiceUrl;
  }

  /**
   * SAP AAE is a third party on a different origin, so none of this app's own request conventions
   * apply to it — all three opt-outs below are load-bearing, each for a failure confirmed live:
   *
   * - SKIP_AUTH: authInterceptor otherwise attaches this app's own AGWA JWT to every outgoing
   *   request, which sent our internal bearer token to an external host on every submit.
   * - SKIP_AUTO_LOGOUT: errorInterceptor treats any 401 as "this app's session expired" and calls
   *   authStore.logout(). SAP rejecting a request it was never going to accept is not our session
   *   expiring — without this, one SAP 401 signs the user out of AGWA mid-submit (and, because
   *   that branch returns EMPTY rather than rethrowing, WwellIrStore's own error handler never
   *   runs, so the workflow id is never recorded and the Submit spinner never clears).
   * - SUPPRESS_GLOBAL_ERROR: WwellIrStore surfaces its own sapTriggerError/timeline toast; without
   *   this the user gets two toasts for the same failure.
   */
  private get sapContext(): HttpContext {
    return new HttpContext().set(SKIP_AUTH, true).set(SKIP_AUTO_LOGOUT, true).set(SUPPRESS_GLOBAL_ERROR, true);
  }

  trigger(request: SapTriggerRequest): Observable<SapTriggerResponse> {
    const params = new HttpParams().set('native_response', 'false');
    return this.http.post<SapTriggerResponse>(`${this.baseUrl}/trigger`, request, { params, context: this.sapContext });
  }

  /** Backs the IR "Approval Workflow" sidebar — `wfObjId` is the raw sap_wf_id minted and stamped
   *  onto the header by the trigger flow (see wwell-ir-sap.config.ts), not the display-only
   *  composite string. */
  getWorkflowStatus(processId: string, wfObjId: number): Observable<SapWorkflowStatusResponse> {
    const params = new HttpParams().set('process_id', processId).set('wf_obj_id', String(wfObjId)).set('native_response', 'false');
    return this.http.get<SapWorkflowStatusResponse>(`${this.baseUrl}/display`, { params, context: this.sapContext });
  }
}

/** Prefers the backend's own `{ message }` (see ApiResponse<T> — every exad_agwa endpoint here
 *  returns that envelope on both success and error) over Angular's generic "Http failure
 *  response for ..." text, so a specific backend validation error (e.g. an Oracle constraint
 *  message) reaches the user instead of a vague fallback. */
function errorMessage(err: unknown, fallback: string): string {
  if (err instanceof HttpErrorResponse) {
    // status 0 means the request never reached a server at all (refused connection, DNS failure,
    // CORS rejection, offline). There is no ApiResponse envelope to prefer, and `err.error` holds
    // the transport's own TypeError — surfacing its "Failed to fetch" as though it were a backend
    // message tells the user nothing, so the caller's own copy wins here.
    if (err.status === 0) return fallback;
    const body: unknown = err.error;
    const backendMessage = body && typeof body === 'object' ? (body as { message?: unknown }).message : undefined;
    if (typeof backendMessage === 'string' && backendMessage) return backendMessage;
    return err.message || fallback;
  }
  return err instanceof Error ? err.message : fallback;
}

/** Like errorMessage above, but prefers the SAP trigger endpoint's own `{ detail: [{ msg }] }`
 *  422 validation shape (SapValidationError) over the generic `{ message }` envelope every other
 *  exad_agwa endpoint here uses — so a real "field X is required" from SAP reaches the user
 *  instead of a vague fallback. */
function sapErrorMessage(err: unknown, fallback: string): string {
  if (err instanceof HttpErrorResponse) {
    const body: unknown = err.error;
    const detail = body && typeof body === 'object' ? (body as { detail?: unknown }).detail : undefined;
    if (Array.isArray(detail)) {
      const messages = detail
        .map(item => (item && typeof item === 'object' ? (item as { msg?: unknown }).msg : undefined))
        .filter((msg): msg is string => typeof msg === 'string' && msg.length > 0);
      if (messages.length > 0) return messages.join('; ');
    }
  }
  return errorMessage(err, fallback);
}

/** A repeat "Save as Draft" on the same IR reuses its ir_seq (see wwell-ir-payload.mapper.ts), so
 *  the response describes the *same* record, not a new one — replace it in place instead of

 *  Also stamps `sapWfId` locally to the id `update_ir_sap_wf_prc` just persisted, so the Approval
 *  Workflow sidebar can be refreshed immediately (see triggerSapWorkflow's own call to

function markSubmittedForReview(list: readonly IntegratedRequirement[], id: string, sapWfId: number): IntegratedRequirement[] {
  return list.map(item => (item.id === id ? { ...item, status: 'submit-for-review' as const, sapWfId } : item));
}

function upsertByIrSeq(list: readonly IntegratedRequirement[], created: IntegratedRequirement): IntegratedRequirement[] {
  const existingIndex = created.irSeq != null ? list.findIndex(item => item.irSeq === created.irSeq) : -1;
  if (existingIndex === -1) return [created, ...list];
  // A brand-new record's `created` (see createIr's own buildIntegratedRequirementFromPayload
  // fallback) has no way to know the record's real approval chain (that's a separate fetch; see
  // loadIrApprovalTimeline), so it always starts a fresh "just initiated" chain. Harmless for a
  // first-time INSERT (nothing to preserve yet), but would otherwise wipe out an already-in-
  // progress/approved record's real chain on every subsequent save. Keep whichever chain this
  // record already had.
  return list.map((item, index) => (index === existingIndex ? { ...created, chain: item.chain } : item));
}

          svc.getEligibleWellsForIr().pipe(
            tapResponse({
              next: linkedRequestWells => {
                console.log('[IR:link] loadLinkedRequestWells — response received', { count: linkedRequestWells.length, linkedRequestWells });
                patchState(store, { linkedRequestWells, linkedRequestWellsLoading: false });
              },
              error: (err: unknown) => {
                console.log('[IR:link] loadLinkedRequestWells — request failed', err);
                patchState(store, {
                  linkedRequestWellsLoading: false,
                  linkedRequestWellsError: errorMessage(err, 'Failed to load wells eligible for IR initiation'),
                });
              },
            }),
          ),
        ),
      ),
    );

    // Stable reference data (not tied to a specific IR selection, unlike loadLinkedRequestWells) —
    // loaded once at list-page init, see WwellIrComponent.ngOnInit.

      pipe(
        tap(() => patchState(store, { gwdOpsMembersLoading: true, gwdOpsMembersError: null })),
        switchMap(() =>
          svc.getGwdOpsMembers().pipe(
            tapResponse({
              next: gwdOpsMembers => patchState(store, { gwdOpsMembers, gwdOpsMembersLoading: false }),
              error: (err: unknown) =>
                patchState(store, { gwdOpsMembersLoading: false, gwdOpsMembersError: errorMessage(err, 'Failed to load GWD ops members') }),
            }),
          ),
        ),
      ),
    );

    // flight fetch for a previously-selected row if the user picks another one before it resolves,
    // so a slow first response can never land after (and clobber) a faster later selection.
    const loadIrDetail = rxMethod<number>(
      pipe(
        tap(() => patchState(store, { irDetailLoading: true, irDetailError: null })),
        switchMap(irSeq =>
          svc.getIrDetailBlocks(irSeq).pipe(
            tapResponse({
              next: blocks => {
                const existing = store.irs().find(item => item.irSeq === irSeq);
                // Nothing to merge onto if the summary row isn't loaded (shouldn't happen from the

                if (!existing) {

                // endpoint doesn't — see wwell-ir-summary.mapper.ts), so this is the first point a
                // real SAP lookup becomes possible for a freshly-selected record.
                loadIrApprovalTimeline({ irSeq, sapWfId: merged.sapWfId });
              },
              error: (err: unknown) =>
                patchState(store, { irDetailLoading: false, irDetailError: errorMessage(err, 'Failed to load integrated requirement detail') }),

    // to look up — that case clears the chain locally instead of calling SAP with nothing to ask
    // about. Same switchMap-cancels-a-stale-fetch reasoning as loadIrDetail applies when the
    // selection changes quickly.
    const loadIrApprovalTimeline = rxMethod<{ irSeq: number; sapWfId: number | null }>(
      pipe(
        tap(() => patchState(store, { irApprovalTimelineLoading: true, irApprovalTimelineError: null })),
        switchMap(({ irSeq, sapWfId }) => {
          // Nothing to look up yet — leave `chain` exactly as it was (the initial local preview;
          // see buildIntegratedRequirementFromPayload) rather than blank it, since chainBorderColor
          // reads it for the detail panel's accent strip in both view and edit mode, not only the
          // view-mode sidebar this gates (wwell-ir.component.html).

            return EMPTY;

            tapResponse({
              next: status => {
                patchState(store, {
                  irs: store.irs().map(item => (item.irSeq === irSeq ? { ...item, chain: mapSapWorkflowStatusToApprovalChain(status) } : item)),

        switchMap(irSeq =>

              next: attachments => {

                  irs: store.irs().map(item => (item.irSeq === irSeq ? { ...item, attachments } : item)),

                });
              },
              error: (err: unknown) =>
                patchState(store, { attachmentsLoading: false, attachmentError: errorMessage(err, 'Failed to load attachments') }),
            }),
          ),
        ),
      ),
    );

    /** Replaces just this slot's entry on the record, leaving the other slot untouched. */
    const withAttachment = (irSeq: number, slot: IrAttachmentSlot, attachment: IrAttachment | null) =>
      store.irs().map(item => {
        if (item.irSeq !== irSeq) return item;
        const others = item.attachments.filter(existing => existing.slot !== slot);
        return { ...item, attachments: attachment ? [...others, attachment] : others };

      loadIrs(): void {
        loadIrs();
      },

      loadIrDetail(irSeq: number): void {
        loadIrDetail(irSeq);
      },

      loadIrAttachments(irSeq: number): void {
        loadIrAttachments(irSeq);
      },

      /**
       * Stages a picked file. Attaching never uploads on its own — the user sends it with the
       * slot's own upload action (`uploadStagedAttachment`), or, for a New IR that has no ir_seq to
       * upload against yet, the first save flushes it via `uploadPendingAttachments`.
       */
      setPendingAttachment(slot: IrAttachmentSlot, file: File): void {
        patchState(store, { pendingAttachments: { ...store.pendingAttachments(), [slot]: file }, attachmentError: null });
      },

      /**
       * Sends the file staged for one slot. Clears the staging only after the server confirms, so a
       * failed upload leaves the file in the slot to retry rather than silently dropping the user's
       * pick — the one thing that would actually lose their work.
       */
      uploadStagedAttachment(irSeq: number, slot: IrAttachmentSlot): void {
        const file = store.pendingAttachments()[slot];
        if (!file) return;

        patchState(store, { busyAttachmentSlot: slot, attachmentUploadPercent: 0, attachmentError: null });
        svc.uploadIrAttachment(irSeq, slot, file).subscribe({
          next: ({ percent, attachment }) => {
            if (!attachment) {
              patchState(store, { attachmentUploadPercent: percent });
              return;
            }
            const remaining = Object.fromEntries(
              Object.entries(store.pendingAttachments()).filter(([key]) => key !== slot),
            ) as Partial<Record<IrAttachmentSlot, File>>;
            patchState(store, {
              irs: withAttachment(irSeq, slot, attachment),
              pendingAttachments: remaining,
              busyAttachmentSlot: null,
              attachmentUploadPercent: null,
            });
          },
          error: (err: unknown) =>
            patchState(store, {
              busyAttachmentSlot: null,
              attachmentUploadPercent: null,
              attachmentError: errorMessage(err, 'Failed to upload the attachment'),
            }),
        });
      },

      removePendingAttachment(slot: IrAttachmentSlot): void {
        const remaining = Object.fromEntries(
          Object.entries(store.pendingAttachments()).filter(([key]) => key !== slot),
        ) as Partial<Record<IrAttachmentSlot, File>>;
        patchState(store, { pendingAttachments: remaining });
      },

      /** Drops every held file — the draft was abandoned, or a different record was selected. */
      clearPendingAttachments(): void {
        patchState(store, { pendingAttachments: {} });
      },

      /**
       * Uploads everything held for a draft that has just been assigned `irSeq`, one slot at a
       * time. Sequential rather than parallel because `busyAttachmentSlot`/`attachmentUploadPercent`
       * describe a single in-flight upload — two at once would fight over both.
       *
       * State is cleared up front so a second save can't re-upload the same files, and a failure on
       * one slot doesn't abort the other: each reports its own error and the chain continues.
       */
      uploadPendingAttachments(irSeq: number): void {
        const pending = Object.entries(store.pendingAttachments()) as [IrAttachmentSlot, File][];
        if (!pending.length) return;
        patchState(store, { pendingAttachments: {} });

        from(pending)
          .pipe(
            concatMap(([slot, file]) => {
              patchState(store, { busyAttachmentSlot: slot, attachmentUploadPercent: 0, attachmentError: null });
              return svc.uploadIrAttachment(irSeq, slot, file).pipe(
                tap(({ percent, attachment }) => {
                  if (!attachment) {
                    patchState(store, { attachmentUploadPercent: percent });
                    return;
                  }
                  patchState(store, {
                    irs: withAttachment(irSeq, slot, attachment),
                    busyAttachmentSlot: null,
                    attachmentUploadPercent: null,
                  });
                }),
                catchError((err: unknown) => {
                  patchState(store, {
                    busyAttachmentSlot: null,
                    attachmentUploadPercent: null,
                    attachmentError: errorMessage(err, `Failed to upload the ${slot} attachment`),
                  });
                  return of(null);
                }),
              );
            }),
          )
          .subscribe();
      },

      /** Percent stays null throughout: a delete has no measurable progress to report. */
      deleteIrAttachment(irSeq: number, slot: IrAttachmentSlot): void {
        patchState(store, { busyAttachmentSlot: slot, attachmentError: null, attachmentUploadPercent: null });
        svc.deleteIrAttachment(irSeq, slot).subscribe({
          next: () => patchState(store, { irs: withAttachment(irSeq, slot, null), busyAttachmentSlot: null }),
          error: (err: unknown) =>
            patchState(store, { busyAttachmentSlot: null, attachmentError: errorMessage(err, 'Failed to remove the attachment') }),
        });
      },

      loadLinkedRequestWells(): void {
        loadLinkedRequestWells();
      },

      loadGwdOpsMembers(): void {
        loadGwdOpsMembers();
      },

      /**
       * No optimistic echo: the backend only ever confirms an `ir_seq` (see WwellIrSaveResult),
       * so this waits out a real follow-up GET (getIrDetailBlocks) and merges *that* onto the
       * record's last known-real state before `irs` ever changes — never onto whatever was just
       * typed. A field with no backing column at all (client, targetAquifer, targetDepthFt... see
       * mergeIrDetailBlocksIntoIntegratedRequirement's own doc comment) simply won't reflect the
       * edit once this settles, since there is nowhere real to read it back from — better an
       * honest revert than a value that only looks saved until the next reload.
       *
       * `creatingIr` stays true for both round-trips (POST, then GET), so the spinner is what
       * covers the extra latency — nothing renders in between.
       */
      createIr(payload: CreateIntegratedRequirementPayload, irBulkPayload: WwellIrBulkSavePayload): void {
        patchState(store, { creatingIr: true, createIrError: null });
        svc
          .createIr(irBulkPayload)
          .pipe(
            delay(WWELL_SAVE_DELAY_MS),
            switchMap(irSeq => {
              if (irSeq === null) return of({ irSeq, blocks: null });
              return svc.getIrDetailBlocks(irSeq).pipe(
                map(blocks => ({ irSeq, blocks })),
                // The save itself already succeeded here — a hiccup on this second, read-only
                // call must not read as "the save failed" (see the `blocks === null` branch below).
                catchError(() => of({ irSeq, blocks: null })),
              );
            }),
          )
          .subscribe({
            next: ({ irSeq, blocks }) => {
              if (irSeq === null) {
                // No identity to file this under at all — shouldn't happen for a genuine backend
                // success, so there's nothing to merge or select; just stop spinning.
                patchState(store, { creatingIr: false });
                return;
              }
              if (blocks === null) {
                // Confirmed saved (a real ir_seq came back) but the follow-up detail fetch failed
                // — record the real identity so a resave correctly UPDATEs instead of re-INSERTing,
                // but leave `irs` untouched rather than show anything unconfirmed.
                patchState(store, {
                  draftIrSeq: irSeq,
                  creatingIr: false,
                  createIrError: 'Saved, but the latest data failed to reload — refresh to see it.',
                });
                return;
              }
              const existing = store.irs().find(item => item.irSeq === irSeq);
              const base =
                existing ?? buildIntegratedRequirementFromPayload(payload, { id: `IR-${irSeq}`, code: `IR-${irSeq}`, irSeq });
              const merged = mergeIrDetailBlocksIntoIntegratedRequirement(base, blocks, store.geologistNameByEmpNum());
              patchState(store, {
                irs: upsertByIrSeq(store.irs(), merged),
                draftIrSeq: irSeq,
                creatingIr: false,
              });
              // A brand-new record's local draft never had a real sapWfId — this GET's own
              // exad_gwd_ir_header block is the first point one could exist (a resave of an
              // already-submitted IR).
              loadIrApprovalTimeline({ irSeq, sapWfId: merged.sapWfId });
            },
            error: (err: unknown) => patchState(store, { creatingIr: false, createIrError: errorMessage(err, 'Failed to create integrated requirement') }),
          });
      },

      /**
       * "Submit for Review" is the SAP workflow submission, in three steps against two backends:
       * mint a workflow id (agwa-planning's get_wf_next_id_fnc) → push the workflow to SAP AAE
       * under that id (SapService) → record the outcome on the IR header (update_ir_sap_wf_prc).
       *
       * There is no separate IR status-transition call: exad_gwd_ir_header has no status column of
       * its own, so the SAP workflow columns this writes ARE the record of having submitted.
       * `updatingIrStatus` stays true across all three so the editor's Submit button spins for the
       * whole action rather than flickering between steps.
       */
      submitIrForReview(ir: IntegratedRequirement): void {
        // A draft that was never saved has no ir_seq to file the workflow under, so there would be
        // nothing to record the outcome against — refuse before calling SAP rather than after.
        if (ir.irSeq == null) {
          patchState(store, { sapTriggerError: 'Save the integrated requirement before submitting it for review.' });
          return;
        }
        patchState(store, { updatingIrStatus: true, sapTriggerError: null });
        triggerSapWorkflow(ir, ir.irSeq);
      },

      /** Deletes the whole IR (header row, cascading to casing/tops/hydrogeology/water — see
       *  WwellIrService.deleteIr) and drops it from `irs` on success. `irSeq` must already be a
       *  persisted record's — there's nothing to delete server-side for a local-only draft. */
      deleteIr(irSeq: number): void {
        patchState(store, { deletingIr: true, deleteIrError: null });
        svc
          .deleteIr(irSeq)
          .pipe(delay(WWELL_DELETE_DELAY_MS))
          .subscribe({
            next: () => patchState(store, { irs: store.irs().filter(item => item.irSeq !== irSeq), deletingIr: false }),
            error: (err: unknown) => patchState(store, { deletingIr: false, deleteIrError: errorMessage(err, 'Failed to delete integrated requirement') }),
          });
      },
    };
  }),
);

import { ApprovalChainStep, ApprovalStepStatus } from '@models/wwell-planning/wwell-planning.model';
import { SapApprovalStep, SapWorkflowStatusResponse } from '@models/sap/sap-workflow-status.model';

/** SAP never sends a real date as `null` — an unreached/unresponded level uses this literal
 *  instead, per the /display contract (see sap-service's README, Angular-Practice's docs/wb-map.md
 *  sibling docs, and the live sample response this mapper is built from). */
const SAP_UNSET_DATE = '0000-00-00';

function isRealDate(value: string | null | undefined): value is string {
  return !!value && value !== SAP_UNSET_DATE;
}

/**
 * A level with a real `response` has been explicitly decided — anything containing "reject"
 * (case-insensitive) counts as a rejection, everything else (SAP's real values aren't documented
 * to us beyond "APPROVED") as an approval. A level SAP has reached (`received_date` real) but not
 * yet decided is the one currently awaiting action — but only the *last* such level in chain
 * order: earlier ones with no response recorded are levels the workflow has already moved past
 * (e.g. the Initiator's own receipt, which is a log entry, not a pending decision), so they read
 * as cleared rather than stuck.
 */
function stepStatus(step: SapApprovalStep, index: number, lastReachedIndex: number): ApprovalStepStatus {
  const response = step.response?.trim();
  if (response) {
    return /reject/i.test(response) ? 'rejected' : 'approved';
  }
  if (!isRealDate(step.received_date)) return 'pending';
  return index === lastReachedIndex ? 'in-progress' : 'approved';
}

function stepDate(step: SapApprovalStep): string | null {
  if (isRealDate(step.response_date)) return step.response_date;
  if (isRealDate(step.received_date)) return step.received_date;
  return null;
}

/** Maps a SAP `GET /display` response onto the shared `ApprovalChainStep[]` shape the IR
 *  "Approval Workflow" sidebar (and the shared `app-wwell-approval-timeline` renderer it uses)
 *  already knows how to draw — see wwell-ir.component.html. */
export function mapSapWorkflowStatusToApprovalChain(status: SapWorkflowStatusResponse): readonly ApprovalChainStep[] {
  const chain = status.approval_chain;
  const lastReachedIndex = chain.reduce((last, step, index) => (isRealDate(step.received_date) ? index : last), -1);

  return chain.map((step, index) => ({
    role: step.approver_title,
    name: step.approver_name,
    status: stepStatus(step, index, lastReachedIndex),
    date: stepDate(step),
  }));
}

/** Every IrXxxRow interface in wwell-ir-payload.model.ts (IrHeaderRow, IrCasingRow, IrTopsRow,
 *  IrHydrogeologyRow, IrWaterRow) uses lowercase snake_case field names mirroring the Oracle
 *  column names — but Oracle's own default column casing (and whatever exad_agwa_queries_pkg's
 *  get_ir_data_prc happens to emit) isn't guaranteed to be lowercase, so a row can come back with
 *  the same fields in UPPERCASE or MixedCase. Since every field this file reads is lowercase by
 *  contract, lowercasing every key up front makes each row match regardless of how the backend
 *  cased it — the same fix as `rowsFor`'s table-name comparison below, applied to row field names. */
function normalizeRowKeys<TRow>(row: Record<string, unknown>): TRow {
  const normalized: Record<string, unknown> = {};
  for (const key of Object.keys(row)) {
    normalized[key.toLowerCase()] = row[key];
  }
  return normalized as TRow;
}

/** Case-insensitive on `candidate.table`: the GET /{irSeq} detail endpoint echoes back
 *  UPPERCASE table names (EXAD_GWD_IR_HEADER...) per exad_agwa_queries_pkg.get_ir_data_prc,
 *  while WwellIrTableName (and the bulk-save write path) uses lowercase — a strict `===` here
 *  never matched, so every block silently resolved to empty and wiped casing/formationTops on
 *  every detail fetch. */
function rowsFor<TRow>(blocks: readonly WwellIrDetailTableBlock[], table: WwellIrTableName): readonly TRow[] {
  const block = blocks.find(candidate => candidate.table.toLowerCase() === table);
  return (block?.data ?? []).map(row => normalizeRowKeys<TRow>(row));

import { IntegratedRequirement, IntegratedRequirementTestType } from '@models/wwell-planning/integrated-requirement.model';
import { todayIsoDate } from '../../shared/wwell-approvals/wwell-approval-request.util';
import { IrHeaderSummary } from './wwell-ir-payload.model';

/**
 * Backend list rows are backed only by exad_gwd_ir_header — the domain fields the UI displays
 * (client, wellName, contacts, casing...) have no column there yet, so they're left blank rather
 * than guessed. Extend once the header table carries those fields (see IrHeaderSummary).
 */
export function mapIrHeaderSummaryToIntegratedRequirement(row: IrHeaderSummary): IntegratedRequirement {
  const code = `IR-${row.irSeq}`;

  return {
    id: code,
    code,
    irSeq: row.irSeq,
    // Neither date column is guaranteed to be populated on exad_gwd_ir_header (see the real
    // payload that motivated this: every row came back with both createdDt and updatedDt null) —
    // fall back to today rather than crashing the whole list map on a null .slice() call.
    updatedDate: (row.updatedDt ?? row.createdDt ?? todayIsoDate()).slice(0, 10),
    // publish_flg is the only status-shaped signal this table carries — collapses the domain
    // model's finer-grained statuses (draft/in-progress/submit-for-review/in-review) into one.
    status: row.publishFlg === 'Y' ? 'approved' : 'in-progress',
    requestReference: {
      // prewap_seq/ltr_no are real exad_gwd_ir_header columns (unlike wellName/coordinates/etc.
      // below) — read straight off the list row so a record's "Link to Request" is known
      // immediately from the list. Water Well Name/Coordinates/Surface Elevation stay blank until
      // detail is fetched (see mergeIrDetailBlocksIntoIntegratedRequirement's exad_rcd_prewap
      // block) — the list endpoint has no join for them.
      linkedRequestId: row.prewapSeq != null ? String(row.prewapSeq) : null,
      letterNumber: row.ltrNo ?? null,
      client: '',
      supportingBusiness: '',
      wellName: '',
      targetAquifer: '',
      targetDepthFt: 0,
      surfaceElevationFtAsl: null,
      coordinates: '',
    },
    formationTops: [],
    topsRemarks: null,
    casing: [],
    programDetails: {
      completion: row.cmplType ?? '',
      openHoleSizeIn: null,
      acidization: '',
      transmission: '',
      developmentTesting: { logging: '', mudLogging: '', sampling: '', flowTest: '', remarks: '' },
    },
    contacts: { geologistPrimary: '', geologistSecondary: null },
    offsetWells: [],
    offsetWellsRemarks: null,
    hydrogeology: { aquifer: '', estimatedTdsPpm: null, h2sPpm: null, estimatedPumpDepthFt: null },
    testProgram: {
      testType: (row.testType as IntegratedRequirementTestType | null) ?? 'flow-test',
      estimatedRateGpm: null,
      estimatedWaterLevelFt: null,
    },
    attachments: [],
    // The list endpoint has no sap_wf_id column to look a real chain up with (only the detail
    // fetch's exad_gwd_ir_header block does — see mergeIrDetailBlocksIntoIntegratedRequirement) —
    // stays empty until a selection's detail fetch resolves one, rather than preview a chain this
    // row can't actually back yet.
    chain: [],
    sapWfId: null,
  };
}

export function mapIrHeaderSummariesToIntegratedRequirements(
  rows: readonly IrHeaderSummary[],

}

      if (message) this.notify.error(message);

      if (message) this.notify.error(message);

    // while in flight (see wwell-ir-editor.component.html's isDeleting()), so nothing needs
    // announcing at click time. A failure is already toasted by the effect above.
    effect(() => {

      const settled = this.wasDeletingIr && !deleting;
      this.wasDeletingIr = deleting;
      if (!settled || this.store.deleteIrError()) return;

      this.notify.info('Integrated requirement deleted.');

      // The deleted record is already gone from store.irs() by this point (WwellIrStore.deleteIr
      // removes it on success) — if it was the one open, fall back to null so the "default to
      // first visible record" effect below picks whatever the list now shows first (or leaves the

      if (this.selectedIrSeq() !== null && !this.store.irs().some(ir => ir.irSeq === this.selectedIrSeq())) {

    // and select an unrelated real record's detail out from under the still-open draft.

    // (casing/tops/hydrogeology) by ir_seq, its primary key.

      // exad_gwd_ir_header block resolves the record's sapWfId (only the detail fetch carries it —
      // see wwell-ir-summary.mapper.ts) — no separate call needed here.

      // exad_gwd_ir_* columns, so they can't ride along in the detail blocks above.
      this.store.loadIrAttachments(irSeq);

      // them surface on whichever record was just selected. (This effect deliberately no-ops on the
      // save transition, see the comment above, so it can't discard files mid-upload.)

    this.store.clearPendingAttachments();

    this.selectedIrSeq.set(null);
    this.previousSelectedIrSeq = null;

    // regardless of whatever mode the previously-open record was left in.
    this.detailMode.set('edit');

    const host = this.detailHost?.nativeElement;

  protected approvedStepCount(chain: readonly ApprovalChainStep[]): number {
    return chain.filter(step => step.status === 'approved').length;

  // ── Unsaved-changes guard — backs three separate "leaving the currently-open record" paths with
  // one prompt: switching the selected row / opening "New IR" / clicking the view (eye) icon (all
  // via guardNavigation below), and a real Angular route change away from this page entirely (via

  // app.routes.ts and unsaved-changes.guard.ts). Browser tab close/refresh is covered separately by
  // the beforeunload listener below, which can't run this same async dialog flow.

  /** Router entry point (ComponentCanDeactivate) — resolves `true` once it's safe to actually leave
   *  the route: immediately if nothing is unsaved, after a confirmed Discard, or after a confirmed
   *  Save has settled. `false` means stay (Cancel, or a save that failed). */
  canDeactivate(): Observable<boolean> {
    const editor = this.editorRef;
    if (!editor?.hasUnsavedChanges()) return of(true);

    return this.confirmDialog
      .openWithResult({
        title: 'Unsaved changes',
        subtitle: 'This Integrated Requirement has section edits that haven’t been saved yet. Save them before leaving?',
        confirmLabel: 'Save & Leave',
        discardLabel: 'Discard Changes',
        cancelLabel: 'Stay Here',
        severity: 'warning',
      })
      .pipe(
        switchMap(result => {
          if (result === 'cancel') return of(false);
          if (result === 'discard') {
            editor.discardUnsavedChanges();
            return of(true);
          }
          return from(editor.saveUnsavedChangesAndWait());
        }),
      );
  }

  /** In-page equivalent of canDeactivate() — same prompt/save/discard flow, but for switching what
   *  this page itself shows (selecting another row, opening a blank "New IR", or the view-icon's
   *  mode switch) rather than leaving the route. `action` only runs once it's actually safe to
   *  proceed. */
  private guardNavigation(action: () => void): void {
    this.canDeactivate()
      .pipe(take(1))
      .subscribe(canLeave => {
        if (canLeave) action();
      });
  }

  /** Native browser navigation (tab close, refresh, typed URL) bypasses the Router entirely, so
   *  canDeactivate() never runs for it — this is the one place that has to use the browser's own
   *  confirmation prompt instead (no custom dialog is possible here; per spec, `returnValue` must be
   *  set and the browser supplies its own generic copy). */
  @HostListener('window:beforeunload', ['$event'])
  protected onBeforeUnload(event: BeforeUnloadEvent): void {
    if (!this.editorRef?.hasUnsavedChanges()) return;
    event.preventDefault();
    event.returnValue = '';
  }
}

/**
 * Works around a confirmed backend defect: an exad_gwd_ir_header INSERT/UPDATE can genuinely
 * succeed while the HTTP response still comes back as a 500, with the real success payload
 * embedded inside the error envelope's own `message` field instead of a proper 2xx — e.g.
 * `{"statusCode":500,"error":true,"message":"{\"status\":\"SUCCESS\",\"ir_seq\":9024}","data":[]}`.
 * Matches only that exact signature (a JSON-parseable message shaped like
 * `{status:"SUCCESS", ir_seq:number}`) and returns the REAL ir_seq the backend actually
 * assigned — never a client-minted placeholder — so the record's identity stays correct and
 * stable across resaves. Any other failure (a genuine validation error, network failure, a
 * differently-shaped message) falls through unmatched and is re-thrown as a real error.
 * Remove once the backend returns a proper 2xx for a save that actually succeeded.
 */
function extractIrSeqFromMisreportedSuccess(err: unknown): number | null {
  if (!(err instanceof HttpErrorResponse)) return null;
  const body: unknown = err.error;
  const message = body && typeof body === 'object' ? (body as { message?: unknown }).message : undefined;
  if (typeof message !== 'string') return null;

  try {
    const parsed: unknown = JSON.parse(message);
    const status = parsed && typeof parsed === 'object' ? (parsed as { status?: unknown }).status : undefined;
    const irSeq = parsed && typeof parsed === 'object' ? (parsed as { ir_seq?: unknown }).ir_seq : undefined;
    return status === 'SUCCESS' && typeof irSeq === 'number' ? irSeq : null;
  } catch {
    return null;
  }
}

    // errorInterceptor's app-wide catchError would show a second, identical one for the same error.

    // brand-new IR (the backend mints one), otherwise the already-known ir_seq of the record being

    // back a fresh ir_seq on that INSERT branch; an UPDATE's `p_return_value` is just the literal
    // `SUCCESS`, no ir_seq, because the caller already has it — see WwellIrSaveResult. Falling back
    // to `knownIrSeq` (not a random mint) below is what keeps a resaved record's identity stable.
    const knownIrSeq = irBulkPayload[0].data?.[0]?.ir_seq ?? null;

    return this.http
      .post<ApiResponse<WwellIrSaveResult>>(`${this.apiUrl}/integrated-requirements`, irBulkPayload, {
        context: new HttpContext().set(SUPPRESS_GLOBAL_ERROR, true),
      })
      .pipe(
        map(res => res.data[0]?.irSeq ?? knownIrSeq),
        catchError((err: unknown) => {
          const realIrSeq = extractIrSeqFromMisreportedSuccess(err);
          return realIrSeq !== null ? of(realIrSeq) : throwError(() => err);
        }),
      );
  }

  /**
   * Deletes the whole IR (header row) by ir_seq. exad_agwa_iud_pkg.save_header's DELETE arm
   * removes just that one row; casing/tops/hydrogeology/water cascade automatically via their
   * `ir_seq ... ON DELETE CASCADE` FK (01-schema.sql), so nothing else needs to be sent — no
   * per-block bulk-save payload here, unlike createIr() above.
   */
  deleteIr(irSeq: number): Observable<void> {
    return this.http
      .delete<ApiResponse<never>>(`${this.apiUrl}/integrated-requirements/${irSeq}`, {
        context: new HttpContext().set(SUPPRESS_GLOBAL_ERROR, true),
      })
      .pipe(map(() => undefined));
  }

  /**
   * Next unique SAP workflow sequence (EXAD_AGWA_QUERIES_PKG.get_wf_next_id_fnc,
   * docs/ir/get_wf_next_id_fnc.sql) — a raw DB sequence value the caller sends to SAP as-is for
   * `workflow_id` and stores as-is via updateIrSapWorkflow below. Live in agwa-planning
   * (SapWorkflowResource) — no offline fallback: a failure here must stop
   * WwellIrStore.submitIrForReview before it mints a fabricated id, not silently hand back one
   * that was never really reserved.
   */
  getNextSapWorkflowSeq(): Observable<number> {
    return this.http.get<ApiResponse<number>>(`${this.apiUrl}/sap-workflow/next-id`).pipe(map(res => res.data[0]));
  }

  /**
   * Persists the SAP workflow id/status onto the IR header
   * (EXAD_AGWA_IUD_PKG.update_ir_sap_wf_prc, docs/ir/update_ir_sap_wf_prc.sql) once the SAP AAE
   * trigger call has been attempted — called on both outcomes (see WwellIrStore.submitIrForReview)
   * so the header always reflects whether the push to SAP actually succeeded. Live in
   * agwa-planning (IntegratedRequirementResource.updateSapWorkflow) — no offline fallback: a
   * failure here is a real, visible error (WwellIrStore's sapTriggerError), not a silently
   * swallowed one, since it would otherwise mask the header never actually getting stamped.
   */
  updateIrSapWorkflow(irSeq: number, wfId: number, wfStatus: string): Observable<void> {
    return this.http
      .patch<ApiResponse<never>>(`${this.apiUrl}/integrated-requirements/${irSeq}/sap-workflow`, { wfId, wfStatus })
      .pipe(map(() => undefined));
  }
}
