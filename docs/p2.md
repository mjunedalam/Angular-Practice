    reader.onload = () => resolve(reader.result as ArrayBuffer);

    const link = document.createElement('a');
    link.href = url;
    link.download = `${this.ir().code}-letter.pdf`;
    link.click();
    URL.revokeObjectURL(url);

   *  already fetches attachment bytes from the same S3 proxy for every other attachment operation
   *  (view/upload/delete), so there's nothing left for agwa-planning to add server-side. Each
   *  attachment is fetched independently so one unreadable file is dropped without losing the
   *  others, and the merge step itself falls back to the unmerged letter on any failure — a broken
   *  appendix should never turn a working download into a failed one. */
  private async withAttachmentsMerged(letterBlob: Blob): Promise<Blob> {
    const attachments = this.ir().attachments;
    if (attachments.length === 0) return letterBlob;

    try {
      const ordered: IrAttachment[] = WWELL_IR_ATTACHMENT_SLOTS.flatMap(config => irAttachmentsFor(attachments, config.slot));
      const fetched = await Promise.all(ordered.map(attachment => this.fetchAttachmentFile(attachment)));
      const files = fetched.filter((file): file is IrLetterAttachmentFile => file !== null);
      const merged = await appendAttachmentsToLetter(await blobToArrayBuffer(letterBlob), files);
      return new Blob([new Uint8Array(merged)], { type: 'application/pdf' });
    } catch {
      return letterBlob;
    }
  }

  private async fetchAttachmentFile(attachment: IrAttachment): Promise<IrLetterAttachmentFile | null> {
    try {
      const file = await firstValueFrom(this.s3.download(attachment.objectKey, attachment.fileName));
      return { label: attachment.label, contentType: attachment.contentType, bytes: await blobToArrayBuffer(file) };
    } catch {
      return null;
    }
  }

  /** `downloadLetterPdf` requests `responseType: 'blob'` so a successful PDF response can be
   *  turned straight into a download link — but Angular's HttpClient applies that same
   *  responseType to error bodies too, so even agwa-planning's JSON error envelope
   *  (`{ message: "Cannot generate IR letter: missing required field(s): ..." }`) arrives as an
   *  unparsed Blob instead of a plain object. Without this, the backend's specific reason (e.g.
   *  a missing Letter Number) was silently discarded in favor of a generic failure message.
   *  Reads it via `FileReader` rather than `Blob.text()` — the latter isn't implemented by
   *  jsdom (this repo's test environment), while `FileReader.readAsText` is. */
  private extractDownloadErrorMessage(err: unknown): Promise<string | null> {
    if (!(err instanceof HttpErrorResponse) || !(err.error instanceof Blob)) return Promise.resolve(null);
    return new Promise(resolve => {
      const reader = new FileReader();
      reader.onload = () => resolve(this.parseErrorMessage(reader.result));
      reader.onerror = () => resolve(null);
      reader.readAsText(err.error as Blob);
    });
  }

  private parseErrorMessage(text: FileReader['result']): string | null {
    if (typeof text !== 'string') return null;
    try {
      const parsed: unknown = JSON.parse(text);
      const message = parsed && typeof parsed === 'object' ? (parsed as { message?: unknown }).message : undefined;
      return typeof message === 'string' ? message : null;
    } catch {
      return null;
    }
  }

  /** Header stat: how many of the editor's 7 rail sections (Wellbore, General, Horizons, Casing,
   *  Hydrogeology, Area Water Wells, Attachments) hold meaningful data, per the same
   *  wwellIrTimelineStepStatus completeness rule the editor's Timeline view already uses — so this
   *  view's "N/7 sections" reading always agrees with what Timeline mode shows for the same record. */
  protected readonly sectionProgress = computed(() => {
    const ir = this.ir();
    const total = WWELL_IR_SECTION_RAIL.length;
    const completed = WWELL_IR_SECTION_RAIL.filter(group => wwellIrTimelineStepStatus(ir, group.key) === 'completed').length;
    return { completed, total, percent: Math.round((completed / total) * 100) };
  });

  protected setTab(tab: WwellIrViewTab): void {
    this.activeTab.set(tab);
  }
}

import { IntegratedRequirement, IntegratedRequirementStatus } from '@models/wwell-planning/integrated-requirement.model';
import { WWELL_STATUS_COLORS, WwellStatusMeta, wwellStatusMeta } from '../shared/wwell-status.config';

/** SAP text that means the workflow was actively killed rather than merely progressing — matched
 *  case-insensitively, the same way wwell-ir-approval.mapper.ts's stepStatus() reads a single
 *  step's own `response`. SAP's exact wording for these isn't part of our contract (the /display
 *  response only documents `wf_status_text` as free text), so this matches on the words themselves
 *  rather than a fixed value list. */
const SAP_TERMINATED_STATUS_TEXT = /reject|cancel|recall/i;
const SAP_APPROVED_STATUS_TEXT = /approve/i;
const SAP_RECALLED_STATUS_TEXT = /recall/i;

/** Every status renders the same view+inline-edit surface (`WwellIrEditorComponent`) now — this
 *  list instead gates the one status-transition action that still needs a distinct status ("Submit
 *  for Review" only makes sense while the record hasn't already been submitted). */
export const WWELL_IR_SUBMITTABLE_STATUSES: readonly IntegratedRequirementStatus[] = ['draft', 'in-progress'];

export const WWELL_IR_STATUS_META: Record<IntegratedRequirementStatus, WwellStatusMeta> = {
  draft: wwellStatusMeta('Draft', WWELL_STATUS_COLORS.slate),
  'in-progress': wwellStatusMeta('In Progress', WWELL_STATUS_COLORS.blue),
  'submit-for-review': { label: 'Submit for Review', bg: 'var(--wp-accent-fill)', color: 'var(--wp-on-accent)', dot: 'var(--wp-on-accent)' },
  'in-review': wwellStatusMeta('In Review', WWELL_STATUS_COLORS.purple),
  approved: wwellStatusMeta('Approved', WWELL_STATUS_COLORS.green),
};

/** True once a record's status badge has something real to show: `status === 'approved'` (a real
 *  fact off publish_flg) or a fetched `sapWfStatusText` (SAP's own current text, or the local
 *  "in progress" placeholder submitIrForReview writes — see WwellIrStore.loadIrApprovalTimeline /
 *  WWELL_IR_SAP_WF_STATUS_IN_PROGRESS). Everything else is the list endpoint's blind "in-progress"
 *  default (no sap_wf_id column to tell a genuine draft from an already-submitted record apart —
 *  see wwell-ir-summary.mapper.ts) and must not be shown as though it were real. Shared by the
 *  list badge, the detail header badge (editor + view), and nowhere else — so all three agree. */
export function hasKnownIrStatus(ir: IntegratedRequirement): boolean {
  return ir.status === 'approved' || ir.sapWfStatusText !== null;
}

/** The status badge's actual colour+label — same fallback rule as the label alone used to follow
 *  (`ir.sapWfStatusText ?? statusMeta[ir.status].label`), but the colour now follows the SAP text
 *  too instead of always trailing `ir.status`. `IntegratedRequirementStatus` has no slot for SAP's
 *  own terminal states (Rejected/Cancelled/Recalled all just leave `ir.status` at whatever it was
 *  pre-submission — see WwellIrStore.loadIrApprovalTimeline), so without this a rejected/cancelled/
 *  recalled record kept showing its old "In Review"/"Submit for Review" colour with only the label
 *  text changing underneath it. Recognised SAP text is shown red (terminated) or green (approved);
 *  anything else keeps riding on `ir.status`'s own colour, only the label swaps, same as before. */
export function irBadgeMeta(ir: IntegratedRequirement): WwellStatusMeta {
  const text = ir.sapWfStatusText;
  const fallback = WWELL_IR_STATUS_META[ir.status];
  if (!text) return fallback;
  if (SAP_TERMINATED_STATUS_TEXT.test(text)) return wwellStatusMeta(text, WWELL_STATUS_COLORS.red);
  if (SAP_APPROVED_STATUS_TEXT.test(text)) return wwellStatusMeta(text, WWELL_STATUS_COLORS.green);
  return { ...fallback, label: text };
}

/** True once SAP's own text says the workflow is done moving — rejected, cancelled, recalled, or
 *  approved — same two patterns irBadgeMeta colours red/green. Gates the editor's "Recall" button
 *  (WwellIrEditorComponent.canRecallIr): a workflow already at a terminal SAP state has nothing left
 *  to recall, same as a real SAP recall would refuse it. */
export function isSapWorkflowTerminal(sapWfStatusText: string | null): boolean {
  return sapWfStatusText !== null && (SAP_TERMINATED_STATUS_TEXT.test(sapWfStatusText) || SAP_APPROVED_STATUS_TEXT.test(sapWfStatusText));
}

/** True only for the recalled subset of `isSapWorkflowTerminal` — unlike Approved/Rejected/Cancelled,
 *  a recall was the user's own action, so the record isn't done: `WwellIrEditorComponent` brings
 *  "Submit for Review" back for it (`canSubmitForReview`), letting them resubmit fresh under a new
 *  SAP workflow id. Deliberately its own predicate rather than reusing `isSapWorkflowTerminal` — the
 *  other three terminal outcomes stay terminal, with no way back from the editor. */
export function isSapWorkflowRecalled(sapWfStatusText: string | null): boolean {
  return sapWfStatusText !== null && SAP_RECALLED_STATUS_TEXT.test(sapWfStatusText);
}

/** Classifies the live SAP status text into the IR letter PDF's three stamp states — the exact
 *  same two patterns `irBadgeMeta` uses to colour the badge red/green, reused here so this is the
 *  one place that decision gets made. Sent directly to agwa-planning (see
 *  `WwellIrService.downloadLetterPdf`'s `stampStatus` param / `LetterPdfRequest.stampStatus`) so
 *  the backend just stamps whatever it's told rather than keeping its own copy of this
 *  classification. `null` when the IR was never submitted (no SAP text at all) — the backend
 *  skips the stamp image entirely in that case instead of defaulting to "Pending". */
export function resolveLetterStampStatus(ir: IntegratedRequirement): 'APPROVED' | 'REJECTED' | 'PENDING' | null {
  const text = ir.sapWfStatusText;
  if (!text) return null;
  if (SAP_TERMINATED_STATUS_TEXT.test(text)) return 'REJECTED';
  if (SAP_APPROVED_STATUS_TEXT.test(text)) return 'APPROVED';
  return 'PENDING';
}

/** Every field the IR form schema marks `required` (wwell-ir-form.schema.ts), outside the
 *  repeatable rows — Casing/Formation Tops/Offset Wells only validate a row once one exists, so
 *  there's no top-level requirement that those sections hold any rows at all. Gates "Submit for
 *  Review": checked against the already-saved `ir` itself, not a live, possibly-still-open form,
 *  because submitting sends exactly this record (WwellIrEditorComponent.submitForReview calls
 *  `store.submitIrForReview(this.ir())`) — an in-progress edit elsewhere doesn't change whether
 *  the saved record is actually ready. The one attachment checked is the Location Map: it must be
 *  uploaded before the record can go for review (Initiation Request is not part of this gate). */
export function wwellIrHasRequiredSubmitFields(ir: IntegratedRequirement): boolean {
  return (
    !!ir.requestReference.linkedRequestId &&
    !!ir.requestReference.letterNumber?.trim() &&
    !!ir.requestReference.wellName.trim() &&
    !!ir.programDetails.completion.trim() &&
    !!ir.contacts.geologistPrimary.trim() &&
    !!ir.hydrogeology.aquifer.trim() &&
    !!ir.testProgram.testType &&
    ir.attachments.some(attachment => attachment.slot === 'location-map')
  );
}

/** The live SAP workflow as printed on the downloaded letter's "SAP Approval Workflow" section. */
export interface WwellIrLetterWorkflow {
  readonly workflowId: string | null;
  readonly workflowStatusText: string | null;
  readonly workflowSteps: readonly { readonly role: string; readonly name: string; readonly status: string; readonly date: string | null }[];
}

/** Sends exactly what the editor/view already show for the record (the chain and status text are
 *  refreshed from SAP on selection), so the letter matches the screen. `null` when the IR was never
 *  submitted — there is no workflow to print yet. */
export function resolveLetterWorkflow(ir: IntegratedRequirement): WwellIrLetterWorkflow | null {
  if (ir.sapWfId == null) return null;
  return {
    workflowId: String(ir.sapWfId),
    workflowStatusText: ir.sapWfStatusText,
    workflowSteps: ir.chain.map(({ role, name, status, date }) => ({ role, name, status, date })),
  };
}

/**
 * Works around a confirmed backend defect: an exad_gwd_ir_header INSERT/UPDATE can genuinely
 * succeed while the HTTP response still comes back as a 500, with the real success payload
 * embedded inside the error envelope's own `message` field instead of a proper 2xx — e.g.
 * `{"statusCode":500,"error":true,"message":"{\"status\":\"SUCCESS\",\"ir_seq\":9024}","data":[]}`.
 * Matches only that exact signature (a JSON-parseable message shaped like
 * `{status:"SUCCESS", ir_seq:number}`) and returns the REAL ir_seq the backend actually
 * assigned — never a client-minted placeholder — so the record's identity stays correct and
 * stable across resaves. Any other failure (a genuine validation error, network failure, a
 * differently-shaped message) falls through unmatched and is re-thrown as a real error.
 * Remove once the backend returns a proper 2xx for a save that actually succeeded.
 */
function extractIrSeqFromMisreportedSuccess(err: unknown): number | null {
  if (!(err instanceof HttpErrorResponse)) return null;
  const body: unknown = err.error;
  const message = body && typeof body === 'object' ? (body as { message?: unknown }).message : undefined;
  if (typeof message !== 'string') return null;

  try {
    const parsed: unknown = JSON.parse(message);
    const status = parsed && typeof parsed === 'object' ? (parsed as { status?: unknown }).status : undefined;
    const irSeq = parsed && typeof parsed === 'object' ? (parsed as { ir_seq?: unknown }).ir_seq : undefined;
    return status === 'SUCCESS' && typeof irSeq === 'number' ? irSeq : null;
  } catch {
    return null;
  }
}

   * whatever a user picks or types is saved on the IR itself, never back to this table.
   */
  getDropdownMenus(): Observable<Record<string, string[]>> {
    return this.http.get<ApiResponse<{ menuName: string; optionValue: string }>>(`${this.apiUrl}/dropdown-menus`).pipe(
      map(res =>
        res.data.reduce<Record<string, string[]>>((byMenu, row) => {
          (byMenu[row.menuName] ??= []).push(row.optionValue);
          return byMenu;
        }, {}),
      ),
    );
  }

  /**
   * ir_seq is the real primary key once an IR is saved — selecting a row in the list passes it
   * here to fetch the full { table, data } detail (casing/tops/hydrogeology) for that one record.
   * No offline-mock fallback here (unlike the other reads above): faking detail for a specific,
   * already-known-real record would misrepresent it rather than degrade gracefully, so a failure
   * surfaces as a real error instead — see WwellIrStore.loadIrDetail.
   */
  getIrDetailBlocks(irSeq: number): Observable<readonly WwellIrDetailTableBlock[]> {
    return this.http
      .get<ApiResponse<WwellIrDetailTableBlock>>(`${this.apiUrl}/integrated-requirements/${irSeq}`)
      .pipe(map(res => res.data));
  }

  /**
   * Downloads the formal "Integrated Requirements" letter as a PDF — rendered server-side
   * (Qute + openhtmltopdf, agwa-planning's IrLetterService) from the same detail data
   * getIrDetailBlocks() above already reads, so the PDF always reflects the last-saved record
   * rather than whatever the Letter tab's own client-side view currently has in memory.
   */
  /** POST (not GET) because the wellbore diagram is rendered client-side (the same D3/SVG
   *  components the Letter tab already shows) and its PNG snapshot rides along as the request
   *  body — the backend has no drawing logic of its own for it anymore. `wellboreImageBase64` is
   *  omitted when the IR has no casing/tops data or the client-side capture failed; the letter
   *  still renders, just without that section. `stampStatus` is the already-resolved
   *  Approved/Rejected/Pending stamp state (see `resolveLetterStampStatus`, wwell-ir.config.ts) —
   *  sent as a plain value so agwa-planning just stamps whatever it's told instead of keeping its
   *  own copy of the SAP-text classification logic. `null` when the IR was never submitted.
   *  `workflow` is the live SAP workflow (id, status text, approval chain) printed as the letter's
   *  "SAP Approval Workflow" section — see `resolveLetterWorkflow`. */
  downloadLetterPdf(
    irSeq: number,
    wellboreImageBase64?: string,
    stampStatus?: string | null,
    workflow?: WwellIrLetterWorkflow,
  ): Observable<Blob> {
    return this.http.post(
      `${this.apiUrl}/integrated-requirements/${irSeq}/letter-pdf`,
      {
        wellboreImageBase64: wellboreImageBase64 ?? null,
        stampStatus: stampStatus ?? null,
        workflowId: workflow?.workflowId ?? null,
        workflowStatusText: workflow?.workflowStatusText ?? null,
        workflowSteps: workflow?.workflowSteps ?? null,
      },
      { responseType: 'blob' },
    );
  }

  /**
   * Returns only the real, backend-confirmed `ir_seq` — never a client-fabricated
   * `IntegratedRequirement` (that was this method's old `buildIrFromPayload`-based "optimistic
   * echo," removed because it kept showing whatever the user had just typed for fields with no
   * backing column at all — client, targetAquifer, targetDepthFt... (see
   * mergeIrDetailBlocksIntoIntegratedRequirement's doc comment for the full list) — with nothing to
   * ever correct it except a full page reload. WwellIrStore.createIr now follows this up with a
   * real GET (getIrDetailBlocks) and merges that onto the record's last known-real state instead.
   */
  createIr(irBulkPayload: WwellIrBulkSavePayload): Observable<number | null> {
    // The backend only accepts the raw exad_gwd_ir_* bulk-save array as the request body (see
    // wwell-ir-payload.model.ts / IntegratedRequirementResource.java's @Consumes(APPLICATION_JSON))
    // — it has no multipart handling and no columns for the richer domain fields below (client,
    // wellName, contacts, attachments...), so those stay client-side only until exad_agwa_iud_pkg's
    // header table grows to carry them. Sending them today just 415s (wrong Content-Type/body shape).
    //
    // No offline-fallback here (unlike getIrs()/getEligibleWellsForIr() above) — a failed save
    // must surface as a real, visible error (WwellIrStore.createIrError), not a fabricated
    // success. A genuine POST failure with no recoverable ir_seq (see extractIrSeqFromMisreportedSuccess
    // below for the one confirmed-defect exception) always re-throws rather than minting a random
    // ir_seq — a client-minted identity on real failure would let a resave silently re-INSERT under
    // a new fake id instead of UPDATE-ing the original row, orphaning the real backend record and
    // duplicating it in the list.
    //
    // SUPPRESS_GLOBAL_ERROR: WwellIrStore.createIr's own error handler (→ createIrError →
    // WwellIrComponent's notify.error) already shows a toast for a genuine failure — without this,
    // errorInterceptor's app-wide catchError would show a second, identical one for the same error.
    // The header row's own `ir_seq` is the identity this save was submitted under — `null` for a
    // brand-new IR (the backend mints one), otherwise the already-known ir_seq of the record being
    // updated. exad_agwa_iud_pkg.update_ir_data_prc (docs/ir/update_ir_data_prc.sql) only echoes
    // back a fresh ir_seq on that INSERT branch; an UPDATE's `p_return_value` is just the literal
    // `SUCCESS`, no ir_seq, because the caller already has it — see WwellIrSaveResult. Falling back
    // to `knownIrSeq` (not a random mint) below is what keeps a resaved record's identity stable.
    const knownIrSeq = irBulkPayload[0].data?.[0]?.ir_seq ?? null;

    return this.http
      .post<ApiResponse<WwellIrSaveResult>>(`${this.apiUrl}/integrated-requirements`, irBulkPayload, {
        context: new HttpContext().set(SUPPRESS_GLOBAL_ERROR, true),
      })
      .pipe(
        map(res => res.data[0]?.irSeq ?? knownIrSeq),
        catchError((err: unknown) => {
          const realIrSeq = extractIrSeqFromMisreportedSuccess(err);
          return realIrSeq !== null ? of(realIrSeq) : throwError(() => err);
        }),
      );
  }

  /**
   * Deletes the whole IR (header row) by ir_seq. exad_agwa_iud_pkg.save_header's DELETE arm
   * removes just that one row; casing/tops/hydrogeology/water cascade automatically via their
   * `ir_seq ... ON DELETE CASCADE` FK (01-schema.sql), so nothing else needs to be sent — no
   * per-block bulk-save payload here, unlike createIr() above.
   */
  deleteIr(irSeq: number): Observable<void> {
    return this.http
      .delete<ApiResponse<never>>(`${this.apiUrl}/integrated-requirements/${irSeq}`, {
        context: new HttpContext().set(SUPPRESS_GLOBAL_ERROR, true),
      })
      .pipe(map(() => undefined));
  }

  /**
   * Next unique SAP workflow sequence (EXAD_AGWA_QUERIES_PKG.get_wf_next_id_fnc,
   * docs/ir/get_wf_next_id_fnc.sql) — a raw DB sequence value the caller sends to SAP as-is for
   * `workflow_id` and stores as-is via updateIrSapWorkflow below. Live in agwa-planning
   * (SapWorkflowResource) — no offline fallback: a failure here must stop
   * WwellIrStore.submitIrForReview before it mints a fabricated id, not silently hand back one
   * that was never really reserved.
   */
  getNextSapWorkflowSeq(): Observable<number> {
    return this.http.get<ApiResponse<number>>(`${this.apiUrl}/sap-workflow/next-id`).pipe(map(res => res.data[0]));
  }

  /**
   * Persists the SAP workflow id/status onto the IR header
   * (EXAD_AGWA_IUD_PKG.update_ir_sap_wf_prc, docs/ir/update_ir_sap_wf_prc.sql) once the SAP AAE
   * trigger call has been attempted — called on both outcomes (see WwellIrStore.submitIrForReview)
   * so the header always reflects whether the push to SAP actually succeeded. Live in
   * agwa-planning (IntegratedRequirementResource.updateSapWorkflow) — no offline fallback: a
   * failure here is a real, visible error (WwellIrStore's sapTriggerError), not a silently
   * swallowed one, since it would otherwise mask the header never actually getting stamped.
   */
  updateIrSapWorkflow(irSeq: number, wfId: number, wfStatus: string): Observable<void> {
    return this.http
      .patch<ApiResponse<never>>(`${this.apiUrl}/integrated-requirements/${irSeq}/sap-workflow`, { wfId, wfStatus })
      .pipe(map(() => undefined));
  }
}
