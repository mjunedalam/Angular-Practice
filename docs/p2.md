export interface RawLinerEntry {
    readonly odSz: string;
    readonly topDepth: number;
    readonly botDepth: number;

/**
 * Depth-only liner list, sorted shallowest-first (ascending botDepth). Source priority:
 * IR data (EXAD_GWD_IR_CASING) drives the list when present; DRLG data (as-drilled) is
 * used on its own otherwise, and always overrides depths when both exist.
 */
export function resolveLinerDepths(casings: ICasingIR[], drlgCasings: IDrlgCsg[]): RawLinerEntry[] {
    return casings.length > 0
        ? rawLinersFromIR(casings, drlgCasings)
        : rawLinersFromDrlg(drlgCasings);
}

/**
 * docs/plan/logic.md `is7InchLiner`: a liner qualifies as the 7" completion target when
 * its own OD (DRLG_CSG.wLnrOdSz, preferred) or planned OD (EXAD_GWD_IR_CASING.csgSize,
 * only for csgType='Liner' rows) is 7". RawLinerEntry.odSz already resolves that same
 * priority (actual wLnrOdSz over planned csgSize) upstream in rawLinersFromIR/
 * rawLinersFromDrlg, and every entry here is already liner-typed, so the spec's OR
 * condition collapses to a single size check on the resolved odSz.
 */
export function is7InchLiner(liner: Pick<RawLinerEntry, 'odSz'>): boolean {
    return parseInchSize(liner.odSz) === SCREEN_LINER_SIZE_IN;
}

/**
 * Index of the liner matching is7InchLiner, searched shallowest-first — the first match
 * wins (docs/plan/logic.md Rule 3: `liners.findIndex(is7InchLiner)`). Null when none
 * qualify.
 */
function resolveTargetLinerIndex(liners: readonly RawLinerEntry[]): number | null {
    const idx = liners.findIndex(is7InchLiner);
    return idx === -1 ? null : idx;
}

export type LinerRuleKind = 'none' | 'single' | 'single-fallback' | 'multi-target' | 'multi-fallback';

export interface LinerRule {
    readonly kind: LinerRuleKind;
    readonly startDepth: number;
    /** Fixed structural bottom for 'single' and 'multi-target' (the target liner's own
     *  shoe). Null for 'none', 'single-fallback', and 'multi-fallback' — those are bounded
     *  by total/current/target depth, data this liner-only function doesn't have, so the
     *  caller resolves them. */
    readonly structuralEndDepth: number | null;
    /** The is7InchLiner match completion elements are drawn around — gets the screen
     *  texture when lsFlg='Y'. Null when no liner qualifies (kind is 'none',
     *  'single-fallback', or 'multi-fallback' — those get the generic Liner Screen shape
     *  instead, see completion.renderer's renderOpenHoleAndScreen). */
    readonly targetLiner: RawLinerEntry | null;
    /** The liner immediately above the interval start, used only to blend Open Hole's
     *  on-screen width between it and the outer casing (see resolveOhHalfWidth). */
    readonly boundaryLiner: RawLinerEntry | null;
}

/**
 * Implements docs/plan/logic.md Rules 1-3 over a depth-only liner list: which liner (if
 * any) is the 7" completion target, and the depth interval Open Hole / Gravel Pack /
 * Perforated Interval share as a result.
 */
export function resolveLinerRule(liners: readonly RawLinerEntry[], lastCasingShoe: number): LinerRule {
    // Rule 1: No Liner
    if (liners.length === 0) {
        return { kind: 'none', startDepth: lastCasingShoe, structuralEndDepth: null, targetLiner: null, boundaryLiner: null };
    }

    // Rule 2: Single Liner
    if (liners.length === 1) {
        const only = liners[0];
        if (!is7InchLiner(only)) {
            // Not the 7" screen liner: still an ordinary liner with open hole/completion
            // below it — draw from its own shoe down to total/current/target depth, same
            // as a no-match multi-liner well (docs/plan/logic.md leaves this case
            // unaddressed; this mirrors Rule 3's fallback rather than drawing nothing).
            return { kind: 'single-fallback', startDepth: only.botDepth, structuralEndDepth: null, targetLiner: null, boundaryLiner: only };
        }
        return { kind: 'single', startDepth: lastCasingShoe, structuralEndDepth: only.botDepth, targetLiner: only, boundaryLiner: only };
    }

    // Rule 3: Multiple Liners
    const targetIndex = resolveTargetLinerIndex(liners);
    if (targetIndex !== null) {
        const target = liners[targetIndex];
        const startDepth = targetIndex === 0 ? lastCasingShoe : liners[targetIndex - 1].botDepth;
        return {
            kind: 'multi-target',
            startDepth,
            structuralEndDepth: target.botDepth,
            targetLiner: target,
            boundaryLiner: targetIndex === 0 ? null : liners[targetIndex - 1],
        };
    }

    // Fallback: multiple liners, none match 7"
    const deepest = liners[liners.length - 1];
    return { kind: 'multi-fallback', startDepth: deepest.botDepth, structuralEndDepth: null, targetLiner: null, boundaryLiner: deepest };
}

function assignHalfWidths(baseHalfWidth: number, count: number): number[] {
    return Array.from({ length: count }, (_, i) =>
        Math.max(22, baseHalfWidth - (i + 1) * LINER_HW_DECREMENT)
    );
}

/**
 * Determines whether a liner qualifies as a 7" pre-perforated liner.
 *
 * Both conditions must pass:
 *   1. The liner's own OD size (DRLG_CSG.wLnrOdSz, preferred over the planned size) is 7".
 *   2. EXAD_GWD_WELL_DESIGN.perforatedFlg is 'Y' (case-insensitive).
 *
 * perforatedRemarks is free text and only ever used for label copy — it must never gate
 * visibility, otherwise perforatedFlg='Y' with a blank/missing remarks field would silently
 * hide a feature the well design explicitly opted into.
 */
function isPrePerforatedLiner(
    odSz: string | null | undefined,
    wellDesign: IWellDesign | null | undefined,
): boolean {
    if (parseInchSize(odSz) !== SCREEN_LINER_SIZE_IN) return false;
    return (wellDesign?.perforatedFlg ?? '').trim().toUpperCase() === 'Y';
}

/** Attaches halfWidth + isScreen/isPrePerfLiner render flags to a depth-only liner list. */
function attachRenderFlags(
    raw: RawLinerEntry[],
    layout: DiagramLayout,
    wellDesign?: IWellDesign | null,
): LinerEntry[] {
    const widths = assignHalfWidths(layout.baseHalfWidth, raw.length);
    const targetIndex = resolveTargetLinerIndex(raw);
    const lsFlgYes = (wellDesign?.lsFlg ?? '').trim().toUpperCase() === 'Y';

    return raw.map((liner, i) => ({
        ...liner,
        halfWidth: widths[i],
        isScreen: i === targetIndex && lsFlgYes,
        isPrePerfLiner: isPrePerforatedLiner(liner.odSz, wellDesign),
    }));
}

/**
 * Extracts liner entries from drlgCasings when no ICasingIR data is available.
 * Deduplicates by wLnrOdSz, sorts shallowest-first.
 */
function rawLinersFromDrlg(drlgCasings: IDrlgCsg[]): RawLinerEntry[] {
    const dedupMap = new Map<string, RawLinerEntry>();
    for (const d of drlgCasings) {
        if (!d.wLnrOdSz || (d.wLnrBotDepth ?? 0) <= 0) continue;
        const existing = dedupMap.get(d.wLnrOdSz);
        if (!existing || d.wLnrBotDepth! > existing.botDepth) {
            dedupMap.set(d.wLnrOdSz, {
                odSz: d.wLnrOdSz,
                topDepth: d.wLnrTopDepth ?? d.wCsgBotDpth,
                botDepth: d.wLnrBotDepth!,
                hasActual: true,
                hasActualTop: (d.wLnrTopDepth ?? 0) > 0,
            });
        }
    }

    return [...dedupMap.values()].sort((a, b) => a.botDepth - b.botDepth);
}

/**
 * Fallback top-of-liner anchor when no matching drlgCasing liner row exists: the shoe of
 * the last (deepest) non-liner casing. Per casing, prefer its actual (as-drilled)
 * DRLG_CSG shoe over its IR (planned) depth; only use the IR depth when no actual record
 * exists for that casing.
 *
 * When the IR is liner-only (no non-liner casing row at all — e.g. a standalone-liner
 * well), there's no IR casing to read a shoe from, so fall back to the deepest actual
 * non-liner casing record instead of defaulting straight to 0.
 */
export function resolveLastCasingShoe(casings: ICasingIR[], drlgCasings: IDrlgCsg[]): number {
    const nonLinerCasings = casings.filter(c => c.csgType !== 'Liner');

    if (nonLinerCasings.length > 0) {
        return nonLinerCasings.reduce((max, c) => {
            const actualShoe = drlgCasings.find(d => d.wCsgOdSz === c.csgSize)?.wCsgBotDpth ?? 0;
            return Math.max(max, actualShoe > 0 ? actualShoe : c.csgDepth);
        }, 0);
    }

    const actualCasingOnly = drlgCasings.filter(d => !d.wLnrOdSz && d.wCsgBotDpth > 0);
    if (actualCasingOnly.length > 0) {
        return Math.max(...actualCasingOnly.map(d => d.wCsgBotDpth));
    }
    return Math.max(0, ...drlgCasings.map(d => d.wCsgBotDpth).filter(v => v > 0));
}

/**
 * Extracts liner entries from ICasingIR + drlgCasings for actual depth override.
 *
 * Source priority per liner:
 *   - DRLG_CSG (actual/as-drilled) overrides depth when a matching wLnrOdSz exists.
 *   - EXAD_GWD_IR_CASING (planned) is the fallback for depth and the primary OD source.
 *
 * Extra DRLG_CSG liners with no corresponding IR entry are appended at the end.
 */
function rawLinersFromIR(casings: ICasingIR[], drlgCasings: IDrlgCsg[]): RawLinerEntry[] {
    // Sorted shallow-to-deep so the chaining below (each un-anchored liner hangs from the
    // one immediately above it) walks the string in physical order, matching
    // buildLinerTopAnchors' identical convention on the IR wizard side.
    const irLiners = [...casings.filter(c => c.csgType === 'Liner')].sort((a, b) => a.csgDepth - b.csgDepth);

    const lastCasingShoe = resolveLastCasingShoe(casings, drlgCasings);

    const matchedOdSizes = new Set<string>();

    // Only the shallowest liner without its own actual top falls back to lastCasingShoe.
    // Every liner after it hangs from the liner immediately above — carried via chainTop,
    // which always holds that previous liner's own resolved (actual-preferred, IR-fallback)
    // botDepth — otherwise two+ stacked liners with no actual data would all render starting
    // at the same depth (the last casing's shoe) instead of hanging from one another.
    let chainTop = lastCasingShoe;
    const irEntries: RawLinerEntry[] = irLiners.map(csg => {
        // Source priority: use DRLG_CSG depth when available; fall back to IR depth.
        const actual = drlgCasings.find(d => d.wLnrOdSz === csg.csgSize) ?? null;
        const hasActualBot = (actual?.wLnrBotDepth ?? 0) > 0;
        const hasActualTop = (actual?.wLnrTopDepth ?? 0) > 0;
        if (actual?.wLnrOdSz) matchedOdSizes.add(actual.wLnrOdSz);
        const botDepth = hasActualBot ? actual!.wLnrBotDepth! : csg.csgDepth;
        // `actual.wCsgBotDpth` is only a meaningful fallback (a real parent-casing shoe) when
        // it's genuinely positive — the wwell-ir wizard's synthetic anchors always hardcode it
        // to 0 (see buildLinerTopAnchors), so this correctly falls through to chainTop for them
        // instead of collapsing every un-anchored liner to depth 0.
        const fallbackTopDepth = (actual?.wCsgBotDpth ?? 0) > 0 ? actual!.wCsgBotDpth : chainTop;
        const topDepth = hasActualTop ? actual!.wLnrTopDepth! : fallbackTopDepth;
        chainTop = botDepth;
        return {
            odSz: actual?.wLnrOdSz ?? csg.csgSize,
            topDepth,
            botDepth,
            hasActual: hasActualBot,
            hasActualTop,
        };
    });

    const extraEntries: RawLinerEntry[] = drlgCasings
        .filter(d => d.wLnrOdSz != null && (d.wLnrBotDepth ?? 0) > 0 && !matchedOdSizes.has(d.wLnrOdSz!))
        .map(d => ({
            odSz: d.wLnrOdSz!,
            topDepth: d.wLnrTopDepth ?? d.wCsgBotDpth,
            botDepth: d.wLnrBotDepth!,
            hasActual: true,
            hasActualTop: (d.wLnrTopDepth ?? 0) > 0,
        }));

    return [...irEntries, ...extraEntries].sort((a, b) => a.botDepth - b.botDepth);
}

function renderLinerShapes({
    rootG,
    defsEl,
    layout,
    liners,
    centerX,
    scale,
    linersStart,
    lastCasingBotDepth,
}: RenderLinersOptions): void {
    const { shoeCurveOffset } = layout;

    liners.forEach((liner, i) => {
        const topPx = scale(liner.topDepth);
        const botPx = scale(liner.botDepth);
        const clipId = `dyn-liner-clip-${i}`;
        const clipRadius = openHoleHalfWidth(liner.halfWidth) + 20;
        const delay = (liners.length - 1 - i) * ANIM.CASING_STAGGER;
        const thisLinerStart = linersStart + delay;
        const casingPath = buildCasingPath(centerX, liner.halfWidth, topPx, botPx, shoeCurveOffset);

        // Planned/estimated liners share the casing look (mainGradient @ 0.40) so they
        // read as the same "planned" style as conductor/casing; actual liners keep the actual fill.
        const fill = liner.hasActual ? 'url(#actualLinerGradient)' : 'url(#linerGradient)';
        const capFill = liner.hasActual ? 'url(#linerCapDepthGradient)' : 'url(#plannedLinerCapDepthGradient)';
        const stroke = LINER_WALL_COLOR;
        const opacity = liner.hasActual ? 0.78 : 0.40;

        const hangerHW = liner.halfWidth;
        const hangerY = (i === 0 ? scale(lastCasingBotDepth) : scale(liners[i - 1].botDepth)) + 4;
        const { left: lh, right: rh } = buildScreenHanger(centerX, hangerY, hangerHW, 14, 26);
        rootG.append('path')
            .attr('class', 'liner-hanger')
            .attr('d', `${lh} ${rh}`)
            .attr('fill', '#000000')
            .attr('stroke', '#000000')
            .attr('stroke-width', '1')
            .attr('stroke-linejoin', 'round')
            .style('opacity', 0)
            .transition()
            .delay(thisLinerStart)
            .duration(Math.round(ANIM.CASING_DURATION / 3))
            .ease(easeCubicInOut)
            .style('opacity', 1);

        const clipRect = defsEl.append('clipPath')
            .attr('class', 'dyn-clip')
            .attr('id', clipId)
            .append('rect')
            .attr('x', centerX - clipRadius - 15)
            .attr('y', topPx - 5)
            .attr('width', clipRadius * 2 + 30)
            .attr('height', 0);

        clipRect.transition()
            .delay(thisLinerStart)
            .duration(ANIM.CASING_DURATION)
            .ease(easeCubicInOut)
            .attr('height', botPx - topPx + shoeCurveOffset + 20);

        rootG.append('path')
            .attr('class', liner.isScreen ? 'liner liner--screen' : 'liner')
            .attr('d', casingPath)
            .attr('clip-path', `url(#${clipId})`)
            .attr('fill', fill)
            .attr('stroke', stroke)
            .attr('stroke-width', '2')
            .style('opacity', opacity);

        rootG.append('ellipse')
            .attr('class', 'liner-top-ellipse')
            .attr('cx', centerX)
            .attr('cy', topPx)
            .attr('rx', liner.halfWidth)
            .attr('ry', 4)
            .attr('clip-path', `url(#${clipId})`)
            .attr('fill', capFill)
            .attr('stroke', stroke)
            .attr('stroke-width', '2')
            .style('opacity', opacity);

        // Darken the part of the liner that is inside the casing to simulate interior shadowing
        const exitPx = i === 0 ? scale(lastCasingBotDepth) : scale(liners[i - 1].botDepth);
        if (exitPx > topPx) {
            // Very light inner dimming, mostly relying on the glass overlay for shadows now
            rootG.append('rect')
                .attr('class', 'liner-interior-shadow')
                .attr('x', centerX - liner.halfWidth)
                .attr('y', topPx)
                .attr('width', liner.halfWidth * 2)
                .attr('height', exitPx - topPx)
                .attr('fill', '#0f172a')
                .style('opacity', 0.15)
                .attr('clip-path', `url(#${clipId})`);

            // Render the actual casing 3D gradient over the liner at semi-transparency.
            // This perfectly simulates the front wall of the casing acting as a glass layer.
            rootG.append('rect')
                .attr('class', 'liner-glass-reflection')
                .attr('x', centerX - liner.halfWidth)
                .attr('y', topPx)
                .attr('width', liner.halfWidth * 2)
                .attr('height', exitPx - topPx)
                .attr('fill', 'url(#mainGradient)')
                .style('opacity', 0.45)
                .attr('clip-path', `url(#${clipId})`);

            // Shadow cast by the casing shoe onto the liner as it exits the hole
            rootG.append('ellipse')
                .attr('cx', centerX)
                .attr('cy', exitPx)
                .attr('rx', liner.halfWidth + 8)
                .attr('ry', 4)
                .attr('fill', 'none')
                .attr('stroke', '#020617')
                .attr('stroke-width', '5')
                .style('opacity', 0.6)
                .attr('clip-path', `url(#${clipId})`);
        }

        if (liner.isScreen) {
            const screenStartPx = topPx + (botPx - topPx) * SCREEN_BLANK_RATIO;
            rootG.append('rect')
                .attr('class', 'liner liner--screen-grid')
                .attr('x', centerX - liner.halfWidth + 3)
                .attr('y', screenStartPx)
                .attr('width', liner.halfWidth * 2 - 6)
                .attr('height', botPx - screenStartPx)
                .attr('clip-path', `url(#${clipId})`)
                .attr('fill', 'url(#gridPattern)')
                .attr('fill-opacity', 0.92)
                .attr('stroke', 'none');
        }
    });
}

function renderLinerLabels({
    rootG,
    liners,
    centerX,
    scale,
    labelsStart,
    animation,
}: RenderLinersOptions): void {
    const innerDelay = labelsStart + (animation.lineStyle === 'draw' ? animation.lineDur : 0);

    liners.forEach((liner) => {
        const topPx = scale(liner.topDepth);
        const botPx = scale(liner.botDepth);
        const midY = topPx + (botPx - topPx) * 0.5;
        const rEdge = centerX + liner.halfWidth;
        const lEnd = rEdge + 26;
        const labelX = lEnd + 8;
        const screenTag = liner.isScreen ? ' (Screen)' : '';
        const prePerfTag = liner.isPrePerfLiner ? ' (Pre Perf Liner)' : '';
        const actualStar = liner.hasActual ? ' *' : '';

        // Any confirmed actual (as-drilled) liner gets the same green label, whether or not
        // it's the 7" screen liner — isScreen no longer forces a separate darker shade.
        const primaryFill = liner.hasActual ? '#22c55e' : '';

        const lg = rootG.append('g').attr('class', 'casing-label');
        lg.append('line').attr('class', 'casing-label__line')
            .attr('x1', rEdge).attr('x2', lEnd).attr('y1', midY).attr('y2', midY);
        lg.append('path').attr('class', 'casing-label__arrow')
            .attr('d', buildArrowHeadRight(lEnd, midY, 6));
        const primaryLabel = lg.append('text').attr('class', 'casing-label__primary')
            .attr('fill', primaryFill)
            .attr('x', labelX).attr('y', midY + 4)
            .text(`${liner.odSz}" Liner @ ${liner.botDepth.toLocaleString()}${screenTag}${prePerfTag}${actualStar}`);
        if (liner.isScreen) primaryLabel.style('font-weight', '900');

        animateLabel(lg, labelsStart, animation);

        // The screen liner's own grid-pattern fill plus the side callout above already
        // identify it as "7" ... (Screen)" — the inner size text on top of that grid
        // pattern is redundant clutter, so skip it when lsFlg='Y' flagged this as the screen.
        if (!liner.isScreen) {
            const innerFill = liner.hasActual ? '#0f2d1a' : '';
            const innerWeight = liner.hasActual ? '800' : 'normal';
            rootG.append('text').attr('class', 'casing-size-inner')
                .attr('x', centerX).attr('y', midY)
                .attr('text-anchor', 'middle')
                .style('opacity', 0)
                .attr('fill', innerFill)
                .style('font-weight', innerWeight)
                .text(`${liner.odSz}"`)
                .transition().delay(innerDelay).duration(animation.fadeDur).ease(easeCubicInOut)
                .style('opacity', 1);
        }

        if (liner.hasActualTop) {
            // TOL (top-of-liner) label on the LEFT side to avoid colliding with the
            // right-side OD/liner labels. Only rendered when topDepth is a genuine top-of-liner
            // value (as-drilled DRLG_CSG data, or an explicitly entered IR TOL) — never for a
            // computed attach-point fallback (last casing/liner shoe), which would misrepresent
            // a guess as a measured depth.
            const lEdge = centerX - liner.halfWidth;
            const tolEnd = lEdge - 56;
            const tolLg = rootG.append('g').attr('class', 'casing-label');
            tolLg.append('line').attr('class', 'casing-label__line')
                .attr('x1', lEdge).attr('x2', tolEnd).attr('y1', topPx).attr('y2', topPx);
            tolLg.append('path').attr('class', 'casing-label__arrow')
                .attr('d', buildArrowHeadLeft(tolEnd, topPx, 6));
            tolLg.append('text').attr('class', 'casing-label__primary')
                .attr('text-anchor', 'end')
                .attr('x', tolEnd - 8).attr('y', topPx + 4)
                .text(`TOL @ ${liner.topDepth.toLocaleString()}`);
            animateLabel(tolLg, labelsStart, animation);
        }

    });
}

import { IntegratedRequirement } from '@models/wwell-planning/wwell-planning.model';

/** Flags a value that's actually meaningful, not just informational — e.g. a nonzero H2S
 *  reading should read as amber/danger regardless of which section it sits in. Omitted (or
 *  `'default'`) renders as a plain field. */
export type RowTone = 'default' | 'warning' | 'danger' | 'success';

export interface KvRow {
  readonly label: string;
  readonly value: string;
  readonly tone?: RowTone;
  /** Renders `value` as a small pill instead of plain text — for fields whose value is one of
   *  a fixed set of options (a completion type, a test type, ...), where a badge reads faster
   *  than inline text at a glance. */
  readonly badge?: boolean;
}

export const EMPTY_VALUE = '—';

export function fmtNum(value: number | null | undefined, unit?: string): string {
  if (value === null || value === undefined) return EMPTY_VALUE;
  return unit ? `${value} ${unit}` : `${value}`;
}

export function fmtStr(value: string | null | undefined): string {
  return value ? value : EMPTY_VALUE;
}

/** Mirrors the IR wizard's "Request Reference" step (wwell-ir-form.schema.ts) field-for-field,
 *  so the detail page's sections line up 1:1 with the stepper the data was captured through. */
export function buildRequestReferenceRows(ir: IntegratedRequirement): readonly KvRow[] {
  const r = ir.requestReference;
  return [
    { label: 'Link to Request', value: fmtStr(r.linkedRequestId) },
    { label: 'Letter Number', value: fmtStr(r.letterNumber) },
    { label: 'BI Number', value: r.client },
    { label: 'Supporting Business', value: r.supportingBusiness },
    { label: 'Water Well Name', value: r.wellName },
    { label: 'Target Aquifer', value: r.targetAquifer, badge: true },
    { label: 'Target Depth', value: fmtNum(r.targetDepthFt, 'ft') },
    { label: 'Surface Elevation', value: `${fmtNum(r.surfaceElevationFtAsl)} ft ASL` },
    { label: 'Coordinates', value: r.coordinates },
  ];
}

/** Mirrors the wizard's "Program Details" step — both its main fields and the nested
 *  Development/Testing subsection. `openHoleSizeIn` only ever means something when the
 *  completion type is 'Open Hole' (see the field's `visibleWhen` gate in the schema), so it's
 *  only included then rather than showing a dangling "—" the rest of the time. */
export function buildProgramDetailsRows(ir: IntegratedRequirement): readonly KvRow[] {
  const p = ir.programDetails;
  const rows: KvRow[] = [{ label: 'Completion', value: p.completion, badge: true }];

  if (p.completion === 'Open Hole') {
    rows.push({ label: 'Open Hole Size', value: fmtStr(p.openHoleSizeIn) });
  }

  rows.push(
    { label: 'Acidization', value: fmtStr(p.acidization), badge: true },
    { label: 'Transmission', value: fmtStr(p.transmission), badge: true },
    { label: 'Logging', value: fmtStr(p.developmentTesting.logging), badge: true },
    { label: 'Mud Logging', value: fmtStr(p.developmentTesting.mudLogging), badge: true },
    { label: 'Sampling', value: fmtStr(p.developmentTesting.sampling), badge: true },
    { label: 'Flow Test', value: fmtStr(p.developmentTesting.flowTest), badge: true },
  );
  return rows;
}

/** Mirrors the wizard's "Estimated Hydrology" step, including its test-type-conditional
 *  fields (SIWHP for a flow test; water level + pump depth for a pump test) — see the
 *  matching `visibleWhen` gates on `siwhpPsi`/`estimatedWaterLevelFt`/`estimatedPumpDepthFt`
 *  in wwell-ir-form.schema.ts. Any nonzero H2S gets `tone: 'warning'` — a toxic-gas reading is
 *  worth flagging on sight, not just listed at the same visual weight as TDS/rate. */
export function buildHydrogeologyRows(ir: IntegratedRequirement): readonly KvRow[] {
  const isFlowTest = ir.testProgram.testType === 'flow-test';
  const h2sPpm = ir.hydrogeology.h2sPpm;
  const rows: KvRow[] = [
    { label: 'Aquifer', value: ir.hydrogeology.aquifer, badge: true },
    { label: 'Test Type', value: isFlowTest ? 'Flow Test' : 'Pump Test', badge: true },
    { label: 'Estimated TDS', value: fmtNum(ir.hydrogeology.estimatedTdsPpm, 'ppm') },
    { label: 'Estimated Rate', value: fmtNum(ir.testProgram.estimatedRateGpm, 'GPM') },
    {
      label: 'H2S',
      value: fmtNum(h2sPpm, 'ppm'),
      tone: h2sPpm ? 'warning' : 'default',
    },
  ];

  if (isFlowTest) {
    rows.push({ label: 'SIWHP', value: fmtNum(ir.testProgram.siwhpPsi, 'psi') });
  } else {
    rows.push(
      { label: 'Estimated Water Level', value: fmtNum(ir.testProgram.estimatedWaterLevelFt, 'ft') },
      { label: 'Estimated Pump Depth', value: fmtNum(ir.hydrogeology.estimatedPumpDepthFt, 'ft') },
    );
  }
  return rows;
}

export function buildGeologistContactRows(ir: IntegratedRequirement): readonly KvRow[] {
  const c = ir.contacts;
  return [
    { label: 'Primary', value: c.geologistPrimary },
    { label: 'Secondary', value: fmtStr(c.geologistSecondary) },
  ];
}

import {
  IntegratedRequirement,
  IntegratedRequirementCasingRow,
  IntegratedRequirementFormationTop,
} from '@models/wwell-planning/wwell-planning.model';
import { ICasingIR } from '@shared/models/wwell/casing-ir.model';
import { IDrlgCsg } from '@shared/models/wwell/drlg-csg.model';
import { ITopsIR } from '@shared/models/wwell/tops-ir.model';
import { IWellDesign } from '@shared/models/wwell/well-design.model';
import { parseInchSize } from '@shared/utils/wellbore-math.util';
import {
  WwellIrFormRawValue,
  WwellIrRawCasingRow,
  WwellIrRawFormationTop,
  WwellIrRawProgramMain,
} from './wwell-ir-form.types';

/**
 * Adapts IR data into the `WellboreDiagramData` shape the shared `app-well-bore-view` component
 * (presentation/well-bore-view) expects — so both the wizard (live, still-being-edited raw form
 * value) and the read-only IR detail page (a submitted `IntegratedRequirement`) can embed that
 * component unmodified: the wizard as an always-animated "Well Design View" step
 * (`buildWellboreDiagramData`), the detail page as a static as-submitted diagram
 * (`buildWellboreDiagramDataFromIntegratedRequirement`).
 *
 * An Integrated Requirement is always a plan, never an as-drilled record, so no `drlgCasings`
 * entry produced here ever carries a real bottom depth (`wCsgBotDpth`/`wLnrBotDepth`).
 * `well-bore-view`'s renderers only switch a casing/liner to the green "actual" gradient when
 * a matching `drlgCasings` row has a real bottom depth — so every shape here always renders
 * with the steel/silver "planned" gradient, even though `drlgCasings` is no longer always
 * empty (see `buildLinerTopAnchors` below).
 */

const LINER_TYPE = 'Liner';

/**
 * A freshly added (or still-mid-edit) Casing/Formation Top row starts out with empty/null
 * field values — but `mat-select`/number inputs still exist and produce SOME value the moment
 * a user makes any single selection (e.g. just picking "Liner" from the Type dropdown before
 * touching Size or Depth). Mapping every row unconditionally meant a half-filled row already
 * rendered a shape/label on the diagram (a "Liner @ 0" annotation with no real size or depth
 * behind it) as soon as it existed, rather than once it actually described something real. These
 * gates require every field the diagram actually draws from before a row counts — reused by both
 * the Subsurface & Construction step's live preview and the Well Design View step (both call
 * `buildWellboreDiagramData`).
 */
function isRenderableCasingRow(row: WwellIrRawCasingRow): boolean {
  return !!row.type && !!row.sizeIn && row.depthFtBgl !== null && row.depthFtBgl !== undefined;
}

function isRenderableFormationTop(row: WwellIrRawFormationTop): boolean {
  return !!row.name && row.depthFt !== null && row.depthFt !== undefined;
}

const COMPLETION_OPEN_HOLE = 'Open Hole';
const COMPLETION_SCREENED = 'Screened';
const COMPLETION_SCREENED_GRAVEL_PACK = 'Screened + Gravel Pack';
const COMPLETION_LINER_PERFORATED = 'Liner + Perforated';

/** Anchors the depth scale to a stable minimum so a shallow/partial entry doesn't stretch
 * across the full drawing height, and so the scale's domain is never degenerate ([0, 0]) before the
 * (required, but non-linear-stepper-skippable) target depth field is filled in. */
const MIN_SCALE_DEPTH_FT = 500;

/**
 * Never used to match a real casing — `casing.renderer.ts` / `wellbore-depth.helpers.ts`
 * match `drlgCasings` rows to casings by `wCsgOdSz === csg.csgSize`, so this sentinel must be
 * a value no user-entered casing size could ever equal. Only `buildLinerTopAnchors`' rows use
 * this; they're read exclusively through `wLnrOdSz`, never `wCsgOdSz`.
 */
const LINER_ANCHOR_SENTINEL = '__wwell_ir_liner_anchor__';

/**
 * `casing.renderer.ts`'s `getCasingTier` derives each casing's on-screen half-width purely
 * from its position in this array via `indexOf` (tier 0 = narrowest, each later tier wider) —
 * it never reads `csgSize` for width, only for the text label. It also paints casings in this
 * same array order, later entries on top, so whichever entry ends up widest is also drawn last
 * — the standard "wide, shallow casing painted over a narrow, deep one" telescoping look.
 *
 * Sorting by SIZE (ascending — smallest first, so the largest entered size lands last/widest)
 * is what makes width actually track the number the user typed, rather than depth order. For
 * well-formed data (size decreasing with depth, real telescoping construction) this produces
 * the exact same order as depth-descending would — size and depth are inversely correlated
 * there, so there's no behavior change for the common case. It only diverges for physically
 * inverted rows (a deeper casing entered wider than a shallower one, flagged separately by
 * `findInvertedCasingSizeRows` in wwell-ir-casing-warnings.util.ts) — there, the wider-but-
 * deeper entry now paints last and can fully cover the narrower-but-shallower one in their
 * overlapping depth range, since two casings can't actually nest that way in reality either.
 * That's an unavoidable consequence of a genuinely invalid combination, not a rendering bug —
 * see `casing.renderer.ts` (off-limits to edit) for why: `buildCasingPath` always draws a
 * casing as a full tube from ground level (0) to its own depth, so the only way to reflect its
 * real entered size is via this array's order.
 *
 * Unparseable sizes sort as 0in (narrowest/first) — the safest fallback, since
 * `isRenderableCasingRow` already guarantees `sizeIn` is non-empty text, just not necessarily
 * a `parseInchSize`-recognizable number.
 */
function mapCasings(rows: readonly WwellIrRawCasingRow[]): ICasingIR[] {
  const casings: ICasingIR[] = rows.map(row => ({
    csgSize: row.sizeIn,
    csgType: row.type,
    csgDepth: row.depthFtBgl ?? 0,
    csgRemarks: row.remarks || null,
  }));
  return [...casings].sort((a, b) => (parseInchSize(a.csgSize) ?? 0) - (parseInchSize(b.csgSize) ?? 0));
}

/**
 * The wizard's "Top Of Liner (TOL)" field (`tolFtBgl`, shown only for Liner rows) is optional —
 * most IR liners never get one filled in. `liner.renderer.ts`'s `rawLinersFromIR` only treats
 * `wLnrTopDepth` as a genuine top-of-liner value (and shows the "TOL @ ..." label/tooltip row)
 * when `LinerEntry.hasActualTop` is true, which it derives straight from whether this anchor's
 * `wLnrTopDepth` is a real positive number — so this must leave `wLnrTopDepth` unset (`null`)
 * whenever the user hasn't entered one, rather than silently filling it with a computed guess
 * that would then get mislabeled as a real measured depth.
 *
 * This builds synthetic `IDrlgCsg` "anchor" entries carrying ONLY `wLnrTopDepth` (when
 * provided) per liner — never `wLnrBotDepth` or a real `wCsgOdSz` — because `rawLinersFromIR`
 * gates `hasActual` (and therefore the green-vs-silver color choice) strictly on
 * `wLnrBotDepth`, and gates the casing-actual lookup strictly on `wCsgOdSz`. Leaving both unset
 * keeps every casing and liner silver.
 *
 * When TOL is left blank, `rawLinersFromIR` falls back to its own `chainTop` chaining instead:
 * the shallowest liner hangs from the deepest structural casing's shoe; each deeper liner hangs
 * from the liner immediately above it (matching the physical assumption already baked into
 * `renderLinerShapes`'s `hangerY`). Casing rows whose type is Conductor, Surface, Intermediate,
 * or Production are all "structural" here — anything that isn't `'Liner'`, exactly matching how
 * casing.renderer.ts itself distinguishes casing from liner (`csgType !== 'Liner'`).
 *
 * An explicit `tolFtBgl` doesn't otherwise perturb that chain: the next liner's fallback top is
 * still this row's own `depthFtBgl` (its shoe), not its `tolFtBgl`, since a liner physically
 * hangs from the casing/liner above it regardless of how far down its own top was entered.
 */
function buildLinerTopAnchors(rows: readonly WwellIrRawCasingRow[]): IDrlgCsg[] {
  const linersShallowToDeep = rows
    .filter(row => row.type === LINER_TYPE)
    .slice()
    .sort((a, b) => (a.depthFtBgl ?? 0) - (b.depthFtBgl ?? 0));

  return linersShallowToDeep.map(row => ({
    wCsgOdSz: LINER_ANCHOR_SENTINEL,
    wCsgBotDpth: 0,
    wLnrOdSz: row.sizeIn,
    wLnrTopDepth: row.tolFtBgl ?? null,
    wLnrBotDepth: null,
  }));
}

function mapGeologicTops(
  rows: readonly WwellIrRawFormationTop[],
  surfaceElevationFtAsl: number | null,
): ITopsIR[] {
  return rows.map(row => {
    const planTvdDepth = row.depthFt ?? 0;
    return {
      stLongCd: row.name,
      planTvdDepth,
      planSsDepth: surfaceElevationFtAsl !== null ? surfaceElevationFtAsl - planTvdDepth : 0,
    };
  });
}

/**
 * Derives completion-element flags/remarks from the "Completion" dropdown (Program
 * Details step) plus, for Open Hole, the conditionally-shown "Open Hole Size" field.
 * `well-bore-view`'s renderers gate Open Hole strictly on a parseable inch size in
 * `ohRemarks`, and Gravel Pack / Liner Screen strictly on non-empty remarks text — the
 * constant fallback labels below exist only to satisfy those gates when the dropdown
 * implies the element but no free-text remarks field exists for it in this form.
 *
 * "Liner + Perforated" only actually renders once a Casing-table Liner row is exactly 7"
 * OD — that 7" convention is hardcoded inside liner.renderer.ts (PRE_PERF_LINER_SIZE_IN)
 * and is intentionally left untouched here.
 */
function buildWellDesignFromForm(programMain: WwellIrRawProgramMain): IWellDesign {
  const completion = programMain.completion;
  const isOpenHole = completion === COMPLETION_OPEN_HOLE;
  const isGravelPack = completion === COMPLETION_SCREENED_GRAVEL_PACK;
  const isScreened = completion === COMPLETION_SCREENED || isGravelPack;
  const isLinerPerforated = completion === COMPLETION_LINER_PERFORATED;

  return {
    name: null,
    prewapSeq: null,
    staWaterLvl: null,
    swLvlTxt: null,
    pumpLvl: null,
    ohFlg: isOpenHole ? 'Y' : 'N',
    ohRemarks: isOpenHole ? (programMain.openHoleSizeIn || null) : null,
    gpFlg: isGravelPack ? 'Y' : 'N',
    gpRemarks: isGravelPack ? 'Gravel Pack' : null,
    perforatedFlg: isLinerPerforated ? 'Y' : 'N',
    perforatedRemarks: isLinerPerforated ? '7" Pre-Perforated Liner' : null,
    lsFlg: isScreened ? 'Y' : 'N',
    lsRemarks: isScreened ? 'Screen' : null,
  };
}

function computeDisplayDepthFt(
  targetDepthFt: number,
  casingRows: readonly WwellIrRawCasingRow[],
  formationRows: readonly WwellIrRawFormationTop[],
): number {
  const depths = [
    targetDepthFt,
    ...casingRows.map(row => row.depthFtBgl ?? 0),
    ...formationRows.map(row => row.depthFt ?? 0),
  ];
  return Math.max(...depths, MIN_SCALE_DEPTH_FT);
}

/** Shared tail for both entry points below — takes already-normalized raw rows so the actual
 *  gating/mapping (isRenderable*, mapCasings, buildLinerTopAnchors, mapGeologicTops,
 *  buildWellDesignFromForm) is written once regardless of whether the source was a live,
 *  still-being-edited form value or a submitted `IntegratedRequirement`. */
function assembleWellboreDiagramData(
  wellName: string,
  targetDepthFt: number,
  casingRows: readonly WwellIrRawCasingRow[],
  formationRows: readonly WwellIrRawFormationTop[],
  surfaceElevationFtAsl: number | null,
  programMain: WwellIrRawProgramMain,
): WellboreDiagramData {
  const renderableCasings = casingRows.filter(isRenderableCasingRow);
  const renderableFormationTops = formationRows.filter(isRenderableFormationTop);

  return {
    wellName,
    totalDepth: targetDepthFt,
    displayDepth: computeDisplayDepthFt(targetDepthFt, casingRows, formationRows),
    casings: mapCasings(renderableCasings),
    drlgCasings: buildLinerTopAnchors(renderableCasings),
    geologicTops: mapGeologicTops(renderableFormationTops, surfaceElevationFtAsl),
    hydrogeology: null,
    prewap: null,
    rigActivity: null,
    currentDepth: 0,
    targetDepth: targetDepthFt,
    mudCirculation: [],
    wellDesign: buildWellDesignFromForm(programMain),
  };
}

/**
 * Gates the *entire* diagram — depth scale, ground level line, and total-depth label included,
 * not just individual casing/formation-top shapes (see `isRenderableCasingRow`/
 * `isRenderableFormationTop` above). Before any row is complete, `well-bore-view` still draws
 * that scaffold around nothing: a bare 0–500ft ruler, a "GROUND LEVEL" line, and a
 * "TOTAL DEPTH: 0 FT" label with no actual construction data behind any of it. Callers should
 * check this first and render an empty state instead of the diagram until it's true.
 */
export function hasWellboreDiagramData(raw: WwellIrFormRawValue): boolean {
  return (
    raw.subsurface.casing.some(isRenderableCasingRow) ||
    raw.subsurface.formationTops.some(isRenderableFormationTop)
  );
}

export function buildWellboreDiagramData(raw: WwellIrFormRawValue): WellboreDiagramData {
  return assembleWellboreDiagramData(
    raw.reference.main.wellName,
    raw.reference.main.targetDepthFt ?? 0,
    raw.subsurface.casing,
    raw.subsurface.formationTops,
    raw.reference.main.surfaceElevationFtAsl,
    raw.programDetails.main,
  );
}

/** `IntegratedRequirementCasingRow`/`FormationTop`/`ProgramDetails` are already-resolved
 *  (non-nullable) counterparts of the wizard's raw row types — these just fill in the raw
 *  shape's optional-text fields (`remarks`/`description`/`openHoleSizeIn`) with `''` in place
 *  of `null`, since a submitted IR's `null` here means the same "nothing entered" as the
 *  wizard's `''`. */
function toRawCasingRows(rows: readonly IntegratedRequirementCasingRow[]): WwellIrRawCasingRow[] {
  return rows.map(row => ({
    type: row.type,
    sizeIn: row.sizeIn,
    depthFtBgl: row.depthFtBgl,
    tolFtBgl: row.tolFtBgl,
    remarks: row.remarks ?? '',
  }));
}

function toRawFormationRows(rows: readonly IntegratedRequirementFormationTop[]): WwellIrRawFormationTop[] {
  return rows.map(row => ({
    name: row.name,
    depthFt: row.depthFt,
    subSeaFtAsl: row.subSeaFtAsl ?? null,
    description: row.description ?? '',
  }));
}

/** Same emptiness gate as `hasWellboreDiagramData`, for the IR detail page. */
export function hasWellboreDiagramDataForIntegratedRequirement(ir: IntegratedRequirement): boolean {
  return (
    toRawCasingRows(ir.casing).some(isRenderableCasingRow) ||
    toRawFormationRows(ir.formationTops).some(isRenderableFormationTop)
  );
}

/** Detail-page counterpart of `buildWellboreDiagramData` — same live `app-well-bore-view`
 *  diagram, fed from a submitted `IntegratedRequirement` instead of the in-progress wizard
 *  form, so the detail page shows the real engineering rendering instead of the lighter
 *  `wwell-ir-design-preview` sketch. */
export function buildWellboreDiagramDataFromIntegratedRequirement(ir: IntegratedRequirement): WellboreDiagramData {
  return assembleWellboreDiagramData(
    ir.requestReference.wellName,
    ir.requestReference.targetDepthFt,
    toRawCasingRows(ir.casing),
    toRawFormationRows(ir.formationTops),
    ir.requestReference.surfaceElevationFtAsl,
    {
      completion: ir.programDetails.completion,
      openHoleSizeIn: ir.programDetails.openHoleSizeIn ?? '',
      acidization: ir.programDetails.acidization,
      transmission: ir.programDetails.transmission,
    },
  );
}

/** The letter's closing signature block — "Prepared by" is always the primary geologist on
 *  record; "Approved by" only names a real approver once the IR has actually cleared the chain
 *  (mirrors `ApprovalChainStep.status` rather than assuming the last chain entry is meaningful

export function letterApprovedByName(ir: IntegratedRequirement): string {
  if (ir.status !== 'approved') return 'Pending Approval';
  const lastApproved = [...ir.chain].reverse().find(step => step.status === 'approved');
  return lastApproved?.name ?? EMPTY_VALUE;
}
