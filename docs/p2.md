  private previewRows<TRow>(
    live: readonly TRow[] | null,
    saved: () => readonly TRow[],
    draftRows: readonly TRow[] | null,
    keyOf: (row: TRow) => string,
  ): readonly TRow[] | null {
    if (!draftRows?.length) return live;
    const rows = live ?? saved();
    const keys = new Set(rows.map(keyOf));
    // "Add" refuses a row whose natural key is already taken (it would overwrite that row) — including by
    // another staged draft row — so a colliding draft is not drawn either.
    const additions = draftRows.filter(draft => {
      const key = keyOf(draft);
      if (keys.has(key)) return false;
      keys.add(key);
      return true;
    });
    return additions.length ? [...rows, ...additions] : live;
  }

  /** Commits every row staged in a "+" add-form's own FormArray into the section's live FormArray — the
   *  same array inline edits and deletes write to, so Save diffs it like any other change. All-or-nothing:
   *  Save matches rows on their natural key, so if ANY staged row's key is already taken (by an existing
   *  row, or by another row in the same batch) the whole commit is refused rather than silently overwriting
   *  that row, or committing some rows and leaving the colliding one stranded in the add-form. */
  private commitNewRows<TRow>(
    add: SectionEditState<FormArray>,
    section: WwellFormSectionConfig,
    commit: {
      readonly existingKeys: readonly string[];
      readonly keyOf: (row: TRow) => string;
      readonly liveArray: () => FormArray | null;
      readonly duplicate: (key: string) => string;
      readonly added: (count: number) => string;
    },
  ): void {
    const form = add.control();
    if (!form || form.invalid || form.length === 0) return;

    const rows = form.getRawValue() as TRow[];
    const keys = new Set(commit.existingKeys);
    for (const row of rows) {
      const key = commit.keyOf(row);
      if (keys.has(key)) {
        this.notify.warn(commit.duplicate(key));
        return;
      }
      keys.add(key);
    }

    const array = commit.liveArray();
    if (!array) return;
    rows.forEach(row => {
      addRepeatableRow(array, section.repeatable?.rowFields ?? [], this.fb);
      array.at(array.length - 1).patchValue(row as object);
    });
    add.close();
    this.notify.info(commit.added(rows.length));
  }

  /**
   * Uploads one staged file for the clicked row only — never any other file staged in this or any
   * other slot, and never the rest of the form. `isPendingUploadBusy` (checked both here and via the
   * row's own `[disabled]`) is what stops a second click on the same row from firing a second request
   * while the first is still in flight; the `[disabled]` binding alone can lag a frame behind a rapid
   * double-click, so the guard is repeated here too.
   *
   * A record that's never been saved has no `ir_seq` to upload the file under yet. The row's Upload
   * button is disabled in that case (see the template), and this guard repeats that rather than
   * silently saving the whole IR as a side effect of an attachment click — that used to auto-save via
   * `saveUnsavedChangesAndWait` and then flush every staged file in every slot, which surprised users
   * by persisting unrelated, still-being-edited form fields and uploading files they hadn't asked to
   * send yet. The user must explicitly save the IR first; once it has an `ir_seq`,
   * `uploadPendingAttachments` (plural) already flushes every staged file as part of that save — see
   * its call site in `wwell-ir.component.ts`.
   */
  protected uploadPendingAttachment(slot: IrAttachmentSlot, item: IrPendingAttachment): void {
    if (this.isPendingUploadBusy(item.id)) return;

    const irSeq = this.ir().irSeq;
    if (irSeq == null) {
      this.notify.info('Save this Integrated Requirement before uploading attachments.');
      return;
    }

    this.store.uploadPendingAttachment(irSeq, slot, item.id);
  }

    );
  }

import { PDFDocument, PDFFont, StandardFonts } from 'pdf-lib';

/** One attachment's raw bytes plus just enough metadata to place it in the merged letter — mirrors
 *  agwa-planning's (now-removed) IrAttachmentPdfAppender.SlotAttachment, since this replaces that
 *  server-side merge with the same behavior run in the browser. */
export interface IrLetterAttachmentFile {
  readonly label: string;
  readonly contentType: string;
  readonly bytes: ArrayBuffer;
}

const PAGE_WIDTH = 595.28; // A4 — matches iText's PageSize.A4 the backend used for the same job.
const PAGE_HEIGHT = 841.89;
const PAGE_MARGIN = 36;
const CAPTION_FONT_SIZE = 11;
const CAPTION_GAP = 12;

/**
 * Returns the rendered letter with each attachment appended as trailing pages — Location Map
 * first, then Initiation Request, each starting on its own fresh page (callers pass `attachments`
 * already in that order; see `irAttachmentsFor`/`WWELL_IR_ATTACHMENT_SLOTS`).
 *
 * A PDF attachment contributes its own pages verbatim; a JPG/PNG is drawn onto a single captioned
 * A4 page — the same two cases agwa-planning's IrAttachmentPdfAppender used to handle server-side,
 * now done here since Angular already holds the attachment bytes (same S3 proxy it uses for
 * upload/list/view/delete) with nothing left for the backend to add.
 *
 * A single unreadable attachment is skipped rather than failing the whole merge — the letter is
 * the document the user asked for; one damaged appendix shouldn't trade a complete-but-imperfect
 * PDF for no PDF at all.
 */
export async function appendAttachmentsToLetter(
  letterPdfBytes: ArrayBuffer,
  attachments: readonly IrLetterAttachmentFile[],
): Promise<Uint8Array> {
  if (attachments.length === 0) {
    return new Uint8Array(letterPdfBytes);
  }

  const target = await PDFDocument.load(letterPdfBytes);
  const captionFont = await target.embedFont(StandardFonts.HelveticaBold);

  for (const attachment of attachments) {
    try {
      if (isPdfContentType(attachment.contentType)) {
        await appendPdfPages(target, attachment.bytes);
      } else {
        await appendImagePage(target, attachment, captionFont);
      }
    } catch {
      // Skipped: a broken image header or unreadable PDF shouldn't fail the whole letter.
    }
  }

  return target.save();
}

function isPdfContentType(contentType: string): boolean {
  return contentType.toLowerCase() === 'application/pdf';
}

async function appendPdfPages(target: PDFDocument, bytes: ArrayBuffer): Promise<void> {
  const source = await PDFDocument.load(bytes);
  const pages = await target.copyPages(source, source.getPageIndices());
  pages.forEach(page => target.addPage(page));
}

/** Draws one image onto a captioned A4 page, scaled to fit inside the margins in both dimensions
 *  (iText's `setAutoScale` equivalent) while keeping its aspect ratio. */
async function appendImagePage(target: PDFDocument, attachment: IrLetterAttachmentFile, captionFont: PDFFont): Promise<void> {
  const image = isPngContentType(attachment.contentType)
    ? await target.embedPng(attachment.bytes)
    : await target.embedJpg(attachment.bytes);

  const page = target.addPage([PAGE_WIDTH, PAGE_HEIGHT]);
  const contentWidth = PAGE_WIDTH - 2 * PAGE_MARGIN;
  const captionY = PAGE_HEIGHT - PAGE_MARGIN - CAPTION_FONT_SIZE;
  const captionWidth = captionFont.widthOfTextAtSize(attachment.label, CAPTION_FONT_SIZE);
  page.drawText(attachment.label, {
    x: PAGE_MARGIN + (contentWidth - captionWidth) / 2,
    y: captionY,
    size: CAPTION_FONT_SIZE,
    font: captionFont,
  });

  const availableHeight = captionY - CAPTION_GAP - PAGE_MARGIN;
  const scale = Math.min(contentWidth / image.width, availableHeight / image.height);
  const drawWidth = image.width * scale;
  const drawHeight = image.height * scale;
  page.drawImage(image, {
    x: PAGE_MARGIN + (contentWidth - drawWidth) / 2,
    y: captionY - CAPTION_GAP - drawHeight,
    width: drawWidth,
    height: drawHeight,
  });
}

function isPngContentType(contentType: string): boolean {
  return contentType.toLowerCase() === 'image/png';
}

   *  others, and the merge step itself falls back to the unmerged letter on any failure — a broken
   *  appendix should never turn a working download into a failed one. */
  private async withAttachmentsMerged(letterBlob: Blob): Promise<Blob> {
    const attachments = this.ir().attachments;
    if (attachments.length === 0) return letterBlob;

    try {

      const merged = await appendAttachmentsToLetter(await blobToArrayBuffer(letterBlob), files);

   *  turned straight into a download link — but Angular's HttpClient applies that same
   *  responseType to error bodies too, so even agwa-planning's JSON error envelope
   *  (`{ message: "Cannot generate IR letter: missing required field(s): ..." }`) arrives as an
   *  unparsed Blob instead of a plain object. Without this, the backend's specific reason (e.g.
   *  a missing Letter Number) was silently discarded in favor of a generic failure message.
   *  Reads it via `FileReader` rather than `Blob.text()` — the latter isn't implemented by
   *  jsdom (this repo's test environment), while `FileReader.readAsText` is. */
  private extractDownloadErrorMessage(err: unknown): Promise<string | null> {
    if (!(err instanceof HttpErrorResponse) || !(err.error instanceof Blob)) return Promise.resolve(null);
    return new Promise(resolve => {
      const reader = new FileReader();
      reader.onload = () => resolve(this.parseErrorMessage(reader.result));
      reader.onerror = () => resolve(null);
      reader.readAsText(err.error as Blob);
    });
  }

  private parseErrorMessage(text: FileReader['result']): string | null {
    if (typeof text !== 'string') return null;
    try {
      const parsed: unknown = JSON.parse(text);
      const message = parsed && typeof parsed === 'object' ? (parsed as { message?: unknown }).message : undefined;
      return typeof message === 'string' ? message : null;
    } catch {
      return null;
    }

    return { completed, total, percent: Math.round((completed / total) * 100) };
  });

    this.activeTab.set(tab);
  }
