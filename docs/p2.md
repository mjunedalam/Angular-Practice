```ts
/**
 * Single source of truth for which files the app lets a user attach. Every file input / dropzone
 * reads its `accept` hint, its type check and its size cap from here.
 *
 * Edit `ALLOWED_UPLOAD_EXTENSIONS` to change what the whole app accepts — the `accept` hint, the
 * hint text and every handler's check follow from that one edit. Set it to `null` to allow any
 * file type. The size cap applies either way.
 *
 * `accept` on an <input type="file"> is only a picker hint, so handlers call `isUploadFileAllowed`
 * on every file regardless; the server remains the real authority.
 */

/** Bare, lower-case extensions (no dot), or `null` to allow any file type. */
export const ALLOWED_UPLOAD_EXTENSIONS: readonly string[] | null = [
  // images
  'jpg', 'jpeg', 'png',
  // documents
  'pdf', 'doc', 'docx', 'txt',
  // spreadsheets
  'xls', 'xlsx', 'xlsm', 'xlsb', 'csv',
];

export const MAX_UPLOAD_SIZE_BYTES = 25 * 1024 * 1024;
export const MAX_UPLOAD_SIZE_LABEL = '25 MB';

/** Value for an <input accept> — `null` (attribute omitted) when any type is allowed. */
export function toAcceptAttribute(extensions: readonly string[] | null = ALLOWED_UPLOAD_EXTENSIONS): string | null {
  return extensions ? extensions.map(ext => `.${ext}`).join(',') : null;
}

/**
 * Extensions that are accepted (picker, validation, server) but deliberately not named in on-screen
 * text — display only; removing one here changes nothing about what can be uploaded.
 */
export const EXTENSIONS_HIDDEN_FROM_LABELS: readonly string[] = ['xls', 'xlsx', 'xlsm', 'xlsb', 'csv'];

/** Dropzone hint text: `PDF · DOC · …`, or `Any file type`. Omits `EXTENSIONS_HIDDEN_FROM_LABELS`. */
export function toExtensionLabel(extensions: readonly string[] | null = ALLOWED_UPLOAD_EXTENSIONS): string {
  if (!extensions) return 'Any file type';
  return extensions
    .filter(ext => !EXTENSIONS_HIDDEN_FROM_LABELS.includes(ext))
    .map(ext => ext.toUpperCase())
    .join(' · ');
}

export function hasAllowedExtension(fileName: string, extensions: readonly string[] | null = ALLOWED_UPLOAD_EXTENSIONS): boolean {
  if (extensions === null) return true;
  const dot = fileName.lastIndexOf('.');
  if (dot < 0) return false;
  return extensions.includes(fileName.slice(dot + 1).toLowerCase());
}

export interface UploadRules {
  readonly extensions?: readonly string[] | null;
  readonly maxBytes?: number;
}

export function isUploadFileAllowed(file: File, rules: UploadRules = {}): boolean {
  const { extensions = ALLOWED_UPLOAD_EXTENSIONS, maxBytes = MAX_UPLOAD_SIZE_BYTES } = rules;
  return hasAllowedExtension(file.name, extensions) && file.size <= maxBytes;
}
```

```ts
import { ApprovalChainStep } from '@models/wwell-planning/wwell-planning.model';

const REQ_NO_MIN = 10000;
const REQ_NO_MAX = 99999;

const UINT32_RANGE = 0x100000000;

// Uniform integer in [min, max] from the CSPRNG (not Math.random — Fortify "Insecure Randomness").
// Rejection sampling discards the biased tail so every value stays equally likely.
function secureRandomInt(min: number, max: number): number {
  const span = max - min + 1;
  const limit = UINT32_RANGE - (UINT32_RANGE % span);
  const buffer = new Uint32Array(1);
  do {
    crypto.getRandomValues(buffer);
  } while (buffer[0] >= limit);
  return min + (buffer[0] % span);
}

export function generateApprovalReqNo(): string {
  return `REQ-${secureRandomInt(REQ_NO_MIN, REQ_NO_MAX)}`;
}

export function todayIsoDate(): string {
  return new Date().toISOString().slice(0, 10);
}

// Freshly submitted requests start with their first reviewer in progress and the rest unassigned.
export function buildInitialApprovalChain(roles: readonly string[], submittedBy: string): ApprovalChainStep[] {
  return roles.map((role, i) => ({
    role,
    name: i === 0 ? submittedBy : 'Pending Assignment',
    status: i === 0 ? 'in-progress' : 'pending',
    date: null,
  }));
}
```

```ts
import { ChangeDetectionStrategy, Component, input, output, signal } from '@angular/core';
import { MatButtonModule } from '@angular/material/button';
import { MatIconModule } from '@angular/material/icon';
import { isUploadFileAllowed, toAcceptAttribute } from '@core/config/file-upload.config';

interface AttachmentItem {
  readonly id: string;
  readonly file: File;
}

@Component({
  selector: 'app-wwell-attachment-dropzone',
  standalone: true,
  imports: [MatButtonModule, MatIconModule],
  templateUrl: './wwell-attachment-dropzone.component.html',
  styleUrl: './wwell-attachment-dropzone.component.scss',
  changeDetection: ChangeDetectionStrategy.OnPush,
})
export class WwellAttachmentDropzoneComponent {
  readonly hint = input('Click to attach files');
  /** When true, hides the dropzone control and shows only a read-only file list (e.g. a "view" dialog mode). */
  readonly disabled = input(false);
  /** When set, renders this fixed list of names read-only instead of the live drag-and-drop attachments. */
  readonly displayNames = input<readonly string[] | null>(null);
  readonly filesChange = output<File[]>();

  protected readonly accept = toAcceptAttribute();
  protected readonly isDragOver = signal(false);
  protected readonly attachments = signal<AttachmentItem[]>([]);

  protected onDragOver(event: DragEvent): void {
    event.preventDefault();
    event.stopPropagation();
    if (this.disabled()) return;
    this.isDragOver.set(true);
  }

  protected onDragLeave(event: DragEvent): void {
    event.preventDefault();
    event.stopPropagation();
    this.isDragOver.set(false);
  }

  protected onDrop(event: DragEvent): void {
    event.preventDefault();
    event.stopPropagation();
    this.isDragOver.set(false);
    if (this.disabled()) return;
    this.addFiles(Array.from(event.dataTransfer?.files ?? []));
  }

  protected onFileInputChange(event: Event): void {
    const input = event.target as HTMLInputElement | null;
    this.addFiles(Array.from(input?.files ?? []));
    if (input) input.value = '';
  }

  protected removeAttachment(id: string): void {
    this.attachments.update(items => items.filter(item => item.id !== id));
    this.emitFiles();
  }

  protected fileSizeLabel(size: number): string {
    if (size < 1024) return `${size} B`;
    if (size < 1024 * 1024) return `${(size / 1024).toFixed(1)} KB`;
    return `${(size / (1024 * 1024)).toFixed(1)} MB`;
  }

  private addFiles(selected: File[]): void {
    if (!selected.length) return;

    const valid = selected.filter(file => isUploadFileAllowed(file));
    if (!valid.length) return;

    this.attachments.update(existing => [
      ...existing,
      ...valid.map((file, index) => ({ file, id: `${file.name}-${file.size}-${existing.length + index}` })),
    ]);
    this.emitFiles();
  }

  private emitFiles(): void {
    this.filesChange.emit(this.attachments().map(item => item.file));
  }
}
```

```ts
import { ChangeDetectionStrategy, Component, OnDestroy, ViewEncapsulation, computed, inject, signal } from '@angular/core';
import { MatButtonModule } from '@angular/material/button';
import { MAT_DIALOG_DATA, MatDialogRef } from '@angular/material/dialog';
import { MatIconModule } from '@angular/material/icon';
import { MatTooltipModule } from '@angular/material/tooltip';
import { DomSanitizer, SafeResourceUrl } from '@angular/platform-browser';

export interface WwellIrAttachmentViewerData {
  readonly file: File;
  /** Slot name ("Location Map") shown above the file name, so the viewer says which slot it's from. */
  readonly label: string;
}

/** What the body renders — images get zoom/rotate, PDFs get the browser's own viewer in a frame. */
type ViewerKind = 'image' | 'pdf' | 'unsupported';

const ZOOM_STEP = 0.25;
const MIN_ZOOM = 0.25;
const MAX_ZOOM = 4;
const QUARTER_TURN_DEGREES = 90;

/**
 * Full-bleed viewer for one IR attachment.
 *
 * Deliberately not the shared FilePreviewDialogComponent: that one is a small draggable panel sized
 * for a quick glance at an upload in a form, and a location map or a signed initiation request needs
 * the opposite — as much of the screen as possible, since the whole point is reading a scanned
 * document. This one takes over the viewport, puts the document on a dark stage so the page edges
 * read clearly, and keeps the chrome to a single header bar.
 *
 * `ViewEncapsulation.None` because the dialog's own panel/backdrop live in the CDK overlay, outside
 * this component's DOM — the `.ir-viewer` panel rules in the stylesheet can't reach them otherwise.
 */
@Component({
  selector: 'app-wwell-ir-attachment-viewer',
  standalone: true,
  imports: [MatButtonModule, MatIconModule, MatTooltipModule],
  templateUrl: './wwell-ir-attachment-viewer.component.html',
  styleUrl: './wwell-ir-attachment-viewer.component.scss',
  changeDetection: ChangeDetectionStrategy.OnPush,
  encapsulation: ViewEncapsulation.None,
})
export class WwellIrAttachmentViewerComponent implements OnDestroy {
  private readonly data = inject<WwellIrAttachmentViewerData>(MAT_DIALOG_DATA);
  private readonly dialogRef = inject(MatDialogRef<WwellIrAttachmentViewerComponent>);
  private readonly sanitizer = inject(DomSanitizer);

  protected readonly label = this.data.label;
  protected readonly fileName = this.data.file.name;
  protected readonly kind: ViewerKind = resolveKind(this.data.file);
  protected readonly typeLabel = this.data.file.name.split('.').pop()?.toUpperCase() ?? 'FILE';
  protected readonly sizeLabel = formatBytes(this.data.file.size);

  protected readonly zoom = signal(1);
  protected readonly rotation = signal(0);

  /** Single transform for both controls, so zooming a rotated page keeps its orientation. */
  protected readonly imageTransform = computed(() => `scale(${this.zoom()}) rotate(${this.rotation()}deg)`);
  protected readonly zoomLabel = computed(() => `${Math.round(this.zoom() * 100)}%`);
  protected readonly canZoomIn = computed(() => this.zoom() < MAX_ZOOM);
  protected readonly canZoomOut = computed(() => this.zoom() > MIN_ZOOM);

  /** Held so it can be revoked on close — an un-revoked object URL pins the whole file in memory. */
  private readonly objectUrl = URL.createObjectURL(this.data.file);
  protected readonly safeUrl: SafeResourceUrl = this.sanitizer.bypassSecurityTrustResourceUrl(this.objectUrl);

  ngOnDestroy(): void {
    URL.revokeObjectURL(this.objectUrl);
  }

  protected zoomIn(): void {
    this.zoom.update(value => Math.min(MAX_ZOOM, value + ZOOM_STEP));
  }

  protected zoomOut(): void {
    this.zoom.update(value => Math.max(MIN_ZOOM, value - ZOOM_STEP));
  }

  /** Back to 100% and upright — one control to undo any amount of fiddling. */
  protected resetView(): void {
    this.zoom.set(1);
    this.rotation.set(0);
  }

  protected rotate(): void {
    this.rotation.update(value => (value + QUARTER_TURN_DEGREES) % 360);
  }

  protected download(): void {
    const link = document.createElement('a');
    link.href = this.objectUrl;
    link.download = this.fileName;
    link.click();
  }

  protected close(): void {
    this.dialogRef.close();
  }
}

function resolveKind(file: File): ViewerKind {
  const extension = file.name.split('.').pop()?.toLowerCase() ?? '';
  if (file.type.startsWith('image/') || ['jpg', 'jpeg', 'png'].includes(extension)) return 'image';
  if (file.type === 'application/pdf' || extension === 'pdf') return 'pdf';
  // Only PDFs and images preview inline; every other allowed type (see core/config/file-upload.config.ts)
  // offers a download rather than rendering a broken frame.
  return 'unsupported';
}

function formatBytes(bytes: number): string {
  if (bytes < 1024) return `${bytes} B`;
  if (bytes < 1024 * 1024) return `${(bytes / 1024).toFixed(1)} KB`;
  return `${(bytes / (1024 * 1024)).toFixed(1)} MB`;
}
```

```ts
import { MAX_UPLOAD_SIZE_BYTES, hasAllowedExtension, toAcceptAttribute } from '@core/config/file-upload.config';
import { IrAttachment, IrAttachmentSlot } from '@models/wwell-planning/ir-attachment.model';

export interface WwellIrAttachmentSlotConfig {
  readonly slot: IrAttachmentSlot;
  /** Matches the IR SECTIONS rail numbering (see wwell-ir-section-rail.config.ts). */
  readonly number: string;
  readonly label: string;
  /** Material icon ligature — topic-specific, the same way the rail's group icons are chosen. */
  readonly icon: string;
  /** Mirrors the schema's own validator on each slot's file field (wwell-ir-form.schema.ts). */
  readonly required: boolean;
  readonly hint: string;
}

/**
 * The IR's attachment slots, in the order they're appended to the letter PDF as trailing pages.
 * Single source of truth for both the editor cards and the letter view, so the two can't drift.
 */
export const WWELL_IR_ATTACHMENT_SLOTS: readonly WwellIrAttachmentSlotConfig[] = [
  {
    slot: 'location-map',
    number: '7.1',
    label: 'Location Map',
    icon: 'map',
    required: true,
    hint: 'Site location map — appended to the letter as its own page.',
  },
  {
    slot: 'initiation-request',
    number: '7.2',
    label: 'Initiation Request',
    icon: 'assignment',
    required: true,
    hint: 'Signed initiation request — appended to the letter after the location map.',
  },
  {
    slot: 'wlda',
    number: '7.3',
    label: 'Well Location and Drilling Approval (WLDA)',
    icon: 'verified',
    required: true,
    hint: 'WLDA document — appended to the letter after the initiation request and sent to SAP on submit.',
  },
];

/**
 * Restricted to what the backend can actually embed as PDF pages (see agwa-planning's
 * IrAttachmentPdfAppender). Office formats would need a headless converter, so they're refused at
 * upload rather than stored as something that silently never appears in the letter.
 */
/** True once every slot has at least one stored file — the gate for Submit for Review. */
export function hasAllRequiredAttachments(attachments: readonly IrAttachment[]): boolean {
  return WWELL_IR_ATTACHMENT_SLOTS.filter(config => config.required).every(config => irAttachmentsFor(attachments, config.slot).length > 0);
}

export const WWELL_IR_ATTACHMENT_ACCEPT = toAcceptAttribute();

/** The `accept` attribute is only a picker hint, so handlers re-check against the same list. */
export function hasAcceptedIrAttachmentExtension(fileName: string): boolean {
  return hasAllowedExtension(fileName);
}

/** Purely a client-side guard now — the S3 object-store proxy enforces no size cap of its own, so
 *  this exists only to fail fast before spending time on an upload that's almost certainly wrong. */
export const WWELL_IR_ATTACHMENT_MAX_BYTES = MAX_UPLOAD_SIZE_BYTES;

/** Every file occupying a slot, oldest-uploaded first — shared by the editor, letter and details
 *  views. A slot can hold any number of attachments, not just one, so callers render this as a list
 *  rather than picking a single result. */
export function irAttachmentsFor(attachments: readonly IrAttachment[], slot: IrAttachmentSlot): readonly IrAttachment[] {
  return attachments.filter(attachment => attachment.slot === slot);
}

/**
 * Recovers the slot from one of this app's own object keys
 * (`ir-attachments/ir-<irSeq>/<slot>/<uuid>`). Anchored on the literal `ir-attachments` segment
 * rather than the key's start, because the real S3 proxy prefixes every key with its own
 * environment segment (e.g. `dev/`) that this app never generates and must simply tolerate.
 * Returns null for a key that doesn't look like one of this app's own.
 */
export function slotFromObjectKey(objectKey: string): IrAttachmentSlot | null {
  const segments = objectKey.split('/');
  const anchor = segments.indexOf('ir-attachments');
  const candidate = anchor >= 0 ? segments[anchor + 2] : undefined;
  const config = WWELL_IR_ATTACHMENT_SLOTS.find(entry => entry.slot === candidate);
  return config ? config.slot : null;
}

/** Display label for a slot ("Location Map") — used to build an `IrAttachment` straight from a
 *  search result, since the proxy itself has no notion of this app's slot labels. */
export function labelForSlot(slot: IrAttachmentSlot): string {
  return WWELL_IR_ATTACHMENT_SLOTS.find(entry => entry.slot === slot)?.label ?? slot;
}
```

```ts
}

function asNumber(value: unknown): number | null {
  return typeof value === 'number' && Number.isFinite(value) ? value : null;
}

/** Turns the Formation Tops "Sub-sea" field into a read-only, calculated one: surface elevation minus the row's depth,
 *  with an (i) tooltip that spells the formula out with the current numbers. */
function withDerivedSubSea(section: WwellFormSectionConfig, elevation: () => number | null): WwellFormSectionConfig {
  const repeatable = section.repeatable;
  if (!repeatable) return section;
  const subSea: Partial<WwellFieldConfig> = {
    readonly: true,
    placeholder: 'auto',
    derived: {
      compute: row => computeSubSeaFt(elevation(), asNumber(row['depthFt'])),
      explain: row => explainSubSea(elevation(), asNumber(row['depthFt'])),
    },
  };
  return {
    ...section,
    repeatable: {
      ...repeatable,
      rowFields: repeatable.rowFields.map(field => (field.key === 'subSeaFtAsl' ? { ...field, ...subSea } : field)),
    },
  };
}

const OFFSET_WELL_PICKER_KEY = 'sourceWell';

/** Prepends the optional "Select Offset Water Well" picker to a section's row fields: it pre-fills the row from a well's
 *  latest record, after which every field can still be edited — or the row typed by hand without picking at all. */
function withOffsetWellPicker(
  section: WwellFormSectionConfig,
  options: Signal<readonly WwellFieldOption[]>,
): WwellFormSectionConfig {
  const repeatable = section.repeatable;
  if (!repeatable) return section;
  const picker: WwellFieldConfig = {
    key: OFFSET_WELL_PICKER_KEY,
    label: 'Select Offset Water Well',
    type: 'select',
    options,
    placeholder: 'Select a well — fills the fields below',
    searchable: true,
    colSpan: 2,
  };
  return { ...section, repeatable: { ...repeatable, rowFields: [picker, ...repeatable.rowFields] } };
}

  private previewRows<TRow>(
    live: readonly TRow[] | null,
    saved: () => readonly TRow[],
    draftRows: readonly TRow[] | null,
    keyOf: (row: TRow) => string,
  ): readonly TRow[] | null {
    if (!draftRows?.length) return live;
    const rows = live ?? saved();
    const keys = new Set(rows.map(keyOf));
    // "Add" refuses a row whose natural key is already taken (it would overwrite that row) — including by
    // another staged draft row — so a colliding draft is not drawn either.
    const additions = draftRows.filter(draft => {
      const key = keyOf(draft);
      if (keys.has(key)) return false;
      keys.add(key);
      return true;
    });
    return additions.length ? [...rows, ...additions] : live;
  }

  /** Watches an offset-well FormArray (the "+" add form, or the section pencil's edit form) and, whenever a row's
   *  "Select Offset Water Well" value changes, fills that row from the picked well. Re-binds each time the form is
   *  opened afresh; the previous subscription is dropped with the previous array. */
  private bindOffsetWellPicker(array: () => FormArray | null): void {
    effect(onCleanup => {
      const live = array();
      if (!live) return;
      const picked = new WeakMap<FormGroup, unknown>();
      const subscription = live.valueChanges.subscribe(() => {
        for (const row of live.controls as FormGroup[]) {
          const value = row.get(OFFSET_WELL_PICKER_KEY)?.value;
          if (picked.get(row) === value) continue;
          picked.set(row, value);
          if (value) this.applyOffsetWellSource(row, value as string);
        }
      });
      onCleanup(() => subscription.unsubscribe());
    });
  }

  /** Commits every row staged in a "+" add-form's own FormArray into the section's live FormArray — the
   *  same array inline edits and deletes write to, so Save diffs it like any other change. All-or-nothing:
   *  Save matches rows on their natural key, so if ANY staged row's key is already taken (by an existing
   *  row, or by another row in the same batch) the whole commit is refused rather than silently overwriting
   *  that row, or committing some rows and leaving the colliding one stranded in the add-form. */
  private commitNewRows<TRow>(
    add: SectionEditState<FormArray>,
    section: WwellFormSectionConfig,
    commit: {
      readonly existingKeys: readonly string[];
      readonly keyOf: (row: TRow) => string;
      /** Last tweak to each row as it is committed (e.g. assigning its save key). */
      readonly prepare?: (row: TRow) => TRow;
      readonly liveArray: () => FormArray | null;
      /** Runs once every staged row is in the live array (e.g. to re-sort it). */
      readonly afterCommit?: (array: FormArray) => void;
      readonly duplicate: (key: string) => string;
      readonly added: (count: number) => string;
    },
  ): void {
    const form = add.control();
    if (!form || form.invalid || form.length === 0) return;

    const rows = form.getRawValue() as TRow[];
    const keys = new Set(commit.existingKeys);
    for (const row of rows) {
      const key = commit.keyOf(row);
      if (keys.has(key)) {
        this.notify.warn(commit.duplicate(key));
        return;
      }
      keys.add(key);
    }

    const array = commit.liveArray();
    if (!array) return;
    rows.forEach(staged => {
      const row = commit.prepare ? commit.prepare(staged) : staged;
      addRepeatableRow(array, section.repeatable?.rowFields ?? [], this.fb);
      array.at(array.length - 1).patchValue(row as object);
    });
    commit.afterCommit?.(array);
    add.close();
    this.notify.info(commit.added(rows.length));
  }

  protected onSlotDragOver(slot: IrAttachmentSlot, event: DragEvent): void {
    if (!event.dataTransfer?.types.includes('Files')) return;
    event.preventDefault();
    event.dataTransfer.dropEffect = 'copy';
    if (this.dragOverSlot() !== slot) this.dragOverSlot.set(slot);
  }

  protected onSlotDragLeave(slot: IrAttachmentSlot, event: DragEvent): void {
    const next = event.relatedTarget as Node | null;
    if (next && (event.currentTarget as HTMLElement).contains(next)) return;
    if (this.dragOverSlot() === slot) this.dragOverSlot.set(null);
  }

  protected onSlotDrop(slot: IrAttachmentSlot, event: DragEvent): void {
    event.preventDefault();
    this.dragOverSlot.set(null);
    const file = event.dataTransfer?.files?.[0];
    if (!file) return;
    if (!hasAcceptedIrAttachmentExtension(file.name)) {
      this.notify.error(`Only ${toExtensionLabel()} files can be attached.`);
      return;
    }
    this.stageAttachment(slot, file);
  }

  protected onAttachmentPicked(slot: IrAttachmentSlot, event: Event): void {
    const input = event.target as HTMLInputElement | null;
    const file = input?.files?.[0];
    if (input) input.value = '';
    if (!file) return;
    this.stageAttachment(slot, file);
  }

  private stageAttachment(slot: IrAttachmentSlot, file: File): void {
    if (!hasAcceptedIrAttachmentExtension(file.name)) {
      this.notify.error(`Unsupported file type. Attach ${toExtensionLabel()}.`);
      return;
    }
    if (file.size > WWELL_IR_ATTACHMENT_MAX_BYTES) {
      this.notify.error('Attachment exceeds the 25 MB limit.');
      return;
    }

    const pendingId = this.store.addPendingAttachment(slot, file);
    // Straight into the upload dialog; dismissing it leaves the file staged on its row.
    this.dialog.open(WwellIrAttachmentUploadDialogComponent, {
      data: { store: this.store, slot, slotLabel: this.slotLabel(slot), slotIcon: this.slotIcon(slot), pendingId, irSeq: this.editorIrSeq() },
      panelClass: 'ir-upload-panel',
      autoFocus: false,
    });
  }

  private saveSettledResolvers: ((succeeded: boolean) => void)[] = [];
  private wasSavingForNavigation = false;

  constructor() {
    this.bindOffsetWellPicker(() => this.offsetWellAdd.control());
    this.bindOffsetWellPicker(() => this.offsetWellsEdit.control());
    effect(onCleanup => {
      const ctrl = this.requestReferenceEdit.control();
      if (!ctrl) return;
      const linkedRequestIdControl = ctrl.get('linkedRequestId');
      const subscription = linkedRequestIdControl?.valueChanges.subscribe((linkedRequestId: string) => {
        console.log('[IR:link] "Link to Request" select — valueChanges fired', { linkedRequestId });
        if (linkedRequestId) this.applyLinkedRequestWell(ctrl, linkedRequestId);
      });
      onCleanup(() => subscription?.unsubscribe());
    });

    effect(() => {
      const id = this.ir().id;
      if (this.previousIrId !== null && id !== this.previousIrId) this.closeAllSections();
      this.previousIrId = id;
    });

    effect(() => {
      if (this.ir().irSeq != null) return;
      untracked(() => this.requestReferenceEdit.open());
    });

    effect(() => {
      const saving = this.store.creatingIr();
      const settled = this.wasSavingForNavigation && !saving;
      this.wasSavingForNavigation = saving;
      if (!settled || this.saveSettledResolvers.length === 0) return;

      const succeeded = !this.store.createIrError();
      const resolvers = this.saveSettledResolvers;
      this.saveSettledResolvers = [];
      resolvers.forEach(resolve => resolve(succeeded));
    });
  }

  public discardUnsavedChanges(): void {
    this.closeAllSections();
  }

  public saveUnsavedChangesAndWait(): Promise<boolean> {
    return new Promise(resolve => {
      if (this.store.creatingIr()) {
        this.saveSettledResolvers.push(resolve);
        return;
      }
      if (this.saveAll()) {
        this.saveSettledResolvers.push(resolve);
      } else {
        resolve(false);
      }
    });
  }

  private applyLinkedRequestWell(referenceGroup: FormGroup, linkedRequestId: string): void {
    console.log('[IR:link] applyLinkedRequestWell — start', { linkedRequestId, eligibleWellsLoaded: this.store.linkedRequestWells().length });
    const well = findLinkedWell(linkedRequestId, this.store.linkedRequestWells());
    if (!well) {
      console.log('[IR:link] applyLinkedRequestWell — no matching eligible well, aborting patch', { linkedRequestId });
      return;
    }

    const hasElevation = linkedWellHasElevation(well);
    const hasCoordinates = linkedWellHasCoordinates(well);
    const hasTargetDepth = linkedWellHasTargetDepth(well);
    const hasTargetAquifer = linkedWellHasTargetAquifer(well);
    const hasSupportingBusiness = linkedWellHasSupportingBusiness(well);

    const patch: Record<string, string | number> = { wellName: well.wellName };
    if (hasElevation) patch['surfaceElevationFtAsl'] = well.surfaceElevation as number;
    patch['coordinates'] = hasCoordinates ? formatLinkedWellCoordinates(well) : '';
    if (hasTargetDepth) patch['targetDepthFt'] = well.targetDepth as number;
    patch['targetAquifer'] = hasTargetAquifer ? (well.targetAquifer as string) : '';
    patch['supportingBusiness'] = hasSupportingBusiness ? (well.supportings as string) : '';
    console.log('[IR:link] applyLinkedRequestWell — patching form group', {
      patch,
      hasElevation,
      hasCoordinates,
      hasTargetDepth,
      hasTargetAquifer,
      hasSupportingBusiness,
    });
    referenceGroup.patchValue(patch);

    referenceGroup.get('wellName')?.disable();
    this.setLinkedFieldLock(referenceGroup, 'surfaceElevationFtAsl', hasElevation);
    console.log('[IR:link] applyLinkedRequestWell — done', { rawValue: referenceGroup.getRawValue() });
  }

  /** `blank` empties every field outright — including those with a schema default (Liner Screen Size) —
   *  instead of restoring that default. Program Details uses it so the whole General step's program
   *  data is deleted; Geologist Contact is a separate section and is never touched by that reset. */
  private resetFlatSection(
    edit: SectionEditState<FormGroup, unknown>,
    section: WwellFormSectionConfig,
    label: string,
    key: 'program' | 'geologist',
    blank = false,
  ): void {
    edit.open();
    const ctrl = edit.control();
    if (!ctrl) return;
    const defaults: Record<string, unknown> = {};
    (section.fields ?? []).forEach(field => {
      defaults[field.key] = blank ? blankValueFor(field) : defaultValueFor(field);
    });
    ctrl.patchValue(defaults);
    // A saved record is deleted for real, right now (the button is a delete icon, not a draft edit). A
    // never-saved draft has nothing stored to delete, so there the form is just emptied.
    if (this.hasSavedIrSeq()) {
      this.deletingSection.set(key);
      // A save that does not start (nothing to save, a validation stop) never raises `isSaving`, so the marker is dropped here.
      if (!this.saveAll()) this.deletingSection.set(null);
      return;
    }
    this.notify.info(`${label} cleared.`);
  }

  private confirmThenAct(
    config: { title: string; rows?: { label: string; value: string }[]; confirmLabel: string },
    onConfirmed: () => void,
  ): void {
    this.confirmDialog
      .open({ title: config.title, rows: config.rows ?? [], confirmLabel: config.confirmLabel, severity: 'danger' })
      .pipe(takeUntilDestroyed(this.destroyRef))
      .subscribe(confirmed => {
        if (confirmed) onConfirmed();
      });
  }

  private confirmThenDelete(title: string, rows: { label: string; value: string }[], onConfirmed: () => void): void {
    this.confirmThenAct({ title, rows, confirmLabel: 'Delete' }, onConfirmed);
  }

  private currentRawFormValue(): WwellIrFormRawValue {
    return mapIntegratedRequirementToRawFormValue(this.ir());
  }

  private saveMergedRaw(raw: WwellIrFormRawValue): void {
    const current = this.ir();
    const status: Extract<IntegratedRequirementStatus, 'draft' | 'in-progress'> =
      current.status === 'draft' || current.status === 'in-progress' ? current.status : 'in-progress';
    const { payload } = mapIrFormValueToSubmission(raw, status);
    const irBulkPayload = buildWwellIrBulkSavePayload(raw, {
      irSeq: current.irSeq ?? null,
      existing: current,
      geologistEmpNumByName: this.store.geologistEmpNumByName(),
    });
    this.store.createIr(payload, irBulkPayload);
  }

  /** Returns whether a save was actually kicked off — `false` on every early-return branch below
   *  (nothing to save, or a blocked validation). `saveUnsavedChangesAndWait` needs this to settle
   *  its own promise immediately in those cases instead of waiting forever for a `creatingIr`
   *  transition that will now never happen. */
  protected saveAll(): boolean {
    // A record whose stored remarks still carry legacy HTML is saved even with nothing edited — the form already
    // holds the cleaned text, and saving is what writes it back.
    if (this.ir().irSeq !== null && !this.hasUnsavedChanges() && !this.ir().hasLegacyMarkup) {
      this.notify.info('No changes to save.');
      return false;
    }

    let raw = this.currentRawFormValue();

    const referenceCtrl = this.requestReferenceEdit.control();
    if (referenceCtrl) raw = { ...raw, reference: { main: referenceCtrl.getRawValue() } };

    // `letterNumber` carries a `Validators.required` in the schema (wwell-ir-form.schema.ts), but
    // nothing elsewhere in this method ever checks form validity before saving — it only gathers
    // raw values. Without this explicit gate, a blank Letter Number saved silently; the asterisk
    // and mat-error only ever showed once the field itself had been touched.
    if (!raw.reference.main.letterNumber?.trim()) {
      this.requestReferenceEdit.open();
      this.requestReferenceEdit.control()?.get('letterNumber')?.markAsTouched();
      this.expandedTimelineSteps.update(current => new Set(current).add('wellbore'));
      this.focusField('ire-sec-wellbore', 'Letter Number');
      this.notify.error('Letter Number is required before saving.');
      return false;
    }

    const programCtrl = this.programEdit.control();
    if (programCtrl) {
      const flat = programCtrl.getRawValue() as WwellIrRawProgramDetails;
      raw = {
        ...raw,
        programDetails: {
          main: {
            completionType: flat.completionType,
            completion: flat.completion,
            openHoleSizeIn: flat.openHoleSizeIn,
            linerSizeIn: flat.linerSizeIn,
            linerDepthFt: flat.linerDepthFt,
            acidization: flat.acidization,
            transmission: flat.transmission,
          },
          developmentTesting: {
            logging: flat.logging,
            mudLogging: flat.mudLogging,
            sampling: flat.sampling,
            flowTest: flat.flowTest,
            remarks: flat.remarks,
          },
        },
      };
    }

    const geologistCtrl = this.geologistContactEdit.control();
    if (geologistCtrl) raw = { ...raw, contacts: { ...raw.contacts, geologist: geologistCtrl.getRawValue() } };

    const hydrogeologyCtrl = this.hydrogeologyEdit.control();
    if (hydrogeologyCtrl) raw = { ...raw, hydrogeology: { main: hydrogeologyCtrl.getRawValue() } };

    const formationTopsCtrl = this.formationTopsEdit.control();
    if (formationTopsCtrl) raw = { ...raw, subsurface: { ...raw.subsurface, formationTops: formationTopsCtrl.getRawValue() } };

    const topsRemarksCtrl = this.topsRemarksEdit.control();
    if (topsRemarksCtrl) raw = { ...raw, subsurface: { ...raw.subsurface, remarks: topsRemarksCtrl.getRawValue() } };

    const casingCtrl = this.casingEdit.control();
    if (casingCtrl) raw = { ...raw, subsurface: { ...raw.subsurface, casing: casingCtrl.getRawValue() } };

    const offsetWellsCtrl = this.offsetWellsEdit.control();
    if (offsetWellsCtrl) raw = { ...raw, contacts: { ...raw.contacts, offsetWells: offsetWellsCtrl.getRawValue() } };

    const offsetRemarksCtrl = this.offsetRemarksEdit.control();
    if (offsetRemarksCtrl) raw = { ...raw, contacts: { ...raw.contacts, remarks: offsetRemarksCtrl.getRawValue() } };

    this.saveMergedRaw(raw);
    this.closeAllSections();
    return true;
  }

  /** Scrolls a labelled field of an open edit form into view and focuses it — what a blocked Save points at. */
  private focusField(sectionId: string, label: string): void {
    setTimeout(() => {
      const fields = Array.from(this.host.nativeElement.querySelectorAll<HTMLElement>(`#${sectionId} mat-form-field`));
      const field = fields.find(candidate => candidate.querySelector('mat-label')?.textContent?.includes(label));
      const input = field?.querySelector<HTMLInputElement | HTMLTextAreaElement>('input, textarea');
      if (!input) return;
      input.scrollIntoView({ block: 'center', behavior: 'smooth' });
      input.focus({ preventScroll: true });
    }, 50);
  }

  private closeAllSections(): void {
    this.flagUnsaved.set(false);
    this.requestReferenceEdit.close();
    this.programEdit.close();
    this.geologistContactEdit.close();
    this.hydrogeologyEdit.close();
    this.formationTopsEdit.close();
    this.topsRemarksEdit.close();
    this.casingEdit.close();
    this.offsetWellsEdit.close();
    this.offsetRemarksEdit.close();
    this.formationTopAdd.close();
    this.casingAdd.close();
    this.offsetWellAdd.close();
  }

  protected submitForReview(): void {
    // The submitted letter/payload is built from the saved record, so unsaved edits would be silently
    // left out — the button is disabled in that state; this guards any other entry point.
    if (this.hasPendingEdits()) return;
    this.store.submitIrForReview(this.ir());
  }

  /** Recalling stops a live SAP workflow — uses the same "confirm, then act" pattern as the other
   *  row-delete actions above, since it's not reversible from here (SAP itself would need a fresh
   *  Submit for Review). */
  protected recallIr(): void {
    const ir = this.ir();
    this.confirmThenAct(
      {
        title: 'Recall this Integrated Requirement?',
        rows: [
          { label: 'IR', value: ir.code },
          { label: 'Well', value: ir.requestReference.wellName || 'Untitled Well' },
        ],
        confirmLabel: 'Recall',
      },
      () => this.store.recallIr(ir),
    );
  }
}
```

```ts
import { ChangeDetectionStrategy, Component, effect, inject, input, signal } from '@angular/core';
import { CommonModule } from '@angular/common';
import { MatDialog } from '@angular/material/dialog';
import { MatIconModule } from '@angular/material/icon';
import { MatButtonModule } from '@angular/material/button';
import { MatTooltipModule } from '@angular/material/tooltip';
import { MatProgressSpinnerModule } from '@angular/material/progress-spinner';
import { UploadFileItem } from '@models/active-wwell/active-wwell-view.model';
import { MAX_UPLOAD_SIZE_BYTES, MAX_UPLOAD_SIZE_LABEL, hasAllowedExtension, toAcceptAttribute, toExtensionLabel } from '@core/config/file-upload.config';
import { WellDocsStore } from '@store/well-docs/well-docs.store';
import { PresDocsService } from '@services/pres-docs.service';
import { NotificationService } from '../notification/notification.service';
import { FilePreviewDialogComponent } from './file-preview-dialog/file-preview-dialog.component';

@Component({
  selector: 'app-file-upload',
  standalone: true,
  imports: [CommonModule, MatIconModule, MatButtonModule, MatTooltipModule, MatProgressSpinnerModule],
  templateUrl: './file-upload.component.html',
  styleUrl: './file-upload.component.scss',
  changeDetection: ChangeDetectionStrategy.OnPush,
})
export class FileUploadComponent {
  readonly epANum = input<number | null>(null);
  readonly date = input<string>('');
  readonly canUpload = input<boolean>(false);

  private readonly dialog = inject(MatDialog);
  private readonly svc = inject(PresDocsService);
  private readonly notify = inject(NotificationService);
  protected readonly docsStore = inject(WellDocsStore);

  protected readonly files = signal<UploadFileItem[]>([]);
  protected readonly accept = toAcceptAttribute();
  protected readonly extensionHint = `${toExtensionLabel()} · Max ${MAX_UPLOAD_SIZE_LABEL}`;
  protected readonly isDragOver = signal(false);
  protected readonly viewLoading = signal<Set<string>>(new Set());

  constructor() {
    // On well or date change: clear pending files + fetch server doc list
    effect(() => {
      const epANum = this.epANum();
      const date = this.date();
      this.files.set([]);
      if (epANum != null && date) {
        this.docsStore.loadDocList({ epANum, date });
      }
    });
  }

  protected onDragOver(event: DragEvent): void {
    event.preventDefault();
    event.stopPropagation();
    this.isDragOver.set(true);
  }

  protected onDragLeave(event: DragEvent): void {
    event.preventDefault();
    event.stopPropagation();
    this.isDragOver.set(false);
  }

  protected onDrop(event: DragEvent): void {
    event.preventDefault();
    event.stopPropagation();
    this.isDragOver.set(false);
    const dropped = Array.from(event.dataTransfer?.files ?? []);
    if (!dropped.length) return;
    this.addFiles(dropped);
  }

  protected onFileChange(event: Event): void {
    const input = event.target as HTMLInputElement | null;
    const selected = Array.from(input?.files ?? []);
    if (!selected.length) return;
    this.addFiles(selected);
    if (input) input.value = '';
  }

  private addFiles(selected: File[]): void {
    const invalidType = selected.filter(f => !hasAllowedExtension(f.name));
    const allowedType = selected.filter(f => hasAllowedExtension(f.name));
    const oversized = allowedType.filter(f => f.size > MAX_UPLOAD_SIZE_BYTES);
    const valid = allowedType.filter(f => f.size <= MAX_UPLOAD_SIZE_BYTES);

    if (invalidType.length) {
      const names = invalidType.map(f => f.name).join(', ');
      this.notify.error(`Unsupported file type and were skipped: ${names}`);
    }

    if (oversized.length) {
      const names = oversized.map(f => f.name).join(', ');
      this.notify.error(`File(s) exceed ${MAX_UPLOAD_SIZE_LABEL} limit and were skipped: ${names}`);
    }

    if (!valid.length) return;
    this.files.update((existing) => [
      ...existing,
      ...valid.map((file, index) => ({
        file,
        id: `${file.name}-${file.size}-${existing.length + index}`,
      })),
    ]);
  }

  protected previewFile(file: File): void {
    this.dialog.open(FilePreviewDialogComponent, {
      data: file,
      width: '60vw',
      height: '70vh',
      maxWidth: '98vw',
      maxHeight: '98vh',
      panelClass: 'file-preview-panel',
      autoFocus: false,
      enterAnimationDuration: '220ms',
    });
  }

  protected viewServerDoc(docName: string): void {
    const epANum = this.epANum();
    const date = this.date();
    if (epANum == null || !date || this.isViewLoading(docName)) return;

    this.setViewLoading(docName, true);
    this.svc.fetchDoc(docName, epANum, date).subscribe({
      next: (blob) => {
        const ext = docName.split('.').pop()?.toLowerCase() ?? '';
        const mimeMap: Record<string, string> = {
          pdf: 'application/pdf',
          jpg: 'image/jpeg', jpeg: 'image/jpeg', png: 'image/png',
          doc: 'application/msword',
          docx: 'application/vnd.openxmlformats-officedocument.wordprocessingml.document',
          xls: 'application/vnd.ms-excel',
          xlsx: 'application/vnd.openxmlformats-officedocument.spreadsheetml.sheet',
        };
        const mime = blob.type && blob.type !== 'application/octet-stream'
          ? blob.type : (mimeMap[ext] ?? 'application/octet-stream');
        const file = new File([blob], docName, { type: mime });
        this.setViewLoading(docName, false);
        this.dialog.open(FilePreviewDialogComponent, {
          data: file,
          width: '60vw',
          height: '70vh',
          maxWidth: '98vw',
          maxHeight: '98vh',
          panelClass: 'file-preview-panel',
          autoFocus: false,
          enterAnimationDuration: '220ms',
        });
      },
      error: () => this.setViewLoading(docName, false),
    });
  }

  protected removeFile(id: string): void {
    this.files.update((items) => items.filter((item) => item.id !== id));
  }

  protected deleteServerDoc(docName: string): void {
    const epANum = this.epANum();
    const uploadDate = this.date();
    if (epANum == null || !uploadDate || !docName) return;
    this.docsStore.removeSingleFile({ uploadDate, epANum, fileName: docName });
  }

  protected submitUpload(): void {
    const epANum = this.epANum();
    const date = this.date();
    const pending = this.files();
    if (epANum == null || !date || !pending.length) return;
    this.docsStore.uploadDocs({ files: pending.map(i => i.file), epANum, date });
    this.files.set([]);
  }

  protected isViewLoading(docName: string): boolean {
    return this.viewLoading().has(docName);
  }

  private setViewLoading(docName: string, loading: boolean): void {
    this.viewLoading.update(set => {
      const next = new Set(set);
      if (loading) {
        next.add(docName);
      } else {
        next.delete(docName);
      }
      return next;
    });
  }

  protected fileSizeLabel(size: number): string {
    if (size < 1024) return `${size} B`;
    if (size < 1024 * 1024) return `${(size / 1024).toFixed(1)} KB`;
    return `${(size / (1024 * 1024)).toFixed(1)} MB`;
  }

  protected fileIcon(file: File): string {
    const ext = file.name.split('.').pop()?.toLowerCase() ?? '';
    const mime = file.type.toLowerCase();
    if (mime.startsWith('image/') || ['jpg', 'jpeg', 'png', 'gif', 'webp'].includes(ext)) return 'image';
    if (mime === 'application/pdf' || ext === 'pdf') return 'picture_as_pdf';
    if (['doc', 'docx'].includes(ext)) return 'article';
    if (['xls', 'xlsx'].includes(ext)) return 'table_chart';
    return 'insert_drive_file';
  }

  protected fileTypeClass(file: File): string {
    const ext = file.name.split('.').pop()?.toLowerCase() ?? '';
    const mime = file.type.toLowerCase();
    if (mime.startsWith('image/') || ['jpg', 'jpeg', 'png', 'gif', 'webp'].includes(ext)) return 'image';
    if (mime === 'application/pdf' || ext === 'pdf') return 'pdf';
    if (['doc', 'docx'].includes(ext)) return 'doc';
    if (['xls', 'xlsx'].includes(ext)) return 'excel';
    return 'other';
  }

  protected fileExt(file: File): string {
    return file.name.split('.').pop()?.toUpperCase() ?? 'FILE';
  }

  protected serverFileIcon(docName: string): string {
    const ext = docName.split('.').pop()?.toLowerCase() ?? '';
    if (['jpg', 'jpeg', 'png', 'gif', 'webp'].includes(ext)) return 'image';

  }

  protected serverFileTypeClass(docName: string): string {
    const ext = docName.split('.').pop()?.toLowerCase() ?? '';
    if (['jpg', 'jpeg', 'png', 'gif', 'webp'].includes(ext)) return 'image';

    return 'other';
  }
```
