/** Schema vocabulary for the wwell-planning dynamic form engine. Add a field = edit a schema array, not a template. */

import type { Signal } from '@angular/core';

export type WwellFieldControlType =
  | 'text'
  | 'number'
  | 'textarea'
  | 'select'
  | 'combo'
  | 'rich-textarea'
  | 'toggle-group'
  | 'file';

export interface WwellFieldOption {
  readonly value: string;
  readonly label: string;
  /** Material icon ligature name — used by 'toggle-group' only. */
  readonly icon?: string;
}

export interface WwellFieldValidatorConfig {
  readonly required?: boolean;
  readonly pattern?: RegExp;
  readonly min?: number;
}

/**
 * Field is only shown (and its control enabled) while a sibling control in the same section
 * matches this condition. Exactly one of `equals` | `oneOf` | `notOneOf` | `includesLine` should
 * be set:
 *  - `equals`: sibling value === this value
 *  - `oneOf`: sibling value is one of this set (e.g. a client tier)
 *  - `notOneOf`: sibling value is absent from this set — the fallback for values outside every
 *    named tier (e.g. a client not yet assigned to a tier keeps the default field set)
 *  - `includesLine`: sibling value, split on `\n` and trimmed, includes this exact line — for a
 *    sibling that's a `type: 'rich-textarea'` checkbox list (see WwellRichTextareaComponent),
 *    whose committed value is every checked option newline-joined rather than a single picked
 *    value, so `equals` can only ever match when exactly one box is checked. `openHoleSizeIn`
 *    (wwell-ir-form.schema.ts) is the motivating case: gated on `completion` (a rich-textarea)
 *    having "Open Hole" checked, which must still hold when other options are checked alongside
 *    it — an `equals` gate there hid (and reset — see DynamicSectionComponent.syncControlEnabled)
 *    this field the instant a second option was checked, silently discarding whatever open hole
 *    size had been typed and leaving `resolveOhHalfWidth` (wellbore-depth.helpers.ts) with no real
 *    diameter to scale Open Hole against — it fell back to a fixed wide default, wider than an
 *    actual entered casing OD.
 */
export interface WwellFieldVisibilityCondition {
  readonly field: string;
  readonly equals?: string;
  readonly oneOf?: readonly string[];
  readonly notOneOf?: readonly string[];
  readonly includesLine?: string;
}

export interface WwellFieldConfig {
  readonly key: string;
  readonly label: string;
  readonly type: WwellFieldControlType;
  readonly placeholder?: string;
  readonly hint?: string;
  readonly suffix?: string;
  /** Used by 'select' | 'combo' | 'toggle-group'. Plain strings are shown as their own label.
   *  A Signal is resolved lazily via `resolveFieldOptions` — for options that come from a store
   *  (e.g. the IR wizard's "Link to Request" picker) rather than being known at schema-definition
   *  time (the schema array itself is a static, module-level constant with no injector). */
  readonly options?: readonly (string | WwellFieldOption)[] | Signal<readonly (string | WwellFieldOption)[]>;
  readonly validators?: WwellFieldValidatorConfig;
  readonly colSpan?: 1 | 2;
  readonly visibleWhen?: WwellFieldVisibilityCondition;
  /** Keeps the field's control in the row/section group (so its value is still carried, validated and saved) but draws no
   *  input for it — for an identifier the user never types, such as the offset well's EP A number. */
  readonly hidden?: boolean;
  /** Overrides the type-based empty default (e.g. pre-selecting a toggle-group option). */
  readonly defaultValue?: string | number;
  /** 'select' only — adds a filter input inside the option panel, narrowing `options` as the user
   *  types. For a picker backed by a long or live-fetched list (e.g. the IR wizard's "Link to
   *  Request", backed by every well eligible for IR initiation) where scrolling an unfiltered
   *  mat-select is impractical; the short, hand-picked lists most 'select' fields use don't need it. */
  readonly searchable?: boolean;
  /** Disables the control — for values derived elsewhere (e.g. a server-computed metric) rather
   *  than typed in directly. Pair with a `placeholder` like 'auto' to signal that visually. */
  readonly readonly?: boolean;
  /** Renders this field's single control value as several read-only labelled boxes side by side
   *  instead of one — e.g. a combined "lat, long" string split into a Latitude box and a
   *  Longitude box. Applies in both edit and review mode; the control itself is untouched (still
   *  one value, one key), this only changes how DynamicFieldComponent draws it. */
  readonly splitDisplay?: (value: string) => readonly { readonly label: string; readonly value: string }[];
}

export interface WwellRepeatableRowConfig {
  readonly key: string;
  readonly addLabel: string;
  readonly rowFields: readonly WwellFieldConfig[];
  /** How many empty rows to seed the array with on load. */
  readonly minRows?: number;
  /** If set, the array itself is invalid until it holds at least this many rows — an empty array is not "complete" just because it has no invalid children. */
  readonly minRequiredRows?: number;
  /** `false` hides `RepeatableGroupComponent`'s own "+ Add Row" button — for a caller that drives
   *  adding rows itself (e.g. a section-head "+" icon that appends a row on every click), so there's
   *  exactly one add affordance instead of two doing the same thing. Defaults to shown (`true`). */
  readonly showAddButton?: boolean;
}

export interface WwellFormSectionConfig {
  readonly key: string;
  readonly title: string | null;
  /** Mutually exclusive with `repeatable`. */
  readonly fields?: readonly WwellFieldConfig[];
  readonly repeatable?: WwellRepeatableRowConfig;
}

export interface WwellFormStepConfig {
  readonly key: string;
  readonly label: string;
  readonly sections: readonly WwellFormSectionConfig[];
}

export function wwellFieldOptionLabel(option: string | WwellFieldOption): string {
  return typeof option === 'string' ? option : option.label;
}

export function wwellFieldOptionValue(option: string | WwellFieldOption): string {
  return typeof option === 'string' ? option : option.value;
}

export function wwellFieldOptionIcon(option: string | WwellFieldOption): string | undefined {
  return typeof option === 'string' ? undefined : option.icon;
}

/** Unwraps a field's `options` (static array or live Signal) into the array consumers iterate. */
export function resolveFieldOptions(field: WwellFieldConfig): readonly (string | WwellFieldOption)[] {
  const options = field.options;
  if (!options) return [];
  return typeof options === 'function' ? options() : options;
}

import { IntegratedRequirement } from '@models/wwell-planning/integrated-requirement.model';
import { toUtmAxes } from '../wwell-ir-coordinates.util';
import { isLinerCompletionType } from '../wwell-ir-form/wwell-ir-completion-type.config';

/** Flags a value that's actually meaningful, not just informational — e.g. a nonzero H2S
 *  reading should read as amber/danger regardless of which section it sits in. Omitted (or
 *  `'default'`) renders as a plain field. */
export type RowTone = 'default' | 'warning' | 'danger' | 'success';

export interface KvRow {
  readonly label: string;
  readonly value: string;
  readonly tone?: RowTone;
  /** Marks a value picked from a fixed set of options (a completion type, a test type, ...).
   *  A multi-select's picks arrive newline-separated in `value` and are shown one per line. */
  readonly options?: boolean;
  /** An identifier or a measurement (an IR number, a depth, a reading): set in the mono face, like the figures elsewhere. */
  readonly mono?: boolean;
}

export const EMPTY_VALUE = '—';

export function fmtNum(value: number | null | undefined, unit?: string): string {
  if (value === null || value === undefined) return EMPTY_VALUE;
  return unit ? `${value} ${unit}` : `${value}`;
}

export function fmtStr(value: string | null | undefined): string {
  return value ? value : EMPTY_VALUE;
}

/** `linkedRequestId` is the raw prewap_seq (e.g. "5008") — resolve it to the well name. Prefers
 *  `r.wellName` (populated from exad_rcd_prewap on a saved IR, or by the picker on a fresh one)
 *  over the "eligible for a new IR" wellOptions lookup: get_ir_wells_prc.sql's eligibility filter
 *  excludes any well that already has an IR header, so the instant this IR is saved its own
 *  linked well permanently drops out of that list and the lookup can never succeed again — the
 *  wellOptions param only still helps a brand-new, not-yet-saved draft. Falls back to the raw id
 *  if neither source has a name yet, rather than showing nothing. */
function resolveLinkedRequestLabel(
  linkedRequestId: string | null,
  wellName: string,
  wellOptions: readonly { readonly value: string; readonly label: string }[],
): string {
  if (!linkedRequestId) return EMPTY_VALUE;
  return wellName || wellOptions.find(option => option.value === linkedRequestId)?.label || linkedRequestId;
}

/** Splits a "Northing X, Easting Y, Zone Z" coordinate string into its own Northing/Easting/UTM
 *  Zone rows — falls back to a single "Coordinates" row when the text doesn't read as that shape
 *  (unset, or hand-typed free text), same fallback `toUtmAxes` uses for the wellbore panel's
 *  Location card. Shown exactly as the raw survey values, no unit conversion. */
function coordinateKvRows(coordinates: string | null | undefined): readonly KvRow[] {
  if (!coordinates) return [{ label: 'Coordinates', value: EMPTY_VALUE, mono: true }];
  return toUtmAxes(coordinates).map(axis => ({
    label: axis.label,
    value: axis.value,
    mono: true,
  }));
}

/** Mirrors the IR wizard's "Request Reference" step (wwell-ir-form.schema.ts) field-for-field,
 *  so the detail page's sections line up 1:1 with the stepper the data was captured through. */
export function buildRequestReferenceRows(
  ir: IntegratedRequirement,
  linkedRequestWellOptions: readonly { readonly value: string; readonly label: string }[] = [],
  /** "Link to Water Well" is only a picker for creating/editing an IR — read-only displays leave it out (the
   *  Water Well Name row below already says which well it is). */
  includeLinkedWell = false,
): readonly KvRow[] {
  const r = ir.requestReference;
  return [
    ...(includeLinkedWell
      ? [{ label: 'Link to Water Well', value: resolveLinkedRequestLabel(r.linkedRequestId, r.wellName, linkedRequestWellOptions) }]
      : []),
    { label: 'Water Well Name', value: fmtStr(r.wellName) },
    { label: 'Letter Number', value: fmtStr(r.letterNumber), mono: true },
    { label: 'Supporting Business', value: fmtStr(r.supportingBusiness) },
    { label: 'Target Aquifer', value: fmtStr(r.targetAquifer), options: true },
    // A brand-new draft is created through the required-number fallback in
    // mapIrFormValueToSubmission (targetDepthFt ?? 0) before the author has entered a real value —
    // treat a literal 0 as "not yet entered" rather than a misleading real reading (see the
    // matching fallback in mapIntegratedRequirementToFormPatch, wwell-ir-edit.mapper.ts).
    { label: 'Target Depth', value: r.targetDepthFt ? fmtNum(r.targetDepthFt, 'ft') : EMPTY_VALUE, mono: true },
    { label: 'Surface Elevation', value: `${fmtNum(r.surfaceElevationFtAsl)} ft ASL`, mono: true },
    ...coordinateKvRows(r.coordinates),
  ];
}

/** Mirrors the schema's "Program Details" section (2.1) — its own rail leaf/edit unit, separate
 *  from Development & Testing (2.2) below. `openHoleSizeIn` only ever means something when the
 *  completion type includes 'Open Hole' (see the field's `visibleWhen` gate in the schema), so
 *  it's only included then rather than showing a dangling "—" the rest of the time.
 *
 *  `completion` is a multi-select — any number of `WWELL_COMPLETION_OPTIONS` can be checked at
 *  once, each becoming its own `\n`-joined line (see wwell-ir-wellbore-diagram.mapper.ts's own
 *  doc comment on this same shape) — so this must check "is 'Open Hole' one of the checked
 *  lines?", the same `includesLine` semantics the schema's `visibleWhen` gate uses
 *  (dynamic-section.component.ts), not a straight equality that only ever matched the
 *  single-selection case and otherwise silently hid an already-entered size. */
export function buildProgramMainRows(ir: IntegratedRequirement): readonly KvRow[] {
  const p = ir.programDetails;
  const rows: KvRow[] = [{ label: 'Completion', value: fmtStr(p.completionType) }];

  // Every pick is drawn inside the open hole, so its size is listed whatever the pick.
  if (p.completionType) {
    rows.push({ label: 'Open Hole Size', value: fmtStr(p.openHoleSizeIn) });
  }
  if (isLinerCompletionType(p.completionType)) {
    rows.push({ label: 'Liner Screen Size', value: fmtStr(p.linerSizeIn) });
    rows.push({ label: 'Liner Depth (ft)', value: p.linerDepthFt != null ? String(p.linerDepthFt) : 'Target depth' });
  }
  rows.push({ label: 'Completion Remarks', value: fmtStr(p.completion), options: true });

  rows.push(
    { label: 'Acidization', value: fmtStr(p.acidization), options: true },
    { label: 'Transmission', value: fmtStr(p.transmission), options: true },
  );
  return rows;
}

/** Mirrors the schema's "Development & Testing" section (2.2) — see buildProgramMainRows' doc
 *  comment for why this is a separate function rather than one combined list. */
export function buildProgramDevRows(ir: IntegratedRequirement): readonly KvRow[] {
  const d = ir.programDetails.developmentTesting;
  const isFlowTest = ir.testProgram.testType === 'flow-test';
  return [
    { label: 'Logging', value: fmtStr(d.logging), options: true },
    { label: 'Mud Logging', value: fmtStr(d.mudLogging), options: true },
    { label: 'Sampling', value: fmtStr(d.sampling), options: true },
    // Hydrology Test Type decides which one applies: a flow test shows Flow Test (FLOW_RMKS) only,
    // a pump test shows Development/Testing (DEV_PUMP_TEST) only.
    isFlowTest
      ? { label: 'Flow Test', value: fmtStr(d.flowTest), options: true }
      : { label: 'Development/Testing', value: fmtStr(d.remarks) },
  ];
}

/** Mirrors the wizard's "Estimated Hydrology" step, including its test-type-conditional
 *  fields (SIWHP for a flow test; water level + pump depth for a pump test) — see the
 *  matching `visibleWhen` gates on `siwhpPsi`/`estimatedWaterLevelFt`/`estimatedPumpDepthFt`
 *  in wwell-ir-form.schema.ts. Any nonzero H2S gets `tone: 'warning'` — a toxic-gas reading is
 *  worth flagging on sight, not just listed at the same visual weight as TDS/rate. */
export function buildHydrogeologyRows(ir: IntegratedRequirement): readonly KvRow[] {
  const isFlowTest = ir.testProgram.testType === 'flow-test';
  const h2sPpm = ir.hydrogeology.h2sPpm;
  const rows: KvRow[] = [
    { label: 'Aquifer', value: fmtStr(ir.hydrogeology.aquifer), options: true },
    { label: 'Test Type', value: isFlowTest ? 'Flow Test' : 'Pump Test', options: true },
    { label: 'Estimated TDS', value: fmtNum(ir.hydrogeology.estimatedTdsPpm, 'ppm'), mono: true },
    { label: 'Estimated Rate', value: fmtNum(ir.testProgram.estimatedRateGpm, 'GPM'), mono: true },
    {
      label: 'H2S',
      value: fmtNum(h2sPpm, 'ppm'),
      tone: h2sPpm ? 'warning' : 'default',
      mono: true,

  private previewRows<TRow>(
    live: readonly TRow[] | null,
    saved: () => readonly TRow[],
    draftRows: readonly TRow[] | null,
    keyOf: (row: TRow) => string,
  ): readonly TRow[] | null {
    if (!draftRows?.length) return live;
    const rows = live ?? saved();
    const keys = new Set(rows.map(keyOf));
    // "Add" refuses a row whose natural key is already taken (it would overwrite that row) — including by
    // another staged draft row — so a colliding draft is not drawn either.
    const additions = draftRows.filter(draft => {
      const key = keyOf(draft);
      if (keys.has(key)) return false;
      keys.add(key);
      return true;
    });
    return additions.length ? [...rows, ...additions] : live;
  }

  /** Commits every row staged in a "+" add-form's own FormArray into the section's live FormArray — the
   *  same array inline edits and deletes write to, so Save diffs it like any other change. All-or-nothing:
   *  Save matches rows on their natural key, so if ANY staged row's key is already taken (by an existing
   *  row, or by another row in the same batch) the whole commit is refused rather than silently overwriting
   *  that row, or committing some rows and leaving the colliding one stranded in the add-form. */
  private commitNewRows<TRow>(
    add: SectionEditState<FormArray>,
    section: WwellFormSectionConfig,
    commit: {
      readonly existingKeys: readonly string[];
      readonly keyOf: (row: TRow) => string;
      readonly liveArray: () => FormArray | null;
      readonly duplicate: (key: string) => string;
      readonly added: (count: number) => string;
    },
  ): void {
    const form = add.control();
    if (!form || form.invalid || form.length === 0) return;

    const rows = form.getRawValue() as TRow[];
    const keys = new Set(commit.existingKeys);
    for (const row of rows) {
      const key = commit.keyOf(row);
      if (keys.has(key)) {
        this.notify.warn(commit.duplicate(key));
        return;
      }
      keys.add(key);
    }

    const array = commit.liveArray();
    if (!array) return;
    rows.forEach(row => {
      addRepeatableRow(array, section.repeatable?.rowFields ?? [], this.fb);
      array.at(array.length - 1).patchValue(row as object);
    });
    add.close();
    this.notify.info(commit.added(rows.length));
  }

  private saveSettledResolvers: ((succeeded: boolean) => void)[] = [];
  private wasSavingForNavigation = false;

  constructor() {
    this.bindOffsetWellPicker(() => this.offsetWellAdd.control());
    this.bindOffsetWellPicker(() => this.offsetWellsEdit.control());
    effect(onCleanup => {
      const ctrl = this.requestReferenceEdit.control();
      if (!ctrl) return;
      const linkedRequestIdControl = ctrl.get('linkedRequestId');
      const subscription = linkedRequestIdControl?.valueChanges.subscribe((linkedRequestId: string) => {
        console.log('[IR:link] "Link to Request" select — valueChanges fired', { linkedRequestId });
        if (linkedRequestId) this.applyLinkedRequestWell(ctrl, linkedRequestId);
      });
      onCleanup(() => subscription?.unsubscribe());
    });

    effect(() => {
      const id = this.ir().id;
      if (this.previousIrId !== null && id !== this.previousIrId) this.closeAllSections();
      this.previousIrId = id;
    });

    effect(() => {
      if (this.ir().irSeq != null) return;
      untracked(() => this.requestReferenceEdit.open());
    });

    effect(() => {
      const saving = this.store.creatingIr();
      const settled = this.wasSavingForNavigation && !saving;
      this.wasSavingForNavigation = saving;
      if (!settled || this.saveSettledResolvers.length === 0) return;

      const succeeded = !this.store.createIrError();
      const resolvers = this.saveSettledResolvers;
      this.saveSettledResolvers = [];
      resolvers.forEach(resolve => resolve(succeeded));
    });
  }

  public discardUnsavedChanges(): void {
    this.closeAllSections();
  }

  public saveUnsavedChangesAndWait(): Promise<boolean> {
    return new Promise(resolve => {
      if (this.store.creatingIr()) {
        this.saveSettledResolvers.push(resolve);
        return;
      }
      if (this.saveAll()) {
        this.saveSettledResolvers.push(resolve);
      } else {
        resolve(false);
      }
    });
  }

  private resetFlatSection(

      defaults[field.key] = blank ? blankValueFor(field) : defaultValueFor(field);
    });
    ctrl.patchValue(defaults);
    // A saved record is deleted for real, right now (the button is a delete icon, not a draft edit). A
    // never-saved draft has nothing stored to delete, so there the form is just emptied.
    if (this.hasSavedIrSeq()) {

      current.status === 'draft' || current.status === 'in-progress' ? current.status : 'in-progress';
    const { payload } = mapIrFormValueToSubmission(raw, status);
    const irBulkPayload = buildWwellIrBulkSavePayload(raw, {
      irSeq: current.irSeq ?? null,
      existing: current,
      geologistEmpNumByName: this.store.geologistEmpNumByName(),
    });

  /** Returns whether a save was actually kicked off — `false` on every early-return branch below
   *  (nothing to save, or a blocked validation). `saveUnsavedChangesAndWait` needs this to settle
   *  its own promise immediately in those cases instead of waiting forever for a `creatingIr`
   *  transition that will now never happen. */
  protected saveAll(): boolean {
    // A record whose stored remarks still carry legacy HTML is saved even with nothing edited — the form already
    // holds the cleaned text, and saving is what writes it back.
    if (this.ir().irSeq !== null && !this.hasUnsavedChanges() && !this.ir().hasLegacyMarkup) {
      this.notify.info('No changes to save.');
      return false;
    }

    let raw = this.currentRawFormValue();

    const referenceCtrl = this.requestReferenceEdit.control();
    if (referenceCtrl) raw = { ...raw, reference: { main: referenceCtrl.getRawValue() } };

    // `letterNumber` carries a `Validators.required` in the schema (wwell-ir-form.schema.ts), but
    // nothing elsewhere in this method ever checks form validity before saving — it only gathers
    // raw values. Without this explicit gate, a blank Letter Number saved silently; the asterisk
    // and mat-error only ever showed once the field itself had been touched.
    if (!raw.reference.main.letterNumber?.trim()) {
      this.requestReferenceEdit.open();
      this.requestReferenceEdit.control()?.get('letterNumber')?.markAsTouched();
      this.notify.error('Letter Number is required before saving.');
      return false;
    }

    const programCtrl = this.programEdit.control();
    if (programCtrl) {
      const flat = programCtrl.getRawValue() as WwellIrRawProgramDetails;
      raw = {
        ...raw,
        programDetails: {
          main: {
            completionType: flat.completionType,
            completion: flat.completion,
            openHoleSizeIn: flat.openHoleSizeIn,
            linerSizeIn: flat.linerSizeIn,
            linerDepthFt: flat.linerDepthFt,
            acidization: flat.acidization,
            transmission: flat.transmission,
          },
          developmentTesting: {
            logging: flat.logging,
            mudLogging: flat.mudLogging,
            sampling: flat.sampling,
            flowTest: flat.flowTest,
            remarks: flat.remarks,
          },
        },
      };
    }

    const geologistCtrl = this.geologistContactEdit.control();
    if (geologistCtrl) raw = { ...raw, contacts: { ...raw.contacts, geologist: geologistCtrl.getRawValue() } };

    const hydrogeologyCtrl = this.hydrogeologyEdit.control();
    if (hydrogeologyCtrl) raw = { ...raw, hydrogeology: { main: hydrogeologyCtrl.getRawValue() } };

    const formationTopsCtrl = this.formationTopsEdit.control();
    if (formationTopsCtrl) raw = { ...raw, subsurface: { ...raw.subsurface, formationTops: formationTopsCtrl.getRawValue() } };

    const topsRemarksCtrl = this.topsRemarksEdit.control();
    if (topsRemarksCtrl) raw = { ...raw, subsurface: { ...raw.subsurface, remarks: topsRemarksCtrl.getRawValue() } };

    const casingCtrl = this.casingEdit.control();
    if (casingCtrl) raw = { ...raw, subsurface: { ...raw.subsurface, casing: casingCtrl.getRawValue() } };

    const offsetWellsCtrl = this.offsetWellsEdit.control();
    if (offsetWellsCtrl) raw = { ...raw, contacts: { ...raw.contacts, offsetWells: offsetWellsCtrl.getRawValue() } };

    const offsetRemarksCtrl = this.offsetRemarksEdit.control();
    if (offsetRemarksCtrl) raw = { ...raw, contacts: { ...raw.contacts, remarks: offsetRemarksCtrl.getRawValue() } };

    this.saveMergedRaw(raw);
    this.closeAllSections();
    return true;
  }

  private closeAllSections(): void {
    this.flagUnsaved.set(false);
    this.requestReferenceEdit.close();
    this.programEdit.close();
    this.geologistContactEdit.close();
    this.hydrogeologyEdit.close();
    this.formationTopsEdit.close();
    this.topsRemarksEdit.close();
    this.casingEdit.close();
    this.offsetWellsEdit.close();
    this.offsetRemarksEdit.close();
    this.formationTopAdd.close();
    this.casingAdd.close();
    this.offsetWellAdd.close();
  }

  protected submitForReview(): void {
    // The submitted letter/payload is built from the saved record, so unsaved edits would be silently
    // left out — the button is disabled in that state; this guards any other entry point.
    if (this.hasPendingEdits()) return;
    this.store.submitIrForReview(this.ir());
  }

  /** Recalling stops a live SAP workflow — uses the same "confirm, then act" pattern as the other
   *  row-delete actions above, since it's not reversible from here (SAP itself would need a fresh
   *  Submit for Review). */
  protected recallIr(): void {
    const ir = this.ir();
    this.confirmThenAct(
      {
        title: 'Recall this Integrated Requirement?',
        rows: [
          { label: 'IR', value: ir.code },
          { label: 'Well', value: ir.requestReference.wellName || 'Untitled Well' },
        ],
        confirmLabel: 'Recall',
      },
      () => this.store.recallIr(ir),
    );
  }
}

 * WwellIrFormRawValue standalone, repeatable rows included. Used by the IR editor's per-section

export function buildWwellIrFormSchema(
  linkedRequestOptions: Signal<readonly WwellFieldOption[]>,
  geologistOptions: Signal<readonly string[]>,
  dropdownMenus: Signal<Readonly<Record<string, readonly string[]>>> = signal({}),
  aquiferOptions: Signal<readonly WwellFieldOption[]> = signal([]),
  formationOptions: Signal<readonly WwellFieldOption[]> = signal([]),
): readonly WwellFormStepConfig[] {
  return WWELL_IR_FORM_SCHEMA.map(step => {
    if (step.key === 'subsurface') {
      return {
        ...step,
        sections: step.sections.map(section =>
          section.key === 'formationTops' && section.repeatable
            ? {
                ...section,
                repeatable: {
                  ...section.repeatable,
                  rowFields: section.repeatable.rowFields.map(field => (field.key === 'name' ? { ...field, options: formationOptions } : field)),
                },
              }
            : section.key === 'casing' && section.repeatable
              ? {
                  ...section,
                  repeatable: {
                    ...section.repeatable,
                    rowFields: section.repeatable.rowFields.map(field => {
                      if (field.key === 'type') return { ...field, options: computed(() => dropdownMenus()[WWELL_IR_CASING_TYPE_MENU] ?? []) };
                      if (field.key === 'sizeIn') return { ...field, options: computed(() => dropdownMenus()[WWELL_IR_CASING_SIZE_MENU] ?? []) };
                      return field;
                    }),
                  },
                }
              : section,
        ),
      };
    }
    if (step.key === 'hydrogeology') {
      return {
        ...step,
        sections: step.sections.map(section => ({
          ...section,
          fields: section.fields?.map(field => (field.key === 'aquifer' ? { ...field, options: aquiferOptions } : field)),
        })),
      };
    }
    if (step.key === 'programDetails') {
      return {
        ...step,
        sections: step.sections.map(section => ({
          ...section,
          fields: section.fields?.map(field => withDropdownMenuOptions(field, dropdownMenus)),
        })),
      };
    }
    if (step.key === 'reference') {
      return {
        ...step,
        sections: step.sections.map(section => {
          if (section.key !== 'main' || !section.fields) return section;
          return {
            ...section,
            fields: section.fields.map(field =>
              field.key === 'linkedRequestId' ? { ...field, options: linkedRequestOptions } : field,
            ),
          };
        }),
      };
    }
    if (step.key === 'contacts') {
      return {
        ...step,
        sections: step.sections.map(section => {
          if (section.key !== 'geologist' || !section.fields) return section;
          return {
            ...section,
            fields: section.fields.map(field =>
              field.key === 'geologistPrimary' || field.key === 'geologistSecondary' ? { ...field, options: geologistOptions } : field,

 *  already present in `existing` is genuinely new and must be sent as INSERT, not UPDATE, or the
 *  backend's `WHERE <key match>` finds nothing and silently drops it. */
export const casingKey = (row: { readonly type: string; readonly sizeIn: string }): string => `${row.type} ${row.sizeIn}`;
export const topKey = (row: { readonly name: string }): string => row.name;

/** Only the columns save_casing's/save_tops' UPDATE branch actually writes (type/size are the
 *  match key itself, not a writable column) — comparing anything else could flag a row "changed"

  existingRow.depthFtBgl === (row.depthFtBgl ?? 0) && (existingRow.remarks ?? null) === (row.remarks || null);

const topRowUnchanged = (existingRow: IntegratedRequirementFormationTop, row: WwellIrRawFormationTop): boolean =>
  existingRow.depthFt === (row.depthFt ?? 0) && (existingRow.subSeaFtAsl ?? null) === (row.subSeaFtAsl ?? null);

function buildCasingRows(
  raw: readonly WwellIrRawCasingRow[],
  irSeq: number | null,
  existing: readonly IntegratedRequirementCasingRow[] | null | undefined,
): IrCasingRow[] {
  const currentRows = raw.filter(row => !isBlankCasingRow(row));

  if (existing == null) {
    const action = resolveRowAction(irSeq);
    return currentRows.map(row => ({
      action,
      ir_seq: irSeq,
      csg_size: row.sizeIn,
      csg_type: row.type,
      csg_depth: row.depthFtBgl ?? 0,
      csg_remarks: row.remarks || null,
    }));
  }

  const existingByKey = new Map(existing.map(row => [casingKey(row), row]));
  const keptKeys = new Set<string>();

  // A row this save didn't actually change is sent as '' (no-op), not a redundant UPDATE — see
  // IrRowAction's doc comment: the backend's per-row action switch then skips the DML entirely
  // instead of re-writing a row with the exact values it already has on every single save.
  const upserts: IrCasingRow[] = currentRows.map(row => {
    const key = casingKey(row);
    const existingRow = existingByKey.get(key);
    if (existingRow) keptKeys.add(key);
    const action: IrRowAction = existingRow ? (casingRowUnchanged(existingRow, row) ? '' : 'UPDATE') : 'INSERT';
    return {
      action,
      ir_seq: existingRow ? irSeq : null,
      csg_size: row.sizeIn,
      csg_type: row.type,
      csg_depth: row.depthFtBgl ?? 0,
      csg_remarks: row.remarks || null,
    };
  });

  const deletes: IrCasingRow[] = existing
    .filter(row => !keptKeys.has(casingKey(row)))
    .map(row => ({
      action: 'DELETE',
      ir_seq: irSeq,
      csg_size: row.sizeIn,
      csg_type: row.type,
      csg_depth: row.depthFtBgl,
      csg_remarks: row.remarks,
    }));

  return [...upserts, ...deletes];
}

 *  aquifer must DELETE the old row, not silently orphan it under a key nothing will ever match again. */
function buildHydrogeologyRows(
  raw: WwellIrRawHydrology,
  irSeq: number | null,
  existing: Pick<IntegratedRequirement, 'hydrogeology' | 'testProgram'> | null | undefined,

      est_productivity: null,
      flowing: null,
      siwhp: null,
      h2s: null,
    });
  }

  if (currentAquifer) {
    const isExisting = existingAquifer === currentAquifer;
    const action: IrRowAction = isExisting

 * is a blanket `WHEN MATCHED THEN UPDATE SET <every column the proc's JSON_TABLE declares>` off
 * this same JSON row — a column this app's payload omits (design_rpt, status, design_phase,
 * w_prim_hid, w_num, sta_water_lvl, tds, completion_type, notes, publish_dt/flg/by, curr_flg) comes
 * back NULL from JSON_TABLE and gets written as NULL, not left alone. If a wd_seq this UPDATE
 * targets already carries real data from that other Well Design workflow, this silently wipes it.
 * That risk already existed for pump_lvl-only saves before this change; sending Completion more
 * often (it's now the primary driver, not just an occasional pump-depth edit) makes it fire more
 * often too. Not fixed here — would need either the backend proc to COALESCE(t.col, h.col) instead
 * of overwriting, or this app reading back and re-sending every other column untouched.
 */
function buildWellDesignRow(
  programMain: WwellIrRawProgramMain,
  pumpDepthFt: number | null,
  prewapSeq: number | null,
  existingWellDesignSeq: number | null | undefined,
  existingPumpDepthFt: number | null | undefined,
  existingProgramDetails: Pick<IntegratedRequirementProgramDetails, 'completionType' | 'completion' | 'openHoleSizeIn' | 'linerSizeIn' | 'linerDepthFt'> | null | undefined,
): IrWellDesignRow | null {
  const wdSeq = existingWellDesignSeq ?? null;
  const flags = deriveCompletionElementFlags(
    programMain.completion,
    programMain.openHoleSizeIn || null,
    programMain.completionType,
    programMain.linerSizeIn || null,
    programMain.linerDepthFt ?? null,
  );
  // IntegratedRequirementWellDesignFlags is camelCase (domain-model shape, shared with the
  // wellbore-diagram mapper); the wire payload below needs update_ir_data_prc's own snake_case
  // column names (see docs/ir/update_ir_data_prc.sql) — translate rather than rename one side.
  const flagColumns = {
    oh_flg: flags.ohFlg,
    oh_remarks: flags.ohRemarks,
    gp_flg: flags.gpFlg,
    gp_remarks: flags.gpRemarks,
    perforated_flg: flags.perforatedFlg,
    perforated_remarks: flags.perforatedRemarks,
    ls_flg: flags.lsFlg,
    ls_remarks: flags.lsRemarks,
  };

  if (wdSeq != null) {
    const unchanged =
      (existingPumpDepthFt ?? null) === pumpDepthFt &&
      (existingProgramDetails?.completionType ?? '') === programMain.completionType &&
      (existingProgramDetails?.completion ?? '') === programMain.completion &&
      (existingProgramDetails?.openHoleSizeIn ?? null) === (programMain.openHoleSizeIn || null) &&
      (existingProgramDetails?.linerSizeIn ?? null) === (programMain.linerSizeIn || null) &&
      (existingProgramDetails?.linerDepthFt ?? null) === (programMain.linerDepthFt ?? null);
    const action: IrRowAction = unchanged ? '' : 'UPDATE';
    return { action, wd_seq: wdSeq, prewap_seq: prewapSeq, pump_lvl: pumpDepthFt, ...flagColumns };
  }

  if ((pumpDepthFt == null && !programMain.completion && !programMain.completionType) || prewapSeq == null) return null;
  return { action: 'INSERT', wd_seq: null, prewap_seq: prewapSeq, pump_lvl: pumpDepthFt, ...flagColumns };
}

  ];
}

import { S3Service } from '@services/s3/s3.service';
import { NotificationService } from '@shared/components/notification/notification.service';
import { WwellIrAttachmentViewerComponent } from '../../wwell-ir-attachment-viewer/wwell-ir-attachment-viewer.component';
import { WellboreDiagramData } from '@models/well-design/wellbore-diagram.model';

import { WellBoreViewComponent } from '../../../../presentation/well-bore-view/well-bore-view.component';
import { WwellIrDepthScaleComponent } from '../../wwell-ir-depth-scale/wwell-ir-depth-scale.component';
import {
  buildWellboreDiagramDataFromIntegratedRequirement,
  hasWellboreDiagramDataForIntegratedRequirement,

import { WWELL_IR_LETTER_DIAGRAM_LAYOUT } from '../../wwell-ir-editor/wwell-ir-preview-diagram-layout';
import { resolveScaleDepthFt } from '../../wwell-ir-depth-scale/wwell-ir-depth-ticks.util';
import { todayIsoDate } from '../../../shared/wwell-approvals/wwell-approval-request.util';
import { WWELL_COMPLETION_TYPE, isLinerCompletionType } from '../../wwell-ir-form/wwell-ir-completion-type.config';
import { captureWellboreSnapshot } from './wwell-ir-wellbore-snapshot.util';

interface LetterTopsRow {
  readonly name: string;
  readonly depthFt: number;
  readonly subSeaFt: number | null;
}

/** "Letter" step of the read-only IR view — a printable-looking rendition of the GWD Integrated
 *  Requirements Letter, populated straight from the submitted record (never a separate document
 *  fetched from the backend — there is no such endpoint yet). Section numbering/labels follow
 *  the reference GWD letter template; every value comes directly off `IntegratedRequirement`
 *  fields rather than the editor's `KvRow` builders, since the letter's own labels ("Completion
 *  Design", "Logging Program"…) intentionally read as prose, not as the wizard's field names.
 *  The attachment list is appended at the end, per the same record's `attachmentNames`.
 *
 *  The Well Bore Diagram section reuses the exact same diagram build (
 *  buildWellboreDiagramDataFromIntegratedRequirement) as the "Well Design" step's own preview, so
 *  the letter's picture of the well always agrees with what that tab shows for the same record. */
/** `"*" + remarks`, or '' when there are none (blank/whitespace); a remark that already starts with "*" is kept as typed. */
function letterFootnote(remarks: string | null | undefined): string {
  const text = remarks?.trim();
  if (!text) return '';
  return text.startsWith('*') ? text : `*${text}`;
}

@Component({
  selector: 'app-wwell-ir-view-letter',
  standalone: true,
  imports: [MatButtonModule, MatIconModule, MatTooltipModule, WellBoreViewComponent, WwellIrDepthScaleComponent],
  templateUrl: './wwell-ir-view-letter.component.html',
  styleUrl: './wwell-ir-view-letter.component.scss',
  changeDetection: ChangeDetectionStrategy.OnPush,
  host: {
    '(window:beforeprint)': 'printing.set(true)',
    '(window:afterprint)': 'printing.set(false)',
  },
})
export class WwellIrViewLetterComponent {
  readonly ir = input.required<IntegratedRequirement>();

  /** The sheet follows the app theme on screen, but paper is white: while the browser is printing it is
   *  pinned back to the light tokens (`.wp-paper`), or a dark-mode print would come out light-on-white.
   *  The PDF download is unaffected either way — it is built server-side, and the diagram in it is
   *  captured from a paper-scoped copy (see wwell-ir-wellbore-snapshot.util.ts). */
  protected readonly printing = signal(false);

  // The masthead's "Date" line is the letter's issue date, not the record's last-modified date
  // (which agwa-planning's own server-rendered irLetter.html stamps as LocalDate.now() too, via
  // IrLetterDataAssembler.issueDate) — independent of whether the record itself has ever been
  // through a real update.
  protected readonly issueDate = todayIsoDate();

  protected readonly attachmentSlots = WWELL_IR_ATTACHMENT_SLOTS;

  protected attachmentsFor(slot: IrAttachmentSlot): readonly IrAttachment[] {
    return irAttachmentsFor(this.ir().attachments, slot);
  }

  private readonly s3 = inject(S3Service);
  private readonly dialog = inject(MatDialog);
  private readonly notify = inject(NotificationService);
  private readonly destroyRef = inject(DestroyRef);

  /** Object key of the attachment currently being fetched for the viewer — disables just that row's button. */
  protected readonly openingKey = signal<string | null>(null);

  /** Object-URL thumbnails of the image attachments (PDFs keep their icon), keyed by object key. */
  protected readonly thumbnails = signal<Readonly<Record<string, string>>>({});
  private readonly requestedThumbnails = new Set<string>();

  constructor() {
    effect(() => {
      for (const attachment of this.ir().attachments) {
        if (!attachment.contentType.startsWith('image/') || this.requestedThumbnails.has(attachment.objectKey)) continue;
        this.requestedThumbnails.add(attachment.objectKey);
        this.s3
          .download(attachment.objectKey, attachment.fileName)
          .pipe(takeUntilDestroyed(this.destroyRef))
          .subscribe({
            next: file => this.thumbnails.update(current => ({ ...current, [attachment.objectKey]: URL.createObjectURL(file) })),
            // A missing thumbnail just leaves the slot icon in place; the View button reports real failures.
            error: () => undefined,
          });
      }
    });
    this.destroyRef.onDestroy(() => Object.values(this.thumbnails()).forEach(url => URL.revokeObjectURL(url)));
  }

  /** Opens a stored attachment in the same full-screen viewer the editor uses. */
  protected viewAttachment(attachment: IrAttachment, label: string): void {
    this.openingKey.set(attachment.objectKey);
    this.s3
      .download(attachment.objectKey, attachment.fileName)
      .pipe(takeUntilDestroyed(this.destroyRef))
      .subscribe({
        next: file => {
          this.openingKey.set(null);
          this.dialog.open(WwellIrAttachmentViewerComponent, {
            data: { file, label },
            panelClass: 'ir-viewer-panel',
            width: '92vw',
            maxWidth: '1500px',
            height: '92vh',
            autoFocus: false,
          });
        },
        error: () => {
          this.openingKey.set(null);
          this.notify.error(`Failed to open ${attachment.fileName}.`);
        },
      });
  }

  private readonly wellboreDiagramRow = viewChild<ElementRef<HTMLElement>>('wellboreDiagramRow');

  /** Public (not protected): WwellIrViewComponent's Download PDF button — hoisted onto the tab bar
   *  so it stays put across tab switches instead of living inside the Letter tab's own scrolling
   *  content — calls this through a `viewChild` reference to get the one piece of this letter it
   *  can't produce itself: a rasterized snapshot of the already-rendered wellbore diagram (see
   *  wwell-ir-wellbore-snapshot.util.ts). The rest of the download (irSeq/loading-state/HTTP call)
   *  lives entirely in the parent now — it already has `ir` as its own input, so there's no risk of
   *  reading a required input before Angular has bound it on a child instance (see git history:
   *  that's exactly what broke when this used to live here and get driven via `viewChild`).
   *  Degrades gracefully — a missing/no-data diagram or a capture failure just means the letter
   *  downloads without that section, not a failed download. */

    if (!this.hasWellboreData()) return undefined;
    const row = this.wellboreDiagramRow()?.nativeElement;
    if (!row) return undefined;
    try {
      return (await captureWellboreSnapshot(row)) ?? undefined;
    } catch {
      return undefined;
    }
  }

   *  raw survey values, no conversion; text that doesn't read as that shape stays one "Coordinates"

  protected readonly coordinateAxes: Signal<readonly UtmAxis[]> = computed(() =>

  protected readonly completionDesign = computed(() => {

    const size = completionType === WWELL_COMPLETION_TYPE.OPEN_HOLE ? openHoleSizeIn : isLinerCompletionType(completionType) ? linerSizeIn : null;

  protected readonly hasTargetDepth: Signal<boolean> = computed(() => (this.ir().requestReference.targetDepthFt ?? 0) > 0);

  protected readonly finalApprover = computed<ApprovalChainStep | null>(() => {

    return value.replace(/ m$/, ''); // the PDF prints the bare survey number; the unit is in the column's meaning
  }

    return elevation === null ? null : Math.round((elevation - depthFt) * 10) / 10;
  }
