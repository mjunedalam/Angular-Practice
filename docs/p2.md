import * as THREE from 'three';

import { ANIM, DiagramLayout, WellboreDiagramData } from '@models/well-design/wellbore-diagram.model';
import { computeCasingHalfWidth } from '@shared/utils/wellbore-math.util';
import {
  resolveCompletionBottomDepth,
  resolveCompletionTopDepth,
  resolveOhHalfWidth,
  shouldRenderGravelPack,
  shouldRenderOpenHole,
} from '../../renderers/wellbore-depth.helpers';
import { buildPipeMesh } from './casing-3d.builder';
import { createGravelMaterial, createRockMaterial, createWaterMaterial } from '../utils/materials.factory';
import { createLabelSprite } from '../utils/label-sprite.util';
import { Depth3DScale, Reveal3DTarget } from '../wellbore-3d.types';

export interface CompletionBuildResult {
  readonly group: THREE.Group;
  readonly reveals: Reveal3DTarget[];
  readonly waterRipple: THREE.Mesh | null;
  readonly drillBit: THREE.Mesh;
  readonly drillString: THREE.Mesh;
  readonly drillBitTargetY: number;
  readonly totalDepthLabel: THREE.Sprite;
  readonly totalDepthY: number;
}

/** Red→amber→green, matching the 2D mud-circulation gradient used on the drill-string arrow. */
function circulationColor(pct: number): string {
  const clamped = Math.max(0, Math.min(100, pct));
  if (clamped === 0) return '#e53e3e';
  if (clamped === 100) return '#16a34a';
  const hue = 30 + ((clamped - 1) / 98) * 25;
  return `hsl(${hue.toFixed(1)}, 90%, 50%)`;
}

function resolveWaterLevelDepth(data: WellboreDiagramData): number | null {
  const wd = data.wellDesign;
  if (!wd) return null;
  const rawTxt = wd.swLvlTxt ?? '';
  const isFlowing = /flow/i.test(rawTxt);
  if (isFlowing) {
    const match = /SIWHP[:\s]*(-?\d+(?:\.\d+)?)/i.exec(rawTxt) ?? /(-?\d+(?:\.\d+)?)\s*psi/i.exec(rawTxt);
    const psi = match ? Number(match[1]) : null;
    return psi !== null ? -(psi * 2.37) : 0;
  }
  return wd.staWaterLvl || null;
}

function buildOpenHoleAndScreen(
  data: WellboreDiagramData,
  layout: DiagramLayout,
  depthScale: Depth3DScale,
  ohStart: number,
  group: THREE.Group,
  reveals: Reveal3DTarget[],
): void {
  if (!data.casings.length || !shouldRenderOpenHole(data.wellDesign)) return;
  if (!hasCompletionInterval(data.casings, data.drlgCasings)) return;

  const ohHW = resolveOhHalfWidth(data, layout);
  const topDepth = resolveCompletionTopDepth(data.casings, data.drlgCasings);
  const bottomDepth = resolveCompletionBottomDepth(data);
  const topY = depthScale.toY(topDepth);
  const bottomY = depthScale.toY(bottomDepth);

  const material = createRockMaterial();
  const mesh = buildPipeMesh(depthScale.toRadius(ohHW), topY, bottomY, material);
  mesh.name = 'open-hole';
  group.add(mesh);

  const plane = new THREE.Plane(new THREE.Vector3(0, 1, 0), -topY);
  material.clippingPlanes = [plane];
  reveals.push({ mesh, clipPlane: plane, topY, bottomY, delay: ohStart, duration: ANIM.OH_DURATION / 1000 });

  const label = createLabelSprite({
    text: data.wellDesign?.ohRemarks ?? `${ohHW.toFixed(0)}px Open Hole`,
    color: '#334155',
    fontSize: 26,
  });
  label.position.set(depthScale.toRadius(ohHW) + 1.4, topY + (bottomY - topY) * 0.2, 0);
  group.add(label);
}

function buildGravelPack(
  data: WellboreDiagramData,
  layout: DiagramLayout,
  depthScale: Depth3DScale,
  gravelStart: number,
  group: THREE.Group,
  reveals: Reveal3DTarget[],
): void {
  const wellDesign = data.wellDesign;
  if (!wellDesign || !shouldRenderGravelPack(wellDesign) || !data.casings.length) return;
  if (!hasCompletionInterval(data.casings, data.drlgCasings)) return;

  const { baseHalfWidth, halfWidthIncrement, linerScreenInset, gpScreenGap, gpFillRatio } = layout;
  const innerHW = computeCasingHalfWidth(0, baseHalfWidth, halfWidthIncrement);
  const ohHW = resolveOhHalfWidth(data, layout);
  const lsHW = Math.min(innerHW, ohHW) - linerScreenInset;
  const gpMaxOuter = ohHW - gpScreenGap;
  const annulusWidth = Math.max(0, (gpMaxOuter - lsHW) * gpFillRatio);
  const innerR = depthScale.toRadius(lsHW);
  const outerR = depthScale.toRadius(lsHW + annulusWidth);
  if (outerR <= innerR) return;

  const topDepth = resolveCompletionTopDepth(data.casings, data.drlgCasings);
  const bottomDepth = resolveCompletionBottomDepth(data);
  const topY = depthScale.toY(topDepth);
  const bottomY = depthScale.toY(bottomDepth);
  const height = Math.max(0.05, topY - bottomY);

  const shape = new THREE.Shape();
  shape.absarc(0, 0, outerR, 0, Math.PI * 2, false);
  const hole = new THREE.Path();
  hole.absarc(0, 0, innerR, 0, Math.PI * 2, true);
  shape.holes.push(hole);

  const geometry = new THREE.ExtrudeGeometry(shape, { depth: height, bevelEnabled: false, curveSegments: 48 });
  geometry.rotateX(-Math.PI / 2);
  geometry.translate(0, 0, 0);

  const material = createGravelMaterial();
  const mesh = new THREE.Mesh(geometry, material);
  mesh.position.y = bottomY;
  mesh.name = 'gravel-pack';
  group.add(mesh);

  const plane = new THREE.Plane(new THREE.Vector3(0, 1, 0), -topY);
  material.clippingPlanes = [plane];
  reveals.push({ mesh, clipPlane: plane, topY, bottomY, delay: gravelStart, duration: ANIM.GRAVEL_DURATION / 1000 });
}

function buildWaterLevel(
  data: WellboreDiagramData,
  layout: DiagramLayout,
  depthScale: Depth3DScale,
  group: THREE.Group,
): THREE.Mesh | null {
  const waterDepth = resolveWaterLevelDepth(data);
  if (waterDepth === null) return null;

  const widestTier = data.casings.length > 0 ? data.casings.length - 1 : 0;
  const innerHW = computeCasingHalfWidth(widestTier, layout.baseHalfWidth, layout.halfWidthIncrement);
  const radius = depthScale.toRadius(innerHW) + 0.6;
  const y = depthScale.toY(waterDepth);

  const disc = new THREE.Mesh(
    new THREE.CircleGeometry(radius, 48),
    createWaterMaterial(),
  );
  disc.rotation.x = -Math.PI / 2;
  disc.position.y = y;
  disc.name = 'water-level';
  group.add(disc);

  const ripple = new THREE.Mesh(
    new THREE.RingGeometry(radius * 0.2, radius * 0.28, 48),
    new THREE.MeshBasicMaterial({ color: '#0ea5e9', transparent: true, opacity: 0.85, side: THREE.DoubleSide }),
  );
  ripple.rotation.x = -Math.PI / 2;
  ripple.position.y = y + 0.02;
  ripple.name = 'water-ripple';
  group.add(ripple);

  const label = createLabelSprite({ text: `Water Level: ${waterDepth.toLocaleString()} ft`, color: '#0ea5e9', fontSize: 26 });
  label.position.set(-(radius + 1.6), y + 0.3, 0);
  group.add(label);

  return ripple;
}

function buildDrillString(
  data: WellboreDiagramData,
  depthScale: Depth3DScale,
  group: THREE.Group,
): { bit: THREE.Mesh; string: THREE.Mesh; targetY: number } {
  const currentDepth = Math.max(0, Math.min(data.currentDepth, data.totalDepth));
  const targetY = depthScale.toY(currentDepth);
  const stringHeight = Math.max(0.001, -targetY);
  const lastPct = data.mudCirculation.length > 0 ? data.mudCirculation[data.mudCirculation.length - 1].pct : 100;

  // Geometry is pre-translated so the mesh's local origin sits at its TOP edge (surface) —
  // animating `scale.y` from 0 → 1 then grows the string straight down from the rig floor,
  // instead of scaling outward from a centered pivot.
  const stringGeom = new THREE.CylinderGeometry(0.18, 0.18, stringHeight, 16);
  stringGeom.translate(0, -stringHeight / 2, 0);
  const stringMat = new THREE.MeshStandardMaterial({ color: circulationColor(lastPct), metalness: 0.6, roughness: 0.4 });
  const stringMesh = new THREE.Mesh(stringGeom, stringMat);
  stringMesh.position.y = 0;
  stringMesh.scale.y = 0;
  stringMesh.name = 'drill-string';
  group.add(stringMesh);

  const bitMat = new THREE.MeshStandardMaterial({ color: '#facc15', metalness: 0.8, roughness: 0.3 });
  const bit = new THREE.Mesh(new THREE.ConeGeometry(0.55, 1.1, 24), bitMat);
  bit.name = 'drill-bit';
  bit.position.y = 0; // rig-floor rest position; the reveal timeline drives it down to targetY
  group.add(bit);

  return { bit, string: stringMesh, targetY };
}

function buildPumpMarker(data: WellboreDiagramData, depthScale: Depth3DScale, group: THREE.Group): void {
  const pumpLvl = data.wellDesign?.pumpLvl;
  if (pumpLvl == null) return;
  const label = createLabelSprite({ text: `Pump Set @ ${pumpLvl.toLocaleString()} ft`, color: '#1e293b', fontSize: 26 });
  label.position.set(0, depthScale.toY(pumpLvl), 0.01);
  group.add(label);
}

export function buildCompletion3D(
  data: WellboreDiagramData,
  layout: DiagramLayout,
  depthScale: Depth3DScale,
  timing: { ohStart: number; gravelStart: number },
): CompletionBuildResult {
  const group = new THREE.Group();
  group.name = 'completion';
  const reveals: Reveal3DTarget[] = [];

  buildOpenHoleAndScreen(data, layout, depthScale, timing.ohStart, group, reveals);
  buildGravelPack(data, layout, depthScale, timing.gravelStart, group, reveals);
  const waterRipple = buildWaterLevel(data, layout, depthScale, group);
  const { bit, string: drillString, targetY } = buildDrillString(data, depthScale, group);
  buildPumpMarker(data, depthScale, group);

  const isBeyondTarget = data.targetDepth > 0 && data.currentDepth > data.targetDepth;
  const labelDepth = isBeyondTarget ? data.currentDepth : data.totalDepth;
  const totalDepthY = depthScale.toY(resolveCompletionBottomDepth(data));
  const totalDepthLabel = createLabelSprite({
    text: `${isBeyondTarget ? 'CURRENT DEPTH' : 'TOTAL DEPTH'}: ${labelDepth.toLocaleString()} FT`,
    color: isBeyondTarget ? '#16a34a' : '#2563eb',
    fontSize: 30,
    fontWeight: '900',
  });
  totalDepthLabel.position.set(0, totalDepthY - 2, 0);
  group.add(totalDepthLabel);

  return {
    group,
    reveals,
    waterRipple,
    drillBit: bit,
    drillString,
    drillBitTargetY: targetY,
    totalDepthLabel,
    totalDepthY,
  };
}

import { easeCubicInOut } from 'd3-ease';
import 'd3-transition';

import { ANIM, DiagramLayout, WellboreDiagramData } from '@models/well-design/wellbore-diagram.model';
import {
  buildArrowHeadLeft,
  buildArrowHeadRight,
  buildGravelPackUPath,
  buildOpenHolePath,
  buildScreenHanger,
  computeCasingHalfWidth,
} from '@shared/utils/wellbore-math.util';
import { animateLabel } from './animation.helpers';
import {
  resolveCompletionBottomDepth,
  resolveCompletionBottomPx,
  resolveCompletionRule,
  resolveCompletionShoeCurvePx,
  resolveCompletionTopDepth,
  resolveCompletionTopPx,
  resolveOhHalfWidth,
  resolvePerforatedIntervalInfo,
  shouldRenderGravelPack,
  shouldRenderLinerScreen,
  shouldRenderOpenHole,
} from './wellbore-depth.helpers';
import { DefsSel, DepthScale, GSel, ResolvedAnimation } from './wellbore-renderer.types';

export interface RenderCompletionElementsOptions {
  readonly rootG: GSel;
  readonly defsEl: DefsSel;
  readonly layout: DiagramLayout;
  readonly data: WellboreDiagramData;
  readonly centerX: number;
  readonly scale: DepthScale;
  readonly ohStart: number;
  readonly ohLabelStart: number;
  readonly hangerStart: number;
  readonly animation: ResolvedAnimation;
}

export function renderCompletionElements(options: RenderCompletionElementsOptions): void {
  renderOpenHoleAndScreen(options);
  renderGravelPackDesign(options);
  renderPrePerforatedLiner(options);
}

function renderOpenHoleAndScreen({
  rootG,
  defsEl,
  layout,
  data,
  centerX,
  scale,
  ohStart,
  ohLabelStart,
  animation,
}: RenderCompletionElementsOptions): void {
  // Only skip when there's no casing data of any kind (planned or as-drilled) to anchor
  // to — a DRLG_CSG-only well (no EXAD_GWD_IR_CASING rows) must still render OH/GP when
  // its flags are 'Y', same as an IR-backed well.
  if (!data.casings.length && !data.drlgCasings.length) return;

  const { wellDesign } = data;
  const shouldDrawOH = shouldRenderOpenHole(wellDesign);
  // When a liner qualifies as the 7" completion target (Rule 2 / Rule 3 standard
  // evaluation), liner.renderer owns all liner rendering (shapes, hangers, and the 7"
  // screen via isScreen). The legacy completion-path screen is only drawn when no liner
  // qualifies (no liner at all, or a fallback well with an ordinary liner / no 7" match)
  // and lsFlg opts in.
  const rule = resolveCompletionRule(data.casings, data.drlgCasings);
  const shouldDrawLS = rule.targetLiner === null && shouldRenderLinerScreen(wellDesign);
  if (!shouldDrawOH && !shouldDrawLS) return;

  const { baseHalfWidth, halfWidthIncrement, linerScreenInset, gpScreenGap } = layout;
  const innerHW = computeCasingHalfWidth(0, baseHalfWidth, halfWidthIncrement);
  const ohHW = resolveOhHalfWidth(data, layout);
  const screenHW = Math.min(innerHW, ohHW) - linerScreenInset;
  const liner = data.casings.find(c => c.csgType === 'Liner');
  const gpCasing = data.casings.find(c => c.csgType === 'Gravel Pack');
  const shoePx = resolveCompletionTopPx(data.casings, data.drlgCasings, scale);
  const completionBottomDepth = resolveCompletionBottomDepth(data);
  // Extend past the resolved depth so the OH border reaches the liner's own rendered
  // shoe curve instead of stopping flat short of it (see resolveCompletionShoeCurvePx).
  const completionBottomPx = scale(completionBottomDepth)
    + resolveCompletionShoeCurvePx(data.casings, data.drlgCasings, layout);
  const hasGP = shouldRenderGravelPack(wellDesign) || !!gpCasing;
  const gpBottomPx = gpCasing ? Math.min(scale(gpCasing.csgDepth), completionBottomPx) : completionBottomPx;
  const screenBottomPx = wellDesign
    ? (shouldDrawLS && hasGP ? gpBottomPx - gpScreenGap : completionBottomPx)
    : (liner ? scale(liner.csgDepth) : completionBottomPx - 15);
  const clipHW = shouldDrawOH ? ohHW : screenHW;
  const clipId = 'dyn-oh-clip';
  const clipRect = defsEl.append('clipPath').attr('class', 'dyn-clip').attr('id', clipId).append('rect')
    .attr('x', centerX - clipHW - 15).attr('y', shoePx - 5).attr('width', clipHW * 2 + 30).attr('height', 0);

  if (shouldDrawOH) {
    rootG.append('path').attr('class', 'open-hole')
      .attr('d', buildOpenHolePath(centerX, ohHW, shoePx, completionBottomPx))
      .attr('clip-path', `url(#${clipId})`);
  }

  if (shouldDrawLS) {
    rootG.append('path').attr('class', 'liner-screen')
      .attr('d', buildOpenHolePath(centerX, screenHW, shoePx, screenBottomPx))
      .attr('clip-path', `url(#${clipId})`);
  }

  clipRect.transition().delay(ohStart).duration(ANIM.OH_DURATION).ease(easeCubicInOut).attr('height', completionBottomPx - shoePx + 5);

  if (shouldDrawOH) {
    const ohLabel = wellDesign?.ohRemarks ?? '8 1/2" Open Hole';
    const ohLY = shoePx + (completionBottomPx - shoePx) * 0.2;
    const ohLG = rootG.append('g').attr('class', 'open-hole-label');
    const ohLineX1 = centerX - ohHW;
    const ohLineX2 = centerX - ohHW - 33;

    ohLG.append('line')
      .attr('class', 'oh-label-line')
      .attr('x1', ohLineX1).attr('x2', ohLineX2).attr('y1', ohLY).attr('y2', ohLY);
    ohLG.append('path')
      .attr('class', 'oh-label-arrow')
      .attr('d', buildArrowHeadLeft(ohLineX2, ohLY, 6));
    ohLG.append('text')
      .attr('class', 'open-hole-text')
      .attr('x', ohLineX2 - 6).attr('y', ohLY + 4).attr('text-anchor', 'end').text(ohLabel);

    animateLabel(ohLG, ohLabelStart, animation);
  }

  if (shouldDrawLS) {
    const shoeDepth = resolveCompletionTopDepth(data.casings, data.drlgCasings);
    const fallbackFt = liner ? liner.csgDepth - shoeDepth : completionBottomDepth - shoeDepth;
    const screenLabel = wellDesign?.lsRemarks ?? (liner?.csgRemarks ?? `+/- ${fallbackFt.toLocaleString()} ft of Screen`);
    const scrDelay = ohLabelStart + ANIM.OVERLAY_FADE + ANIM.SEQ_GAP;
    const scrLG = rootG.append('g').attr('class', 'screen-label');
    const scrX1 = centerX + screenHW;
    const scrX2 = centerX + screenHW + 33;

    scrLG.append('line')
      .attr('class', 'screen-label-line')
      .attr('x1', scrX1).attr('x2', scrX2).attr('y1', screenBottomPx).attr('y2', screenBottomPx);
    scrLG.append('path')
      .attr('d', buildArrowHeadRight(scrX2, screenBottomPx, 6));
    scrLG.append('text')
      .attr('class', 'screen-label-text')
      .attr('x', scrX2 + 6).attr('y', screenBottomPx + 4).text(screenLabel);

    animateLabel(scrLG, scrDelay, animation);
  }
}

function renderGravelPackDesign({
  rootG,
  defsEl,
  layout,
  data,
  centerX,
  scale,
  ohStart,
  ohLabelStart,
  animation,
}: RenderCompletionElementsOptions): void {
  const wellDesign = data.wellDesign;
  if (!wellDesign || !shouldRenderGravelPack(wellDesign)) return;
  if (!data.casings.length && !data.drlgCasings.length) return;

  const { baseHalfWidth, halfWidthIncrement, linerScreenInset, gpScreenGap, gpFillRatio } = layout;
  const innerHW = computeCasingHalfWidth(0, baseHalfWidth, halfWidthIncrement);
  const ohHW = resolveOhHalfWidth(data, layout);
  const lsHW = Math.min(innerHW, ohHW) - linerScreenInset;
  const gpMaxOuter = ohHW - gpScreenGap;
  const annulusWidth = Math.max(0, (gpMaxOuter - lsHW) * gpFillRatio);
  const gpOuterEdge = lsHW + annulusWidth;
  const annulusCenter = lsHW + annulusWidth / 2;
  const startDepthPx = resolveCompletionTopPx(data.casings, data.drlgCasings, scale);
  // Extend past the resolved depth so the GP border reaches the liner's own rendered
  // shoe curve instead of stopping flat short of it (see resolveCompletionShoeCurvePx).
  const completionBottomPx = resolveCompletionBottomPx(data, scale)
    + resolveCompletionShoeCurvePx(data.casings, data.drlgCasings, layout);

  const clipId = 'dyn-gp-design-clip';
  const clipRect = defsEl.append('clipPath').attr('class', 'dyn-clip').attr('id', clipId).append('rect')
    .attr('x', centerX - gpOuterEdge).attr('y', startDepthPx)
    .attr('width', gpOuterEdge * 2).attr('height', 0);

  rootG.append('path')
    .attr('class', 'gravelHole')
    .attr('d', buildGravelPackUPath(centerX, annulusCenter, startDepthPx, completionBottomPx))
    .attr('stroke', 'url(#gravelpattern)')
    .attr('stroke-width', String(annulusWidth))
    .style('fill', 'none')
    .attr('clip-path', `url(#${clipId})`);

  clipRect.transition()
    .delay(ohStart)
    .duration(ANIM.GRAVEL_DURATION)
    .ease(easeCubicInOut)
    .attr('height', completionBottomPx - startDepthPx);

  const labelY = startDepthPx + (completionBottomPx - startDepthPx) * 0.5;
  const rEdge = centerX + gpOuterEdge;
  const lEnd = rEdge + 26;
  const lg = rootG.append('g').attr('class', 'gp-design-label');

  lg.append('line')
    .attr('class', 'screen-label-line')
    .attr('x1', rEdge).attr('x2', lEnd).attr('y1', labelY).attr('y2', labelY);
  lg.append('path').attr('d', buildArrowHeadRight(lEnd, labelY, 6));
  lg.append('text')
    .attr('class', 'screen-label-text')
    .attr('x', lEnd + 6).attr('y', labelY + 4).text(wellDesign.gpRemarks ?? 'Gravel Pack');

  animateLabel(lg, ohLabelStart, animation);
}

function renderPrePerforatedLiner({
  rootG,
  defsEl,
  layout,
  data,
  centerX,
  scale,
  ohStart,
}: RenderCompletionElementsOptions): void {
  const wellDesign = data.wellDesign;
  const perfInfo = resolvePerforatedIntervalInfo(wellDesign, data.casings, data.drlgCasings);
  if (!wellDesign || !perfInfo) return;

  const { openHoleHwMargin } = layout;
  const ohHW = resolveOhHalfWidth(data, layout);
  const perfHwInset = Math.max(12, Math.min(20, openHoleHwMargin * 0.65));
  const prePerfHW = Math.max(10, ohHW - perfHwInset);
  // Same shared interval as Open Hole / Gravel Pack (docs/plan/logic.md Rules 1-3),
  // extended to reach the liner's own rendered shoe curve (see resolveCompletionShoeCurvePx).
  const perfTopPx = scale(resolveCompletionTopDepth(data.casings, data.drlgCasings));
  const perfBottomPx = resolveCompletionBottomPx(data, scale)
    + resolveCompletionShoeCurvePx(data.casings, data.drlgCasings, layout);

  const clipId = 'dyn-perf-clip';
  const clipRect = defsEl.append('clipPath').attr('class', 'dyn-clip').attr('id', clipId).append('rect')
    .attr('x', centerX - prePerfHW - 6).attr('y', perfTopPx - 5)
    .attr('width', (prePerfHW + 6) * 2).attr('height', 0);

  const perfGroup = rootG.append('g')
    .attr('class', 'perf-wall-group')
    .attr('clip-path', `url(#${clipId})`);

  // Same outline as open hole (left wall, bottom shoe, right wall in one continuous path) —
  // just a bigger dash gap and a thicker, darker stroke so it reads as its own element.
  perfGroup.append('path')
    .attr('class', 'pre-perf-liner')
    .attr('d', buildOpenHolePath(centerX, prePerfHW, perfTopPx, perfBottomPx));
  // linerOdSz comes straight from DRLG_CSG and may already carry a trailing inch mark
  // (e.g. `7"`) — strip it so the tooltip never doubles up as `7"" Pre Perf Liner`. Null
  // (Rule 1 / Rule 3 fallback — no single liner anchors the interval) falls back to a
  // generic label.
  const perfTitle = perfInfo.linerOdSz
    ? `${perfInfo.linerOdSz.replace(/["']/g, '').trim()}" Pre Perf Liner`
    : (wellDesign.perforatedRemarks ?? 'Perforated Interval');
  perfGroup.append('title').text(perfTitle);

  clipRect.transition()
    .delay(ohStart)
    .duration(ANIM.OH_DURATION)
    .ease(easeCubicInOut)
    .attr('height', perfBottomPx - perfTopPx + 10);
}

export function renderScreenHanger({
  rootG,
  layout,
  data,
  centerX,
  scale,
  hangerStart,
}: RenderCompletionElementsOptions): void {
  if (!data.casings.length && !data.drlgCasings.length) return;
  // Liner hangers are drawn per-liner by liner.renderer (for both DRLG and IR sources).
  // This function no longer draws a hanger for the liner itself — that produced a
  // duplicate OUTER hanger set. It only draws the well-design liner-screen hanger
  // (lsFlg='Y') for a screen that has no corresponding liner geometry.
  if (!shouldRenderLinerScreen(data.wellDesign)) return;
  // Only when no liner qualifies as the 7" completion target: liner.renderer already
  // draws a hanger for that target liner itself (isScreen), so this generic bracket would
  // duplicate it. When no liner qualifies (no liner at all, or a fallback well with an
  // ordinary liner / no 7" match), there's no liner geometry for the implied screen, so
  // this draws its hanger in place of one.
  if (resolveCompletionRule(data.casings, data.drlgCasings).targetLiner !== null) return;

  const { baseHalfWidth, halfWidthIncrement, linerScreenInset } = layout;
  const innerHW = computeCasingHalfWidth(0, baseHalfWidth, halfWidthIncrement);
  const ohHW = resolveOhHalfWidth(data, layout);
  const screenHW = Math.min(innerHW, ohHW) - linerScreenInset;
  const hangerG = rootG.append('g').attr('class', 'screen-hangers').style('opacity', 0);

  const screenY = scale(resolveCompletionTopDepth(data.casings, data.drlgCasings));
  const { left, right } = buildScreenHanger(centerX, screenY, screenHW);

export type LinerRuleKind = 'none' | 'single' | 'single-fallback' | 'multi-target' | 'multi-fallback';

export interface LinerRule {
    readonly kind: LinerRuleKind;
    readonly startDepth: number;
    /** Fixed structural bottom for 'single' and 'multi-target' (the target liner's own
     *  shoe). Null for 'none', 'single-fallback', and 'multi-fallback' — those are bounded
     *  by total/current/target depth, data this liner-only function doesn't have, so the
     *  caller resolves them. */
    readonly structuralEndDepth: number | null;
    /** The is7InchLiner match completion elements are drawn around — gets the screen
     *  texture when lsFlg='Y'. Null when no liner qualifies (kind is 'none',
     *  'single-fallback', or 'multi-fallback' — those get the generic Liner Screen shape
     *  instead, see completion.renderer's renderOpenHoleAndScreen). */
    readonly targetLiner: RawLinerEntry | null;
    /** The liner immediately above the interval start, used only to blend Open Hole's
     *  on-screen width between it and the outer casing (see resolveOhHalfWidth). */
    readonly boundaryLiner: RawLinerEntry | null;

/**
 * Implements docs/plan/logic.md Rules 1-3 over a depth-only liner list: which liner (if
 * any) is the 7" completion target, and the depth interval Open Hole / Gravel Pack /
 * Perforated Interval share as a result.
 */
export function resolveLinerRule(liners: readonly RawLinerEntry[], lastCasingShoe: number): LinerRule {
    // Rule 1: No Liner
    if (liners.length === 0) {
        return { kind: 'none', startDepth: lastCasingShoe, structuralEndDepth: null, targetLiner: null, boundaryLiner: null };
    }

    // Rule 2: Single Liner
    if (liners.length === 1) {
        const only = liners[0];
        if (!is7InchLiner(only)) {
            // Not the 7" screen liner: still an ordinary liner with open hole/completion
            // below it — draw from its own shoe down to total/current/target depth, same
            // as a no-match multi-liner well (docs/plan/logic.md leaves this case
            // unaddressed; this mirrors Rule 3's fallback rather than drawing nothing).
            return { kind: 'single-fallback', startDepth: only.botDepth, structuralEndDepth: null, targetLiner: null, boundaryLiner: only };
        }
        return { kind: 'single', startDepth: lastCasingShoe, structuralEndDepth: only.botDepth, targetLiner: only, boundaryLiner: only };
    }

    // Rule 3: Multiple Liners
    const targetIndex = resolveTargetLinerIndex(liners);
    if (targetIndex !== null) {
        const target = liners[targetIndex];
        const startDepth = targetIndex === 0 ? lastCasingShoe : liners[targetIndex - 1].botDepth;
        return {
            kind: 'multi-target',
            startDepth,
            structuralEndDepth: target.botDepth,
            targetLiner: target,
            boundaryLiner: targetIndex === 0 ? null : liners[targetIndex - 1],
        };
    }

    // Fallback: multiple liners, none match 7"
    const deepest = liners[liners.length - 1];
    return { kind: 'multi-fallback', startDepth: deepest.botDepth, structuralEndDepth: null, targetLiner: null, boundaryLiner: deepest };
}

 * Both conditions must pass:

/**
 * Extracts liner entries from drlgCasings when no ICasingIR data is available.
 * Deduplicates by wLnrOdSz, sorts shallowest-first.
 */
function rawLinersFromDrlg(drlgCasings: IDrlgCsg[]): RawLinerEntry[] {
    const dedupMap = new Map<string, RawLinerEntry>();
    for (const d of drlgCasings) {
        if (!d.wLnrOdSz || (d.wLnrBotDepth ?? 0) <= 0) continue;
        const existing = dedupMap.get(d.wLnrOdSz);
        if (!existing || d.wLnrBotDepth! > existing.botDepth) {
            dedupMap.set(d.wLnrOdSz, {
                odSz: d.wLnrOdSz,
                topDepth: d.wLnrTopDepth ?? d.wCsgBotDpth,
                botDepth: d.wLnrBotDepth!,
                hasActual: true,
            });
        }
    }

    return [...dedupMap.values()].sort((a, b) => a.botDepth - b.botDepth);
}

/**
 * Extracts liner entries from ICasingIR + drlgCasings for actual depth override.
 *
 * Source priority per liner:
 *   - DRLG_CSG (actual/as-drilled) overrides depth when a matching wLnrOdSz exists.
 *   - EXAD_GWD_IR_CASING (planned) is the fallback for depth and the primary OD source.
 *
 * Extra DRLG_CSG liners with no corresponding IR entry are appended at the end.
 */
function rawLinersFromIR(casings: ICasingIR[], drlgCasings: IDrlgCsg[]): RawLinerEntry[] {
    const irLiners = casings.filter(c => c.csgType === 'Liner');

    // Fallback top depth when no drlgCasing match exists: shoe of the deepest non-liner casing
    const lastCasingShoe = casings
        .filter(c => c.csgType !== 'Liner')
        .reduce((max, c) => Math.max(max, c.csgDepth), 0);

    const matchedOdSizes = new Set<string>();

    const irEntries: RawLinerEntry[] = irLiners.map(csg => {
        // Source priority: use DRLG_CSG depth when available; fall back to IR depth.
        const actual = drlgCasings.find(d => d.wLnrOdSz === csg.csgSize) ?? null;
        const hasActualBot = (actual?.wLnrBotDepth ?? 0) > 0;
        const hasActualTop = (actual?.wLnrTopDepth ?? 0) > 0;
        if (actual?.wLnrOdSz) matchedOdSizes.add(actual.wLnrOdSz);
        const fallbackTopDepth = actual != null ? actual.wCsgBotDpth : lastCasingShoe;
        return {
            odSz: actual?.wLnrOdSz ?? csg.csgSize,
            topDepth: hasActualTop ? actual!.wLnrTopDepth! : fallbackTopDepth,
            botDepth: hasActualBot ? actual!.wLnrBotDepth! : csg.csgDepth,
            hasActual: hasActualBot,
        };
    });

    const extraEntries: RawLinerEntry[] = drlgCasings
        .filter(d => d.wLnrOdSz != null && (d.wLnrBotDepth ?? 0) > 0 && !matchedOdSizes.has(d.wLnrOdSz!))
        .map(d => ({
            odSz: d.wLnrOdSz!,
            topDepth: d.wLnrTopDepth ?? d.wCsgBotDpth,
            botDepth: d.wLnrBotDepth!,
            hasActual: true,
        }));

    return [...irEntries, ...extraEntries].sort((a, b) => a.botDepth - b.botDepth);
}

export function resolveShoeDepth(casings: ICasingIR[], drlgCasings: IDrlgCsg[]): number {
  const irNonLiner = casings.filter(c => c.csgType !== 'Liner' && c.csgType !== 'Gravel Pack');
  const deepest = irNonLiner[0];
  if (!deepest) return drlgCasings[0]?.wCsgBotDpth ?? 0;

  const actual = drlgCasings.find(d => d.wCsgOdSz === deepest.csgSize) ?? null;
  return (actual !== null && actual.wCsgBotDpth > 0 ? actual.wCsgBotDpth : null) ?? deepest.csgDepth;
}

/** Central entry point for docs/plan/logic.md Rules 1-3 (see liner.renderer's resolveLinerRule). */

  const lastCasingShoe = resolveShoeDepth(casings, drlgCasings);

}

export function resolveCompletionBottomDepth(data: WellboreDiagramData): number {
  const rule = resolveCompletionRule(data.casings, data.drlgCasings);

  let naturalBottom: number;
  switch (rule.kind) {
    case 'none':
    case 'single-fallback': {
      // Rule 1, and Rule 2's non-7" liner: end = total depth, extended past it while
      // actively drilling beyond target.
      const isCurrentDepthBeyondTarget = data.targetDepth > 0 && data.currentDepth > data.targetDepth;
      naturalBottom = isCurrentDepthBeyondTarget
        ? Math.max(rule.startDepth, Math.min(data.totalDepth, rule.startDepth + OVERRUN_COMPLETION_LENGTH_FT))
        : data.totalDepth;
      break;
    }
    case 'single':
    case 'multi-target':
      // Rule 2 / Rule 3 standard evaluation: end = the 7" target liner's own shoe depth.
      naturalBottom = rule.structuralEndDepth ?? rule.startDepth;
      break;
    case 'multi-fallback': {
      // Rule 3 fallback: end = max(current depth, target depth), clamped to [start, TD].
      const fallbackEnd = Math.max(data.currentDepth || 0, data.targetDepth || 0);
      naturalBottom = Math.max(rule.startDepth, Math.min(data.totalDepth, fallbackEnd));
      break;
    }
  }

  return Math.min(naturalBottom, data.displayDepth);
}

export function resolveCompletionBottomPx(data: WellboreDiagramData, scale: DepthScale): number {
  return scale(resolveCompletionBottomDepth(data));
}

/**
 * Extra pixels Open Hole / Gravel Pack / Perforated Interval should extend past their
 * resolved bottom depth so their border reaches the liner's own rendered shoe curve.
 * buildCasingPath draws the liner's shoe as a bulge that dips shoeCurveOffset px below
 * its flat bottom depth (see liner.renderer's renderLinerShapes) — without this, those
 * flat-bottomed completion paths visually stop short, leaving a bare gap at the tip.
 *
 * Only applies when the interval's END is bounded by a liner's own shoe (Rule 2 / Rule 3
 * standard evaluation, resolveLinerRule kind 'single' | 'multi-target'). The other kinds
 * all terminate in open rock (total depth, or current/target depth) with no liner shoe
 * to match at the bottom, so no curve applies there.
 */
export function resolveCompletionShoeCurvePx(
  casings: ICasingIR[],
  drlgCasings: IDrlgCsg[],
  layout: DiagramLayout,
): number {
  const kind = resolveCompletionRule(casings, drlgCasings).kind;
  return kind === 'single' || kind === 'multi-target' ? layout.shoeCurveOffset : 0;
}

function isFlagYes(flg: string | null | undefined): boolean {
  return (flg ?? '').trim().toUpperCase() === 'Y';
}

/**
 * Resolves whether the Perforated Interval element (EXAD_GWD_WELL_DESIGN.perforatedFlg)
 * should render, and which liner (if any) to reference in its label.
 *
 * perforatedRemarks is free text and only affects label copy elsewhere — it must never
 * gate visibility (see isPrePerforatedLiner in liner.renderer for the matching rationale).
 *
 * Per docs/plan/logic.md, perforatedFlg is drawn across the same shared interval as
 * ohFlg/gpFlg (resolveCompletionTopDepth/resolveCompletionBottomDepth) — every liner shape
 * (Rules 1-3) resolves to some interval, so this only gates on the flag itself.
 */
export function resolvePerforatedIntervalInfo(
  wellDesign: IWellDesign | null | undefined,
  casings: ICasingIR[],
  drlgCasings: IDrlgCsg[],

  if (!isFlagYes(wellDesign?.perforatedFlg)) return null;

  const rule = resolveCompletionRule(casings, drlgCasings);

}

  wellDesign: IWellDesign | null | undefined,
  casings: ICasingIR[] = [],
  drlgCasings: IDrlgCsg[] = [],
): boolean {
  return resolvePerforatedIntervalInfo(wellDesign, casings, drlgCasings) !== null;
}

export function resolveOhHalfWidth(data: WellboreDiagramData, layout: DiagramLayout): number {
  const { baseHalfWidth, halfWidthIncrement, openHoleHwMargin } = layout;
  const innerHW = computeCasingHalfWidth(0, baseHalfWidth, halfWidthIncrement);

  if (!isFlagYes(data.wellDesign?.ohFlg)) return innerHW + openHoleHwMargin;

  const ohDiameter = parseInchSize(data.wellDesign?.ohRemarks);
  if (ohDiameter === null) return innerHW + openHoleHwMargin;

  // innerHW is the on-screen half-width of the deepest structural casing (tier 0) — the
  // casing the anchor liner hangs from — so its real OD is the outer bound OH must stay inside.
  const structuralCasings = data.casings.filter(
    c => c.csgType !== 'Liner' && c.csgType !== 'Gravel Pack',
  );
  const outerOD = parseInchSize(structuralCasings[0]?.csgSize ?? null);

  // When liners exist, OH must render strictly outside the boundary liner it surrounds and
  // inside the casing it hangs from (e.g. a 12 1/4" OH must sit outside a 7" liner and
  // inside a 13 3/8" casing). Per-liner on-screen widths (anchorLiner.halfWidth, innerHW)
  // are a cosmetic decrement scheme with no fixed relationship to real inches, so scaling
  // one of them by a raw diameter ratio can overshoot the casing or undershoot the liner.
  // Instead, interpolate between the two known on-screen widths using the real diameter
  // ratio as the blend factor — this guarantees OH always lands between them.
  // boundaryLiner (see resolveLinerRule) is the liner immediately above the completion
  // interval's start — for Rule 3's fallback that's the deepest liner itself.
  const liners = resolveLiners(data.casings, data.drlgCasings, layout, data.wellDesign);
  const boundary = resolveCompletionRule(data.casings, data.drlgCasings).boundaryLiner;
  const anchorLiner = boundary !== null
    ? liners.find(l => l.odSz === boundary.odSz && l.botDepth === boundary.botDepth) ?? null
    : null;

  if (anchorLiner !== null) {
    const linerOD = parseInchSize(anchorLiner.odSz);
    if (linerOD !== null && linerOD > 0) {
      if (outerOD !== null && outerOD > linerOD) {
        const blend = Math.min(1, Math.max(0, (ohDiameter - linerOD) / (outerOD - linerOD)));
        return anchorLiner.halfWidth + blend * (innerHW - anchorLiner.halfWidth);
      }
      // No known casing OD to blend against: fall back to a direct ratio, clamped so OH
      // never renders outside the innermost casing's on-screen width.
      return Math.min(anchorLiner.halfWidth * (ohDiameter / linerOD), innerHW);
    }
  }

  // No liners: scale relative to innermost structural casing
  if (outerOD !== null && outerOD > 0 && ohDiameter < outerOD) {
    return innerHW * (ohDiameter / outerOD);
  }

  return innerHW + openHoleHwMargin;
}

export function getCasingTier(csg: ICasingIR, casings: ICasingIR[]): number {
  const structuralCasings = casings.filter(item => item.csgType !== 'Gravel Pack');
  const idx = structuralCasings.indexOf(csg);
  return idx === -1 ? 0 : idx;
}
