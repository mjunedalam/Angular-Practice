import { easeCubicInOut } from 'd3-ease';
import 'd3-transition';

import { ANIM, DiagramLayout, WellboreDiagramData } from '@models/well-design/wellbore-diagram.model';
import {
  buildArrowHeadLeft,
  buildArrowHeadRight,
  buildGravelPackUPath,
  buildOpenHolePath,
  buildScreenHanger,
  computeCasingHalfWidth,

import { animateLabel } from './animation.helpers';
import {
  resolveCompletionBottomDepth,
  resolveCompletionBottomPx,
  resolveCompletionRule,
  resolveCompletionShoeCurvePx,
  resolveCompletionTopDepth,
  resolveCompletionTopPx,
  resolveOhHalfWidth,
  resolvePerforatedIntervalInfo,
  shouldRenderGravelPack,
  shouldRenderLinerScreen,
  shouldRenderOpenHole,
} from './wellbore-depth.helpers';
import { DefsSel, DepthScale, GSel, ResolvedAnimation } from './wellbore-renderer.types';

export interface RenderCompletionElementsOptions {
  readonly rootG: GSel;
  readonly defsEl: DefsSel;
  readonly layout: DiagramLayout;
  readonly data: WellboreDiagramData;
  readonly centerX: number;
  readonly scale: DepthScale;
  readonly ohStart: number;
  readonly ohLabelStart: number;
  readonly hangerStart: number;
  readonly animation: ResolvedAnimation;
}

export function renderCompletionElements(options: RenderCompletionElementsOptions): void {
  renderOpenHoleAndScreen(options);
  renderGravelPackDesign(options);
  renderPrePerforatedLiner(options);
}

function renderOpenHoleAndScreen({
  rootG,
  defsEl,
  layout,
  data,
  centerX,
  scale,
  ohStart,
  ohLabelStart,
  animation,
}: RenderCompletionElementsOptions): void {
  // Only skip when there's no casing data of any kind (planned or as-drilled) to anchor
  // to — a DRLG_CSG-only well (no EXAD_GWD_IR_CASING rows) must still render OH/GP when
  // its flags are 'Y', same as an IR-backed well.
  if (!data.casings.length && !data.drlgCasings.length) return;

  const { wellDesign } = data;
  const shouldDrawOH = shouldRenderOpenHole(wellDesign);
  // When a liner qualifies as the 7" completion target (Rule 2 / Rule 3 standard
  // evaluation), liner.renderer owns all liner rendering (shapes, hangers, and the 7"
  // screen via isScreen). The legacy completion-path screen is only drawn when no liner
  // qualifies (no liner at all, or a fallback well with an ordinary liner / no 7" match)
  // and lsFlg opts in.
  const rule = resolveCompletionRule(data.casings, data.drlgCasings);
  const shouldDrawLS = rule.targetLiner === null && shouldRenderLinerScreen(wellDesign);
  if (!shouldDrawOH && !shouldDrawLS) return;

  const { baseHalfWidth, halfWidthIncrement, linerScreenInset, gpScreenGap } = layout;
  const innerHW = computeCasingHalfWidth(0, baseHalfWidth, halfWidthIncrement);
  const ohHW = resolveOhHalfWidth(data, layout);
  const screenHW = Math.min(innerHW, ohHW) - linerScreenInset;
  const liner = data.casings.find(c => c.csgType === 'Liner');
  const gpCasing = data.casings.find(c => c.csgType === 'Gravel Pack');
  const shoePx = resolveCompletionTopPx(data.casings, data.drlgCasings, scale);
  const completionBottomDepth = resolveCompletionBottomDepth(data);
  // Extend past the resolved depth so the OH border reaches the liner's own rendered
  // shoe curve instead of stopping flat short of it (see resolveCompletionShoeCurvePx).
  const completionBottomPx = scale(completionBottomDepth)
    + resolveCompletionShoeCurvePx(data.casings, data.drlgCasings, layout);
  const hasGP = shouldRenderGravelPack(wellDesign) || !!gpCasing;
  const gpBottomPx = gpCasing ? Math.min(scale(gpCasing.csgDepth), completionBottomPx) : completionBottomPx;
  const screenBottomPx = wellDesign
    ? (shouldDrawLS && hasGP ? gpBottomPx - gpScreenGap : completionBottomPx)
    : (liner ? scale(liner.csgDepth) : completionBottomPx - 15);
  const clipHW = shouldDrawOH ? ohHW : screenHW;
  const clipId = 'dyn-oh-clip';
  const clipRect = defsEl.append('clipPath').attr('class', 'dyn-clip').attr('id', clipId).append('rect')
    .attr('x', centerX - clipHW - 15).attr('y', shoePx - 5).attr('width', clipHW * 2 + 30).attr('height', 0);

  if (shouldDrawOH) {
    rootG.append('path').attr('class', 'open-hole')
      .attr('d', buildOpenHolePath(centerX, ohHW, shoePx, completionBottomPx))
      .attr('clip-path', `url(#${clipId})`);
  }

  if (shouldDrawLS) {
    rootG.append('path').attr('class', 'liner-screen')
      .attr('d', buildOpenHolePath(centerX, screenHW, shoePx, screenBottomPx))
      .attr('clip-path', `url(#${clipId})`);
  }

  clipRect.transition().delay(ohStart).duration(ANIM.OH_DURATION).ease(easeCubicInOut).attr('height', completionBottomPx - shoePx + 5);

  if (shouldDrawOH) {
    const ohLabel = wellDesign?.ohRemarks ?? '8 1/2" Open Hole';
    const ohLY = shoePx + (completionBottomPx - shoePx) * 0.2;
    const ohLG = rootG.append('g').attr('class', 'open-hole-label');
    const ohLineX1 = centerX - ohHW;
    const ohLineX2 = centerX - ohHW - 33;

    ohLG.append('line')
      .attr('class', 'oh-label-line')
      .attr('x1', ohLineX1).attr('x2', ohLineX2).attr('y1', ohLY).attr('y2', ohLY);
    ohLG.append('path')
      .attr('class', 'oh-label-arrow')
      .attr('d', buildArrowHeadLeft(ohLineX2, ohLY, 6));
    ohLG.append('text')
      .attr('class', 'open-hole-text')
      .attr('x', ohLineX2 - 6).attr('y', ohLY + 4).attr('text-anchor', 'end').text(ohLabel);

    animateLabel(ohLG, ohLabelStart, animation);
  }

  if (shouldDrawLS) {
    const shoeDepth = resolveCompletionTopDepth(data.casings, data.drlgCasings);
    const fallbackFt = liner ? liner.csgDepth - shoeDepth : completionBottomDepth - shoeDepth;
    const screenLabel = wellDesign?.lsRemarks ?? (liner?.csgRemarks ?? `+/- ${fallbackFt.toLocaleString()} ft of Screen`);
    const scrDelay = ohLabelStart + ANIM.OVERLAY_FADE + ANIM.SEQ_GAP;
    const scrLG = rootG.append('g').attr('class', 'screen-label');
    const scrX1 = centerX + screenHW;
    const scrX2 = centerX + screenHW + 33;

    scrLG.append('line')
      .attr('class', 'screen-label-line')
      .attr('x1', scrX1).attr('x2', scrX2).attr('y1', screenBottomPx).attr('y2', screenBottomPx);
    scrLG.append('path')
      .attr('d', buildArrowHeadRight(scrX2, screenBottomPx, 6));
    scrLG.append('text')
      .attr('class', 'screen-label-text')
      .attr('x', scrX2 + 6).attr('y', screenBottomPx + 4).text(screenLabel);

    animateLabel(scrLG, scrDelay, animation);
  }
}

function renderGravelPackDesign({
  rootG,
  defsEl,
  layout,
  data,
  centerX,
  scale,
  ohStart,
  ohLabelStart,
  animation,
}: RenderCompletionElementsOptions): void {
  const wellDesign = data.wellDesign;
  if (!wellDesign || !shouldRenderGravelPack(wellDesign)) return;
  if (!data.casings.length && !data.drlgCasings.length) return;

  const { baseHalfWidth, halfWidthIncrement, linerScreenInset, gpScreenGap, gpFillRatio } = layout;
  const innerHW = computeCasingHalfWidth(0, baseHalfWidth, halfWidthIncrement);
  const ohHW = resolveOhHalfWidth(data, layout);
  const lsHW = Math.min(innerHW, ohHW) - linerScreenInset;
  const gpMaxOuter = ohHW - gpScreenGap;
  const annulusWidth = Math.max(0, (gpMaxOuter - lsHW) * gpFillRatio);
  const gpOuterEdge = lsHW + annulusWidth;
  const annulusCenter = lsHW + annulusWidth / 2;
  const startDepthPx = resolveCompletionTopPx(data.casings, data.drlgCasings, scale);
  // Extend past the resolved depth so the GP border reaches the liner's own rendered
  // shoe curve instead of stopping flat short of it (see resolveCompletionShoeCurvePx).
  const completionBottomPx = resolveCompletionBottomPx(data, scale)
    + resolveCompletionShoeCurvePx(data.casings, data.drlgCasings, layout);

  const clipId = 'dyn-gp-design-clip';
  const clipRect = defsEl.append('clipPath').attr('class', 'dyn-clip').attr('id', clipId).append('rect')
    .attr('x', centerX - gpOuterEdge).attr('y', startDepthPx)
    .attr('width', gpOuterEdge * 2).attr('height', 0);

  rootG.append('path')
    .attr('class', 'gravelHole')
    .attr('d', buildGravelPackUPath(centerX, annulusCenter, startDepthPx, completionBottomPx))
    .attr('stroke', 'url(#gravelpattern)')
    .attr('stroke-width', String(annulusWidth))
    .style('fill', 'none')
    .attr('clip-path', `url(#${clipId})`);

  clipRect.transition()
    .delay(ohStart)
    .duration(ANIM.GRAVEL_DURATION)
    .ease(easeCubicInOut)
    .attr('height', completionBottomPx - startDepthPx);

  const labelY = startDepthPx + (completionBottomPx - startDepthPx) * 0.5;
  const rEdge = centerX + gpOuterEdge;
  const lEnd = rEdge + 26;
  const lg = rootG.append('g').attr('class', 'gp-design-label');

  lg.append('line')
    .attr('class', 'screen-label-line')
    .attr('x1', rEdge).attr('x2', lEnd).attr('y1', labelY).attr('y2', labelY);
  lg.append('path').attr('d', buildArrowHeadRight(lEnd, labelY, 6));
  lg.append('text')
    .attr('class', 'screen-label-text')
    .attr('x', lEnd + 6).attr('y', labelY + 4).text(wellDesign.gpRemarks ?? 'Gravel Pack');

  animateLabel(lg, ohLabelStart, animation);
}

function renderPrePerforatedLiner({
  rootG,
  defsEl,
  layout,
  data,
  centerX,
  scale,
  ohStart,
  ohLabelStart,
  animation,
}: RenderCompletionElementsOptions): void {
  const wellDesign = data.wellDesign;
  const perfInfo = resolvePerforatedIntervalInfo(wellDesign, data.casings, data.drlgCasings);
  if (!wellDesign || !perfInfo) return;

  const { openHoleHwMargin } = layout;
  const ohHW = resolveOhHalfWidth(data, layout);
  const perfHwInset = Math.max(12, Math.min(20, openHoleHwMargin * 0.65));
  const prePerfHW = Math.max(10, ohHW - perfHwInset);
  // Same shared interval as Open Hole / Gravel Pack (docs/plan/logic.md Rules 1-3),
  // extended to reach the liner's own rendered shoe curve (see resolveCompletionShoeCurvePx).
  const perfTopPx = scale(resolveCompletionTopDepth(data.casings, data.drlgCasings));
  const perfBottomPx = resolveCompletionBottomPx(data, scale)
    + resolveCompletionShoeCurvePx(data.casings, data.drlgCasings, layout);

  const clipId = 'dyn-perf-clip';
  const clipRect = defsEl.append('clipPath').attr('class', 'dyn-clip').attr('id', clipId).append('rect')
    .attr('x', centerX - prePerfHW - 6).attr('y', perfTopPx - 5)
    .attr('width', (prePerfHW + 6) * 2).attr('height', 0);

  const perfGroup = rootG.append('g')
    .attr('class', 'perf-wall-group')
    .attr('clip-path', `url(#${clipId})`);

  // Same outline as open hole (left wall, bottom shoe, right wall in one continuous path) —
  // just a bigger dash gap and a thicker, darker stroke so it reads as its own element.
  perfGroup.append('path')
    .attr('class', 'pre-perf-liner')
    .attr('d', buildOpenHolePath(centerX, prePerfHW, perfTopPx, perfBottomPx));
  // linerOdSz comes straight from DRLG_CSG and may already carry a trailing inch mark
  // (e.g. `7"`) — strip it so the tooltip never doubles up as `7"" Pre Perf Liner`. Null
  // (Rule 1 / Rule 3 fallback — no single liner anchors the interval) falls back to a
  // generic label.
  const perfTitle = perfInfo.linerOdSz
    ? `${perfInfo.linerOdSz.replace(/["']/g, '').trim()}" Pre Perf Liner`
    : (wellDesign.perforatedRemarks ?? 'Perforated Interval');
  perfGroup.append('title').text(perfTitle);

  clipRect.transition()
    .delay(ohStart)
    .duration(ANIM.OH_DURATION)
    .ease(easeCubicInOut)
    .attr('height', perfBottomPx - perfTopPx + 10);

  // Visible on-diagram label — every sibling completion element (Open Hole, Gravel Pack,
  // Liner Screen) gets a leader-line text label; this one previously only carried a native
  // <title> tooltip, which is invisible until hover, so perforatedRemarks effectively never
  // showed up even when populated. Prefer the remarks text itself over perfTitle's generic
  // "Pre Perf Liner"/"Perforated Interval" fallback.
  const perfLabel = wellDesign.perforatedRemarks ?? perfTitle;
  const perfLY = perfTopPx + (perfBottomPx - perfTopPx) * 0.7;
  const perfLG = rootG.append('g').attr('class', 'perf-label');
  const perfLineX1 = centerX - prePerfHW;
  const perfLineX2 = perfLineX1 - 33;

  perfLG.append('line')
    .attr('class', 'oh-label-line')
    .attr('x1', perfLineX1).attr('x2', perfLineX2).attr('y1', perfLY).attr('y2', perfLY);
  perfLG.append('path')
    .attr('class', 'oh-label-arrow')
    .attr('d', buildArrowHeadLeft(perfLineX2, perfLY, 6));
  perfLG.append('text')
    .attr('class', 'open-hole-text')
    .attr('x', perfLineX2 - 6).attr('y', perfLY + 4).attr('text-anchor', 'end').text(perfLabel);

  animateLabel(perfLG, ohLabelStart, animation);
}

export function renderScreenHanger({
  rootG,
  layout,
  data,
  centerX,
  scale,
  hangerStart,
}: RenderCompletionElementsOptions): void {
  if (!data.casings.length && !data.drlgCasings.length) return;
  // Liner hangers are drawn per-liner by liner.renderer (for both DRLG and IR sources).
  // This function no longer draws a hanger for the liner itself — that produced a
  // duplicate OUTER hanger set. It only draws the well-design liner-screen hanger
  // (lsFlg='Y') for a screen that has no corresponding liner geometry.
  if (!shouldRenderLinerScreen(data.wellDesign)) return;
  // Only when no liner qualifies as the 7" completion target: liner.renderer already
  // draws a hanger for that target liner itself (isScreen), so this generic bracket would
  // duplicate it. When no liner qualifies (no liner at all, or a fallback well with an
  // ordinary liner / no 7" match), there's no liner geometry for the implied screen, so
  // this draws its hanger in place of one.
  if (resolveCompletionRule(data.casings, data.drlgCasings).targetLiner !== null) return;

  const { baseHalfWidth, halfWidthIncrement, linerScreenInset } = layout;
  const innerHW = computeCasingHalfWidth(0, baseHalfWidth, halfWidthIncrement);
  const ohHW = resolveOhHalfWidth(data, layout);
  const screenHW = Math.min(innerHW, ohHW) - linerScreenInset;
  const hangerG = rootG.append('g').attr('class', 'screen-hangers').style('opacity', 0);

  const screenY = scale(resolveCompletionTopDepth(data.casings, data.drlgCasings));
  const { left, right } = buildScreenHanger(centerX, screenY, screenHW);
  hangerG.append('path').attr('class', 'screen-hanger').attr('stroke', '#000000').attr('stroke-width', '1.5').attr('fill', '#000000').attr('d', left);
  hangerG.append('path').attr('class', 'screen-hanger').attr('stroke', '#000000').attr('stroke-width', '1.5').attr('fill', '#000000').attr('d', right);

  hangerG.transition().delay(hangerStart).duration(ANIM.HANGER_DURATION).ease(easeCubicInOut).style('opacity', 1);
}

import { computed, inject } from '@angular/core';
import { HttpErrorResponse } from '@angular/common/http';
import { patchState, signalStore, withComputed, withMethods, withState } from '@ngrx/signals';
import { rxMethod } from '@ngrx/signals/rxjs-interop';
import { tapResponse } from '@ngrx/operators';
import { pipe, switchMap, tap } from 'rxjs';
import { CreateIntegratedRequirementPayload, IntegratedRequirement } from '@models/wwell-planning/integrated-requirement.model';
import { mergeIrDetailBlocksIntoIntegratedRequirement } from '../wwell-ir-form/wwell-ir-detail-response.mapper';
import { WwellIrBulkSavePayload } from '../wwell-ir-form/wwell-ir-payload.model';
import { WwellIrService } from '../wwell-ir.service';
import { WwellIrStoreState, initialWwellIrState } from './wwell-ir.state';

/** Prefers the backend's own `{ message }` (see ApiResponse<T> — every exad_agwa endpoint here
 *  returns that envelope on both success and error) over Angular's generic "Http failure
 *  response for ..." text, so a specific backend validation error (e.g. an Oracle constraint
 *  message) reaches the user instead of a vague fallback. */
function errorMessage(err: unknown, fallback: string): string {
  if (err instanceof HttpErrorResponse) {
    const body: unknown = err.error;
    const backendMessage = body && typeof body === 'object' ? (body as { message?: unknown }).message : undefined;
    if (typeof backendMessage === 'string' && backendMessage) return backendMessage;
    return err.message || fallback;
  }
  return err instanceof Error ? err.message : fallback;
}

/** A repeat "Save as Draft" on the same IR reuses its ir_seq (see wwell-ir-payload.mapper.ts), so
 *  the response describes the *same* record, not a new one — replace it in place instead of
 *  prepending a duplicate. Falls back to prepend for a genuinely new IR (irSeq not seen yet). */
function upsertByIrSeq(list: readonly IntegratedRequirement[], created: IntegratedRequirement): IntegratedRequirement[] {
  const existingIndex = created.irSeq != null ? list.findIndex(item => item.irSeq === created.irSeq) : -1;
  if (existingIndex === -1) return [created, ...list];
  // `created` is rebuilt client-side from what was just submitted (see
  // WwellIrService.buildIrFromPayload) — it has no way to know the record's real approval chain
  // (that's a separate, still-mocked endpoint; see loadIrApprovalTimeline), so every resave mints
  // a brand new "just initiated" chain. Harmless for a first-time INSERT (nothing to preserve
  // yet), but a repeat save on an already-in-progress/approved record — including every single
  // inline section edit on the detail page — would otherwise wipe out its real chain state on
  // every save. Keep whichever chain this record already had.
  return list.map((item, index) => (index === existingIndex ? { ...created, chain: item.chain } : item));
}

export const WwellIrStore = signalStore(
  withState<WwellIrStoreState>(initialWwellIrState),

  withComputed(({ irsLoading, linkedRequestWells, linkedRequestWellsLoading, irsError, linkedRequestWellsError, createIrError, updateIrStatusError, irDetailError, irApprovalTimelineError, deleteIrError }) => ({
    isLoading: computed(() => irsLoading() || linkedRequestWellsLoading()),
    /** Trimmed {value,label} view of linkedRequestWells for the "Link to Request" picker itself —
     *  see wwell-ir-editor.component.ts's `applyLinkedRequestWell` for where the full record
     *  (looked up from linkedRequestWells by this same value) drives the auto-populate. */
    linkedRequestWellOptions: computed(() => linkedRequestWells().map(well => ({ value: String(well.prewapSeq), label: well.wellName }))),
    activeErrors: computed(() =>
      [irsError(), linkedRequestWellsError(), createIrError(), updateIrStatusError(), irDetailError(), irApprovalTimelineError(), deleteIrError()].filter(
        (message): message is string => message !== null,
      ),
    ),
    hasError: computed(
      () =>
        irsError() !== null ||
        linkedRequestWellsError() !== null ||
        createIrError() !== null ||
        updateIrStatusError() !== null ||
        irDetailError() !== null ||
        irApprovalTimelineError() !== null ||
        deleteIrError() !== null,
    ),
  })),

  withMethods((store, svc = inject(WwellIrService)) => {
    const loadIrs = rxMethod<void>(
      pipe(
        tap(() => patchState(store, { irsLoading: true, irsError: null })),
        switchMap(() =>
          svc.getIrs().pipe(
            tapResponse({
              next: irs => patchState(store, { irs, irsLoading: false }),
              error: (err: unknown) => patchState(store, { irsLoading: false, irsError: errorMessage(err, 'Failed to load integrated requirements') }),
            }),
          ),
        ),
      ),
    );

    const loadLinkedRequestWells = rxMethod<void>(
      pipe(
        tap(() => {
          console.log('[IR:link] loadLinkedRequestWells — fetching GET /wells/eligible-for-ir');
          patchState(store, { linkedRequestWellsLoading: true, linkedRequestWellsError: null });
        }),
        switchMap(() =>
          svc.getEligibleWellsForIr().pipe(
            tapResponse({
              next: linkedRequestWells => {
                console.log('[IR:link] loadLinkedRequestWells — response received', { count: linkedRequestWells.length, linkedRequestWells });
                patchState(store, { linkedRequestWells, linkedRequestWellsLoading: false });
              },
              error: (err: unknown) => {
                console.log('[IR:link] loadLinkedRequestWells — request failed', err);
                patchState(store, {
                  linkedRequestWellsLoading: false,
                  linkedRequestWellsError: errorMessage(err, 'Failed to load wells eligible for IR initiation'),
                });
              },
            }),
          ),
        ),
      ),
    );

    // Keyed by ir_seq (the real primary key once an IR is saved) — switchMap cancels a still-in-
    // flight fetch for a previously-selected row if the user picks another one before it resolves,
    // so a slow first response can never land after (and clobber) a faster later selection.
    const loadIrDetail = rxMethod<number>(
      pipe(
        tap(() => patchState(store, { irDetailLoading: true, irDetailError: null })),
        switchMap(irSeq =>
          svc.getIrDetailBlocks(irSeq).pipe(
            tapResponse({
              next: blocks => {
                const existing = store.irs().find(item => item.irSeq === irSeq);
                // Nothing to merge onto if the summary row isn't loaded (shouldn't happen from the
                // normal select-a-row flow) — just clear the loading flag and leave state as-is.
                if (!existing) {
                  patchState(store, { irDetailLoading: false });
                  return;
                }
                const merged = mergeIrDetailBlocksIntoIntegratedRequirement(existing, blocks);
                patchState(store, { irs: upsertByIrSeq(store.irs(), merged), irDetailLoading: false });
              },
              error: (err: unknown) =>
                patchState(store, { irDetailLoading: false, irDetailError: errorMessage(err, 'Failed to load integrated requirement detail') }),
            }),
          ),
        ),
      ),
    );

    // Fires right after a save settles (see createIr below) to replace the client-side optimistic
    // echo (WwellIrService.buildIrFromPayload — built from what was *submitted*, not what the
    // database actually stored: server-computed defaults, triggers, or a concurrent edit would all
    // be invisible until this runs) with the real persisted row. Deliberately does not touch
    // `irDetailLoading` — unlike loadIrDetail above, it must never flip WwellIrComponent's
    // `isSelecting()` (which swaps the whole detail panel for a loading skeleton): the user just
    // watched this exact record save, so re-showing it as a blank loading state a moment later
    // would be jarring, not reassuring. A failure here is silent by design — the save itself
    // already succeeded and is already reflected via the optimistic echo, so a background refresh
    // hiccup shouldn't surface as if the save had failed.
    const refreshIrDetail = rxMethod<number>(
      pipe(
        switchMap(irSeq =>
          svc.getIrDetailBlocks(irSeq).pipe(
            tapResponse({
              next: blocks => {
                const existing = store.irs().find(item => item.irSeq === irSeq);
                if (!existing) return;
                const merged = mergeIrDetailBlocksIntoIntegratedRequirement(existing, blocks);
                patchState(store, { irs: upsertByIrSeq(store.irs(), merged) });
              },
              error: () => undefined,
            }),
          ),
        ),
      ),
    );

    // Separate call from loadIrDetail above — mirrors the future real API shape (approval process
    // gets its own endpoint), mocked client-side in the meantime. Same switchMap-cancels-a-stale-
    // fetch reasoning applies when the selection changes quickly.
    const loadIrApprovalTimeline = rxMethod<number>(
      pipe(
        tap(() => patchState(store, { irApprovalTimelineLoading: true, irApprovalTimelineError: null })),
        switchMap(irSeq =>
          svc.getIrApprovalTimeline(irSeq).pipe(
            tapResponse({
              next: chain => {
                patchState(store, {
                  irs: store.irs().map(item => (item.irSeq === irSeq ? { ...item, chain } : item)),
                  irApprovalTimelineLoading: false,
                });
              },
              error: (err: unknown) =>
                patchState(store, { irApprovalTimelineLoading: false, irApprovalTimelineError: errorMessage(err, 'Failed to load approval timeline') }),
            }),
          ),
        ),
      ),
    );

    return {
      loadIrs(): void {
        loadIrs();
      },

      loadIrDetail(irSeq: number): void {
        loadIrDetail(irSeq);
      },

      loadIrApprovalTimeline(irSeq: number): void {
        loadIrApprovalTimeline(irSeq);
      },

      loadLinkedRequestWells(): void {
        loadLinkedRequestWells();
      },

      createIr(
        payload: CreateIntegratedRequirementPayload,
        irBulkPayload: WwellIrBulkSavePayload,
        attachmentNames: readonly string[] = [],
      ): void {
        patchState(store, { creatingIr: true, createIrError: null });
        svc.createIr(payload, irBulkPayload, attachmentNames).subscribe({
          next: created => {
            patchState(store, {
              irs: upsertByIrSeq(store.irs(), created),
              // Falls back to whatever was already known so an UPDATE response missing irSeq never clears it.
              draftIrSeq: created.irSeq ?? store.draftIrSeq(),
              creatingIr: false,
            });
            // Every save (create or update) immediately shows the optimistic echo above, then
            // quietly swaps in the real persisted row once it lands — see refreshIrDetail's own
            // comment for why this never blocks or reloads the visible UI.
            if (created.irSeq != null) refreshIrDetail(created.irSeq);
          },
          error: (err: unknown) => patchState(store, { creatingIr: false, createIrError: errorMessage(err, 'Failed to create integrated requirement') }),
        });
      },

      submitIrForReview(ir: IntegratedRequirement): void {
        patchState(store, { updatingIrStatus: true, updateIrStatusError: null });
        svc.submitIrForReview(ir).subscribe({
          next: updated => patchState(store, { irs: store.irs().map(item => (item.id === updated.id ? updated : item)), updatingIrStatus: false }),
          error: (err: unknown) =>
            patchState(store, { updatingIrStatus: false, updateIrStatusError: errorMessage(err, 'Failed to submit integrated requirement for review') }),
        });
      },

      /** Deletes the whole IR (header row, cascading to casing/tops/hydrogeology/water — see
       *  WwellIrService.deleteIr) and drops it from `irs` on success. `irSeq` must already be a
       *  persisted record's — there's nothing to delete server-side for a local-only draft. */
      deleteIr(irSeq: number): void {
        patchState(store, { deletingIr: true, deleteIrError: null });
        svc.deleteIr(irSeq).subscribe({
          next: () => patchState(store, { irs: store.irs().filter(item => item.irSeq !== irSeq), deletingIr: false }),
          error: (err: unknown) => patchState(store, { deletingIr: false, deleteIrError: errorMessage(err, 'Failed to delete integrated requirement') }),
        });
      },
    };
  }),
);

   *  list (see store.loadLinkedRequestWells()) — selecting a link never itself triggers a
   *  network call.

   *  request record, so letting someone edit or clear them here would silently diverge from (or
   *  desync) that source. Also reused at initial section-open for an already-linked record (see
   *  requestReferenceEdit above) — re-locks the same fields to the same values, so it's a no-op
   *  data-wise there. */
  private applyLinkedRequestWell(referenceGroup: FormGroup, linkedRequestId: string): void {
    console.log('[IR:link] applyLinkedRequestWell — start', { linkedRequestId, eligibleWellsLoaded: this.store.linkedRequestWells().length });
    const well = findLinkedWell(linkedRequestId, this.store.linkedRequestWells());
    if (!well) {
      console.log('[IR:link] applyLinkedRequestWell — no matching eligible well, aborting patch', { linkedRequestId });
      return;

    const hasElevation = linkedWellHasElevation(well);
    const hasCoordinates = linkedWellHasCoordinates(well);

    const patch: Record<string, string | number> = { wellName: well.wellName };
    if (hasElevation) patch['surfaceElevationFtAsl'] = well.surfaceElevation as number;
    if (hasCoordinates) patch['coordinates'] = formatLinkedWellCoordinates(well);
    console.log('[IR:link] applyLinkedRequestWell — patching form group', { patch, hasElevation, hasCoordinates });

    this.setLinkedFieldLock(referenceGroup, 'coordinates', hasCoordinates);
    console.log('[IR:link] applyLinkedRequestWell — done', { rawValue: referenceGroup.getRawValue() });

        if (confirmed) onConfirmed();

   *  open sections' slices overlaid, so a section nobody touched this time round is resent
   *  unchanged (and diffed to a DML no-op by buildWwellIrBulkSavePayload's `existing` comparison)
   *  rather than dropped.
   *
   *  Deliberately fires no success toast — the request is still in flight here, so anything shown
   *  now would claim success before the backend has answered (and would still show next to the
   *  error toast when the save actually fails). WwellIrComponent reports the real outcome once
   *  store.creatingIr() settles.

   *  Status handling: the save payload type only ever accepts 'draft' | 'in-progress' (see
   *  CreateIntegratedRequirementPayload) — there's no contract for "edit this field but keep it
   *  Approved/In Review". Editing a draft/in-progress record keeps its current status; editing
   *  anything further along pulls it back to 'in-progress' (not 'draft', since it was already
   *  worked past that point) so it's clear the record needs to move through review again. */
  private saveMergedRaw(raw: WwellIrFormRawValue, attachmentNames?: readonly string[]): void {
    const current = this.ir();
    const status: Extract<IntegratedRequirementStatus, 'draft' | 'in-progress'> =
      current.status === 'draft' || current.status === 'in-progress' ? current.status : 'in-progress';
    const { payload } = mapIrFormValueToSubmission(raw, status);
    const irBulkPayload = buildWwellIrBulkSavePayload(raw, { irSeq: current.irSeq ?? null, existing: current });
    this.store.createIr(payload, irBulkPayload, attachmentNames ?? current.attachmentNames);
  }

  /** Global footer "Save" — merges every currently-open section's live form value (plus
   *  attachments, if that's open too) into a single raw value and saves it in one call, instead of
   *  requiring the user to click each open section's own Save button separately. A section that
   *  isn't open contributes its existing saved value unchanged, exactly as it would for any one
   *  section's own save (see saveMergedRaw / currentRawFormValue). This is also what turns a brand
   *  new local-only draft (ir().irSeq === null) into a real backend record on the very first click,
   *  even if the user never opened a single section. */
  protected saveAll(): void {
    let raw = this.currentRawFormValue();

    const referenceCtrl = this.requestReferenceEdit.control();
    if (referenceCtrl) raw = { ...raw, reference: { main: referenceCtrl.getRawValue() } };

    const programMainCtrl = this.programMainEdit.control();
    if (programMainCtrl) raw = { ...raw, programDetails: { ...raw.programDetails, main: programMainCtrl.getRawValue() } };

    const programDevCtrl = this.programDevEdit.control();
    if (programDevCtrl) {
      raw = { ...raw, programDetails: { ...raw.programDetails, developmentTesting: programDevCtrl.getRawValue() } };
    }

    const geologistCtrl = this.geologistContactEdit.control();
    if (geologistCtrl) raw = { ...raw, contacts: { ...raw.contacts, geologist: geologistCtrl.getRawValue() } };

    const hydrogeologyCtrl = this.hydrogeologyEdit.control();
    if (hydrogeologyCtrl) raw = { ...raw, hydrogeology: { main: hydrogeologyCtrl.getRawValue() } };

    const formationTopsCtrl = this.formationTopsEdit.control();
    if (formationTopsCtrl) raw = { ...raw, subsurface: { ...raw.subsurface, formationTops: formationTopsCtrl.getRawValue() } };

    const casingCtrl = this.casingEdit.control();
    if (casingCtrl) raw = { ...raw, subsurface: { ...raw.subsurface, casing: casingCtrl.getRawValue() } };

    const offsetWellsCtrl = this.offsetWellsEdit.control();
    if (offsetWellsCtrl) {
      raw = {
        ...raw,
        contacts: {
          ...raw.contacts,
          offsetWells: this.offsetWellsArrayControl(offsetWellsCtrl).getRawValue(),
          remarks: this.offsetWellsRemarksControl(offsetWellsCtrl).getRawValue(),
        },
      };
    }

    const attachmentNames = this.attachmentsEditing()
      ? [...this.attachmentsDraftNames(), ...this.attachmentsNewFiles().map(file => file.name)]
      : undefined;

    this.saveMergedRaw(raw, attachmentNames);
    this.closeAllSections();
  }

  private closeAllSections(): void {
    this.requestReferenceEdit.close();
    this.programMainEdit.close();
    this.programDevEdit.close();
    this.geologistContactEdit.close();
    this.hydrogeologyEdit.close();
    this.formationTopsEdit.close();
    this.casingEdit.close();
    this.offsetWellsEdit.close();
    this.closeAttachmentsEdit();
  }

  protected openAttachmentsEdit(): void {
    this.attachmentsDraftNames.set(this.ir().attachmentNames);
    this.attachmentsNewFiles.set([]);
    this.attachmentsEditing.set(true);
  }

  protected closeAttachmentsEdit(): void {
    this.attachmentsEditing.set(false);
  }

  protected removeAttachmentName(name: string): void {
    this.attachmentsDraftNames.update(names => names.filter(existing => existing !== name));
  }

  protected removeNewAttachmentFile(file: File): void {
    this.attachmentsNewFiles.update(files => files.filter(existing => existing !== file));
  }

  protected onAttachmentFilesPicked(event: Event): void {
    const input = event.target as HTMLInputElement;
    const picked = input.files ? Array.from(input.files) : [];
    this.attachmentsNewFiles.update(files => [...files, ...picked]);
    input.value = '';
  }

  protected submitForReview(): void {
    this.store.submitIrForReview(this.ir());
  }
}

/** Inverse of wwell-ir-payload.mapper.ts's resolveTransmissionCode: exad_gwd_ir_header.trns_type
 *  is VARCHAR2(10), too short for the full label, so a known label is stored as its short code
 *  ('Pipeline' → 'PL'). Decode it back to the label the form's option list expects, otherwise the

 *  text that was truncated to fit) is passed through as-is. */
function decodeTransmission(code: string | null | undefined): string | null {
  if (!code) return null;
  const label = Object.keys(WWELL_TRANSMISSION_CODE_BY_LABEL).find(key => WWELL_TRANSMISSION_CODE_BY_LABEL[key] === code);
  return label ?? code;

/** Every IrXxxRow interface in wwell-ir-payload.model.ts (IrHeaderRow, IrCasingRow, IrTopsRow,
 *  IrHydrogeologyRow, IrWaterRow) uses lowercase snake_case field names mirroring the Oracle
 *  column names — but Oracle's own default column casing (and whatever exad_agwa_queries_pkg's
 *  get_ir_data_prc happens to emit) isn't guaranteed to be lowercase, so a row can come back with
 *  the same fields in UPPERCASE or MixedCase. Since every field this file reads is lowercase by
 *  contract, lowercasing every key up front makes each row match regardless of how the backend
 *  cased it — the same fix as `rowsFor`'s table-name comparison below, applied to row field names. */
function normalizeRowKeys<TRow>(row: Record<string, unknown>): TRow {
  const normalized: Record<string, unknown> = {};
  for (const key of Object.keys(row)) {
    normalized[key.toLowerCase()] = row[key];
  }
  return normalized as TRow;
}

/** Case-insensitive on `candidate.table`: the GET /{irSeq} detail endpoint echoes back
 *  UPPERCASE table names (EXAD_GWD_IR_HEADER...) per exad_agwa_queries_pkg.get_ir_data_prc,
 *  while WwellIrTableName (and the bulk-save write path) uses lowercase — a strict `===` here
 *  never matched, so every block silently resolved to empty and wiped casing/formationTops on
 *  every detail fetch. */
function rowsFor<TRow>(blocks: readonly WwellIrDetailTableBlock[], table: WwellIrTableName): readonly TRow[] {
  const block = blocks.find(candidate => candidate.table.toLowerCase() === table);
  return (block?.data ?? []).map(row => normalizeRowKeys<TRow>(row));
}

 * real casing/tops/hydrogeology/program data from a GET /{irSeq} detail fetch. Merges onto

 * supportingBusiness, targetAquifer, targetDepthFt, contacts, offsetWells, attachments...) have no
 * backing column on exad_gwd_ir_header — see wwell-ir-summary.mapper.ts for the same limitation on

/** Wire format for the IR bulk-save call — mirrors docs/wwell-planning/ir/ir.json 1:1 (table-per-block, row-per-action). */

/** New row → INSERT. Changed persisted row → UPDATE. Removed persisted row → DELETE. Persisted
 *  row the user didn't touch this save → '' (no-op) — sent instead of a redundant UPDATE so the
 *  backend's per-row action switch (EXAD_AGWA_IUD_PKG.save_casing/save_tops/save_hydrogeology)
 *  simply doesn't match any branch and skips the DML, rather than re-writing a row with the exact
 *  values it already has on every save. */
export type IrRowAction = 'INSERT' | 'UPDATE' | 'DELETE' | '';

export type WwellIrTableName =
  | 'exad_gwd_ir_header'
  | 'exad_gwd_ir_casing'
  | 'exad_gwd_ir_hydrogeology'
  | 'exad_gwd_ir_tops'
  | 'exad_gwd_ir_water'
  | 'exad_rcd_prewap';

interface IrRow {
  readonly action: IrRowAction;
  /** null until the backend assigns one on first save; echo it back on every later save. */
  readonly ir_seq: number | null;
}

export interface IrHeaderRow extends IrRow {
  readonly prewap_seq: number | null;
  readonly ltr_no: string | null;
  readonly screen: string;
  readonly sampling: string | null;
  readonly logging: string | null;
  readonly tops_remarks: string | null;
  readonly publish_dt: string | null;
  readonly publish_flg: 'Y' | 'N';
  readonly curr_flg: 'Y' | 'N';

    effect(() => {

      this.store.loadIrApprovalTimeline(irSeq);

      // server-side between selections in a long-lived session (another user initiates an IR for
      // a well, a well newly becomes eligible...) — no cached/one-time snapshot, always the
      // current server state for whichever record is being opened.
      this.store.loadLinkedRequestWells();
    });

    // replays this across the whole list — see `rowsSignature`'s comment above.

      if (!host) return;

      const items = Array.from(host.querySelectorAll<HTMLElement>('.wwell-ir__row'));
      if (items.length) playIfMotionAllowed(buildListRowsRevealTimeline(items));
    });

/**
 * Works around a confirmed backend defect: an exad_gwd_ir_header INSERT/UPDATE can genuinely
 * succeed while the HTTP response still comes back as a 500, with the real success payload
 * embedded inside the error envelope's own `message` field instead of a proper 2xx — e.g.
 * `{"statusCode":500,"error":true,"message":"{\"status\":\"SUCCESS\",\"ir_seq\":9024}","data":[]}`.
 * Matches only that exact signature (a JSON-parseable message shaped like
 * `{status:"SUCCESS", ir_seq:number}`) and returns the REAL ir_seq the backend actually
 * assigned — never a client-minted placeholder — so the record's identity stays correct and
 * stable across resaves. Any other failure (a genuine validation error, network failure, a
 * differently-shaped message) falls through unmatched and is re-thrown as a real error.
 * Remove once the backend returns a proper 2xx for a save that actually succeeded.
 */
function extractIrSeqFromMisreportedSuccess(err: unknown): number | null {
  if (!(err instanceof HttpErrorResponse)) return null;
  const body: unknown = err.error;
  const message = body && typeof body === 'object' ? (body as { message?: unknown }).message : undefined;
  if (typeof message !== 'string') return null;

  try {
    const parsed: unknown = JSON.parse(message);
    const status = parsed && typeof parsed === 'object' ? (parsed as { status?: unknown }).status : undefined;
    const irSeq = parsed && typeof parsed === 'object' ? (parsed as { ir_seq?: unknown }).ir_seq : undefined;
    return status === 'SUCCESS' && typeof irSeq === 'number' ? irSeq : null;
  } catch {
    return null;
  }
}

    // — it has no multipart handling and no columns for the richer domain fields below (client,
    // wellName, contacts, attachments...), so those stay client-side only until exad_agwa_iud_pkg's
    // header table grows to carry them. Sending them today just 415s (wrong Content-Type/body shape).
    //
    // No offline-fallback here (unlike getIrs()/getEligibleWellsForIr() above) — a failed save
    // must surface as a real, visible error (WwellIrStore.createIrError), not a fabricated
    // success. A genuine POST failure with no recoverable ir_seq (see extractIrSeqFromMisreportedSuccess
    // below for the one confirmed-defect exception) always re-throws rather than minting a random
    // ir_seq — a client-minted identity on real failure would let a resave silently re-INSERT under
    // a new fake id instead of UPDATE-ing the original row, orphaning the real backend record and
    // duplicating it in the list.
    //
    // SUPPRESS_GLOBAL_ERROR: WwellIrStore.createIr's own error handler (→ createIrError →
    // WwellIrComponent's notify.error) already shows a toast for a genuine failure — without this,
    // errorInterceptor's app-wide catchError would show a second, identical one for the same error.
    // The header row's own `ir_seq` is the identity this save was submitted under — `null` for a
    // brand-new IR (the backend mints one), otherwise the already-known ir_seq of the record being
    // updated. exad_agwa_iud_pkg.update_ir_data_prc (docs/ir/update_ir_data_prc.sql) only echoes
    // back a fresh ir_seq on that INSERT branch; an UPDATE's `p_return_value` is just the literal
    // `SUCCESS`, no ir_seq, because the caller already has it — see WwellIrSaveResult. Falling back
    // to `knownIrSeq` (not a random mint) below is what keeps a resaved record's identity stable.
    const knownIrSeq = irBulkPayload[0].data?.[0]?.ir_seq ?? null;

    return this.http
      .post<ApiResponse<WwellIrSaveResult>>(`${this.apiUrl}/integrated-requirements`, irBulkPayload, {

        catchError((err: unknown) => {
          const realIrSeq = extractIrSeqFromMisreportedSuccess(err);
          return realIrSeq !== null ? of(this.buildIrFromPayload(payload, realIrSeq, attachmentNames)) : throwError(() => err);
        }),
      );
  }

    return this.http
      .delete<ApiResponse<never>>(`${this.apiUrl}/integrated-requirements/${irSeq}`, {

      .pipe(map(() => undefined));

  }

   *  back an ir_seq (see WwellIrSaveResult), never the full IntegratedRequirement. `irSeq` is
   *  already fully resolved by the caller before this runs (the response's own value on INSERT,
   *  the already-known `knownIrSeq` on UPDATE — see createIr() above), so it's used as-is: no

   *  re-INSERT under a new fake id instead of UPDATE-ing the original row. */

    const code = `IR-${irSeq}`;
    return buildIntegratedRequirementFromPayload(payload, { id: code, code, irSeq }, attachmentNames);
  }

export interface ActionConfirmDialogRow {
  label: string;
  value: string;
}

/**
 * Plain generic confirm/reject gate — resets, row deletes, whole-record deletes, any yes/no
 * action with nothing to do with sending an email. Deliberately its own component/service, not a
 * variant of ConfirmDialogService (which is scoped to morning-report's "Send Daily Report" flow

  title: string;
  subtitle?: string;
  rows?: ActionConfirmDialogRow[];
  confirmLabel?: string;
  cancelLabel?: string;
  severity?: 'info' | 'warning' | 'danger';
  /** Short footer note. Defaults per `severity` (see DEFAULT_HINT) when omitted; pass '' to show
   *  no hint at all. */
  hint?: string;
}

  info: '',
  warning: 'Please review before continuing.',
  danger: 'This action cannot be undone.',

  selector: 'app-action-confirm-dialog',
  standalone: true,
  templateUrl: './action-confirm-dialog.component.html',
  styleUrl: './action-confirm-dialog.component.scss',
  changeDetection: ChangeDetectionStrategy.OnPush,
  encapsulation: ViewEncapsulation.None,

  private readonly dialogRef = inject(MatDialogRef<ActionConfirmDialogComponent>);

  protected readonly confirmLabel = this.data.confirmLabel ?? 'Confirm';
