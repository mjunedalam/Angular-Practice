```typescript
import gsap from 'gsap';

const STEPS_REVEAL_STAGGER_S = 0.07;
const STEPS_REVEAL_DURATION_S = 0.3;
const STEPS_REVEAL_SLIDE_PX = 12;
const NODE_REVEAL_DURATION_S = 0.32;
const NODE_SCALE_FROM = 0.5;

/**
 * Paused — caller decides when to play via `playIfMotionAllowed`. Steps fade/slide in on
 * the same stagger as before; each step's node circle additionally pops in from a smaller
 * scale on a slight overlap (`-=` offset) so the rail reads as "lighting up" step by step
 * rather than the whole card just fading in at once.
 */
export function buildApprovalStepsRevealTimeline(items: readonly HTMLElement[]): gsap.core.Timeline {
  const timeline = gsap.timeline({ paused: true }).fromTo(
    items,
    { opacity: 0, y: STEPS_REVEAL_SLIDE_PX },
    { opacity: 1, y: 0, duration: STEPS_REVEAL_DURATION_S, ease: 'power3.out', stagger: STEPS_REVEAL_STAGGER_S },
  );

  const nodes = items
    .map(item => item.querySelector<HTMLElement>('.approval-timeline__node'))
    .filter((node): node is HTMLElement => node !== null);

  if (nodes.length) {
    timeline.fromTo(
      nodes,
      { scale: NODE_SCALE_FROM },
      { scale: 1, duration: NODE_REVEAL_DURATION_S, ease: 'back.out(2.2)', stagger: STEPS_REVEAL_STAGGER_S },
      `-=${STEPS_REVEAL_DURATION_S}`,
    );
  }

  return timeline;
}
```

```typescript
  'in-progress': wwellStatusMeta('In Progress', WWELL_STATUS_COLORS.blue),

export const WWELL_WLDA_CHAIN_ROLES: readonly string[] = [
  'Hydrogeologist',
  'Field Supervisor',
  'Dept. Manager',
  'SAP / Upper Mgmt',
];

export const WWELL_LAND_AFFAIRS_CHAIN_ROLES: readonly string[] = ['Land Surveyor', 'Land Affairs Officer', 'Manager'];

export const WWELL_IR_CHAIN_ROLES: readonly string[] = ['Hydrogeologist', 'Lead Hydrogeologist', 'Manager'];

export type ChainCompletionStatus = 'complete' | 'partial' | 'none';

/** 'complete' once every step is approved, 'none' while zero steps are approved yet
 * (whether pending or in-progress), 'partial' anywhere in between. Rejected steps don't
 * count as approved, so a rejection anywhere keeps the chain out of 'complete'. */
export function getChainCompletionStatus(chain: readonly ApprovalChainStep[]): ChainCompletionStatus {
  if (chain.length === 0) return 'none';

  const approvedCount = chain.filter(step => step.status === 'approved').length;
  if (approvedCount === chain.length) return 'complete';
  if (approvedCount === 0) return 'none';
  return 'partial';
}

/** Border accent per completion state — reused for both the master-list row's left border
 * and every detail-panel card border, so a record's approval progress reads at a glance
 * without opening the timeline itself. */
export const WWELL_CHAIN_COMPLETION_COLOR: Record<ChainCompletionStatus, string> = {
  complete: WWELL_STATUS_COLORS.green.dot,
  partial: WWELL_STATUS_COLORS.amber.dot,
  none: WWELL_STATUS_COLORS.slateMuted.dot,
};

export function chainCompletionColor(chain: readonly ApprovalChainStep[]): string {
  return WWELL_CHAIN_COMPLETION_COLOR[getChainCompletionStatus(chain)];
}

/** 4 letters + dash + a 2-4 digit "hundreds" suffix, e.g. KHAJ-300, RXYD-100. */
export const WWELL_APPROVAL_WELL_NAME_PATTERN = /^[A-Za-z]{4}-\d{2,4}$/;

/** Free-form "24.1751°N, 47.2951°E" style pair; hemisphere letters are optional. */
```

## `src/app/features/wwell-planning/shared/wwell-pending-selection.util.ts`

```typescript
import { Signal, computed, signal } from '@angular/core';

export interface PendingSelection {
  readonly pendingId: Signal<string | null>;
  readonly isSelecting: Signal<boolean>;
  /** No-ops if `id` already matches `currentId` — clicking the already-selected row shouldn't
   * replay the loading transition. */
  select(id: string, currentId: string | null, onResolved: (id: string) => void): void;
  /** Cancels any in-flight timer — call from `ngOnDestroy` so a resolved selection can't land
   * on a signal after the component (and its view) is gone. */
  clear(): void;
}

/** Brief shimmer-loading transition between selecting a row and showing its detail — the data
 * itself is already in memory, so this is a deliberate, short UX beat rather than a real wait. */
export function createPendingSelection(delayMs: number): PendingSelection {
  const pendingId = signal<string | null>(null);
  const isSelecting = computed(() => pendingId() !== null);
  let timeoutId: ReturnType<typeof setTimeout> | undefined;

  return {
    pendingId,
    isSelecting,
    select(id, currentId, onResolved) {
      if (id === currentId) return;
      if (timeoutId) clearTimeout(timeoutId);
      pendingId.set(id);
      timeoutId = setTimeout(() => {
        onResolved(id);
        pendingId.set(null);
      }, delayMs);
    },
    clear() {
      if (timeoutId) clearTimeout(timeoutId);
      timeoutId = undefined;
    },
  };
}
```

```typescript
import {
  ChangeDetectionStrategy,
  Component,
  ElementRef,
  OnDestroy,
  OnInit,
  ViewChild,
  computed,
  effect,
  inject,
  signal,
} from '@angular/core';
import { FormsModule } from '@angular/forms';
import { ActivatedRoute, Router } from '@angular/router';
import { MatButtonModule } from '@angular/material/button';
import { MatIconModule } from '@angular/material/icon';
import { MatSnackBar } from '@angular/material/snack-bar';
import { ApprovalChainStep, IntegratedRequirement } from '@models/wwell-planning/wwell-planning.model';
import { buildDetailRevealTimeline } from '../shared/dynamic-form/detail-reveal.animation';
import { playIfMotionAllowed } from '../shared/dynamic-form/gsap-motion.util';
import { buildListRowsRevealTimeline } from '../shared/dynamic-form/list-reveal.animation';
import { chainCompletionColor } from '../shared/wwell-approvals/wwell-approvals.config';
import { WwellApprovalMiniChainComponent } from '../shared/wwell-approvals/wwell-approval-timeline/wwell-approval-mini-chain.component';
import { WWELL_DETAIL_LOADING_MS, WwellDetailLoadingComponent } from '../shared/wwell-detail-loading/wwell-detail-loading.component';
import { createPendingSelection } from '../shared/wwell-pending-selection.util';
import { WwellPlanningStore } from '../store/wwell-planning.store';
import { WwellIrDetailComponent } from './wwell-ir-detail/wwell-ir-detail.component';
import { WWELL_IR_STATUS_META } from './wwell-ir.config';

const SNACKBAR_DURATION_MS = 3000;
const SKELETON_ROW_COUNT = 6;

@Component({
  selector: 'app-wwell-ir',
  standalone: true,
  imports: [
    FormsModule,
    MatButtonModule,
    MatIconModule,
    WwellIrDetailComponent,
    WwellApprovalMiniChainComponent,
    WwellDetailLoadingComponent,
  ],
  templateUrl: './wwell-ir.component.html',
  styleUrl: './wwell-ir.component.scss',
  changeDetection: ChangeDetectionStrategy.OnPush,
})
export class WwellIrComponent implements OnInit, OnDestroy {
  protected readonly store = inject(WwellPlanningStore);
  private readonly router = inject(Router);
  private readonly route = inject(ActivatedRoute);
  private readonly snackBar = inject(MatSnackBar);

  protected readonly isLoading = this.store.integratedRequirementsLoading;
  protected readonly loadError = this.store.integratedRequirementsError;
  protected readonly isSubmittingForReview = this.store.updatingIntegratedRequirementStatus;

  protected readonly searchTerm = signal('');
  protected readonly selectedId = signal<string | null>(null);

  protected readonly rows = computed(() => {
    const term = this.searchTerm().trim().toLowerCase();
    const all = this.store.integratedRequirements();
    if (!term) return all;

    return all.filter(
      ir =>
        ir.code.toLowerCase().includes(term) ||
        ir.requestReference.wellName.toLowerCase().includes(term) ||
        (ir.requestReference.linkedRequestId ?? '').toLowerCase().includes(term) ||
        ir.requestReference.client.toLowerCase().includes(term),
    );
  });

  // Deliberately looks up the *unfiltered* store list (not `rows()`) so narrowing the search
  // never drops the current selection just because it fell outside the filtered results.
  protected readonly selectedIr = computed<IntegratedRequirement | null>(
    () => this.store.integratedRequirements().find(ir => ir.id === this.selectedId()) ?? null,
  );

  protected readonly statusMeta = WWELL_IR_STATUS_META;

  protected readonly skeletonRows = Array.from({ length: SKELETON_ROW_COUNT }, (_, index) => index);

  private readonly pendingSelection = createPendingSelection(WWELL_DETAIL_LOADING_MS);
  protected readonly pendingId = this.pendingSelection.pendingId;
  protected readonly isSelecting = this.pendingSelection.isSelecting;

  @ViewChild('detailHost') private readonly detailHost?: ElementRef<HTMLElement>;
  @ViewChild('listHost') private readonly listHost?: ElementRef<HTMLElement>;
  private previousSelectedId: string | null = null;

  constructor() {
    effect(() => {
      const message = this.store.createIntegratedRequirementError();
      if (message) {
        this.snackBar.open(message, 'Dismiss', { duration: SNACKBAR_DURATION_MS });
      }
    });

    effect(() => {
      const message = this.store.updateIntegratedRequirementStatusError();
      if (message) {
        this.snackBar.open(message, 'Dismiss', { duration: SNACKBAR_DURATION_MS });
      }
    });

    // Default to the first record once the list loads — mirrors the reference app, which
    // always has something selected in the detail pane rather than starting on an empty state.
    effect(() => {
      const all = this.store.integratedRequirements();
      if (all.length > 0 && this.selectedId() === null) {
        this.selectedId.set(all[0].id);
      }
    });

    effect(() => {
      const id = this.selectedId();
      if (id === null || id === this.previousSelectedId) return;

      const host = this.detailHost?.nativeElement;
      if (host) playIfMotionAllowed(buildDetailRevealTimeline(host));
      this.previousSelectedId = id;
    });

    // Re-plays whenever the visible rows actually change (initial load finishing, or the
    // search term narrowing/widening the result set) — not on every keystroke that leaves
    // the result set untouched, since `rows()` only changes reference when it does.
    effect(() => {
      const currentRows = this.rows();
      if (this.isLoading() || currentRows.length === 0) return;

      const host = this.listHost?.nativeElement;
      if (!host) return;

      const items = Array.from(host.querySelectorAll<HTMLElement>('.wwell-ir__row'));
      if (items.length) playIfMotionAllowed(buildListRowsRevealTimeline(items));
    });
  }

  ngOnInit(): void {
    this.store.loadIntegratedRequirements();
  }

  ngOnDestroy(): void {
    this.pendingSelection.clear();
  }

  protected onSearchChange(term: string): void {
    this.searchTerm.set(term);
  }

  protected selectIr(id: string): void {
    this.pendingSelection.select(id, this.selectedId(), resolvedId => this.selectedId.set(resolvedId));
  }

  protected openNewIrPage(): void {
    this.router.navigate(['new'], { relativeTo: this.route });
  }

  protected chainBorderColor(chain: readonly ApprovalChainStep[]): string {
    return chainCompletionColor(chain);
  }

  protected submitForReview(ir: IntegratedRequirement): void {
    this.store.submitIntegratedRequirementForReview(ir);
  }
}
```

```typescript
      if (message) {
        this.snackBar.open(message, 'Dismiss', { duration: SNACKBAR_DURATION_MS });
      }
    });

    // Default to the first record once the list loads — mirrors the reference app, which
    // always has something selected in the detail pane rather than starting on an empty state.
    effect(() => {
      const all = this.store.landAffairsApprovals();
      if (all.length > 0 && this.selectedId() === null) {
        this.selectedId.set(all[0].id);
      }
    });

    effect(() => {
      const id = this.selectedId();
      if (id === null || id === this.previousSelectedId) return;

      const host = this.detailHost?.nativeElement;
      if (host) playIfMotionAllowed(buildDetailRevealTimeline(host));
      this.previousSelectedId = id;
    });

    // Re-plays whenever the visible rows actually change (initial load finishing, or the
    // search term narrowing/widening the result set) — not on every keystroke that leaves
    // the result set untouched, since `rows()` only changes reference when it does.
    effect(() => {
      const currentRows = this.rows();
      if (this.isLoading() || currentRows.length === 0) return;

      const host = this.listHost?.nativeElement;
      if (!host) return;

      const items = Array.from(host.querySelectorAll<HTMLElement>('.wla__row'));
      if (items.length) playIfMotionAllowed(buildListRowsRevealTimeline(items));
    });
  }

  ngOnInit(): void {
    this.store.loadLandAffairsApprovals();
  }

  ngOnDestroy(): void {
    this.pendingSelection.clear();
  }

  protected onSearchChange(term: string): void {
    this.searchTerm.set(term);
  }

  protected selectRow(id: string): void {
    this.pendingSelection.select(id, this.selectedId(), resolvedId => this.selectedId.set(resolvedId));
  }

  protected openLandAffairsRequestDialog(): void {
    this.dialog
      .open<WwellLandAffairsRequestFormComponent, unknown, WwellLandAffairsRequestFormResult>(
        WwellLandAffairsRequestFormComponent,
        { panelClass: 'wwell-land-affairs-request-form-panel', maxWidth: '95vw', autoFocus: false },
      )
      .afterClosed()
      .subscribe(result => {
        if (!result) return;
        if (result.mode === 'draft') {
          this.snackBar.open('Land Affairs request saved as draft.', 'Dismiss', { duration: SNACKBAR_DURATION_MS });
          return;
        }
        this.store.submitLandAffairsApproval(result.payload, result.files);
        this.snackBar.open('Land Affairs approval request submitted.', 'Dismiss', { duration: SNACKBAR_DURATION_MS });
      });
  }

  protected isDownloading(id: string): boolean {
    return this.downloadingIds().has(id);
  }

  protected isChainComplete(chain: readonly ApprovalChainStep[]): boolean {
    return chain.length > 0 && chain.every(step => step.status === 'approved');
  }

  protected chainBorderColor(chain: readonly ApprovalChainStep[]): string {
    return chainCompletionColor(chain);
  }

  protected downloadLandAffairsPdf(row: LandAffairsApproval): void {
    this.downloadPdf(row.id, `${row.reqNo}-Land-Affairs.pdf`, this.svc.getLandAffairsApprovalPdf(row));
  }

  private downloadPdf(id: string, fileName: string, request$: Observable<Blob>): void {
    if (this.isDownloading(id)) return;

    this.setDownloading(id, true);
    request$.subscribe({
      next: blob => {
        const url = URL.createObjectURL(blob);
        const anchor = document.createElement('a');
        anchor.href = url;
        anchor.download = fileName;
        document.body.appendChild(anchor);
        anchor.click();
        document.body.removeChild(anchor);
        setTimeout(() => URL.revokeObjectURL(url), DOWNLOAD_URL_REVOKE_DELAY_MS);
        this.setDownloading(id, false);
      },
      error: () => {
        this.setDownloading(id, false);
        this.snackBar.open('Failed to download the approval PDF.', 'Dismiss', { duration: SNACKBAR_DURATION_MS });
      },
    });
  }

  private setDownloading(id: string, isDownloading: boolean): void {
    const next = new Set(this.downloadingIds());
    if (isDownloading) {
      next.add(id);
    } else {
      next.delete(id);
    }
    this.downloadingIds.set(next);
  }

  private filterRows(rows: readonly LandAffairsApproval[]): LandAffairsApproval[] {
    const term = this.searchTerm().trim().toLowerCase();
    if (!term) return [...rows];

    return rows.filter(
      row =>
        row.reqNo.toLowerCase().includes(term) ||
        row.client.toLowerCase().includes(term) ||
        row.wellName.toLowerCase().includes(term),
    );
  }
```

```typescript
import {
  ChangeDetectionStrategy,
  Component,
  ElementRef,
  OnDestroy,
  OnInit,
  ViewChild,
  computed,
  effect,
  inject,
  signal,
} from '@angular/core';
import { FormsModule } from '@angular/forms';
import { MatButtonModule } from '@angular/material/button';
import { MatDialog } from '@angular/material/dialog';
import { MatIconModule } from '@angular/material/icon';
import { MatSnackBar } from '@angular/material/snack-bar';
import { WwellPlanningRequest } from '@models/wwell-planning/wwell-planning.model';
import { buildDetailRevealTimeline } from '../shared/dynamic-form/detail-reveal.animation';
import { playIfMotionAllowed } from '../shared/dynamic-form/gsap-motion.util';
import { buildListRowsRevealTimeline } from '../shared/dynamic-form/list-reveal.animation';
import { WWELL_DETAIL_LOADING_MS, WwellDetailLoadingComponent } from '../shared/wwell-detail-loading/wwell-detail-loading.component';
import { createPendingSelection } from '../shared/wwell-pending-selection.util';
import { WwellPlanningStore } from '../store/wwell-planning.store';
import { WWELL_REQUEST_PRIORITY_META, WWELL_REQUEST_STATUS_META } from './wwell-requests.config';
import { WwellRequestFormComponent, WwellRequestFormResult } from './wwell-request-form/wwell-request-form.component';

const SNACKBAR_DURATION_MS = 3000;
const SKELETON_ROW_COUNT = 6;

@Component({
  selector: 'app-wwell-requests',
  standalone: true,
  imports: [FormsModule, MatButtonModule, MatIconModule, WwellDetailLoadingComponent],
  templateUrl: './wwell-requests.component.html',
  styleUrl: './wwell-requests.component.scss',
  changeDetection: ChangeDetectionStrategy.OnPush,
})
export class WwellRequestsComponent implements OnInit, OnDestroy {
  protected readonly store = inject(WwellPlanningStore);
  private readonly dialog = inject(MatDialog);
  private readonly snackBar = inject(MatSnackBar);

  protected readonly isLoading = this.store.requestsLoading;
  protected readonly isCreating = this.store.creatingRequest;
  protected readonly loadError = this.store.requestsError;

  protected readonly statusMeta = WWELL_REQUEST_STATUS_META;
  protected readonly priorityMeta = WWELL_REQUEST_PRIORITY_META;

  protected readonly searchTerm = signal('');
  protected readonly selectedId = signal<string | null>(null);

  protected readonly skeletonRows = Array.from({ length: SKELETON_ROW_COUNT }, (_, index) => index);

  protected readonly rows = computed(() => this.filterRows(this.store.requests()));

  // Looks up the *unfiltered* store list (not `rows()`) so narrowing the search never drops
  // the current selection just because it fell outside the filtered results — mirrors wwell-ir.
  protected readonly selectedRow = computed<WwellPlanningRequest | null>(
    () => this.store.requests().find(row => row.id === this.selectedId()) ?? null,
  );

  protected readonly totalCount = computed(() => this.store.requests().length);

  private readonly pendingSelection = createPendingSelection(WWELL_DETAIL_LOADING_MS);
  protected readonly pendingId = this.pendingSelection.pendingId;
  protected readonly isSelecting = this.pendingSelection.isSelecting;

  @ViewChild('detailHost') private readonly detailHost?: ElementRef<HTMLElement>;
  @ViewChild('listHost') private readonly listHost?: ElementRef<HTMLElement>;
  private previousSelectedId: string | null = null;

  constructor() {
    effect(() => {
      const message = this.store.createRequestError();
      if (message) {
        this.snackBar.open(message, 'Dismiss', { duration: SNACKBAR_DURATION_MS });
      }
    });

    // Default to the first record once the list loads — mirrors the reference app, which
    // always has something selected in the detail pane rather than starting on an empty state.
    effect(() => {
      const all = this.store.requests();
      if (all.length > 0 && this.selectedId() === null) {
        this.selectedId.set(all[0].id);
      }
    });

    effect(() => {
      const id = this.selectedId();
      if (id === null || id === this.previousSelectedId) return;

      const host = this.detailHost?.nativeElement;
      if (host) playIfMotionAllowed(buildDetailRevealTimeline(host));
      this.previousSelectedId = id;
    });

    // Re-plays whenever the visible rows actually change (initial load finishing, or the
    // search term narrowing/widening the result set) — not on every keystroke that leaves
    // the result set untouched, since `rows()` only changes reference when it does.
    effect(() => {
      const currentRows = this.rows();
      if (this.isLoading() || currentRows.length === 0) return;

      const host = this.listHost?.nativeElement;
      if (!host) return;

      const items = Array.from(host.querySelectorAll<HTMLElement>('.wreq__row'));
      if (items.length) playIfMotionAllowed(buildListRowsRevealTimeline(items));
    });
  }

  ngOnInit(): void {
    this.store.loadRequests();
  }

  ngOnDestroy(): void {
    this.pendingSelection.clear();
  }

  protected onSearchChange(term: string): void {
    this.searchTerm.set(term);
  }

  protected selectRow(id: string): void {
    this.pendingSelection.select(id, this.selectedId(), resolvedId => this.selectedId.set(resolvedId));
  }

  protected coordinatesLabel(row: WwellPlanningRequest): string {
    return row.finalCoordinates ? `${row.initialCoordinates} → ${row.finalCoordinates}` : row.initialCoordinates;
  }

  protected openNewRequestDialog(): void {
    this.dialog
      .open<WwellRequestFormComponent, unknown, WwellRequestFormResult | undefined>(WwellRequestFormComponent, {
        data: { mode: 'create' },
        panelClass: 'wwell-request-form-panel',
        maxWidth: '95vw',
        autoFocus: false,
      })
      .afterClosed()
      .subscribe(result => {
        if (!result) return;
        this.store.createRequest(result.payload, result.files);
      });
  }

  protected openViewDialog(request: WwellPlanningRequest): void {
    this.dialog.open(WwellRequestFormComponent, {
      data: { mode: 'view', request },
      panelClass: 'wwell-request-form-panel',
      maxWidth: '95vw',
      autoFocus: false,
      enterAnimationDuration: '0ms',
      exitAnimationDuration: '0ms',
    });
  }

  private filterRows(rows: readonly WwellPlanningRequest[]): WwellPlanningRequest[] {
    const term = this.searchTerm().trim().toLowerCase();
    if (!term) return [...rows];

    return rows.filter(
      row =>
        row.id.toLowerCase().includes(term) ||
        row.client.toLowerCase().includes(term) ||
        row.wellName.toLowerCase().includes(term) ||
        row.title.toLowerCase().includes(term),
    );
  }
}
```

```typescript
const SKELETON_ROW_COUNT = 6;

@Component({
  selector: 'app-wwell-wlda-approvals',

  // the current selection just because it fell outside the filtered results — mirrors wwell-ir.

      const currentRows = this.rows();

      const host = this.listHost?.nativeElement;
      if (!host) return;

      const items = Array.from(host.querySelectorAll<HTMLElement>('.wlda__row'));
      if (items.length) playIfMotionAllowed(buildListRowsRevealTimeline(items));
    });
  }

  ngOnInit(): void {
    this.store.loadWldaApprovals();
  }

  ngOnDestroy(): void {
    this.pendingSelection.clear();
  }

  protected onSearchChange(term: string): void {
    this.searchTerm.set(term);
  }

  protected selectRow(id: string): void {
    this.pendingSelection.select(id, this.selectedId(), resolvedId => this.selectedId.set(resolvedId));
  }

  protected openWldaRequestDialog(): void {
    this.dialog
      .open<WwellWldaRequestFormComponent, unknown, WwellWldaRequestFormResult>(WwellWldaRequestFormComponent, {
        panelClass: 'wwell-wlda-request-form-panel',
        maxWidth: '95vw',
        autoFocus: false,
      })
      .afterClosed()
      .subscribe(result => {
        if (!result) return;
        if (result.mode === 'draft') {
          this.snackBar.open('WLDA request saved as draft.', 'Dismiss', { duration: SNACKBAR_DURATION_MS });
          return;
        }
        this.store.submitWldaApproval(result.payload, result.files);
        this.snackBar.open('WLDA approval request submitted.', 'Dismiss', { duration: SNACKBAR_DURATION_MS });
      });
  }

  protected isDownloading(id: string): boolean {
    return this.downloadingIds().has(id);
  }

  protected isChainComplete(chain: readonly ApprovalChainStep[]): boolean {
    return chain.length > 0 && chain.every(step => step.status === 'approved');
  }

  protected chainBorderColor(chain: readonly ApprovalChainStep[]): string {
    return chainCompletionColor(chain);
  }

  protected downloadWldaPdf(row: WldaApproval): void {
    this.downloadPdf(row.id, `${row.reqNo}-WLDA.pdf`, this.svc.getWldaApprovalPdf(row));
  }

  private downloadPdf(id: string, fileName: string, request$: Observable<Blob>): void {
    if (this.isDownloading(id)) return;

    this.setDownloading(id, true);
    request$.subscribe({
      next: blob => {
        const url = URL.createObjectURL(blob);
        const anchor = document.createElement('a');
        anchor.href = url;
        anchor.download = fileName;
        document.body.appendChild(anchor);
        anchor.click();
        document.body.removeChild(anchor);
        setTimeout(() => URL.revokeObjectURL(url), DOWNLOAD_URL_REVOKE_DELAY_MS);
        this.setDownloading(id, false);
      },
      error: () => {
        this.setDownloading(id, false);
        this.snackBar.open('Failed to download the approval PDF.', 'Dismiss', { duration: SNACKBAR_DURATION_MS });
      },
    });
  }

  private setDownloading(id: string, isDownloading: boolean): void {
    const next = new Set(this.downloadingIds());
    if (isDownloading) {
      next.add(id);
    } else {
      next.delete(id);
    }
    this.downloadingIds.set(next);
  }

  private filterRows(rows: readonly WldaApproval[]): WldaApproval[] {
    const term = this.searchTerm().trim().toLowerCase();
    if (!term) return [...rows];

    return rows.filter(
      row =>
        row.reqNo.toLowerCase().includes(term) ||
        row.client.toLowerCase().includes(term) ||
        row.wellName.toLowerCase().includes(term),
    );
  }
}
```
