import { ChangeDetectionStrategy, Component, computed, input, signal } from '@angular/core';
import { toObservable, toSignal } from '@angular/core/rxjs-interop';
import { FormControl, ReactiveFormsModule } from '@angular/forms';
import { MatAutocompleteModule } from '@angular/material/autocomplete';
import { MatButtonToggleModule } from '@angular/material/button-toggle';
import { MatFormFieldModule } from '@angular/material/form-field';
import { MatIconModule } from '@angular/material/icon';
import { MatInputModule } from '@angular/material/input';
import { MatSelectModule } from '@angular/material/select';
import { switchMap, startWith } from 'rxjs';
import { WwellAttachmentDropzoneComponent } from '../../wwell-attachment-dropzone/wwell-attachment-dropzone.component';
import {
  WwellFieldConfig,
  resolveFieldOptions,
  wwellFieldOptionIcon,
  wwellFieldOptionLabel,
  wwellFieldOptionValue,
} from '../wwell-dynamic-form.types';

export type WwellDynamicFieldMode = 'edit' | 'review';

@Component({
  selector: 'app-dynamic-field',
  standalone: true,
  imports: [
    ReactiveFormsModule,
    MatFormFieldModule,
    MatInputModule,
    MatSelectModule,
    MatButtonToggleModule,
    MatAutocompleteModule,
    MatIconModule,
    WwellAttachmentDropzoneComponent,
  ],
  templateUrl: './dynamic-field.component.html',
  styleUrl: './dynamic-field.component.scss',
  changeDetection: ChangeDetectionStrategy.OnPush,
})
export class DynamicFieldComponent {
  readonly config = input.required<WwellFieldConfig>();
  readonly control = input.required<FormControl>();
  readonly mode = input<WwellDynamicFieldMode>('edit');
  /** Class applied to this field's portaled select/autocomplete overlay panel, so a host page can theme it. */
  readonly selectPanelClass = input('');

  protected readonly optionLabel = wwellFieldOptionLabel;
  protected readonly optionValue = wwellFieldOptionValue;
  protected readonly optionIcon = wwellFieldOptionIcon;

  /** Resolves the config's `options` (static array or live Signal) once per change — every
   *  options `@for` loop below reads this instead of `config().options` directly. */
  protected readonly resolvedOptions = computed(() => resolveFieldOptions(this.config()));

  /** Backs the filter input a `searchable: true` 'select' field projects into its own mat-select
   *  panel (see the template) — mat-select has no built-in search. Stays '' for every other
   *  field, since nothing renders an input to change it, so `filteredOptions` below is then just
   *  `resolvedOptions()` unfiltered. */
  protected readonly optionSearchTerm = signal('');

  protected readonly filteredOptions = computed(() => {
    const term = this.optionSearchTerm().trim().toLowerCase();
    if (!term) return this.resolvedOptions();
    return this.resolvedOptions().filter(opt => this.optionLabel(opt).toLowerCase().includes(term));
  });

  /** Clears the filter on close so reopening the panel always starts from the full list instead
   *  of whatever was last typed. */
  protected onSelectOpenedChange(opened: boolean): void {
    if (!opened) this.optionSearchTerm.set('');
  }

  protected onOptionSearchInput(event: Event): void {
    this.optionSearchTerm.set((event.target as HTMLInputElement).value);
  }

  /**
   * `control()` is a Signal input, but the bound FormControl's own `.value` mutates in place
   * on every keystroke and is never itself signal-tracked — reading it directly inside a
   * `computed()` (as `reviewValue` below used to) only evaluates once, at first read, and then
   * freezes forever: the Attachments & Review step's tiles kept showing '—' even after the
   * matching field was filled in on an earlier step. Bridging through `valueChanges` (and
   * re-subscribing whenever a *different* control is bound) keeps this genuinely reactive.
   */
  // No `initialValue`/`requireSync`: reading `this.control()` synchronously in a field
  // initializer (for an `initialValue`) fails the compiler's NG8118 check — inputs aren't bound
  // yet at that point — and `toObservable` doesn't emit synchronously either (its effect is
  // scheduled, not immediate), so `requireSync` throws NG0601 at construction instead. Left
  // without either, `liveValue()` is simply `undefined` for the first tick, which `reviewValue`
  // below already treats the same as an empty value.
  private readonly liveValue = toSignal(
    toObservable(this.control).pipe(switchMap(control => control.valueChanges.pipe(startWith(control.value)))),
  );

  protected readonly reviewValue = computed(() => {
    const value = this.liveValue();
    const cfg = this.config();

    if (cfg.type === 'file') {
      const files = value as File[] | null;
      return files?.length ? `${files.length} file(s) attached` : '—';
    }
    if (cfg.type === 'select' || cfg.type === 'toggle-group') {
      const match = this.resolvedOptions().find(opt => wwellFieldOptionValue(opt) === value);
      return match ? wwellFieldOptionLabel(match) : (value || '—');
    }
    return value === null || value === undefined || value === '' ? '—' : String(value);
  });

  protected readonly isReviewValueEmpty = computed(() => this.reviewValue() === '—');

  protected onFilesChange(files: File[]): void {
    this.control().setValue(files);
    this.control().markAsTouched();
  }

  protected errorMessage(): string | null {
    const control = this.control();
    if (!control.touched || control.valid) return null;
    if (control.hasError('required')) return `${this.config().label} is required`;
    if (control.hasError('pattern')) return `${this.config().label} format is invalid`;
    if (control.hasError('min')) return `${this.config().label} is below the minimum`;
    return 'Invalid value';
  }
}

import type { Signal } from '@angular/core';

export type WwellFieldControlType = 'text' | 'number' | 'textarea' | 'select' | 'combo' | 'toggle-group' | 'file';

export interface WwellFieldOption {
  readonly value: string;
  readonly label: string;
  /** Material icon ligature name — used by 'toggle-group' only. */
  readonly icon?: string;
}

export interface WwellFieldValidatorConfig {
  readonly required?: boolean;
  readonly pattern?: RegExp;
  readonly min?: number;
}

/**
 * Field is only shown (and its control enabled) while a sibling control in the same section
 * matches this condition. Exactly one of `equals` | `oneOf` | `notOneOf` should be set:
 *  - `equals`: sibling value === this value
 *  - `oneOf`: sibling value is one of this set (e.g. a client tier)
 *  - `notOneOf`: sibling value is absent from this set — the fallback for values outside every
 *    named tier (e.g. a client not yet assigned to a tier keeps the default field set)
 */
export interface WwellFieldVisibilityCondition {
  readonly field: string;
  readonly equals?: string;
  readonly oneOf?: readonly string[];
  readonly notOneOf?: readonly string[];
}

export interface WwellFieldConfig {
  readonly key: string;
  readonly label: string;
  readonly type: WwellFieldControlType;
  readonly placeholder?: string;

import gsap from 'gsap';

const COLLAPSE_ENTER_DURATION_S = 0.2;
const COLLAPSE_EXIT_DURATION_S = 0.15;
const COLLAPSE_SLIDE_PX = 6;

/**
 * Compositor-only (`opacity` + `transform`) — deliberately never touches `height`. A section here
 * can hold several mat-form-fields or a multi-row table, and tweening `height` (even to/from GSAP's
 * measured 'auto') forces a synchronous layout recalculation on every single frame, which is what
 * made this read as stuttering/heavy instead of smooth. `transform`/`opacity` are compositor-only —
 * the browser can run them entirely on the GPU with zero layout work (see the project's own
 * performance rules: prefer transform/opacity, avoid animating height/width/margin/padding).
 *
 * The tradeoff: surrounding content still shifts in the same frame the element mounts/unmounts
 * (no smooth height "push") rather than over the animation's duration — invisible in practice
 * since it lines up with the very start of the fade, and far lighter than a stuttering alternative.
 */
export function buildCollapseEnterTimeline(host: HTMLElement, onComplete: () => void): gsap.core.Timeline {
  return gsap
    .timeline({ paused: true, onComplete })
    .fromTo(
      host,
      { opacity: 0, y: -COLLAPSE_SLIDE_PX },
      { opacity: 1, y: 0, duration: COLLAPSE_ENTER_DURATION_S, ease: 'power1.out' },
    );
}

/** `onComplete` resolves `animate.leave`'s own `animationComplete()` — Angular only detaches the
 *  node once that runs, so the exit tween is guaranteed to finish on screen first regardless of
 *  what triggered the removal (a click, a signal flip elsewhere, whatever). */
export function buildCollapseExitTimeline(host: HTMLElement, onComplete: () => void): gsap.core.Timeline {
  return gsap
    .timeline({ paused: true, onComplete })
    .to(host, { opacity: 0, y: -COLLAPSE_SLIDE_PX, duration: COLLAPSE_EXIT_DURATION_S, ease: 'power1.in' });
}

import { Directive, ElementRef, inject } from '@angular/core';
import type { AnimationCallbackEvent } from '@angular/core';
import { playIfMotionAllowed } from '../dynamic-form/gsap-motion.util';
import { buildCollapseEnterTimeline, buildCollapseExitTimeline } from './wwell-collapse-animate.animation';

/**
 * Drop-in smooth open/close for any block Angular's control flow (`@if`/`@for`) adds or removes
 * from the DOM — a section swapping from its read display to an edit form, a timeline step
 * expanding, a filter row appearing. Without this, `@if` just snaps the DOM in and out instantly,
 * which is the "tabs" in this editor (section edit toggles, timeline step accordions) reading as
 * abrupt rather than smooth.
 *
 * Hooks Angular's native `animate.enter`/`animate.leave` (fired right before insertion / right
 * before removal — see @angular/core's `AnimationCallbackEvent`) to run a lightweight, compositor-
 * only (opacity + transform, never height — see wwell-collapse-animate.animation.ts) GSAP tween
 * instead of nothing. `animate.leave`'s `animationComplete()` is what actually lets Angular
 * detach the node, so the exit tween is guaranteed to finish on screen before the element
 * disappears, no matter what triggered the removal. Respects prefers-reduced-motion via the
 * shared `playIfMotionAllowed` (jumps straight to the end state, still resolving
 * `animationComplete()` either way, so a reduced-motion user is never left with an undetached
 * node stuck mid-removal).
 *
 * Usage: add the `appWwellCollapseAnimate` attribute directly to whichever element is the root of
 * the `@if`/`@else` branch that should transition — e.g. both the read-display root and the
 * edit-form root of a section, so switching between them cross-fades instead of swapping instantly.
 */
@Directive({
  selector: '[appWwellCollapseAnimate]',
  standalone: true,
  host: {
    '(animate.enter)': 'onEnter($event)',
    '(animate.leave)': 'onLeave($event)',
  },
})
export class WwellCollapseAnimateDirective {
  private readonly hostEl = inject<ElementRef<HTMLElement>>(ElementRef).nativeElement;

  protected onEnter(event: AnimationCallbackEvent): void {
    playIfMotionAllowed(buildCollapseEnterTimeline(this.hostEl, () => event.animationComplete()));
  }

  protected onLeave(event: AnimationCallbackEvent): void {
    playIfMotionAllowed(buildCollapseExitTimeline(this.hostEl, () => event.animationComplete()));
  }
}

import { computed, inject } from '@angular/core';
import { HttpErrorResponse } from '@angular/common/http';
import { patchState, signalStore, withComputed, withMethods, withState } from '@ngrx/signals';
import { rxMethod } from '@ngrx/signals/rxjs-interop';
import { tapResponse } from '@ngrx/operators';
import { pipe, switchMap, tap } from 'rxjs';
import { CreateIntegratedRequirementPayload, IntegratedRequirement } from '@models/wwell-planning/integrated-requirement.model';
import { mergeIrDetailBlocksIntoIntegratedRequirement } from '../wwell-ir-form/wwell-ir-detail-response.mapper';
import { WwellIrBulkSavePayload } from '../wwell-ir-form/wwell-ir-payload.model';
import { WwellIrService } from '../wwell-ir.service';
import { WwellIrStoreState, initialWwellIrState } from './wwell-ir.state';

/** Prefers the backend's own `{ message }` (see ApiResponse<T> — every exad_agwa endpoint here
 *  returns that envelope on both success and error) over Angular's generic "Http failure
 *  response for ..." text, so a specific backend validation error (e.g. an Oracle constraint
 *  message) reaches the user instead of a vague fallback. */
function errorMessage(err: unknown, fallback: string): string {
  if (err instanceof HttpErrorResponse) {
    const body: unknown = err.error;
    const backendMessage = body && typeof body === 'object' ? (body as { message?: unknown }).message : undefined;
    if (typeof backendMessage === 'string' && backendMessage) return backendMessage;
    return err.message || fallback;
  }
  return err instanceof Error ? err.message : fallback;
}

/** A repeat "Save as Draft" on the same IR reuses its ir_seq (see wwell-ir-payload.mapper.ts), so
 *  the response describes the *same* record, not a new one — replace it in place instead of
 *  prepending a duplicate. Falls back to prepend for a genuinely new IR (irSeq not seen yet). */
function upsertByIrSeq(list: readonly IntegratedRequirement[], created: IntegratedRequirement): IntegratedRequirement[] {
  const existingIndex = created.irSeq != null ? list.findIndex(item => item.irSeq === created.irSeq) : -1;
  if (existingIndex === -1) return [created, ...list];
  // `created` is rebuilt client-side from what was just submitted (see
  // WwellIrService.buildIrFromPayload) — it has no way to know the record's real approval chain
  // (that's a separate, still-mocked endpoint; see loadIrApprovalTimeline), so every resave mints
  // a brand new "just initiated" chain. Harmless for a first-time INSERT (nothing to preserve
  // yet), but a repeat save on an already-in-progress/approved record — including every single
  // inline section edit on the detail page — would otherwise wipe out its real chain state on
  // every save. Keep whichever chain this record already had.
  return list.map((item, index) => (index === existingIndex ? { ...created, chain: item.chain } : item));
}

export const WwellIrStore = signalStore(
  withState<WwellIrStoreState>(initialWwellIrState),

  withComputed(({ irsLoading, linkedRequestWells, linkedRequestWellsLoading, irsError, linkedRequestWellsError, createIrError, updateIrStatusError, irDetailError, irApprovalTimelineError }) => ({
    isLoading: computed(() => irsLoading() || linkedRequestWellsLoading()),
    /** Trimmed {value,label} view of linkedRequestWells for the "Link to Request" picker itself —
     *  see wwell-ir-editor.component.ts's `applyLinkedRequestWell` for where the full record
     *  (looked up from linkedRequestWells by this same value) drives the auto-populate. */
    linkedRequestWellOptions: computed(() => linkedRequestWells().map(well => ({ value: String(well.prewapSeq), label: well.wellName }))),
    activeErrors: computed(() =>
      [irsError(), linkedRequestWellsError(), createIrError(), updateIrStatusError(), irDetailError(), irApprovalTimelineError()].filter(
        (message): message is string => message !== null,
      ),
    ),
    hasError: computed(
      () =>
        irsError() !== null ||
        linkedRequestWellsError() !== null ||
        createIrError() !== null ||
        updateIrStatusError() !== null ||
        irDetailError() !== null ||
        irApprovalTimelineError() !== null,
    ),
  })),

  withMethods((store, svc = inject(WwellIrService)) => {
    const loadIrs = rxMethod<void>(
      pipe(
        tap(() => patchState(store, { irsLoading: true, irsError: null })),
        switchMap(() =>
          svc.getIrs().pipe(
            tapResponse({
              next: irs => patchState(store, { irs, irsLoading: false }),
              error: (err: unknown) => patchState(store, { irsLoading: false, irsError: errorMessage(err, 'Failed to load integrated requirements') }),
            }),
          ),
        ),
      ),
    );

    const loadLinkedRequestWells = rxMethod<void>(
      pipe(
        tap(() => patchState(store, { linkedRequestWellsLoading: true, linkedRequestWellsError: null })),
        switchMap(() =>
          svc.getEligibleWellsForIr().pipe(
            tapResponse({
              next: linkedRequestWells => patchState(store, { linkedRequestWells, linkedRequestWellsLoading: false }),
              error: (err: unknown) =>
                patchState(store, {
                  linkedRequestWellsLoading: false,
                  linkedRequestWellsError: errorMessage(err, 'Failed to load wells eligible for IR initiation'),
                }),
            }),
          ),
        ),
      ),
    );

    // Keyed by ir_seq (the real primary key once an IR is saved) — switchMap cancels a still-in-
    // flight fetch for a previously-selected row if the user picks another one before it resolves,
    // so a slow first response can never land after (and clobber) a faster later selection.
    const loadIrDetail = rxMethod<number>(
      pipe(
        tap(() => patchState(store, { irDetailLoading: true, irDetailError: null })),
        switchMap(irSeq =>
          svc.getIrDetailBlocks(irSeq).pipe(
            tapResponse({
              next: blocks => {
                const existing = store.irs().find(item => item.irSeq === irSeq);
                // Nothing to merge onto if the summary row isn't loaded (shouldn't happen from the
                // normal select-a-row flow) — just clear the loading flag and leave state as-is.
                if (!existing) {
                  patchState(store, { irDetailLoading: false });
                  return;
                }
                const merged = mergeIrDetailBlocksIntoIntegratedRequirement(existing, blocks);
                patchState(store, { irs: upsertByIrSeq(store.irs(), merged), irDetailLoading: false });
              },
              error: (err: unknown) =>
                patchState(store, { irDetailLoading: false, irDetailError: errorMessage(err, 'Failed to load integrated requirement detail') }),
            }),
          ),
        ),
      ),
    );

    // Fires right after a save settles (see createIr below) to replace the client-side optimistic
    // echo (WwellIrService.buildIrFromPayload — built from what was *submitted*, not what the
    // database actually stored: server-computed defaults, triggers, or a concurrent edit would all
    // be invisible until this runs) with the real persisted row. Deliberately does not touch
    // `irDetailLoading` — unlike loadIrDetail above, it must never flip WwellIrComponent's
    // `isSelecting()` (which swaps the whole detail panel for a loading skeleton): the user just
    // watched this exact record save, so re-showing it as a blank loading state a moment later
    // would be jarring, not reassuring. A failure here is silent by design — the save itself
    // already succeeded and is already reflected via the optimistic echo, so a background refresh
    // hiccup shouldn't surface as if the save had failed.
    const refreshIrDetail = rxMethod<number>(
      pipe(
        switchMap(irSeq =>
          svc.getIrDetailBlocks(irSeq).pipe(
            tapResponse({
              next: blocks => {
                const existing = store.irs().find(item => item.irSeq === irSeq);
                if (!existing) return;
                const merged = mergeIrDetailBlocksIntoIntegratedRequirement(existing, blocks);
                patchState(store, { irs: upsertByIrSeq(store.irs(), merged) });
              },
              error: () => undefined,
            }),
          ),
        ),
      ),
    );

    // Separate call from loadIrDetail above — mirrors the future real API shape (approval process
    // gets its own endpoint), mocked client-side in the meantime. Same switchMap-cancels-a-stale-
    // fetch reasoning applies when the selection changes quickly.
    const loadIrApprovalTimeline = rxMethod<number>(
      pipe(
        tap(() => patchState(store, { irApprovalTimelineLoading: true, irApprovalTimelineError: null })),
        switchMap(irSeq =>
          svc.getIrApprovalTimeline(irSeq).pipe(
            tapResponse({
              next: chain => {
                patchState(store, {
                  irs: store.irs().map(item => (item.irSeq === irSeq ? { ...item, chain } : item)),
                  irApprovalTimelineLoading: false,
                });
              },
              error: (err: unknown) =>
                patchState(store, { irApprovalTimelineLoading: false, irApprovalTimelineError: errorMessage(err, 'Failed to load approval timeline') }),
            }),
          ),
        ),
      ),
    );

    return {
      loadIrs(): void {

      },

      loadIrDetail(irSeq: number): void {
        loadIrDetail(irSeq);
      },

      loadIrApprovalTimeline(irSeq: number): void {
        loadIrApprovalTimeline(irSeq);
      },

      loadLinkedRequestWells(): void {
        loadLinkedRequestWells();
      },

      createIr(
        payload: CreateIntegratedRequirementPayload,
        irBulkPayload: WwellIrBulkSavePayload,
        attachmentNames: readonly string[] = [],
      ): void {
        patchState(store, { creatingIr: true, createIrError: null });
        svc.createIr(payload, irBulkPayload, attachmentNames).subscribe({
          next: created => {
            patchState(store, {
              irs: upsertByIrSeq(store.irs(), created),
              // Falls back to whatever was already known so an UPDATE response missing irSeq never clears it.
              draftIrSeq: created.irSeq ?? store.draftIrSeq(),
              creatingIr: false,
            });
            // Every save (create or update) immediately shows the optimistic echo above, then
            // quietly swaps in the real persisted row once it lands — see refreshIrDetail's own
            // comment for why this never blocks or reloads the visible UI.
            if (created.irSeq != null) refreshIrDetail(created.irSeq);
          },
          error: (err: unknown) => patchState(store, { creatingIr: false, createIrError: errorMessage(err, 'Failed to create integrated requirement') }),
        });
      },

      submitIrForReview(ir: IntegratedRequirement): void {
        patchState(store, { updatingIrStatus: true, updateIrStatusError: null });
        svc.submitIrForReview(ir).subscribe({
          next: updated => patchState(store, { irs: store.irs().map(item => (item.id === updated.id ? updated : item)), updatingIrStatus: false }),
          error: (err: unknown) =>
            patchState(store, { updatingIrStatus: false, updateIrStatusError: errorMessage(err, 'Failed to submit integrated requirement for review') }),
        });
      },
    };
  }),

  // pencil is clicked, so opening a section always starts from the latest saved values. ─────────
  protected readonly requestReferenceEdit = new SectionEditState(() => {
    const group = buildAndPatchGroup(
      this.referenceMainSection,
      this.fb,
      mapIntegratedRequirementToFormPatch(this.ir()).reference.main,
    );
    // Record already has a linked request (opened for edit, not just picked this session) — lock
    // its populated fields immediately instead of waiting for a valueChanges emission that already
    // happened during patchValue() above, before anything was subscribed to hear it.
    const linkedRequestId = group.get('linkedRequestId')?.value as string | null;
    if (linkedRequestId) this.applyLinkedRequestWell(group, linkedRequestId);
    return group;
  });
  protected readonly programMainEdit = new SectionEditState(() =>
    buildAndPatchGroup(this.programMainSection, this.fb, mapIntegratedRequirementToFormPatch(this.ir()).programDetails.main),
  );
  protected readonly programDevEdit = new SectionEditState(() =>
    buildAndPatchGroup(
      this.programDevSection,
      this.fb,
      mapIntegratedRequirementToFormPatch(this.ir()).programDetails.developmentTesting,
    ),
  );
  protected readonly geologistContactEdit = new SectionEditState(() =>
    buildAndPatchGroup(this.geologistSection, this.fb, mapIntegratedRequirementToFormPatch(this.ir()).contacts.geologist),
  );
  protected readonly hydrogeologyEdit = new SectionEditState(() =>
    buildAndPatchGroup(this.hydrogeologySection, this.fb, mapIntegratedRequirementToFormPatch(this.ir()).hydrogeology.main),
  );
  protected readonly formationTopsEdit = new SectionEditState<FormArray, WwellIrRawFormationTop[]>(() =>
    buildAndPatchArray(this.formationTopsSection, this.fb, this.ir().formationTops, formationTopRowPatch),
  );
  protected readonly casingEdit = new SectionEditState<FormArray, WwellIrRawCasingRow[]>(() =>
    buildAndPatchArray(this.casingSection, this.fb, this.ir().casing, casingRowPatch),
  );
  protected readonly offsetWellsEdit = new SectionEditState(() => {
    const current = this.ir();
    return this.fb.group({
      offsetWells: buildAndPatchArray(this.offsetWellsSection, this.fb, current.offsetWells, offsetWellRowPatch),
      remarks: buildAndPatchGroup(this.offsetRemarksSection, this.fb, { offsetWellsRemarks: current.offsetWellsRemarks ?? '' }),
    });
  });

  // Attachments has no schema-form counterpart worth reusing — both upload slots save into one
  // flat `attachmentNames` list on the domain record (see wwell-ir-section-rail.config.ts's
  // "attachments" entry), and a File can't be reconstructed from a saved name either way — so
  // this is a plain add/remove-by-name editor instead of a SectionEditState/dynamic-section instance.
  protected readonly attachmentsEditing = signal(false);
  protected readonly attachmentsDraftNames = signal<readonly string[]>([]);
  protected readonly attachmentsNewFiles = signal<readonly File[]>([]);

  // One entry per flattened rail leaf, in the same order — see wwell-ir-editor.component.html.
  @ViewChildren('sectionEl', { read: ElementRef }) private readonly sectionEls?: QueryList<ElementRef<HTMLElement>>;
  // The internally-scrolling region (header/footer stay fixed) — see wwell-ir-editor.component.scss.
  @ViewChild('body', { read: ElementRef }) private readonly bodyEl?: ElementRef<HTMLElement>;

  // Tracks which record this editor instance is currently showing — see the identity-change
  // effect below, which is the only reader/writer of this outside the constructor.
  private previousIrId: string | null = null;

  constructor() {
    // The "Link to Request" select lives inside requestReferenceEdit's own lazily-built
    // FormGroup (a fresh instance every time that section's pencil is clicked — see
    // SectionEditState.open()), so this can't be a one-time subscription set up in the
    // constructor: it has to re-subscribe to whichever FormGroup instance is *currently* open,
    // and stop when that section closes (control() goes back to null) or is reopened (a new
    // FormGroup replaces the old one). `effect()`'s cleanup callback handles both.
    effect(onCleanup => {
      const ctrl = this.requestReferenceEdit.control();
      if (!ctrl) return;
      const linkedRequestIdControl = ctrl.get('linkedRequestId');
      const subscription = linkedRequestIdControl?.valueChanges.subscribe((linkedRequestId: string) => {
        if (linkedRequestId) this.applyLinkedRequestWell(ctrl, linkedRequestId);
      });
      onCleanup(() => subscription?.unsubscribe());
    });

    // WwellIrComponent reuses this same editor instance across selections — its `@else if
    // (selectedIr(); as ir)` branch never changes just because the *bound* record does (switching
    // list rows, or opening a blank "New IR" draft), so nothing here gets recreated for free. A
    // section already open at that moment keeps the FormGroup it already built — SectionEditState
    // .open() is a no-op once editing (see its own doc comment) — so without this, an open
    // section keeps showing the *previous* record's values after the identity underneath it
    // changes. Closing everything forces every section back to its read display, which re-derives
    // fresh from whatever `ir()` now is on next open. Compares `id` (not the object reference,
    // which also changes on every in-place update from a save) so an update to the *same* record
    // never closes sections the user still has open mid-edit.
    effect(() => {
      const id = this.ir().id;
      if (this.previousIrId !== null && id !== this.previousIrId) this.closeAllSections();
      this.previousIrId = id;
    });
  }

  ngOnInit(): void {
    this.store.loadLinkedRequestWells();
  }

  ngAfterViewInit(): void {
    if (this.bodyEl && this.sectionEls) this.rail.attach(this.bodyEl, this.sectionEls);
  }

  /** Rail click: leaf index is the flattened `railLeaves` index, which is also the `#sectionEl`
   *  DOM order — see wwellIrRailLeaves()'s doc comment. */
  protected leafIndex(leafKey: string): number {
    return this.railLeaves.findIndex(leaf => leaf.key === leafKey);
  }

  protected scrollToLeaf(leafKey: string): void {
    this.rail.scrollToIndex(this.leafIndex(leafKey));
  }

  /** Group header click (e.g. "2.0 General") scrolls to its first child. */
  protected scrollToGroupStart(group: WwellIrRailGroup): void {
    const key = group.leaf?.key ?? group.children?.[0]?.key;
    if (key) this.scrollToLeaf(key);
  }

  protected isLeafActive(leafKey: string): boolean {
    return this.rail.activeIndex() === this.leafIndex(leafKey);
  }

  protected subSeaFt(depthFt: number): number | null {
    const elevation = this.ir().requestReference.surfaceElevationFtAsl;
    return elevation === null ? null : elevation - depthFt;
  }

  /** Selecting a "Link to Request" option auto-fills whichever Request Reference fields that
   *  eligible-well record actually has data for — BI Number/Supporting Business/Target
   *  Aquifer/Target Depth have no source there (exad_rcd_prewap doesn't carry them) and stay
   *  manually entered. Purely a client-side lookup against the already-loaded eligible-wells
   *  list (see store.loadLinkedRequestWells()) — selecting a link never itself triggers a
   *  network call.
   *
   *  Whichever fields the link actually supplies are then locked read-only: they track the source
   *  request record, so letting someone edit or clear them here would silently diverge from (or
   *  desync) that source. Also reused at initial section-open for an already-linked record (see
   *  requestReferenceEdit above) — re-locks the same fields to the same values, so it's a no-op
   *  data-wise there. */
  private applyLinkedRequestWell(referenceGroup: FormGroup, linkedRequestId: string): void {
    const well = this.store.linkedRequestWells().find(candidate => String(candidate.prewapSeq) === linkedRequestId);
    if (!well) return;

    const hasElevation = well.surfaceElevation != null;
    const hasCoordinates = well.surfaceNorth != null && well.surfaceEast != null;

    const patch: Record<string, string | number> = { wellName: well.wellName };
    if (hasElevation) patch['surfaceElevationFtAsl'] = well.surfaceElevation;
    if (hasCoordinates) patch['coordinates'] = `${well.surfaceNorth}°N, ${well.surfaceEast}°E`;
    referenceGroup.patchValue(patch);

    referenceGroup.get('wellName')?.disable();
    this.setLinkedFieldLock(referenceGroup, 'surfaceElevationFtAsl', hasElevation);
    this.setLinkedFieldLock(referenceGroup, 'coordinates', hasCoordinates);
  }

  /** Locks/unlocks one Request Reference field depending on whether the currently-linked request
   *  actually supplied it — switching to a different linked request that lacks, say, coordinates
   *  hands that field back for manual entry rather than leaving it stuck disabled and empty. */
  private setLinkedFieldLock(group: FormGroup, key: string, locked: boolean): void {
    const control = group.get(key);
    if (!control) return;
    if (locked) control.disable();
    else control.enable();
  }

  protected offsetWellsArrayControl(ctrl: FormGroup): FormArray {
    return ctrl.get('offsetWells') as FormArray;
  }

  protected offsetWellsRemarksControl(ctrl: FormGroup): FormGroup {
    return ctrl.get('remarks') as FormGroup;
  }

  private currentRawFormValue(): WwellIrFormRawValue {
    return mapIntegratedRequirementToRawFormValue(this.ir());
  }

  /** Shared save path: re-derives the payload/bulk-save arrays from the *current* record with the
   *  open sections' slices overlaid, so a section nobody touched this time round is resent
   *  unchanged (and diffed to a DML no-op by buildWwellIrBulkSavePayload's `existing` comparison)
   *  rather than dropped.
   *
   *  Deliberately fires no success toast — the request is still in flight here, so anything shown
   *  now would claim success before the backend has answered (and would still show next to the
   *  error toast when the save actually fails). WwellIrComponent reports the real outcome once
   *  store.creatingIr() settles.
   *
   *  Status handling: the save payload type only ever accepts 'draft' | 'in-progress' (see
   *  CreateIntegratedRequirementPayload) — there's no contract for "edit this field but keep it
   *  Approved/In Review". Editing a draft/in-progress record keeps its current status; editing
   *  anything further along pulls it back to 'in-progress' (not 'draft', since it was already
   *  worked past that point) so it's clear the record needs to move through review again. */
  private saveMergedRaw(raw: WwellIrFormRawValue, attachmentNames?: readonly string[]): void {
    const current = this.ir();
    const status: Extract<IntegratedRequirementStatus, 'draft' | 'in-progress'> =
      current.status === 'draft' || current.status === 'in-progress' ? current.status : 'in-progress';
    const { payload } = mapIrFormValueToSubmission(raw, status);
    const irBulkPayload = buildWwellIrBulkSavePayload(raw, { irSeq: current.irSeq ?? null, existing: current });
    this.store.createIr(payload, irBulkPayload, attachmentNames ?? current.attachmentNames);
  }

  /** Global footer "Save" — merges every currently-open section's live form value (plus
   *  attachments, if that's open too) into a single raw value and saves it in one call, instead of
   *  requiring the user to click each open section's own Save button separately. A section that
   *  isn't open contributes its existing saved value unchanged, exactly as it would for any one
   *  section's own save (see saveMergedRaw / currentRawFormValue). This is also what turns a brand
   *  new local-only draft (ir().irSeq === null) into a real backend record on the very first click,
   *  even if the user never opened a single section. */
  protected saveAll(): void {
    let raw = this.currentRawFormValue();

    const referenceCtrl = this.requestReferenceEdit.control();
    if (referenceCtrl) raw = { ...raw, reference: { main: referenceCtrl.getRawValue() } };

    const programMainCtrl = this.programMainEdit.control();
    if (programMainCtrl) raw = { ...raw, programDetails: { ...raw.programDetails, main: programMainCtrl.getRawValue() } };

    const programDevCtrl = this.programDevEdit.control();
    if (programDevCtrl) {
      raw = { ...raw, programDetails: { ...raw.programDetails, developmentTesting: programDevCtrl.getRawValue() } };
    }

    const geologistCtrl = this.geologistContactEdit.control();
    if (geologistCtrl) raw = { ...raw, contacts: { ...raw.contacts, geologist: geologistCtrl.getRawValue() } };

    const hydrogeologyCtrl = this.hydrogeologyEdit.control();
    if (hydrogeologyCtrl) raw = { ...raw, hydrogeology: { main: hydrogeologyCtrl.getRawValue() } };

    const formationTopsCtrl = this.formationTopsEdit.control();
    if (formationTopsCtrl) raw = { ...raw, subsurface: { ...raw.subsurface, formationTops: formationTopsCtrl.getRawValue() } };

    const casingCtrl = this.casingEdit.control();
    if (casingCtrl) raw = { ...raw, subsurface: { ...raw.subsurface, casing: casingCtrl.getRawValue() } };

    const offsetWellsCtrl = this.offsetWellsEdit.control();
    if (offsetWellsCtrl) {
      raw = {
        ...raw,
        contacts: {
          ...raw.contacts,
          offsetWells: this.offsetWellsArrayControl(offsetWellsCtrl).getRawValue(),
          remarks: this.offsetWellsRemarksControl(offsetWellsCtrl).getRawValue(),
        },
      };
    }

    const attachmentNames = this.attachmentsEditing()
      ? [...this.attachmentsDraftNames(), ...this.attachmentsNewFiles().map(file => file.name)]
      : undefined;

    this.saveMergedRaw(raw, attachmentNames);
    this.closeAllSections();
  }

  private closeAllSections(): void {
    this.requestReferenceEdit.close();
    this.programMainEdit.close();
    this.programDevEdit.close();
    this.geologistContactEdit.close();
    this.hydrogeologyEdit.close();
    this.formationTopsEdit.close();
    this.casingEdit.close();
    this.offsetWellsEdit.close();
    this.closeAttachmentsEdit();
  }

  protected openAttachmentsEdit(): void {
    this.attachmentsDraftNames.set(this.ir().attachmentNames);
    this.attachmentsNewFiles.set([]);
    this.attachmentsEditing.set(true);
  }

  protected closeAttachmentsEdit(): void {
    this.attachmentsEditing.set(false);
  }

  protected removeAttachmentName(name: string): void {
    this.attachmentsDraftNames.update(names => names.filter(existing => existing !== name));
  }

  protected removeNewAttachmentFile(file: File): void {
    this.attachmentsNewFiles.update(files => files.filter(existing => existing !== file));
  }

  protected onAttachmentFilesPicked(event: Event): void {
    const input = event.target as HTMLInputElement;
    const picked = input.files ? Array.from(input.files) : [];
    this.attachmentsNewFiles.update(files => [...files, ...picked]);
    input.value = '';
  }

  protected submitForReview(): void {
    this.store.submitIrForReview(this.ir());
  }
}

import { IntegratedRequirement, IntegratedRequirementTestType } from '@models/wwell-planning/integrated-requirement.model';
import { WWELL_TRANSMISSION_CODE_BY_LABEL } from '../../shared/wwell-planning-reference-data.config';
import {
  IrCasingRow,
  IrHeaderRow,
  IrHydrogeologyRow,
  IrTopsRow,
  IrWaterRow,
  WwellIrDetailTableBlock,
  WwellIrTableName,
} from './wwell-ir-payload.model';

/** Inverse of wwell-ir-payload.mapper.ts's resolveTransmissionCode: exad_gwd_ir_header.trns_type
 *  is VARCHAR2(10), too short for the full label, so a known label is stored as its short code
 *  ('Pipeline' → 'PL'). Decode it back to the label the form's option list expects, otherwise the
 *  select has no matching option and silently renders blank. Anything unrecognised (free-typed
 *  text that was truncated to fit) is passed through as-is. */
function decodeTransmission(code: string | null | undefined): string | null {
  if (!code) return null;
  const label = Object.keys(WWELL_TRANSMISSION_CODE_BY_LABEL).find(key => WWELL_TRANSMISSION_CODE_BY_LABEL[key] === code);
  return label ?? code;
}

/** Every IrXxxRow interface in wwell-ir-payload.model.ts (IrHeaderRow, IrCasingRow, IrTopsRow,
 *  IrHydrogeologyRow, IrWaterRow) uses lowercase snake_case field names mirroring the Oracle
 *  column names — but Oracle's own default column casing (and whatever exad_agwa_queries_pkg's
 *  get_ir_data_prc happens to emit) isn't guaranteed to be lowercase, so a row can come back with
 *  the same fields in UPPERCASE or MixedCase. Since every field this file reads is lowercase by
 *  contract, lowercasing every key up front makes each row match regardless of how the backend
 *  cased it — the same fix as `rowsFor`'s table-name comparison below, applied to row field names. */
function normalizeRowKeys<TRow>(row: Record<string, unknown>): TRow {
  const normalized: Record<string, unknown> = {};
  for (const key of Object.keys(row)) {
    normalized[key.toLowerCase()] = row[key];
  }
  return normalized as TRow;
}

/** Case-insensitive on `candidate.table`: the GET /{irSeq} detail endpoint echoes back
 *  UPPERCASE table names (EXAD_GWD_IR_HEADER...) per exad_agwa_queries_pkg.get_ir_data_prc,
 *  while WwellIrTableName (and the bulk-save write path) uses lowercase — a strict `===` here
 *  never matched, so every block silently resolved to empty and wiped casing/formationTops on
 *  every detail fetch. */
function rowsFor<TRow>(blocks: readonly WwellIrDetailTableBlock[], table: WwellIrTableName): readonly TRow[] {
  const block = blocks.find(candidate => candidate.table.toLowerCase() === table);
  return (block?.data ?? []).map(row => normalizeRowKeys<TRow>(row));
}

/** Inverse of the `h2s: '${ppm} ppm'` formatting in wwell-ir-payload.mapper.ts. */
function parseH2sPpm(h2s: string | null | undefined): number | null {
  if (!h2s) return null;
  const value = parseFloat(h2s);
  return Number.isFinite(value) ? value : null;
}

/**
 * Enriches an already-known IntegratedRequirement (e.g. from the GET-all summary list) with the
 * real casing/tops/hydrogeology/program data from a GET /{irSeq} detail fetch. Merges onto
 * `existing` rather than building from scratch because most domain-only fields (client, wellName,
 * supportingBusiness, targetAquifer, targetDepthFt, surfaceElevationFtAsl, coordinates, contacts,
 * offsetWells, attachments...) have no backing column on exad_gwd_ir_header — see
 * wwell-ir-summary.mapper.ts for the same limitation on the list side, and
 * docs/ir/EXAD_GWD_IR_HEADER.md for the table's real column set. `linkedRequestId`/`letterNumber`
 * (prewap_seq/ltr_no) and the Development & Testing fields (sampling/logging/mud_rmks/flow_rmks)
 * DO have real columns, so those are read back from `header` below rather than left to the
 * optimistic echo. Field mappings mirror buildWwellIrBulkSavePayload in wwell-ir-payload.mapper.ts
 * exactly (this is that function's inverse) — keep the two in sync if either changes.
 */
export function mergeIrDetailBlocksIntoIntegratedRequirement(
  existing: IntegratedRequirement,
  blocks: readonly WwellIrDetailTableBlock[],
): IntegratedRequirement {
  const header = rowsFor<IrHeaderRow>(blocks, 'exad_gwd_ir_header')[0] as IrHeaderRow | undefined;
  const casingRows = rowsFor<IrCasingRow>(blocks, 'exad_gwd_ir_casing');
  const topsRows = rowsFor<IrTopsRow>(blocks, 'exad_gwd_ir_tops');
  const hydroRow = rowsFor<IrHydrogeologyRow>(blocks, 'exad_gwd_ir_hydrogeology')[0] as IrHydrogeologyRow | undefined;
  const waterRows = rowsFor<IrWaterRow>(blocks, 'exad_gwd_ir_water');

  return {
    ...existing,
    requestReference: {
      ...existing.requestReference,
      // prewap_seq/ltr_no are the only two Request Reference columns exad_gwd_ir_header actually
      // has (see docs/ir/EXAD_GWD_IR_HEADER.md) — everything else on this section (wellName,
      // client, targetAquifer...) has no backing column and stays whatever `existing` already
      // carries (the optimistic save-time echo), same limitation this function's own doc comment
      // above describes.
      linkedRequestId: header?.prewap_seq != null ? String(header.prewap_seq) : existing.requestReference.linkedRequestId,
      letterNumber: header?.ltr_no ?? existing.requestReference.letterNumber,
    },
    formationTops: topsRows.map(row => ({
      name: row.st_long_cd,
      depthFt: row.plan_tvd_dpth,
      subSeaFtAsl: row.plan_ss_dpth ?? null,
    })),
    // exad_gwd_ir_water has no well-name column — epANum is the only identifier that round-trips
    // (see IrWaterRow / docs/ir/p2-ir.md). `temp` has no form field, so it is simply not read.
    offsetWells: waterRows.map(row => ({
      epANum: row.ep_a_num,
      distanceKm: row.distanct ?? null,
      location: row.direction ?? null,
      aquifer: row.aquifer ?? null,
      totalDepthFt: row.td ?? null,
      testDurationMin: row.test_duration ?? null,
      rateGpm: row.rate ?? null,
      rpm: row.rpm ?? null,
      swlFt: row.swl ?? null,
      pwlFt: row.final_pwl ?? null,
      drawdownFt: row.drawdown ?? null,
      specificCapacity: row.spec_capacity ?? null,
      tdsPpm: row.field_tds ?? null,
      h2sPpm: row.h2s ?? null,
    })),
    // tolFtBgl has no column yet (buildWwellIrBulkSavePayload never writes it either) — stays null.
    casing: casingRows.map(row => ({
      type: row.csg_type,
      sizeIn: row.csg_size,
      depthFtBgl: row.csg_depth,
      tolFtBgl: null,
      remarks: row.csg_remarks ?? null,
    })),
    programDetails: {
      ...existing.programDetails,
      completion: header?.completion ?? header?.cmpl_type ?? existing.programDetails.completion,
      // Both are written by buildWwellIrBulkSavePayload but were never read back here, so
      // Transmission/Acidization rendered blank after any reload even though the values were in
      // the DB the whole time — the same class of gap that hid Logging/Sampling before.
      acidization: header?.acidization ?? existing.programDetails.acidization,
      transmission: decodeTransmission(header?.trns_type) ?? existing.programDetails.transmission,
      developmentTesting: {
        logging: header?.logging ?? existing.programDetails.developmentTesting.logging,
        mudLogging: header?.mud_rmks ?? existing.programDetails.developmentTesting.mudLogging,
        sampling: header?.sampling ?? existing.programDetails.developmentTesting.sampling,
        flowTest: header?.flow_rmks ?? existing.programDetails.developmentTesting.flowTest,
      },
    },
    hydrogeology: hydroRow
      ? {
          aquifer: hydroRow.st_long_cd,
          estimatedTdsPpm: hydroRow.est_water_quality,
          h2sPpm: parseH2sPpm(hydroRow.h2s),
          // Not persisted anywhere in the wire contract yet — carry forward whatever was known.
          estimatedPumpDepthFt: existing.hydrogeology.estimatedPumpDepthFt,
        }
      : existing.hydrogeology,
    testProgram: {

/**
 * Carries the previously-assigned ir_seq (if any) so a repeat save UPDATEs instead of
 * re-INSERTing, plus (when known) the record as it was last loaded/saved — used to diff each
 * repeatable row's OWN action instead of inheriting one blanket action for the whole payload.
 * `existing` is intentionally distinguished from "omitted": omitting it (or passing `null`) falls
 * back to the old whole-payload action (every row shares `resolveRowAction(irSeq)`) — the only
 * safe choice when the caller has no snapshot to diff against, since guessing would otherwise
 * risk re-labeling an already-persisted row as a fresh INSERT (duplicate) or a still-current row
 * as DELETE (data loss).
 */
export interface WwellIrSaveContext {
  readonly irSeq: number | null;
  readonly existing?: Pick<
    IntegratedRequirement,
    'casing' | 'formationTops' | 'hydrogeology' | 'testProgram' | 'offsetWells'
  > | null;
}

const NEW_IR_CONTEXT: WwellIrSaveContext = { irSeq: null, existing: null };

function resolveRowAction(irSeq: number | null): IrRowAction {
  return irSeq === null ? 'INSERT' : 'UPDATE';
}

const isBlankCasingRow = (row: WwellIrRawCasingRow): boolean =>
  !row.type && !row.sizeIn && !row.remarks && row.depthFtBgl == null;

const isBlankFormationTop = (row: WwellIrRawFormationTop): boolean =>
  !row.name && !row.description && row.depthFt == null && row.subSeaFtAsl == null;

/** Natural keys the backend's UPDATE/DELETE branches match repeatable rows by (see
 *  EXAD_AGWA_IUD_PKG.save_casing/save_tops in 05-agwa-iud-package.sql) — a row whose key isn't
 *  already present in `existing` is genuinely new and must be sent as INSERT, not UPDATE, or the
 *  backend's `WHERE <key match>` finds nothing and silently drops it. */
const casingKey = (row: { readonly type: string; readonly sizeIn: string }): string => `${row.type} ${row.sizeIn}`;
const topKey = (row: { readonly name: string }): string => row.name;

/** Only the columns save_casing's/save_tops' UPDATE branch actually writes (type/size are the
 *  match key itself, not a writable column) — comparing anything else could flag a row "changed"
 *  when nothing the backend would touch actually differs. */
const casingRowUnchanged = (existingRow: IntegratedRequirementCasingRow, row: WwellIrRawCasingRow): boolean =>
  existingRow.depthFtBgl === (row.depthFtBgl ?? 0) && (existingRow.remarks ?? null) === (row.remarks || null);

const topRowUnchanged = (existingRow: IntegratedRequirementFormationTop, row: WwellIrRawFormationTop): boolean =>
  existingRow.depthFt === (row.depthFt ?? 0) && (existingRow.subSeaFtAsl ?? null) === (row.subSeaFtAsl ?? null);

function buildCasingRows(
  raw: readonly WwellIrRawCasingRow[],
  irSeq: number | null,
  existing: readonly IntegratedRequirementCasingRow[] | null | undefined,
): IrCasingRow[] {
  const currentRows = raw.filter(row => !isBlankCasingRow(row));

  if (existing == null) {
    const action = resolveRowAction(irSeq);
    return currentRows.map(row => ({
      action,
      ir_seq: irSeq,
      csg_size: row.sizeIn,
      csg_type: row.type,
      csg_depth: row.depthFtBgl ?? 0,
      csg_remarks: row.remarks || null,
    }));
  }

  const existingByKey = new Map(existing.map(row => [casingKey(row), row]));
  const keptKeys = new Set<string>();

  // A row this save didn't actually change is sent as '' (no-op), not a redundant UPDATE — see
  // IrRowAction's doc comment: the backend's per-row action switch then skips the DML entirely
  // instead of re-writing a row with the exact values it already has on every single save.
  const upserts: IrCasingRow[] = currentRows.map(row => {
    const key = casingKey(row);
    const existingRow = existingByKey.get(key);
    if (existingRow) keptKeys.add(key);
    const action: IrRowAction = existingRow ? (casingRowUnchanged(existingRow, row) ? '' : 'UPDATE') : 'INSERT';
    return {
      action,
      ir_seq: existingRow ? irSeq : null,
      csg_size: row.sizeIn,
      csg_type: row.type,
      csg_depth: row.depthFtBgl ?? 0,
      csg_remarks: row.remarks || null,
    };
  });

  const deletes: IrCasingRow[] = existing
    .filter(row => !keptKeys.has(casingKey(row)))
    .map(row => ({
      action: 'DELETE',
      ir_seq: irSeq,
      csg_size: row.sizeIn,
      csg_type: row.type,
      csg_depth: row.depthFtBgl,
      csg_remarks: row.remarks,
    }));

  return [...upserts, ...deletes];
}

 *  est_sta_water_lvl/est_productivity come from the form's (and domain model's) testProgram
 *  section, not hydrogeology, so both are needed to tell "unchanged" from "changed". */
const hydrogeologyRowUnchanged = (
  existingHydro: IntegratedRequirementHydrogeology,
  existingTestProgram: IntegratedRequirementTestProgram,
  raw: WwellIrRawHydrology,
): boolean =>
  existingTestProgram.estimatedWaterLevelFt === raw.estimatedWaterLevelFt &&
  existingHydro.estimatedTdsPpm === raw.estimatedTdsPpm &&
  existingTestProgram.estimatedRateGpm === raw.estimatedRateGpm &&
  existingHydro.h2sPpm === raw.h2sPpm;

/** Only ever 0 or 1 row (the form has a single hydrogeology section, not a repeatable list), keyed
 *  by aquifer (st_long_cd) the same way the backend matches it — so switching to a different
 *  aquifer must DELETE the old row, not silently orphan it under a key nothing will ever match again. */
function buildHydrogeologyRows(
  raw: WwellIrRawHydrology,
  irSeq: number | null,
  existing: Pick<IntegratedRequirement, 'hydrogeology' | 'testProgram'> | null | undefined,
): IrHydrogeologyRow[] {
  const currentAquifer = raw.aquifer || null;

  if (existing == null) {
    if (!currentAquifer) return [];
    const action = resolveRowAction(irSeq);
    return [
      {
        action,
        ir_seq: irSeq,
        st_long_cd: currentAquifer,
        est_sta_water_lvl: raw.estimatedWaterLevelFt,
        est_water_quality: raw.estimatedTdsPpm,
        est_productivity: raw.estimatedRateGpm,
        flowing: null,
        h2s: raw.h2sPpm != null ? `${raw.h2sPpm} ppm` : null,
      },
    ];
  }

  const existingAquifer = existing.hydrogeology.aquifer || null;
  const rows: IrHydrogeologyRow[] = [];

  if (existingAquifer && existingAquifer !== currentAquifer) {
    rows.push({
      action: 'DELETE',
      ir_seq: irSeq,
      st_long_cd: existingAquifer,
      est_sta_water_lvl: null,
      est_water_quality: null,
      est_productivity: null,
      flowing: null,
      h2s: null,
    });
  }

  if (currentAquifer) {
    const isExisting = existingAquifer === currentAquifer;
    const action: IrRowAction = isExisting
      ? hydrogeologyRowUnchanged(existing.hydrogeology, existing.testProgram, raw)
        ? ''
        : 'UPDATE'
      : 'INSERT';
    rows.push({
      action,
      ir_seq: isExisting ? irSeq : null,
      st_long_cd: currentAquifer,
      est_sta_water_lvl: raw.estimatedWaterLevelFt,
      est_water_quality: raw.estimatedTdsPpm,
      est_productivity: raw.estimatedRateGpm,
      flowing: null,
      h2s: raw.h2sPpm != null ? `${raw.h2sPpm} ppm` : null,
    });
  }

  return rows;
}

/** New row → INSERT. Changed persisted row → UPDATE. Removed persisted row → DELETE. Persisted
 *  row the user didn't touch this save → '' (no-op) — sent instead of a redundant UPDATE so the
 *  backend's per-row action switch (EXAD_AGWA_IUD_PKG.save_casing/save_tops/save_hydrogeology)
 *  simply doesn't match any branch and skips the DML, rather than re-writing a row with the exact
 *  values it already has on every save. */
export type IrRowAction = 'INSERT' | 'UPDATE' | 'DELETE' | '';

export type WwellIrTableName =
  | 'exad_gwd_ir_header'
  | 'exad_gwd_ir_casing'
  | 'exad_gwd_ir_hydrogeology'
  | 'exad_gwd_ir_tops'
  | 'exad_gwd_ir_water';

  readonly action: IrRowAction;
  /** null until the backend assigns one on first save; echo it back on every later save. */
  readonly ir_seq: number | null;

 * Remove once the backend returns a proper 2xx for a save that actually succeeded.
 */
function extractIrSeqFromMisreportedSuccess(err: unknown): number | null {
  if (!(err instanceof HttpErrorResponse)) return null;
  const body: unknown = err.error;
  const message = body && typeof body === 'object' ? (body as { message?: unknown }).message : undefined;
  if (typeof message !== 'string') return null;

  try {
    const parsed: unknown = JSON.parse(message);
    const status = parsed && typeof parsed === 'object' ? (parsed as { status?: unknown }).status : undefined;
    const irSeq = parsed && typeof parsed === 'object' ? (parsed as { ir_seq?: unknown }).ir_seq : undefined;
    return status === 'SUCCESS' && typeof irSeq === 'number' ? irSeq : null;
  } catch {
    return null;
  }
}

@Injectable({ providedIn: 'root' })
export class WwellIrService {
  private readonly http = inject(HttpClient);
  private readonly extConfigService = inject(ExternalConfigService);

  private get apiUrl(): string {
    const serverUrl = this.extConfigService.settings.planningServiceUrl ?? '';
    return `${serverUrl}/planning/api/v1`;
  }

  getIrs(): Observable<IntegratedRequirement[]> {
    return this.http
      .get<ApiResponse<IrHeaderSummary>>(`${this.apiUrl}/integrated-requirements`)
      .pipe(map(res => mapIrHeaderSummariesToIntegratedRequirements(res.data)));
  }

  /**
   * Wells still eligible to have an IR initiated (backed by exad_rcd_prewap via
   * exad_agwa_queries_pkg.get_ir_wells_prc) — feeds the "New IR" wizard's Request Reference
   * "Link to Request" picker. Returns the full record (not just {value,label}) so selecting one
   * can also auto-populate Water Well Name/Surface Elevation/Coordinates — see WwellIrStore's
   * `linkedRequestWellOptions` computed for the trimmed picker-option view derived from this.
   */
  getEligibleWellsForIr(): Observable<IrEligibleWell[]> {
    return this.http.get<ApiResponse<IrEligibleWell>>(`${this.apiUrl}/wells/eligible-for-ir`).pipe(map(res => res.data));
  }

  /**
   * ir_seq is the real primary key once an IR is saved — selecting a row in the list passes it
   * here to fetch the full { table, data } detail (casing/tops/hydrogeology) for that one record.
   * No offline-mock fallback here (unlike the other reads above): faking detail for a specific,
   * already-known-real record would misrepresent it rather than degrade gracefully, so a failure
   * surfaces as a real error instead — see WwellIrStore.loadIrDetail.
   */
  getIrDetailBlocks(irSeq: number): Observable<readonly WwellIrDetailTableBlock[]> {
    return this.http
      .get<ApiResponse<WwellIrDetailTableBlock>>(`${this.apiUrl}/integrated-requirements/${irSeq}`)
      .pipe(map(res => res.data));
  }

  /**
   * The approval-process API is a separate call from the detail fetch above and doesn't exist yet
   * — selecting a row fires both, this one just has nothing real to hit. Mocked client-side (with
   * an artificial delay so it reads as its own network round-trip) until a real
   * `GET .../integrated-requirements/{irSeq}/approval-timeline` endpoint replaces it.
   */
  getIrApprovalTimeline(irSeq: number): Observable<readonly ApprovalChainStep[]> {
    return of(mockApprovalChainForIrSeq(irSeq)).pipe(delay(300));
  }

  createIr(
    payload: CreateIntegratedRequirementPayload,
    irBulkPayload: WwellIrBulkSavePayload,
    attachmentNames: readonly string[] = [],
  ): Observable<IntegratedRequirement> {
    // The backend only accepts the raw exad_gwd_ir_* bulk-save array as the request body (see
    // wwell-ir-payload.model.ts / IntegratedRequirementResource.java's @Consumes(APPLICATION_JSON))
    // — it has no multipart handling and no columns for the richer domain fields below (client,
    // wellName, contacts, attachments...), so those stay client-side only until exad_agwa_iud_pkg's
    // header table grows to carry them. Sending them today just 415s (wrong Content-Type/body shape).
    //
    // No offline-fallback here (unlike getIrs()/getEligibleWellsForIr() above) — a failed save
    // must surface as a real, visible error (WwellIrStore.createIrError), not a fabricated
    // success. A genuine POST failure with no recoverable ir_seq (see extractIrSeqFromMisreportedSuccess
    // below for the one confirmed-defect exception) always re-throws rather than minting a random
    // ir_seq — a client-minted identity on real failure would let a resave silently re-INSERT under
    // a new fake id instead of UPDATE-ing the original row, orphaning the real backend record and
    // duplicating it in the list.
    //
    // SUPPRESS_GLOBAL_ERROR: WwellIrStore.createIr's own error handler (→ createIrError →
    // WwellIrComponent's notify.error) already shows a toast for a genuine failure — without this,
    // errorInterceptor's app-wide catchError would show a second, identical one for the same error.
    // The header row's own `ir_seq` is the identity this save was submitted under — `null` for a
    // brand-new IR (the backend mints one), otherwise the already-known ir_seq of the record being
    // updated. exad_agwa_iud_pkg.update_ir_data_prc (docs/ir/update_ir_data_prc.sql) only echoes
    // back a fresh ir_seq on that INSERT branch; an UPDATE's `p_return_value` is just the literal
    // `SUCCESS`, no ir_seq, because the caller already has it — see WwellIrSaveResult. Falling back
    // to `knownIrSeq` (not a random mint) below is what keeps a resaved record's identity stable.
    const knownIrSeq = irBulkPayload[0].data?.[0]?.ir_seq ?? null;

    return this.http
      .post<ApiResponse<WwellIrSaveResult>>(`${this.apiUrl}/integrated-requirements`, irBulkPayload, {
        context: new HttpContext().set(SUPPRESS_GLOBAL_ERROR, true),
      })
      .pipe(
        map(res => this.buildIrFromPayload(payload, res.data[0]?.irSeq ?? knownIrSeq, attachmentNames)),
        catchError((err: unknown) => {
          const realIrSeq = extractIrSeqFromMisreportedSuccess(err);
          return realIrSeq !== null ? of(this.buildIrFromPayload(payload, realIrSeq, attachmentNames)) : throwError(() => err);
        }),
      );
  }

  submitIrForReview(ir: IntegratedRequirement): Observable<IntegratedRequirement> {
    return this.http
      .post<ApiResponse<IntegratedRequirement>>(`${this.apiUrl}/integrated-requirements/${ir.id}/submit-for-review`, {})
      .pipe(
        map(res => res.data[0]),
        // TODO: remove fallback once the submit-for-review endpoint is live.
        catchError(() => of({ ...ir, status: 'submit-for-review' as const, updatedDate: todayIsoDate() })),
      );
  }

  /** Builds the domain object locally from what was submitted — the backend only ever echoes
   *  back an ir_seq (see WwellIrSaveResult), never the full IntegratedRequirement. `irSeq` is
   *  already fully resolved by the caller before this runs (the response's own value on INSERT,
   *  the already-known `knownIrSeq` on UPDATE — see createIr() above), so it's used as-is: no
   *  client-minted placeholder, since a fake identity here is exactly what caused the resave to
   *  re-INSERT under a new fake id instead of UPDATE-ing the original row. */
  private buildIrFromPayload(
    payload: CreateIntegratedRequirementPayload,
    irSeq: number | null,
    attachmentNames: readonly string[],
  ): IntegratedRequirement {
    // ir_seq-derived, matching mapIrHeaderSummaryToIntegratedRequirement's format — this same
    // record must display the same code whether it was just created or reloaded from the list.
    const code = `IR-${irSeq}`;
    return buildIntegratedRequirementFromPayload(payload, { id: code, code, irSeq }, attachmentNames);
  }
}

import {

import { Router, RouterOutlet, NavigationEnd } from '@angular/router';
import { filter } from 'rxjs';
import { TopNavbarComponent } from './header/top-navbar.component';
import { LeftSidenavComponent } from './sidebar/left-sidenav.component';
import { LoaderService } from '@shared/components/global-loader/loader.service';

@Component({
  selector: 'app-layout',
  standalone: true,
  imports: [TopNavbarComponent, LeftSidenavComponent, RouterOutlet],
  templateUrl: './layout.component.html',
  styleUrl: './layout.component.scss',
  changeDetection: ChangeDetectionStrategy.OnPush,
})
export class LayoutComponent {
  private  readonly router           = inject(Router);
  private  readonly loader           = inject(LoaderService);
  protected readonly sidenavCollapsed = signal<boolean>(false);
  protected readonly activeRoute      = signal<string>('home');

  private static readonly AUTO_COLLAPSE_ROUTES = new Set([
    'water-wells-overview',
    'integrated-requirements',
  ]);

  constructor() {
    const initialSegment = this.routeSegment(this.router.url);
    this.activeRoute.set(initialSegment);
    if (LayoutComponent.AUTO_COLLAPSE_ROUTES.has(initialSegment)) {
      this.sidenavCollapsed.set(true);
    }

    // Keep active nav item in sync with browser URL
    this.router.events
      .pipe(filter((e): e is NavigationEnd => e instanceof NavigationEnd))
      .subscribe((e) => {
        const segment = this.routeSegment(e.urlAfterRedirects);
        this.activeRoute.set(segment);
        if (LayoutComponent.AUTO_COLLAPSE_ROUTES.has(segment)) {
          this.sidenavCollapsed.set(true);
        }
      });

    // Fire after the first full render cycle — child ngOnInit calls (which register
    // boot tasks) have already run by this point, so completeBootIfReady() sees the
    // correct task count before deciding whether to close the native bar.
    afterNextRender(() => this.loader.markAppShellReady());
  }

  protected toggleSidenav(): void {
    this.sidenavCollapsed.update((v) => !v);
  }

  protected onNavItemClick(id: string): void {
    this.activeRoute.set(id);
    this.router.navigate(['main', id]);
  }

  private routeSegment(url: string): string {
    return url.split('?')[0].split('/').filter(Boolean)[1] ?? 'home';
  }
}
