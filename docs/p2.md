import * as THREE from 'three';

import { ANIM, DiagramLayout, WellboreDiagramData } from '@models/well-design/wellbore-diagram.model';
import { computeCasingHalfWidth } from '@shared/utils/wellbore-math.util';
import {
  resolveCompletionBottomDepth,
  resolveCompletionTopDepth,
  resolveOhHalfWidth,
  shouldRenderGravelPack,
  shouldRenderOpenHole,
} from '../../renderers/wellbore-depth.helpers';
import { buildPipeMesh } from './casing-3d.builder';
import { createGravelMaterial, createRockMaterial, createWaterMaterial } from '../utils/materials.factory';
import { createLabelSprite } from '../utils/label-sprite.util';
import { Depth3DScale, Reveal3DTarget } from '../wellbore-3d.types';

export interface CompletionBuildResult {
  readonly group: THREE.Group;
  readonly reveals: Reveal3DTarget[];
  readonly waterRipple: THREE.Mesh | null;
  readonly drillBit: THREE.Mesh;
  readonly drillString: THREE.Mesh;
  readonly drillBitTargetY: number;
  readonly totalDepthLabel: THREE.Sprite;
  readonly totalDepthY: number;
}

/** Red→amber→green, matching the 2D mud-circulation gradient used on the drill-string arrow. */
function circulationColor(pct: number): string {
  const clamped = Math.max(0, Math.min(100, pct));
  if (clamped === 0) return '#e53e3e';
  if (clamped === 100) return '#16a34a';
  const hue = 30 + ((clamped - 1) / 98) * 25;
  return `hsl(${hue.toFixed(1)}, 90%, 50%)`;
}

function resolveWaterLevelDepth(data: WellboreDiagramData): number | null {
  const wd = data.wellDesign;
  if (!wd) return null;
  const rawTxt = wd.swLvlTxt ?? '';
  const isFlowing = /flow/i.test(rawTxt);
  if (isFlowing) {
    const match = /SIWHP[:\s]*(-?\d+(?:\.\d+)?)/i.exec(rawTxt) ?? /(-?\d+(?:\.\d+)?)\s*psi/i.exec(rawTxt);
    const psi = match ? Number(match[1]) : null;
    return psi !== null ? -(psi * 2.37) : 0;
  }
  return wd.staWaterLvl || null;
}

function buildOpenHoleAndScreen(
  data: WellboreDiagramData,
  layout: DiagramLayout,
  depthScale: Depth3DScale,
  ohStart: number,
  group: THREE.Group,
  reveals: Reveal3DTarget[],
): void {
  if (!data.casings.length || !shouldRenderOpenHole(data.wellDesign)) return;

  const ohHW = resolveOhHalfWidth(data, layout);
  const topDepth = resolveCompletionTopDepth(data.casings, data.drlgCasings);
  const bottomDepth = resolveCompletionBottomDepth(data);
  const topY = depthScale.toY(topDepth);
  const bottomY = depthScale.toY(bottomDepth);

  const material = createRockMaterial();
  const mesh = buildPipeMesh(depthScale.toRadius(ohHW), topY, bottomY, material);
  mesh.name = 'open-hole';
  group.add(mesh);

  const plane = new THREE.Plane(new THREE.Vector3(0, 1, 0), -topY);
  material.clippingPlanes = [plane];
  reveals.push({ mesh, clipPlane: plane, topY, bottomY, delay: ohStart, duration: ANIM.OH_DURATION / 1000 });

  const label = createLabelSprite({
    text: data.wellDesign?.ohRemarks ?? `${ohHW.toFixed(0)}px Open Hole`,
    color: '#334155',
    fontSize: 26,
  });
  label.position.set(depthScale.toRadius(ohHW) + 1.4, topY + (bottomY - topY) * 0.2, 0);
  group.add(label);
}

function buildGravelPack(
  data: WellboreDiagramData,
  layout: DiagramLayout,
  depthScale: Depth3DScale,
  gravelStart: number,
  group: THREE.Group,
  reveals: Reveal3DTarget[],
): void {
  const wellDesign = data.wellDesign;
  if (!wellDesign || !shouldRenderGravelPack(wellDesign) || !data.casings.length) return;

  const { baseHalfWidth, halfWidthIncrement, linerScreenInset, gpScreenGap, gpFillRatio } = layout;
  const innerHW = computeCasingHalfWidth(0, baseHalfWidth, halfWidthIncrement);
  const ohHW = resolveOhHalfWidth(data, layout);
  const lsHW = Math.min(innerHW, ohHW) - linerScreenInset;
  const gpMaxOuter = ohHW - gpScreenGap;
  const annulusWidth = Math.max(0, (gpMaxOuter - lsHW) * gpFillRatio);
  const innerR = depthScale.toRadius(lsHW);
  const outerR = depthScale.toRadius(lsHW + annulusWidth);
  if (outerR <= innerR) return;

  const topDepth = resolveCompletionTopDepth(data.casings, data.drlgCasings);
  const bottomDepth = resolveCompletionBottomDepth(data);
  const topY = depthScale.toY(topDepth);
  const bottomY = depthScale.toY(bottomDepth);
  const height = Math.max(0.05, topY - bottomY);

  const shape = new THREE.Shape();
  shape.absarc(0, 0, outerR, 0, Math.PI * 2, false);
  const hole = new THREE.Path();
  hole.absarc(0, 0, innerR, 0, Math.PI * 2, true);
  shape.holes.push(hole);

  const geometry = new THREE.ExtrudeGeometry(shape, { depth: height, bevelEnabled: false, curveSegments: 48 });
  geometry.rotateX(-Math.PI / 2);
  geometry.translate(0, 0, 0);

  const material = createGravelMaterial();
  const mesh = new THREE.Mesh(geometry, material);
  mesh.position.y = bottomY;
  mesh.name = 'gravel-pack';
  group.add(mesh);

  const plane = new THREE.Plane(new THREE.Vector3(0, 1, 0), -topY);
  material.clippingPlanes = [plane];
  reveals.push({ mesh, clipPlane: plane, topY, bottomY, delay: gravelStart, duration: ANIM.GRAVEL_DURATION / 1000 });
}

function buildWaterLevel(
  data: WellboreDiagramData,
  layout: DiagramLayout,
  depthScale: Depth3DScale,
  group: THREE.Group,
): THREE.Mesh | null {
  const waterDepth = resolveWaterLevelDepth(data);
  if (waterDepth === null) return null;

  const widestTier = data.casings.length > 0 ? data.casings.length - 1 : 0;
  const innerHW = computeCasingHalfWidth(widestTier, layout.baseHalfWidth, layout.halfWidthIncrement);
  const radius = depthScale.toRadius(innerHW) + 0.6;
  const y = depthScale.toY(waterDepth);

  const disc = new THREE.Mesh(
    new THREE.CircleGeometry(radius, 48),
    createWaterMaterial(),
  );
  disc.rotation.x = -Math.PI / 2;
  disc.position.y = y;
  disc.name = 'water-level';
  group.add(disc);

  const ripple = new THREE.Mesh(
    new THREE.RingGeometry(radius * 0.2, radius * 0.28, 48),
    new THREE.MeshBasicMaterial({ color: '#0ea5e9', transparent: true, opacity: 0.85, side: THREE.DoubleSide }),
  );
  ripple.rotation.x = -Math.PI / 2;
  ripple.position.y = y + 0.02;
  ripple.name = 'water-ripple';
  group.add(ripple);

  const label = createLabelSprite({ text: `Water Level: ${waterDepth.toLocaleString()} ft`, color: '#0ea5e9', fontSize: 26 });
  label.position.set(-(radius + 1.6), y + 0.3, 0);
  group.add(label);

  return ripple;
}

function buildDrillString(
  data: WellboreDiagramData,
  depthScale: Depth3DScale,
  group: THREE.Group,
): { bit: THREE.Mesh; string: THREE.Mesh; targetY: number } {
  const currentDepth = Math.max(0, Math.min(data.currentDepth, data.totalDepth));
  const targetY = depthScale.toY(currentDepth);
  const stringHeight = Math.max(0.001, -targetY);
  const lastPct = data.mudCirculation.length > 0 ? data.mudCirculation[data.mudCirculation.length - 1].pct : 100;

  // Geometry is pre-translated so the mesh's local origin sits at its TOP edge (surface) —
  // animating `scale.y` from 0 → 1 then grows the string straight down from the rig floor,
  // instead of scaling outward from a centered pivot.
  const stringGeom = new THREE.CylinderGeometry(0.18, 0.18, stringHeight, 16);
  stringGeom.translate(0, -stringHeight / 2, 0);
  const stringMat = new THREE.MeshStandardMaterial({ color: circulationColor(lastPct), metalness: 0.6, roughness: 0.4 });
  const stringMesh = new THREE.Mesh(stringGeom, stringMat);
  stringMesh.position.y = 0;
  stringMesh.scale.y = 0;
  stringMesh.name = 'drill-string';
  group.add(stringMesh);

  const bitMat = new THREE.MeshStandardMaterial({ color: '#facc15', metalness: 0.8, roughness: 0.3 });
  const bit = new THREE.Mesh(new THREE.ConeGeometry(0.55, 1.1, 24), bitMat);
  bit.name = 'drill-bit';
  bit.position.y = 0; // rig-floor rest position; the reveal timeline drives it down to targetY
  group.add(bit);

  return { bit, string: stringMesh, targetY };
}

function buildPumpMarker(data: WellboreDiagramData, depthScale: Depth3DScale, group: THREE.Group): void {
  const pumpLvl = data.wellDesign?.pumpLvl;
  if (pumpLvl == null) return;
  const label = createLabelSprite({ text: `Pump Set @ ${pumpLvl.toLocaleString()} ft`, color: '#1e293b', fontSize: 26 });
  label.position.set(0, depthScale.toY(pumpLvl), 0.01);
  group.add(label);
}

export function buildCompletion3D(
  data: WellboreDiagramData,
  layout: DiagramLayout,
  depthScale: Depth3DScale,
  timing: { ohStart: number; gravelStart: number },
): CompletionBuildResult {
  const group = new THREE.Group();
  group.name = 'completion';
  const reveals: Reveal3DTarget[] = [];

  buildOpenHoleAndScreen(data, layout, depthScale, timing.ohStart, group, reveals);
  buildGravelPack(data, layout, depthScale, timing.gravelStart, group, reveals);
  const waterRipple = buildWaterLevel(data, layout, depthScale, group);
  const { bit, string: drillString, targetY } = buildDrillString(data, depthScale, group);
  buildPumpMarker(data, depthScale, group);

  const isBeyondTarget = data.targetDepth > 0 && data.currentDepth > data.targetDepth;
  const labelDepth = isBeyondTarget ? data.currentDepth : data.totalDepth;
  const totalDepthY = depthScale.toY(resolveCompletionBottomDepth(data));
  const totalDepthLabel = createLabelSprite({
    text: `${isBeyondTarget ? 'CURRENT DEPTH' : 'TOTAL DEPTH'}: ${labelDepth.toLocaleString()} FT`,
    color: isBeyondTarget ? '#16a34a' : '#2563eb',
    fontSize: 30,
    fontWeight: '900',
  });
  totalDepthLabel.position.set(0, totalDepthY - 2, 0);
  group.add(totalDepthLabel);

  return {
    group,
    reveals,
    waterRipple,
    drillBit: bit,
    drillString,
    drillBitTargetY: targetY,
    totalDepthLabel,
    totalDepthY,
  };
}

import { pointer } from 'd3-selection';
import { easeCubicInOut } from 'd3-ease';
import 'd3-transition';

import { ANIM, DiagramLayout } from '@models/well-design/wellbore-diagram.model';
import { ICasingIR } from '@shared/models/wwell/casing-ir.model';
import { IDrlgCsg } from '@shared/models/wwell/drlg-csg.model';
import { IWellDesign } from '@shared/models/wwell/well-design.model';
import {
    buildArrowHeadRight,
    buildCasingPath,
    buildGravelPackUPath,
    casingGradientId,
    computeCasingHalfWidth,
    openHoleHalfWidth,
} from '@shared/utils/wellbore-math.util';
import { animateLabel } from './animation.helpers';
import { getCasingTier, resolveCompletionTopDepth } from './wellbore-depth.helpers';
import { resolveLiners, renderLiners, LinerEntry } from './liner.renderer';
import { DefsSel, DepthScale, GSel, ResolvedAnimation } from './wellbore-renderer.types';

const CSG_TOOLTIP_GAP = 18;
const CSG_TOOLTIP_W = 260;
const CSG_TOOLTIP_FONT = 'DM Sans, sans-serif';

export interface RenderCasingElementsOptions {
    readonly rootG: GSel;
    readonly defsEl: DefsSel;
    readonly layout: DiagramLayout;
    readonly casings: ICasingIR[];
    readonly drlgCasings: IDrlgCsg[];
    readonly centerX: number;
    readonly scale: DepthScale;
    readonly casingsStart: number;
    readonly linersStart: number;
    readonly gravelStart: number;
    readonly labelsStart: number;
    readonly openHoleHalfWidthPx: number;
    readonly animation: ResolvedAnimation;
    readonly wellDesign?: IWellDesign | null;
}

export function renderCasingElements(options: RenderCasingElementsOptions): void {
    const { casings, drlgCasings, layout, wellDesign, rootG, defsEl, centerX, scale, linersStart, labelsStart, animation } = options;
    const liners = resolveLiners(casings, drlgCasings, layout, wellDesign);
    const lastCasingBotDepth = resolveLastCasingBotDepth(casings, drlgCasings);

    renderCasings(options);
    renderLiners({ rootG, defsEl, layout, liners, centerX, scale, linersStart, labelsStart, animation, lastCasingBotDepth });
    renderCasingLabels(options);
    addCasingLinerTooltips(options, liners);
}

function resolveLastCasingBotDepth(casings: ICasingIR[], drlgCasings: IDrlgCsg[]): number {
    // Find the deepest non-liner casing shoe — that is the 13 3/8" (or innermost casing) shoe
    if (casings.length === 0) {
        // drlgCasings only: wCsgBotDpth is the parent casing shoe; pick the shallowest
        // non-zero value that represents the innermost casing (smallest wCsgBotDpth among actual casing records)
        const casingOnly = drlgCasings.filter(d => !d.wLnrOdSz && d.wCsgBotDpth > 0);
        if (casingOnly.length > 0) {
            return Math.max(...casingOnly.map(d => d.wCsgBotDpth));
        }
        // fallback: deepest wCsgBotDpth across all records
        return Math.max(...drlgCasings.map(d => d.wCsgBotDpth).filter(v => v > 0), 0);
    }

    const nonLiners = casings.filter(c => c.csgType !== 'Liner');
    if (nonLiners.length === 0) return 0;

    // Pick the non-liner casing with the deepest shoe (largest depth value)
    const deepest = nonLiners.reduce((max, c) => c.csgDepth > max.csgDepth ? c : max, nonLiners[0]);
    const actual = drlgCasings.find(d => d.wCsgOdSz === deepest.csgSize);
    return (actual?.wCsgBotDpth ?? 0) > 0 ? actual!.wCsgBotDpth : deepest.csgDepth;
}

function renderCasings({
    rootG,
    defsEl,
    layout,
    casings,
    drlgCasings,
    centerX,
    scale,
    casingsStart,
    gravelStart,
    openHoleHalfWidthPx,
}: RenderCasingElementsOptions): void {
    if (casings.length === 0) {
        renderCasingsFromActual({ rootG, defsEl, layout, drlgCasings, centerX, scale, casingsStart });
        return;
    }

    const { baseHalfWidth, halfWidthIncrement, shoeCurveOffset } = layout;
    const completionTopPx = scale(resolveCompletionTopDepth(casings, drlgCasings));

    casings.forEach((csg, i) => {
        if (csg.csgType === 'Liner') return;
        const tier = getCasingTier(csg, casings);
        const hw = computeCasingHalfWidth(tier, baseHalfWidth, halfWidthIncrement);
        const isConductor = csg.csgType.toLowerCase() === 'conductor';
        const isGP = csg.csgType === 'Gravel Pack';
        const actualCsg = drlgCasings.find(d => d.wCsgOdSz === csg.csgSize) ?? null;
        const hasActual = actualCsg !== null && actualCsg.wCsgBotDpth > 0;
        const displayDepth = hasActual ? actualCsg!.wCsgBotDpth : csg.csgDepth;
        const bottomPx = scale(displayDepth);
        const clipId = `dyn-casing-clip-${i}`;
        const animOrder = casings.length - 1 - i;
        const delay = animOrder * ANIM.CASING_STAGGER;
        const gpInnerHW = computeCasingHalfWidth(0, layout.baseHalfWidth, layout.halfWidthIncrement);
        const gpLsHW = Math.min(gpInnerHW, openHoleHalfWidthPx) - layout.linerScreenInset;
        const gpMaxOuter = openHoleHalfWidthPx - layout.gpScreenGap;
        const gpAnnulusWidth = Math.max(0, (gpMaxOuter - gpLsHW) * layout.gpFillRatio);
        const gpOuterEdge = gpLsHW + gpAnnulusWidth;
        const gpAnnulusCenter = gpLsHW + gpAnnulusWidth / 2;
        const clipRadius = isGP ? gpOuterEdge : openHoleHalfWidth(hw) + 20;

        const clipRect = defsEl
            .append('clipPath')
            .attr('class', 'dyn-clip')
            .attr('id', clipId)
            .append('rect')
            .attr('x', centerX - clipRadius - (isGP ? 0 : 15))
            .attr('y', isGP ? completionTopPx : -5)
            .attr('width', clipRadius * 2 + (isGP ? 0 : 30))
            .attr('height', 0);

        if (isGP) {
            const gravelBottomPx = scale(csg.csgDepth);
            rootG.append('path')
                .attr('class', 'gravelHole')
                .attr('d', buildGravelPackUPath(centerX, gpAnnulusCenter, completionTopPx, gravelBottomPx))
                .attr('stroke', 'url(#gravelpattern)')
                .attr('stroke-width', String(gpAnnulusWidth))
                .style('fill', 'none')
                .attr('clip-path', `url(#${clipId})`);
            clipRect.transition()
                .delay(gravelStart)
                .duration(ANIM.GRAVEL_DURATION)
                .ease(easeCubicInOut)
                .attr('height', gravelBottomPx - completionTopPx);
            return;
        }

        const casingD = buildCasingPath(centerX, hw, 0, bottomPx, shoeCurveOffset);

        if (hasActual) {
            rootG.append('path')
                .attr('class', 'casing')
                .attr('d', casingD)
                .attr('clip-path', `url(#${clipId})`)
                .attr('fill', 'url(#actualCasingGradient)')
                .attr('stroke', '#000000')
                .attr('stroke-width', '5')
                .style('opacity', 0.78);
        } else {
            const plannedFillGrad = isConductor
                ? 'url(#mainGradient)'
                : `url(#${casingGradientId(csg.csgType)})`;
            rootG.append('path')
                .attr('class', 'casing')
                .attr('d', casingD)
                .attr('clip-path', `url(#${clipId})`)
                .attr('fill', plannedFillGrad)
                .attr('stroke', '#000000')
                .attr('stroke-width', '5')
                .style('opacity', isConductor ? 0.75 : 0.40);
        }

        clipRect.transition()
            .delay(casingsStart + delay)
            .duration(ANIM.CASING_DURATION)
            .ease(easeCubicInOut)
            .attr('height', bottomPx + shoeCurveOffset + 20);
    });
}

function uniqueCasingsByOd(drlgCasings: IDrlgCsg[]): IDrlgCsg[] {
    const map = new Map<string, IDrlgCsg>();
    for (const d of drlgCasings) {
        if (d.wCsgBotDpth <= 0) continue;
        const existing = map.get(d.wCsgOdSz);
        if (!existing || d.wCsgBotDpth > existing.wCsgBotDpth) {
            map.set(d.wCsgOdSz, d);
        }
    }
    return [...map.values()].sort((a, b) => b.wCsgBotDpth - a.wCsgBotDpth);
}

function renderCasingsFromActual({
    rootG,
    defsEl,
    layout,
    drlgCasings,
    centerX,
    scale,
    casingsStart,
}: Pick<RenderCasingElementsOptions, 'rootG' | 'defsEl' | 'layout' | 'drlgCasings' | 'centerX' | 'scale' | 'casingsStart'>): void {
    const { baseHalfWidth, halfWidthIncrement, shoeCurveOffset } = layout;
    const unique = uniqueCasingsByOd(drlgCasings);

    unique.forEach((actual, i) => {
        const hw = computeCasingHalfWidth(i, baseHalfWidth, halfWidthIncrement);
        const animOrder = unique.length - 1 - i;
        const delay = animOrder * ANIM.CASING_STAGGER;
        const bottomPx = scale(actual.wCsgBotDpth);
        const clipId = `dyn-casing-clip-actual-${i}`;
        const clipRadius = openHoleHalfWidth(hw) + 20;
        const clipRect = defsEl.append('clipPath')
            .attr('class', 'dyn-clip').attr('id', clipId).append('rect')
            .attr('x', centerX - clipRadius - 15).attr('y', -5)
            .attr('width', clipRadius * 2 + 30).attr('height', 0);

        rootG.append('path')
            .attr('class', 'casing')
            .attr('d', buildCasingPath(centerX, hw, 0, bottomPx, shoeCurveOffset))
            .attr('clip-path', `url(#${clipId})`)
            .attr('fill', 'url(#actualCasingGradient)')
            .attr('stroke', '#000000')
            .attr('stroke-width', '2')
            .style('opacity', 0.78);

        clipRect.transition()
            .delay(casingsStart + delay).duration(ANIM.CASING_DURATION).ease(easeCubicInOut)
            .attr('height', bottomPx + shoeCurveOffset + 20);
    });
}

function renderCasingLabels({
    rootG,
    layout,
    casings,
    drlgCasings,
    centerX,
    scale,
    labelsStart,
    openHoleHalfWidthPx,
    animation,
}: RenderCasingElementsOptions): void {
    if (casings.length === 0) {
        renderCasingLabelsFromActual({ rootG, layout, drlgCasings, centerX, scale, labelsStart, animation });
        return;
    }

    const { baseHalfWidth, halfWidthIncrement, linerScreenInset, gpScreenGap, gpFillRatio } = layout;
    const gpLabelInnerHW = computeCasingHalfWidth(0, baseHalfWidth, halfWidthIncrement);
    const gpLabelLsHW = Math.min(gpLabelInnerHW, openHoleHalfWidthPx) - linerScreenInset;
    const gpLabelMaxOuter = openHoleHalfWidthPx - gpScreenGap;
    const gpLabelAnnulus = Math.max(0, (gpLabelMaxOuter - gpLabelLsHW) * gpFillRatio);
    const gpLabelOuterEdge = gpLabelLsHW + gpLabelAnnulus;

    const sortedShoeDepths = casings
        .filter(c => c.csgType !== 'Liner' && c.csgType !== 'Gravel Pack')
        .map(c => {
            const act = drlgCasings.find(d => d.wCsgOdSz === c.csgSize) ?? null;
            return (act !== null && act.wCsgBotDpth > 0) ? act.wCsgBotDpth : c.csgDepth;
        })
        .sort((a, b) => a - b);

    casings.forEach((csg) => {
        if (csg.csgType === 'Liner') return;
        const tier = getCasingTier(csg, casings);
        const hw = computeCasingHalfWidth(tier, baseHalfWidth, halfWidthIncrement);
        const actualCsg = drlgCasings.find(d => d.wCsgOdSz === csg.csgSize) ?? null;
        const hasActual = actualCsg !== null && actualCsg.wCsgBotDpth > 0;
        const displayDepth = hasActual ? actualCsg!.wCsgBotDpth : csg.csgDepth;
        const displaySize = hasActual ? actualCsg!.wCsgOdSz : csg.csgSize;
        const shoePx = scale(displayDepth);
        const labelYPx = csg.csgType === 'Gravel Pack' ? scale(csg.csgDepth) : shoePx;
        const rEdge = csg.csgType === 'Gravel Pack'
            ? centerX + gpLabelOuterEdge
            : centerX + hw;
        const lEnd = rEdge + 26;
        const labelX = lEnd + 8;
        const lg = rootG.append('g').attr('class', 'casing-label');

        lg.append('line')
            .attr('class', 'casing-label__line')
            .attr('x1', rEdge).attr('x2', lEnd)
            .attr('y1', labelYPx).attr('y2', labelYPx);
        lg.append('path')
            .attr('class', 'casing-label__arrow')
            .attr('d', buildArrowHeadRight(lEnd, labelYPx, 6));

        if (csg.csgType === 'Gravel Pack') {
            lg.append('text')
                .attr('class', 'casing-label__primary')
                .attr('x', labelX).attr('y', labelYPx + 4)
                .text(csg.csgRemarks || 'Gravel Pack');
        } else {
            const primaryEl = lg.append('text')
                .attr('class', 'casing-label__primary')
                .attr('x', labelX).attr('y', shoePx + 4);
            if (hasActual) {
                primaryEl
                    .attr('fill', '#22c55e')
                    .text(`${displaySize}" ${csg.csgType} @ ${displayDepth.toLocaleString()} *`);
            } else {
                primaryEl.text(`${csg.csgSize}" ${csg.csgType} @ ${csg.csgDepth.toLocaleString()}`);
                if (csg.csgRemarks) {
                    lg.append('text')
                        .attr('class', 'casing-label__secondary')
                        .attr('x', labelX).attr('y', shoePx + 24)
                        .text(`(${csg.csgRemarks})`);
                }
            }

            const innerDelay = labelsStart + (animation.lineStyle === 'draw' ? animation.lineDur : 0);
            const rank = sortedShoeDepths.indexOf(displayDepth);
            const topDepth = rank > 0 ? sortedShoeDepths[rank - 1] : 0;
            const innerMidY = scale((topDepth + displayDepth) / 2);
            rootG.append('text')
                .attr('class', 'casing-size-inner')
                .attr('x', centerX).attr('y', innerMidY)
                .attr('text-anchor', 'middle')
                .attr('dominant-baseline', 'central')

function buildCasingTooltipController(
    tooltipLayer: GSel,
    layout: DiagramLayout,
    centerX: number,

    interface Row { label: string; value: string; hilite?: boolean; }

    let isVisible = false;
    let lastSide: 'left' | 'right' | null = null;

    const hide = (): void => {
        isVisible = false;
        lastSide = null;
        tooltipLayer.interrupt().transition().duration(120).style('opacity', 0);
    };

    const renderCard = (
        title: string,
        badge: string,
        accent: string,
        rows: readonly Row[],
        hw: number,
        mouseX: number,
        mouseY: number,
    ): void => {
        const ROW_H = 28;
        const tooltipH = 60 + rows.length * ROW_H + 16;
        const tooltipW = CSG_TOOLTIP_W;
        const onRight = mouseX > centerX;
        const side: 'left' | 'right' = onRight ? 'right' : 'left';
        const casingEdgeX = onRight ? centerX + hw : centerX - hw;
        const mutedAccent = accent === '#16a34a' ? 'rgba(22,163,74,0.14)' : 'rgba(37,99,235,0.14)';

        const rawTooltipX = onRight
            ? casingEdgeX + CSG_TOOLTIP_GAP

        const tooltipX = Math.max(8, Math.min(rawTooltipX, layout.wellboreViewWidth - tooltipW - 8));
        const tooltipY = Math.max(4, Math.min(mouseY - tooltipH / 2, layout.drawingHeight - tooltipH - 4));
        const connY = Math.min(Math.max(mouseY, tooltipY + 18), tooltipY + tooltipH - 18);

        tooltipLayer.selectAll('*').remove();
        tooltipLayer.raise();

        // Connector: elbow from casing edge to tooltip connector-side edge
        const connTipX = onRight ? tooltipX : tooltipX + tooltipW;
        const connMidX = onRight ? tooltipX - 8 : tooltipX + tooltipW + 8;
        tooltipLayer.append('path')
            .attr('d', `M${casingEdgeX},${mouseY} H${connMidX} Q${connTipX},${mouseY} ${connTipX},${connY} H${connTipX}`)
            .attr('fill', 'none').attr('stroke', accent)
            .attr('stroke-width', 1.4).attr('stroke-dasharray', '4,3')
            .attr('stroke-linecap', 'round').attr('opacity', 0.65);

        // Anchor dot on casing edge
        tooltipLayer.append('circle')
            .attr('cx', casingEdgeX).attr('cy', mouseY)
            .attr('r', 3.2).attr('fill', '#fff')
            .attr('stroke', accent).attr('stroke-width', 1.6);

        // Card background
        tooltipLayer.append('rect')
            .attr('x', tooltipX).attr('y', tooltipY)
            .attr('width', tooltipW).attr('height', tooltipH)
            .attr('rx', 8).attr('fill', '#ffffff')
            .attr('stroke', '#e2e8f0').attr('stroke-width', 1.2)
            .attr('filter', 'drop-shadow(0 8px 16px rgba(15,23,42,0.18))');

        // Accent rail on the connector side of the card
        const railX = onRight ? tooltipX : tooltipX + tooltipW - 5;
        tooltipLayer.append('rect')
            .attr('x', railX).attr('y', tooltipY)
            .attr('width', 5).attr('height', tooltipH)
            .attr('rx', 2).attr('fill', accent);

        // Badge
        const badgeW = badge.length * 8.5 + 16;
        const badgeX = tooltipX + 12;
        tooltipLayer.append('rect')
            .attr('x', badgeX).attr('y', tooltipY + 11)
            .attr('width', badgeW).attr('height', 22)
            .attr('rx', 11).attr('fill', mutedAccent);
        tooltipLayer.append('text')

        tooltipLayer.append('text')

            .attr('text-anchor', 'end')
            .attr('font-size', '17').attr('font-family', CSG_TOOLTIP_FONT)
            .attr('font-weight', '800').attr('fill', '#0f172a').text(title);

        // Divider
        tooltipLayer.append('line')

            .attr('y1', tooltipY + 44).attr('y2', tooltipY + 44)
            .attr('stroke', '#e2e8f0').attr('stroke-width', 1);

        });

        const sideChanged = side !== lastSide;
        lastSide = side;
        if (!isVisible || sideChanged) {
            tooltipLayer.interrupt().style('opacity', 0).transition().duration(150).style('opacity', 1);
            isVisible = true;
        } else {
            tooltipLayer.style('opacity', 1);

    };

        actual: IDrlgCsg | null,
        hw: number,

        const typeName = csg?.csgType ?? 'Casing';
        const typeLabel = typeName.toLowerCase() === 'conductor' ? 'CONDUCTOR' : typeName.toUpperCase();

            if (csg?.csgRemarks) rows.push({ label: 'Remark', value: csg.csgRemarks });

        } else if (actual) {
            rows.push({ label: 'actual bottom depth', value: `${actual.wCsgBotDpth.toLocaleString()} ft`, hilite: true });
        }

        renderCard(`${size}" ${typeName}`, badge, accent, rows, hw, mouseX, mouseY);
    };

    const showForLiner = (liner: LinerEntry, mouseX: number, mouseY: number): void => {
        const length = liner.botDepth - liner.topDepth;

            { label: 'Bottom Depth', value: `${liner.botDepth.toLocaleString()} ft`, hilite: liner.hasActual },
            { label: 'Length', value: `${length.toLocaleString()} ft` },
        ];

    };

    return { showForCasing, showForLiner, hide };
}

export function renderScreenHanger({
  rootG,
  layout,
  data,
  centerX,
  scale,
  hangerStart,
}: RenderCompletionElementsOptions): void {
  if (!data.casings.length && !data.drlgCasings.length) return;
  // Liner hangers are drawn per-liner by liner.renderer (for both DRLG and IR sources).
  // This function no longer draws a hanger for the liner itself — that produced a
  // duplicate OUTER hanger set. It only draws the well-design liner-screen hanger
  // (lsFlg='Y') for a screen that has no corresponding liner geometry.
  if (!shouldRenderLinerScreen(data.wellDesign)) return;
  const hasDrlgLiners = data.drlgCasings.some(d => d.wLnrOdSz && (d.wLnrBotDepth ?? 0) > 0);
  if (hasDrlgLiners) return;

  const { baseHalfWidth, halfWidthIncrement, linerScreenInset } = layout;
  const innerHW = computeCasingHalfWidth(0, baseHalfWidth, halfWidthIncrement);
  const ohHW = resolveOhHalfWidth(data, layout);
  const screenHW = Math.min(innerHW, ohHW) - linerScreenInset;
  const hangerG = rootG.append('g').attr('class', 'screen-hangers').style('opacity', 0);

  const screenY = scale(resolveCompletionTopDepth(data.casings, data.drlgCasings));
  const { left, right } = buildScreenHanger(centerX, screenY, screenHW);
  hangerG.append('path').attr('class', 'screen-hanger').attr('stroke', '#000000').attr('stroke-width', '1.5').attr('fill', '#000000').attr('d', left);
  hangerG.append('path').attr('class', 'screen-hanger').attr('stroke', '#000000').attr('stroke-width', '1.5').attr('fill', '#000000').attr('d', right);

  hangerG.transition().delay(hangerStart).duration(ANIM.HANGER_DURATION).ease(easeCubicInOut).style('opacity', 1);
}

    if (sorted.length === 0) return null;
    return sorted.length === 1 ? sorted[0] : sorted[sorted.length - 2];

/**
 * Extracts liner entries from drlgCasings when no ICasingIR data is available.
 * Deduplicates by wLnrOdSz, sorts shallowest-first.
 */
function rawLinersFromDrlg(drlgCasings: IDrlgCsg[]): RawLinerEntry[] {
    const dedupMap = new Map<string, RawLinerEntry>();
    for (const d of drlgCasings) {
        if (!d.wLnrOdSz || (d.wLnrBotDepth ?? 0) <= 0) continue;
        const existing = dedupMap.get(d.wLnrOdSz);
        if (!existing || d.wLnrBotDepth! > existing.botDepth) {
            dedupMap.set(d.wLnrOdSz, {
                odSz: d.wLnrOdSz,
                topDepth: d.wLnrTopDepth ?? d.wCsgBotDpth,
                botDepth: d.wLnrBotDepth!,
                hasActual: true,
            });
        }
    }

    return [...dedupMap.values()].sort((a, b) => a.botDepth - b.botDepth);
}

/**
 * Extracts liner entries from ICasingIR + drlgCasings for actual depth override.
 *
 * Source priority per liner:
 *   - DRLG_CSG (actual/as-drilled) overrides depth when a matching wLnrOdSz exists.
 *   - EXAD_GWD_IR_CASING (planned) is the fallback for depth and the primary OD source.
 *
 * Extra DRLG_CSG liners with no corresponding IR entry are appended at the end.
 */
function rawLinersFromIR(casings: ICasingIR[], drlgCasings: IDrlgCsg[]): RawLinerEntry[] {
    const irLiners = casings.filter(c => c.csgType === 'Liner');

    // Fallback top depth when no drlgCasing match exists: shoe of the deepest non-liner casing
    const lastCasingShoe = casings
        .filter(c => c.csgType !== 'Liner')
        .reduce((max, c) => Math.max(max, c.csgDepth), 0);

    const matchedOdSizes = new Set<string>();

    const irEntries: RawLinerEntry[] = irLiners.map(csg => {
        // Source priority: use DRLG_CSG depth when available; fall back to IR depth.
        const actual = drlgCasings.find(d => d.wLnrOdSz === csg.csgSize) ?? null;
        const hasActualBot = (actual?.wLnrBotDepth ?? 0) > 0;
        const hasActualTop = (actual?.wLnrTopDepth ?? 0) > 0;
        if (actual?.wLnrOdSz) matchedOdSizes.add(actual.wLnrOdSz);
        const fallbackTopDepth = actual != null ? actual.wCsgBotDpth : lastCasingShoe;
        return {
            odSz: actual?.wLnrOdSz ?? csg.csgSize,
            topDepth: hasActualTop ? actual!.wLnrTopDepth! : fallbackTopDepth,
            botDepth: hasActualBot ? actual!.wLnrBotDepth! : csg.csgDepth,
            hasActual: hasActualBot,
        };
    });

    const extraEntries: RawLinerEntry[] = drlgCasings
        .filter(d => d.wLnrOdSz != null && (d.wLnrBotDepth ?? 0) > 0 && !matchedOdSizes.has(d.wLnrOdSz!))
        .map(d => ({
            odSz: d.wLnrOdSz!,
            topDepth: d.wLnrTopDepth ?? d.wCsgBotDpth,
            botDepth: d.wLnrBotDepth!,
            hasActual: true,
        }));

    return [...irEntries, ...extraEntries].sort((a, b) => a.botDepth - b.botDepth);
}

import { DiagramLayout, WellboreDiagramData } from '@models/well-design/wellbore-diagram.model';
import { ICasingIR } from '@shared/models/wwell/casing-ir.model';
import { IDrlgCsg } from '@shared/models/wwell/drlg-csg.model';
import { IWellDesign } from '@shared/models/wwell/well-design.model';
import { computeCasingHalfWidth, parseInchSize } from '@shared/utils/wellbore-math.util';
import { resolveAnchorLiner, resolveLinerDepths, resolveLiners } from './liner.renderer';
import { DepthScale } from './wellbore-renderer.types';

const OVERRUN_COMPLETION_LENGTH_FT = 1200;

export function resolveShoeDepth(casings: ICasingIR[], drlgCasings: IDrlgCsg[]): number {
  const irNonLiner = casings.filter(c => c.csgType !== 'Liner' && c.csgType !== 'Gravel Pack');
  const deepest = irNonLiner[0];
  if (!deepest) return drlgCasings[0]?.wCsgBotDpth ?? 0;

  const actual = drlgCasings.find(d => d.wCsgOdSz === deepest.csgSize) ?? null;
  return (actual !== null && actual.wCsgBotDpth > 0 ? actual.wCsgBotDpth : null) ?? deepest.csgDepth;
}

/**
 * Depth Open Hole / Gravel Pack / Pre-Perf start from:
 *   - 2+ liners: below the anchor (second-deepest) liner's bottom — the deepest liner is
 *     physically run into that open-hole continuation on its own.
 *   - Exactly 1 liner: there's no separate pipe liner above it — a lone liner is always
 *     run into the drilled hole below the casing shoe, so completion elements must start
 *     at the CASING SHOE (not the liner's own bottom) to visually surround it. This is a
 *     geometric fact, independent of EXAD_GWD_WELL_DESIGN.lsFlg — lsFlg only controls
 *     whether the liner gets the decorative "screen" grid pattern/label (see
 *     isLastLinerScreen in liner.renderer), not where it physically sits.
 *   - No liners: the casing shoe.
 */
export function resolveCompletionTopDepth(casings: ICasingIR[], drlgCasings: IDrlgCsg[]): number {
  const liners = resolveLinerDepths(casings, drlgCasings);
  if (liners.length <= 1) return resolveShoeDepth(casings, drlgCasings);

  return liners[liners.length - 2].botDepth;
}

export function resolveCompletionTopPx(
  casings: ICasingIR[],
  drlgCasings: IDrlgCsg[],
  scale: DepthScale,
): number {
  return scale(resolveCompletionTopDepth(casings, drlgCasings));
}

export function resolveCompletionBottomDepth(data: WellboreDiagramData): number {
  const completionTopDepth = resolveCompletionTopDepth(data.casings, data.drlgCasings);
  const isCurrentDepthBeyondTarget = data.targetDepth > 0 && data.currentDepth > data.targetDepth;

  const naturalBottom = isCurrentDepthBeyondTarget
    ? Math.max(completionTopDepth, Math.min(data.totalDepth, completionTopDepth + OVERRUN_COMPLETION_LENGTH_FT))
    : data.totalDepth;

  return Math.min(naturalBottom, data.displayDepth);
}

export function resolveCompletionBottomPx(data: WellboreDiagramData, scale: DepthScale): number {
  return scale(resolveCompletionBottomDepth(data));
}

function isFlagYes(flg: string | null | undefined): boolean {
  return (flg ?? '').trim().toUpperCase() === 'Y';
}

const PRE_PERF_LINER_SIZE_IN = 7;

export interface PrePerforatedLinerDepth {
  readonly odSz: string;
  readonly topDepth: number;
  readonly botDepth: number;
}

/**
 * Resolves the pre-perforated liner overlay depth range.
 *
 * Both conditions must pass:
 *   1. EXAD_GWD_WELL_DESIGN.perforatedFlg is 'Y' (case-insensitive).
 *   2. The anchor liner (see resolveAnchorLiner) has a 7" OD.
 *
 * perforatedRemarks is free text and only affects label copy elsewhere — it must never
 * gate visibility (see isPrePerforatedLiner in liner.renderer for the matching rationale).
 *
 * The returned depths come from that anchor liner's own top/bottom so the overlay is
 * drawn only across its interval — and, with 2+ liners, from the same liner Open Hole
 * and Gravel Pack are drawn below, not the deepest (screen) liner.
 */
export function resolvePrePerforatedLinerDepth(
  wellDesign: IWellDesign | null | undefined,
  drlgCasings: IDrlgCsg[],
  casings: ICasingIR[] = [],
): PrePerforatedLinerDepth | null {
  if (!isFlagYes(wellDesign?.perforatedFlg)) return null;

  const anchor = resolveAnchorLiner(casings, drlgCasings);
  if (!anchor || parseInchSize(anchor.odSz) !== PRE_PERF_LINER_SIZE_IN) return null;

  return { odSz: anchor.odSz, topDepth: anchor.topDepth, botDepth: anchor.botDepth };
}

export function shouldRenderPrePerforatedLiner(
  wellDesign: IWellDesign | null | undefined,
  drlgCasings: IDrlgCsg[] = [],
  casings: ICasingIR[] = [],
): boolean {
  return resolvePrePerforatedLinerDepth(wellDesign, drlgCasings, casings) !== null;
}

export function resolveOhHalfWidth(data: WellboreDiagramData, layout: DiagramLayout): number {
  const { baseHalfWidth, halfWidthIncrement, openHoleHwMargin } = layout;
  const innerHW = computeCasingHalfWidth(0, baseHalfWidth, halfWidthIncrement);

  if (!isFlagYes(data.wellDesign?.ohFlg)) return innerHW + openHoleHwMargin;

  const ohDiameter = parseInchSize(data.wellDesign?.ohRemarks);
  if (ohDiameter === null) return innerHW + openHoleHwMargin;

  // innerHW is the on-screen half-width of the deepest structural casing (tier 0) — the
  // casing the anchor liner hangs from — so its real OD is the outer bound OH must stay inside.
  const structuralCasings = data.casings.filter(
    c => c.csgType !== 'Liner' && c.csgType !== 'Gravel Pack',
  );
  const outerOD = parseInchSize(structuralCasings[0]?.csgSize ?? null);

  // When liners exist, OH must render strictly outside the anchor liner it surrounds and
  // inside the casing it hangs from (e.g. a 12 1/4" OH must sit outside a 7" liner and
  // inside a 13 3/8" casing). Per-liner on-screen widths (anchorLiner.halfWidth, innerHW)
  // are a cosmetic decrement scheme with no fixed relationship to real inches, so scaling
  // one of them by a raw diameter ratio can overshoot the casing or undershoot the liner.
  // Instead, interpolate between the two known on-screen widths using the real diameter
  // ratio as the blend factor — this guarantees OH always lands between them.
  const liners = resolveLiners(data.casings, data.drlgCasings, layout, data.wellDesign);
  const anchorLiner = liners.length > 1 ? liners[liners.length - 2] : (liners[0] ?? null);

  if (anchorLiner !== null) {
    const linerOD = parseInchSize(anchorLiner.odSz);
    if (linerOD !== null && linerOD > 0) {
      if (outerOD !== null && outerOD > linerOD) {
        const blend = Math.min(1, Math.max(0, (ohDiameter - linerOD) / (outerOD - linerOD)));
        return anchorLiner.halfWidth + blend * (innerHW - anchorLiner.halfWidth);
      }
      // No known casing OD to blend against: fall back to a direct ratio, clamped so OH
      // never renders outside the innermost casing's on-screen width.
      return Math.min(anchorLiner.halfWidth * (ohDiameter / linerOD), innerHW);
    }
  }

  // No liners: scale relative to innermost structural casing
  if (outerOD !== null && outerOD > 0 && ohDiameter < outerOD) {
    return innerHW * (ohDiameter / outerOD);
  }

  return innerHW + openHoleHwMargin;
}

export function getCasingTier(csg: ICasingIR, casings: ICasingIR[]): number {
  const structuralCasings = casings.filter(item => item.csgType !== 'Gravel Pack');
  const idx = structuralCasings.indexOf(csg);
  return idx === -1 ? 0 : idx;
}
