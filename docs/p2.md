export type WwellDynamicFieldMode = 'edit' | 'review';

@Component({
  selector: 'app-dynamic-field',
  standalone: true,
  imports: [
    ReactiveFormsModule,
    MatFormFieldModule,
    MatInputModule,
    MatSelectModule,
    MatButtonToggleModule,

  readonly config = input.required<WwellFieldConfig>();
  readonly control = input.required<FormControl>();
  readonly mode = input<WwellDynamicFieldMode>('edit');
  /** Class applied to this field's portaled select/autocomplete overlay panel, so a host page can theme it. */
  readonly selectPanelClass = input('');

  protected readonly optionLabel = wwellFieldOptionLabel;
  protected readonly optionValue = wwellFieldOptionValue;
  protected readonly optionIcon = wwellFieldOptionIcon;

  /** Resolves the config's `options` (static array or live Signal) once per change — every
   *  options `@for` loop below reads this instead of `config().options` directly. */
  protected readonly resolvedOptions = computed(() => resolveFieldOptions(this.config()));

  /**
   * `control()` is a Signal input, but the bound FormControl's own `.value` mutates in place
   * on every keystroke and is never itself signal-tracked — reading it directly inside a
   * `computed()` (as `reviewValue` below used to) only evaluates once, at first read, and then
   * freezes forever: the Attachments & Review step's tiles kept showing '—' even after the
   * matching field was filled in on an earlier step. Bridging through `valueChanges` (and
   * re-subscribing whenever a *different* control is bound) keeps this genuinely reactive.
   */
  // No `initialValue`/`requireSync`: reading `this.control()` synchronously in a field
  // initializer (for an `initialValue`) fails the compiler's NG8118 check — inputs aren't bound
  // yet at that point — and `toObservable` doesn't emit synchronously either (its effect is
  // scheduled, not immediate), so `requireSync` throws NG0601 at construction instead. Left
  // without either, `liveValue()` is simply `undefined` for the first tick, which `reviewValue`
  // below already treats the same as an empty value.
  private readonly liveValue = toSignal(
    toObservable(this.control).pipe(switchMap(control => control.valueChanges.pipe(startWith(control.value)))),
  );

  protected readonly reviewValue = computed(() => {
    const value = this.liveValue();
    const cfg = this.config();

    if (cfg.type === 'file') {
      const files = value as File[] | null;
      return files?.length ? `${files.length} file(s) attached` : '—';
    }
    if (cfg.type === 'select' || cfg.type === 'toggle-group') {
      const match = this.resolvedOptions().find(opt => wwellFieldOptionValue(opt) === value);
      return match ? wwellFieldOptionLabel(match) : (value || '—');
    }
    return value === null || value === undefined || value === '' ? '—' : String(value);
  });

  protected readonly isReviewValueEmpty = computed(() => this.reviewValue() === '—');

  protected onFilesChange(files: File[]): void {
    this.control().setValue(files);
    this.control().markAsTouched();
  }

  protected errorMessage(): string | null {
    const control = this.control();
    if (!control.touched || control.valid) return null;
    if (control.hasError('required')) return `${this.config().label} is required`;
    if (control.hasError('pattern')) return `${this.config().label} format is invalid`;
    if (control.hasError('min')) return `${this.config().label} is below the minimum`;
    return 'Invalid value';
  }
}

import {
  AfterViewInit,
  ChangeDetectionStrategy,
  ChangeDetectorRef,
  Component,
  ElementRef,
  OnDestroy,
  QueryList,
  ViewChildren,
  computed,
  effect,
  inject,
  input,
  signal,
} from '@angular/core';
import { FormArray, FormBuilder, FormControl, FormGroup } from '@angular/forms';
import { MatButtonModule } from '@angular/material/button';
import { MatIconModule } from '@angular/material/icon';
import { Subscription } from 'rxjs';
import {
  buildConditionalFieldEnterTimeline,
  buildConditionalFieldExitTimeline,
  setConditionalFieldVisibility,
} from '../dynamic-section/conditional-field.animation';
import { DynamicFieldComponent, WwellDynamicFieldMode } from '../dynamic-field/dynamic-field.component';
import { playIfMotionAllowed } from '../gsap-motion.util';
import { addRepeatableRow, defaultValueFor } from '../wwell-dynamic-form.builder';
import { WwellFieldConfig, WwellRepeatableRowConfig } from '../wwell-dynamic-form.types';
import { buildRowEnterTimeline, buildRowExitTimeline } from './repeatable-row.animation';

@Component({
  selector: 'app-repeatable-group',
  standalone: true,
  imports: [DynamicFieldComponent, MatButtonModule, MatIconModule],
  templateUrl: './repeatable-group.component.html',
  styleUrl: './repeatable-group.component.scss',
  changeDetection: ChangeDetectionStrategy.OnPush,
})
export class RepeatableGroupComponent implements AfterViewInit, OnDestroy {
  private readonly fb = inject(FormBuilder);
  private readonly cdr = inject(ChangeDetectorRef);

  readonly config = input.required<WwellRepeatableRowConfig>();
  readonly formArray = input.required<FormArray>();
  readonly mode = input<WwellDynamicFieldMode>('edit');
  readonly selectPanelClass = input('');

  @ViewChildren('rowHost') private readonly rowHosts?: QueryList<ElementRef<HTMLElement>>;

  /**
   * Row-level `visibleWhen` support — `app-dynamic-section` implements this for flat
   * `section().fields`, but never for `section().repeatable` rows, so it's ported here
   * rather than duplicated per-feature. Each row gets its own independent visibility
   * state, resolved against that row's own `FormGroup` (not the whole array), because a
   * repeatable row's condition (e.g. "this row's Type is Liner") only ever depends on a
   * sibling field within the *same* row.
   */
  @ViewChildren('rowFieldHost') private readonly rowFieldHosts?: QueryList<ElementRef<HTMLElement>>;

  /** Only fields with `visibleWhen` grow a `#rowFieldHost` wrapper — the template renders
   * exactly this many per row, in this order, so `row index * length + field index` maps
   * a (row, field) pair onto the flat, document-order `rowFieldHosts` query list. */
  protected readonly conditionalRowFields = computed(() => this.config().rowFields.filter(field => field.visibleWhen));

  /** Bumped only on user-triggered `addRow()` so the effect below can play an entrance animation for it. */
  private readonly addedRowTick = signal(0);

  /** Tracks last-applied visibility per (row, field key) so we only animate on an actual flip. Keyed by row
   * object identity (not index) so it survives rows being added/removed/reordered without misattribution. */
  private readonly lastVisibility = new Map<FormGroup, Map<string, boolean>>();
  private valueChangesSubscription?: Subscription;
  private rowFieldHostsSubscription?: Subscription;

  constructor() {
    effect(() => {
      const tick = this.addedRowTick();
      if (tick === 0) return;
      const last = this.rowHosts?.last;
      if (last) playIfMotionAllowed(buildRowEnterTimeline(last.nativeElement));
    });
  }

  ngAfterViewInit(): void {
    this.syncConditionalRowFields();
    this.valueChangesSubscription = this.formArray().valueChanges.subscribe(() => this.onFormArrayChanged());
    // Re-sync whenever the query list's elements change (row added/removed) — a freshly
    // added row's #rowFieldHost isn't queryable until after this fires.
    this.rowFieldHostsSubscription = this.rowFieldHosts?.changes.subscribe(() => this.syncConditionalRowFields());
    // See `onFormArrayChanged`'s doc comment — the very first render can be just as stale as
    // any later one, for the same reason, so it needs the same forced re-check.
    this.cdr.detectChanges();
  }

  ngOnDestroy(): void {
    this.valueChangesSubscription?.unsubscribe();
    this.rowFieldHostsSubscription?.unsubscribe();
  }

  protected rows(): FormGroup[] {
    return this.formArray().controls as FormGroup[];
  }

  /**
   * Pushing/removing a row (or editing any field in one) mutates the `formArray` input's own
   * `FormArray` object in place — the input's own signal reference never changes, so Angular's
   * standard OnPush triggers (changed input, event sourced from this component's own template,
   * a *directly template-read* signal changing) never fire for a `RepeatableGroupComponent`
   * instance that didn't cause the mutation itself. That's invisible for the normal "edit"
   * instance (its own "+ Add"/remove buttons are template-sourced click events, so zone.js
   * re-checks it for free) but left every *other* instance of the same section — concretely,
   * the read-only copy the Attachments & Review step renders for the same FormArray — stuck
   * forever showing whatever row count existed when it first mounted. Confirmed live: filling
   * in an Offset Water Well and opening the review step showed a completely empty section,
   * with the underlying FormArray provably holding the right row the whole time. Forcing a
   * synchronous `detectChanges()` here — rather than the usual `markForCheck()` — is deliberate:
   * this component sits inside a `mat-expansion-panel`, and Material detaches a collapsed
   * panel's content from the ambient change-detection walk, so merely flagging "needs a check"
   * isn't enough when nothing ever walks back down to it on its own.
   */
  private onFormArrayChanged(): void {
    this.syncConditionalRowFields();
    this.cdr.detectChanges();
  }

  protected fieldControl(row: FormGroup, key: string): FormControl {
    return row.get(key) as FormControl;
  }

  protected addRow(): void {
    addRepeatableRow(this.formArray(), this.config().rowFields, this.fb);
    this.addedRowTick.update(tick => tick + 1);
  }

  protected removeRow(index: number): void {
    const row = this.rows()[index];
    const host = this.rowHosts?.get(index);
    if (!host) {
      this.formArray().removeAt(index);
      this.lastVisibility.delete(row);
      return;
    }

    playIfMotionAllowed(buildRowExitTimeline(host.nativeElement, () => {
      this.formArray().removeAt(index);
      this.lastVisibility.delete(row);
    }));
  }

  private syncConditionalRowFields(): void {
    const conditionalFields = this.conditionalRowFields();
    if (!conditionalFields.length) return;

    this.rows().forEach((row, rowIndex) => {
      conditionalFields.forEach((field, fieldIndex) => {
        const host = this.rowFieldHosts?.get(rowIndex * conditionalFields.length + fieldIndex)?.nativeElement;
        if (host) this.applyFieldVisibility(row, field, host);
      });
    });
  }

  private applyFieldVisibility(row: FormGroup, field: WwellFieldConfig, host: HTMLElement): void {
    const rowVisibility = this.lastVisibility.get(row) ?? new Map<string, boolean>();
    this.lastVisibility.set(row, rowVisibility);

    const isVisible = this.isFieldVisible(row, field);
    const seenBefore = rowVisibility.has(field.key);
    const wasVisible = rowVisibility.get(field.key);
    rowVisibility.set(field.key, isVisible);

    if (!seenBefore) {
      setConditionalFieldVisibility(host, isVisible);
      this.syncControlEnabled(row, field, isVisible);
      return;
    }
    if (isVisible === wasVisible) return;

    if (isVisible) {
      this.syncControlEnabled(row, field, true);
      playIfMotionAllowed(buildConditionalFieldEnterTimeline(host));
    } else {
      playIfMotionAllowed(buildConditionalFieldExitTimeline(host, () => this.syncControlEnabled(row, field, false)));
    }
  }

  private isFieldVisible(row: FormGroup, field: WwellFieldConfig): boolean {
    const condition = field.visibleWhen;
    if (!condition) return true;

    const sibling = row.get(condition.field);
    const value = sibling?.value;
    if (condition.equals !== undefined) return value === condition.equals;
    if (condition.oneOf) return condition.oneOf.includes(value);
    if (condition.notOneOf) return !condition.notOneOf.includes(value);
    return true;
  }

  /** Hidden fields are disabled (excluded from validity/`.value`) and reset so a stale value from another row/mode never submits. */
  private syncControlEnabled(row: FormGroup, field: WwellFieldConfig, isVisible: boolean): void {
    const control = this.fieldControl(row, field.key);
    if (isVisible) {
      if (control.disabled) control.enable({ emitEvent: false });
      return;
    }
    if (control.enabled) {
      control.reset(defaultValueFor(field), { emitEvent: false });
      control.disable({ emitEvent: false });
    }
  }
}

import { FormArray, FormBuilder, FormControl, FormGroup, ValidatorFn, Validators } from '@angular/forms';
import {
  WwellFieldConfig,
  WwellFormSectionConfig,
  WwellFormStepConfig,
  WwellRepeatableRowConfig,
} from './wwell-dynamic-form.types';

/** Empty `File[]` isn't falsy to Angular's default `required` validator — needs its own check. */
const requiredFileList: ValidatorFn = control => (control.value?.length ? null : { required: true });

/**
 * `Validators.minLength` looks like the obvious fit for `minRequiredRows`, but Angular's own
 * `isEmptyInputValue` check makes it treat a *zero-length* array as "nothing to validate yet"
 * and returns null (valid) — the same opt-out that lets `minLength` coexist with an optional
 * field. That silently defeated `minRequiredRows`'s whole point (see its doc comment): a
 * freshly-loaded step with zero rows showed up as complete (green check) in both the step rail
 * and the Attachments & Review completion count. This validator has no such exemption — an
 * array shorter than `min`, including empty, is always invalid.
 */
function minRowsValidator(min: number): ValidatorFn {
  return control => ((control.value?.length ?? 0) < min ? { minRows: { requiredLength: min, actualLength: control.value?.length ?? 0 } } : null);
}

export function defaultValueFor(field: WwellFieldConfig): unknown {
  if (field.defaultValue !== undefined) return field.defaultValue;
  if (field.type === 'file') return [];
  if (field.type === 'number') return null;
  return '';
}

function buildValidators(field: WwellFieldConfig): ValidatorFn[] {
  const config = field.validators;
  if (!config) return [];

  const validators: ValidatorFn[] = [];
  if (config.required) {
    validators.push(field.type === 'file' ? requiredFileList : Validators.required);
  }
  if (config.pattern) validators.push(Validators.pattern(config.pattern));
  if (config.min !== undefined) validators.push(Validators.min(config.min));
  return validators;
}

export function buildFieldControl(field: WwellFieldConfig, fb: FormBuilder): FormControl {
  return fb.control({ value: defaultValueFor(field), disabled: !!field.readonly }, buildValidators(field));
}

export function buildRepeatableRowGroup(rowFields: readonly WwellFieldConfig[], fb: FormBuilder): FormGroup {
  const group = fb.group({});
  rowFields.forEach(field => group.addControl(field.key, buildFieldControl(field, fb)));
  return group;
}

export function addRepeatableRow(array: FormArray, rowFields: readonly WwellFieldConfig[], fb: FormBuilder): void {
  array.push(buildRepeatableRowGroup(rowFields, fb));
}

function buildRepeatableArray(config: WwellRepeatableRowConfig, fb: FormBuilder): FormArray {
  const arrayValidators = config.minRequiredRows ? [minRowsValidator(config.minRequiredRows)] : [];
  const array = fb.array<FormGroup>([], arrayValidators);
  const minRows = config.minRows ?? 0;
  for (let i = 0; i < minRows; i++) {
    addRepeatableRow(array, config.rowFields, fb);
  }
  return array;
}

export function buildSectionGroup(section: WwellFormSectionConfig, fb: FormBuilder): FormGroup | FormArray {
  if (section.repeatable) return buildRepeatableArray(section.repeatable, fb);

  const group = fb.group({});
  (section.fields ?? []).forEach(field => group.addControl(field.key, buildFieldControl(field, fb)));
  return group;
}

export function buildWwellDynamicForm(schema: readonly WwellFormStepConfig[], fb: FormBuilder): FormGroup {
  const root = fb.group({});
  schema.forEach(step => {
    const stepGroup = fb.group({});
    step.sections.forEach(section => stepGroup.addControl(section.key, buildSectionGroup(section, fb)));
    root.addControl(step.key, stepGroup);
  });
  return root;
}

/** Schema vocabulary for the wwell-planning dynamic form engine. Add a field = edit a schema array, not a template. */

import type { Signal } from '@angular/core';

export type WwellFieldControlType = 'text' | 'number' | 'textarea' | 'select' | 'combo' | 'toggle-group' | 'file';

export interface WwellFieldOption {
  readonly value: string;
  readonly label: string;
  /** Material icon ligature name — used by 'toggle-group' only. */
  readonly icon?: string;
}

export interface WwellFieldValidatorConfig {
  readonly required?: boolean;
  readonly pattern?: RegExp;
  readonly min?: number;
}

/**
 * Field is only shown (and its control enabled) while a sibling control in the same section
 * matches this condition. Exactly one of `equals` | `oneOf` | `notOneOf` should be set:
 *  - `equals`: sibling value === this value
 *  - `oneOf`: sibling value is one of this set (e.g. a client tier)
 *  - `notOneOf`: sibling value is absent from this set — the fallback for values outside every
 *    named tier (e.g. a client not yet assigned to a tier keeps the default field set)
 */
export interface WwellFieldVisibilityCondition {
  readonly field: string;
  readonly equals?: string;
  readonly oneOf?: readonly string[];
  readonly notOneOf?: readonly string[];
}

export interface WwellFieldConfig {
  readonly key: string;
  readonly label: string;
  readonly type: WwellFieldControlType;
  readonly placeholder?: string;
  readonly hint?: string;
  readonly suffix?: string;
  /** Used by 'select' | 'combo' | 'toggle-group'. Plain strings are shown as their own label.
   *  A Signal is resolved lazily via `resolveFieldOptions` — for options that come from a store
   *  (e.g. the IR wizard's "Link to Request" picker) rather than being known at schema-definition
   *  time (the schema array itself is a static, module-level constant with no injector). */
  readonly options?: readonly (string | WwellFieldOption)[] | Signal<readonly (string | WwellFieldOption)[]>;
  readonly validators?: WwellFieldValidatorConfig;
  readonly colSpan?: 1 | 2;
  readonly visibleWhen?: WwellFieldVisibilityCondition;
  /** Overrides the type-based empty default (e.g. pre-selecting a toggle-group option). */
  readonly defaultValue?: string | number;
  /** Disables the control — for values derived elsewhere (e.g. a server-computed metric) rather
   *  than typed in directly. Pair with a `placeholder` like 'auto' to signal that visually. */
  readonly readonly?: boolean;
}

export interface WwellRepeatableRowConfig {
  readonly key: string;
  readonly addLabel: string;
  readonly rowFields: readonly WwellFieldConfig[];
  /** How many empty rows to seed the array with on load. */
  readonly minRows?: number;
  /** If set, the array itself is invalid until it holds at least this many rows — an empty array is not "complete" just because it has no invalid children. */
  readonly minRequiredRows?: number;
}

export interface WwellFormSectionConfig {
  readonly key: string;
  readonly title: string | null;
  /** Mutually exclusive with `repeatable`. */
  readonly fields?: readonly WwellFieldConfig[];
  readonly repeatable?: WwellRepeatableRowConfig;
}

export interface WwellFormStepConfig {
  readonly key: string;
  readonly label: string;
  readonly sections: readonly WwellFormSectionConfig[];
}

export function wwellFieldOptionLabel(option: string | WwellFieldOption): string {
  return typeof option === 'string' ? option : option.label;
}

export function wwellFieldOptionValue(option: string | WwellFieldOption): string {
  return typeof option === 'string' ? option : option.value;
}

export function wwellFieldOptionIcon(option: string | WwellFieldOption): string | undefined {
  return typeof option === 'string' ? undefined : option.icon;
}

/** Unwraps a field's `options` (static array or live Signal) into the array consumers iterate. */
export function resolveFieldOptions(field: WwellFieldConfig): readonly (string | WwellFieldOption)[] {
  const options = field.options;
  if (!options) return [];
  return typeof options === 'function' ? options() : options;
}

/**
 * Returns a copy of WWELL_IR_FORM_SCHEMA with the 'reference'/'main' section's 'linkedRequestId'
 * field wired to `linkedRequestOptions` instead of its static `[]` placeholder. The schema array
 * itself is a module-level constant (no injector available at definition time), so a store-backed
 * Signal can only be attached here, once, by the component that actually has one — see
 * WwellIrNewPageComponent's `schema` field.
 */
export function buildWwellIrFormSchema(
  linkedRequestOptions: Signal<readonly WwellFieldOption[]>,
): readonly WwellFormStepConfig[] {
  return WWELL_IR_FORM_SCHEMA.map(step => {
    if (step.key !== 'reference') return step;
    return {
      ...step,
      sections: step.sections.map(section => {
        if (section.key !== 'main' || !section.fields) return section;
        return {
          ...section,
          fields: section.fields.map(field =>
            field.key === 'linkedRequestId' ? { ...field, options: linkedRequestOptions } : field,
          ),
        };
      }),
    };
  });
}

   * are tracked), so it only ever redraws when `diagramData` gets a NEW object reference.
   * Angular Material's horizontal stepper renders every step's content upfront and keeps it
   * mounted (hidden inactive steps get `visibility: hidden` + an off-screen `transform`, not
   * `display: none`/destroyed) — so the very first redraw (with its clip-path "grow" reveal
   * transition, see casing.renderer.ts's `renderCasings`) runs while this step is still
   * hidden and transformed off-screen, and has already finished — collapsed — by the time
   * the user actually navigates here. Tracking `activeStepIndex` below forces a brand-new
   * `WellboreDiagramData` object (and therefore a fresh redraw + reveal animation) every time
   * the active step changes, so it's guaranteed to run once while genuinely visible. This is
   * also a wwell-ir-local fix, per the standing rule not to modify anything under
   * `presentation/well-bore-view/`.
   */
  protected readonly wellboreDiagramData: Signal<WellboreDiagramData> = computed(() => {
    this.activeStepIndex();
    return buildWellboreDiagramData(this.rawFormValue());
  });

  // Gates the diagram card's scale/ground-level/total-depth chrome, not just individual
  // casing/formation-top shapes — see `hasWellboreDiagramData`'s own doc comment.
  protected readonly hasWellboreData: Signal<boolean> = computed(() =>
    hasWellboreDiagramData(this.rawFormValue()),
  );

  // Surfaces cases where a deeper casing's entered size is larger than a shallower casing's —
  // see `findInvertedCasingSizeRows`'s doc comment for why the diagram can never draw that
  // combination correctly.
  protected readonly casingSizeWarnings: Signal<readonly CasingSizeWarning[]> = computed(() =>
    findInvertedCasingSizeRows(this.rawFormValue().subsurface.casing),
  );

  // `wellboreDiagramData().totalDepth` falls back to 0 when Target Depth (Request Reference,
  // a required field) hasn't been filled in yet — reachable via the stepper's free (non-linear)
  // navigation. `well-bore-view` always draws its "TOTAL DEPTH: 0 FT" label regardless (see the
  // standing rule against editing anything under `presentation/well-bore-view/`), so this instead
  // drives a page-local CSS hook (`.nip__wellbore-diagram-row--no-depth` in
  // wwell-ir-new-page.component.scss) that hides that one D3-rendered label group.
  protected readonly hasTargetDepth: Signal<boolean> = computed(
    () => (this.rawFormValue().reference.main.targetDepthFt ?? 0) > 0,
  );
  @ViewChildren('stepContent') private readonly stepContents?: QueryList<ElementRef<HTMLElement>>;
  @ViewChildren('reviewCard', { read: ElementRef }) private readonly reviewCards?: QueryList<ElementRef<HTMLElement>>;

  private previousStepIndex = 0;
  private wasSaving = false;
  private reviewRevealed = false;
  // Which submitAs() call is currently in flight, and whether it targeted an already-saved draft
  // (UPDATE) or a brand-new one (INSERT) — the success effect below reads both to decide the
  // notification wording and whether to stay on this page or exit the wizard.
  private lastSubmitStatus: 'draft' | 'in-progress' | null = null;
  private lastSubmitWasUpdate = false;
  private editFormPopulated = false;

  constructor() {
    effect(() => {
      const irSeq = this.editIrSeq();
      if (irSeq === null) {
        // A brand-new visit to this page must never inherit a previous draft's ir_seq.
        this.store.resetIrDraft();
        return;
      }
      // Every Save/Submit for the rest of this visit must UPDATE this record, not INSERT a new one.
      this.store.startEditingIr(irSeq);
    });

    effect(() => {
      if (this.editFormPopulated) return;
      const ir = this.editingIr();
      if (!ir) return;
      this.populateFormForEdit(ir);
      this.editFormPopulated = true;
    });

    effect(() => {
      const index = this.activeStepIndex();
      const host = this.stepContents?.get(index);
      if (!host) return;

      const direction: WwellStepDirection = index >= this.previousStepIndex ? 'forward' : 'backward';
      playIfMotionAllowed(buildStepTransitionTimeline(host.nativeElement, direction));
      this.previousStepIndex = index;
    });

    effect(() => {
      // Keyed on the 'attachments' step itself (not "last step") — Well Design View is now
      // appended after it, so the review UI is no longer on the final step.
      const isReviewStep = WWELL_IR_FORM_SCHEMA[this.activeStepIndex()]?.key === 'attachments';
      if (!isReviewStep || this.reviewRevealed) return;

      const cards = this.reviewCards?.map(card => card.nativeElement) ?? [];
      if (!cards.length) return;

      playIfMotionAllowed(buildReviewRevealTimeline(cards));
      this.reviewRevealed = true;
    });

    effect(() => {
      const saving = this.isSaving();
      const error = this.store.createIrError();

      if (error) {
        this.notify.error(error);
      } else if (this.wasSaving && !saving) {
        this.onSaveSucceeded();
      }
      this.wasSaving = saving;
    });
  }

  ngOnInit(): void {
    this.store.loadLinkedRequestWells();
  }

  /**
   * A draft save notifies and stays on the wizard — drafts are a checkpoint for continued
   * editing, not a workflow exit, so navigating away would strand the author mid-form and force
   * them to reopen the record to keep going. A full Submit still notifies and exits, since that
   * completes the intake step.
   */
  private onSaveSucceeded(): void {
    // Look up by the ir_seq that was actually just saved — never assume index 0. That was only
    // ever true for a brand-new INSERT (upsertByIrSeq prepends those); an UPDATE to an existing
    // record replaces it *in place* at its original index, so index 0 can point at a totally
    // different (and possibly stale) record.
    const savedIrSeq = this.store.draftIrSeq();
    const code = this.store.irs().find(ir => ir.irSeq === savedIrSeq)?.code;

    if (this.lastSubmitStatus === 'draft') {
      const verb = this.lastSubmitWasUpdate ? 'updated' : 'saved';
      this.notify.info(code ? `Draft ${verb} — ${code}.` : `Draft ${verb}.`);
      return;
    }

    this.notify.info(code ? `${code} submitted for review.` : 'Integrated requirement submitted for review.');
    this.navigateToList();
  }

  // 'edit/:irSeq' is one route config with a *compound* path, so its ActivatedRoute carries two
  // URL segments (['edit', irSeq]) on a single node. Angular's relative `['..']` pops only the
  // last URL segment of the current node, not "up to the parent route" — from here that lands on
  // '/main/integrated-requirements/edit' (matches no route, falls through to the '**' redirect to
  // login). Anchoring at `this.route.parent` instead sidesteps this regardless of how many URL
  // segments the current leaf route (this one, or the single-segment 'new') consumed.
  private navigateToList(): void {
    this.router.navigate(['.'], { relativeTo: this.route.parent });
  }

  protected stepGroup(stepKey: string): FormGroup {
    return this.form.get(stepKey) as FormGroup;
  }

  protected sectionControl(stepKey: string, sectionKey: string): FormGroup | FormArray {
    return this.stepGroup(stepKey).get(sectionKey) as FormGroup | FormArray;
  }

  /** Fills every simple field from the saved record, then grows+fills each repeatable section's
   *  FormArray to match. Attachments are left empty — a File object can't be reconstructed from
   *  just the saved attachment names, so re-attaching (if needed) is a manual step in edit mode. */
  private populateFormForEdit(ir: IntegratedRequirement): void {
    const patch = mapIntegratedRequirementToFormPatch(ir);
    this.stepGroup('reference').patchValue(patch.reference);
    this.stepGroup('programDetails').patchValue(patch.programDetails);
    this.stepGroup('contacts').patchValue({ geologist: patch.contacts.geologist, remarks: patch.contacts.remarks });
    this.stepGroup('hydrogeology').patchValue(patch.hydrogeology);

    this.populateRepeatableSection('subsurface', 'formationTops', ir.formationTops, formationTopRowPatch);
    this.populateRepeatableSection('subsurface', 'casing', ir.casing, casingRowPatch);
    this.populateRepeatableSection('contacts', 'offsetWells', ir.offsetWells, offsetWellRowPatch);
  }

  private populateRepeatableSection<TDomainRow>(
    stepKey: string,
    sectionKey: string,
    rows: readonly TDomainRow[],
    toRowPatch: (row: TDomainRow) => object,
  ): void {
    const array = this.sectionControl(stepKey, sectionKey) as FormArray;
    const rowFields = this.repeatableRowFields(stepKey, sectionKey);
    rows.forEach((row, index) => {
      if (index >= array.length) addRepeatableRow(array, rowFields, this.fb);
      array.at(index).patchValue(toRowPatch(row));
    });
  }

  private repeatableRowFields(stepKey: string, sectionKey: string): readonly WwellFieldConfig[] {
    const section = WWELL_IR_FORM_SCHEMA.find(step => step.key === stepKey)?.sections.find(s => s.key === sectionKey);
    return section?.repeatable?.rowFields ?? [];
  }

  protected isStepValid(stepKey: string): boolean {
    return this.stepGroup(stepKey).valid;
  }

  /** `reviewSteps` is `schema` with 'attachments'/'wellDesignView' filtered out, so its own
   *  `$index` only happens to line up with the real stepper index while those two live at the
   *  very end of the schema — resolving by key instead of trusting that ordering is what the
   *  review cards' "Edit" button uses to jump back to the right step. */
  protected stepIndexOf(stepKey: string): number {
    return this.schema.findIndex(step => step.key === stepKey);
  }

  protected reviewCompletionSummary(): { done: number; total: number } {
    const total = this.reviewSteps.length;
    const done = this.reviewSteps.filter(step => this.isStepValid(step.key)).length;
    return { done, total };
  }

  protected onReviewCardOpened(index: number): void {
    this.reviewCards?.get(index)?.nativeElement.scrollIntoView({ behavior: 'smooth', block: 'nearest' });
  }

  protected onStepSelectionChange(index: number): void {
    this.activeStepIndex.set(index);
  }

  protected cancel(): void {
    this.navigateToList();
  }

  protected saveAsDraft(): void {
    this.submitAs('draft');
  }

  // TEMPORARY: validation gate disabled per explicit request — Submit saves whatever is filled
  // in across all steps, same as Save as Draft, instead of blocking on invalid/incomplete fields.
  protected submit(): void {
    this.submitAs('in-progress');
  }

  private submitAs(status: 'draft' | 'in-progress'): void {
    const raw = this.form.getRawValue() as WwellIrFormRawValue;
    const { payload, files } = mapIrFormValueToSubmission(raw, status);
    const knownIrSeq = this.store.draftIrSeq();
    // Reusing the remembered ir_seq turns a repeat "Save as Draft" into an UPDATE instead of a
    // duplicate INSERT. `existing` is what this record looked like as of the last save/load — the
    // payload mapper diffs the current form against it so each casing/tops/hydrogeology row gets
    // its own real INSERT/UPDATE/DELETE (a row added on this pass is genuinely new even though the
    // IR itself already has an ir_seq) instead of one blanket action for the whole submission.
    const existing = knownIrSeq === null ? null : (this.store.irs().find(ir => ir.irSeq === knownIrSeq) ?? null);
    const irBulkPayload = buildWwellIrBulkSavePayload(raw, { irSeq: knownIrSeq, existing });
    this.lastSubmitStatus = status;
    this.lastSubmitWasUpdate = knownIrSeq !== null;
    this.store.createIr(payload, irBulkPayload, files);
  }
}
