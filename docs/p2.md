import { HttpErrorResponse } from '@angular/common/http';
import { computed, inject } from '@angular/core';
import { patchState, signalStore, withComputed, withHooks, withMethods, withState } from '@ngrx/signals';
import { Observable, catchError, forkJoin, last, map, of, switchMap } from 'rxjs';
import { WldaAttachment, WldaDraft, WldaLetter, WldaPrewap } from '@models/wwell-planning/wlda.model';
import { S3Service } from '@services/s3/s3.service';
import { NotificationService } from '@shared/components/notification/notification.service';
import { AuthStore } from 'src/app/features/auth/store/auth.store';
import { WldaService } from '../wlda.service';
import {
  WLDA_EMPTY_DRAFT,
  draftOfLetter,
  isWldaDraftDirty,
  toWldaCreatePayload,
  toWldaUpdatePayload,
  findPlanForWell,
  validateWldaDraft,
  wldaFieldErrors,
  wldaCode,
  wldaKey,
  wldaS3Prefix,
} from '../wlda.util';

/** `empty` = nothing open; `view` = a saved letter, read-only; `create` / `edit` = the draft is live. */
export type WldaMode = 'empty' | 'view' | 'create' | 'edit';

interface WldaState {
  readonly letters: readonly WldaLetter[];
  readonly prewaps: readonly WldaPrewap[];
  /** The letter being authored (create) or changed (edit — then `selectedKey` is also set). */
  readonly draft: WldaDraft | null;
  /** `wldaKey` of the saved letter open on the right, or null. */
  readonly selectedKey: string | null;
  /** Files stored in S3 for the opened letter. */
  readonly storedAttachments: readonly WldaAttachment[];
  readonly loading: boolean;
  readonly saving: boolean;
  readonly deleting: boolean;
  readonly loadError: string | null;
  /** The one-file upload dialog for a saved letter: in flight, byte progress (null = unknown), last failure. */
  readonly uploadingFile: boolean;
  readonly uploadPercent: number | null;
  readonly uploadError: string | null;
}

const initialState: WldaState = {
  letters: [],
  prewaps: [],
  draft: null,
  selectedKey: null,
  storedAttachments: [],
  loading: false,
  saving: false,
  deleting: false,
  loadError: null,
  uploadingFile: false,
  uploadPercent: null,
  uploadError: null,
};

/** Prefers the backend's own `{ message }` over Angular's generic "Http failure response" text. */
function errorMessage(err: unknown, fallback: string): string {
  if (err instanceof HttpErrorResponse) {
    const body: unknown = err.error;
    const message = body && typeof body === 'object' ? (body as { message?: unknown }).message : undefined;
    return err.status !== 0 && typeof message === 'string' && message ? message : fallback;
  }
  return err instanceof Error ? err.message : fallback;
}

/** Copies the plan's details into the draft; `overwrite` (an explicit well link) replaces typed values, otherwise only blanks are filled. */
function fillFromPlan(draft: WldaDraft, plan: WldaPrewap, overwrite = false): WldaDraft {
  const blank = (value: string) => overwrite || value.trim() === '';
  return {
    ...draft,
    prewapSeq: plan.prewapSeq,
    field: overwrite ? (plan.wPrimHid ?? '') : draft.field,
    wellNumber: overwrite ? (plan.wNum ?? '') : draft.wellNumber,
    waterWellName: blank(draft.waterWellName) ? `${plan.wPrimHid ?? ''}-${plan.wNum ?? ''}` : draft.waterWellName,
    reservoir: blank(draft.reservoir) ? (plan.targetForm ?? '') : draft.reservoir,
    wellType: blank(draft.wellType) ? (plan.wellType ?? '') : draft.wellType,
    totalDepth: blank(draft.totalDepth) ? (plan.tdMd == null ? '' : String(plan.tdMd)) : draft.totalDepth,
    svYutmCord: blank(draft.svYutmCord) ? (plan.surfaceNorth == null ? '' : String(plan.surfaceNorth)) : draft.svYutmCord,
    svXutmCord: blank(draft.svXutmCord) ? (plan.surfaceEast == null ? '' : String(plan.surfaceEast)) : draft.svXutmCord,
    svUtmZn: blank(draft.svUtmZn) ? (plan.svUtmZn == null ? '' : String(plan.svUtmZn)) : draft.svUtmZn,
  };
}

export const WldaStore = signalStore(
  withState<WldaState>(initialState),

  withComputed(({ letters, prewaps, draft, selectedKey }) => {
    const selectedLetter = computed(() => letters().find(letter => wldaKey(letter) === selectedKey()) ?? null);
    const draftPlan = computed(() => {
      const current = draft();
      if (!current) return null;
      if (selectedLetter() || current.prewapSeq !== null) return prewaps().find(plan => plan.prewapSeq === current.prewapSeq) ?? null;
      return findPlanForWell(prewaps(), current.field, current.wellNumber);
    });
    return {
      selectedLetter,
      draftPlan,
      draftFieldErrors: computed(() => {
        const current = draft();
        return current ? wldaFieldErrors(current, selectedLetter() ? 'edit' : 'create', draftPlan()) : {};
      }),
      draftErrors: computed(() => {
        const current = draft();
        return current ? validateWldaDraft(current, selectedLetter() ? 'edit' : 'create', draftPlan()) : [];
      }),
      mode: computed<WldaMode>(() => {
        if (draft()) return selectedLetter() ? 'edit' : 'create';
        return selectedLetter() ? 'view' : 'empty';
      }),
      isDirty: computed(() => {
        const current = draft();
        return current ? isWldaDraftDirty(current, selectedLetter()) : false;
      }),
    };
  }),

  withMethods((store, wldaService = inject(WldaService), s3 = inject(S3Service), authStore = inject(AuthStore), notify = inject(NotificationService)) => {
    const networkId = () => authStore.networkId() ?? '';

    /** Uploads each file under the letter's prefix; resolves to the names that failed (letter stays saved). */
    function uploadFiles(files: readonly File[], prewapSeq: number, ltrId: number): Observable<string[]> {
      if (files.length === 0) return of([]);
      const prefix = wldaS3Prefix(prewapSeq, ltrId);
      return forkJoin(
        files.map(file =>
          s3.store(file, prefix, networkId(), file.name).pipe(
            last(),
            map((): string | null => null),
            catchError(() => of(file.name)),
          ),
        ),
      ).pipe(map(outcomes => outcomes.filter((name): name is string => name !== null)));
    }

    /** Deletes stored objects by key; resolves to the names that failed. */
    function deleteStored(keys: readonly string[], names: ReadonlyMap<string, string>): Observable<string[]> {
      if (keys.length === 0) return of([]);
      return forkJoin(
        keys.map(key =>
          s3.delete(key, networkId()).pipe(
            map((): string | null => null),
            catchError(() => of(names.get(key) ?? key)),
          ),
        ),
      ).pipe(map(outcomes => outcomes.filter((name): name is string => name !== null)));
    }

    function loadStored(letter: WldaLetter, key: string): void {
      s3.search(`${wldaS3Prefix(letter.prewapSeq, letter.ltrId)}/`).subscribe({
        next: res => {
          if (store.selectedKey() !== key) return;
          patchState(store, {
            storedAttachments: res.objects.map(object => ({ key: object.OBJ_KEY, name: object.ORIGINAL_FILENAME, pendingIndex: null })),
          });
        },
        error: () => {
          if (store.selectedKey() === key) notify.error('Could not load the letter attachments');
        },
      });
    }

    /** Re-reads one letter (single-letter GET) and its files, e.g. after a save. */
    /** Re-reads one letter (single-letter GET) and its files, e.g. after a save. `reopenEdit` re-bases the open draft on the fresh row. */
    function refreshLetter(prewapSeq: number, ltrId: number, reopenEdit = false): void {
      const key = wldaKey({ prewapSeq, ltrId });
      wldaService.getLetter(prewapSeq, ltrId).subscribe({
        next: fresh => {
          patchState(store, {
            letters: store.letters().map(item => (wldaKey(item) === key ? fresh : item)),
            ...(reopenEdit && store.selectedKey() === key ? { draft: draftOfLetter(fresh) } : {}),
          });
          loadStored(fresh, key);
        },
        error: (err: unknown) => notify.error(errorMessage(err, 'Failed to refresh the WLDA letter')),
      });
    }

    return {
      /** Uploads one file straight to the opened letter's S3 folder (edit mode) and lists it on success. */
      uploadLetterFile(file: File): void {
        const letter = store.selectedLetter();
        if (!letter || store.uploadingFile()) return;
        patchState(store, { uploadingFile: true, uploadPercent: null, uploadError: null });
        s3.store(file, wldaS3Prefix(letter.prewapSeq, letter.ltrId), networkId(), file.name).subscribe({
          next: ({ percent, response }) => {
            if (!response) {
              patchState(store, { uploadPercent: percent });
              return;
            }
            patchState(store, {
              uploadingFile: false,
              uploadPercent: null,
              storedAttachments: [...store.storedAttachments(), { key: response.key, name: file.name, pendingIndex: null }],
            });
            notify.info(`${file.name} uploaded.`);
          },
          error: (err: unknown) =>
            patchState(store, { uploadingFile: false, uploadPercent: null, uploadError: errorMessage(err, `Failed to upload ${file.name}`) }),
        });
      },

      /** Clears a previous upload failure before the next file's dialog opens. */
      resetUpload(): void {
        patchState(store, { uploadError: null, uploadPercent: null });
      },

      loadAll(): void {
        patchState(store, { loading: true, loadError: null });
        forkJoin({ prewaps: wldaService.getPrewaps(), letters: wldaService.getLetters() }).subscribe({
          next: ({ prewaps, letters }) => patchState(store, { prewaps, letters, loading: false }),
          error: (err: unknown) => patchState(store, { loading: false, loadError: errorMessage(err, 'Failed to load WLDA letters') }),
        });
      },

      startDraft(): void {
        patchState(store, { draft: WLDA_EMPTY_DRAFT, selectedKey: null, storedAttachments: [] });
      },

      /** Switches the opened letter from view to edit. */
      startEdit(): void {
        const letter = store.selectedLetter();
        if (letter) patchState(store, { draft: draftOfLetter(letter) });
      },

      patchDraft(patch: Partial<WldaDraft>): void {
        const current = store.draft();
        if (!current) return;
        let next: WldaDraft = { ...current, ...patch };
        // Typing a different Field / Well Number un-links the chosen well; the typed values decide again.
        if (!store.selectedLetter() && ('field' in patch || 'wellNumber' in patch)) next = { ...next, prewapSeq: null };
        // Creating: once Field + Well Number land on a plan, fill the details it already knows (blank fields only).
        if (!store.selectedLetter() && ('field' in patch || 'wellNumber' in patch)) {
          const before = findPlanForWell(store.prewaps(), current.field, current.wellNumber);
          const plan = findPlanForWell(store.prewaps(), next.field, next.wellNumber);
          if (plan && plan !== before) next = fillFromPlan(next, plan);
        }
        patchState(store, { draft: next });
      },

      /** "Link to Water Well": pick a pre-well plan and populate the form from it (null clears the link). */
      linkWell(prewapSeq: number | null): void {
        const current = store.draft();
        if (!current || store.selectedLetter()) return;
        const plan = store.prewaps().find(item => item.prewapSeq === prewapSeq);
        patchState(store, { draft: plan ? fillFromPlan(current, plan, true) : { ...current, prewapSeq: null } });
      },

      /** Drops the draft: a create goes back to nothing open, an edit back to viewing the saved letter. */
      discardDraft(): void {
        patchState(store, { draft: null });
      },

      /** Opens a saved letter: the listed row at once, then a fresh single-letter GET and its S3 files. */
      select(key: string | null): void {
        patchState(store, { selectedKey: key, storedAttachments: [], draft: null });
        const letter = store.letters().find(item => wldaKey(item) === key);
        if (letter && key) refreshLetter(letter.prewapSeq, letter.ltrId);
      },

      /** Saves the letter first (it owns the LTR_ID the S3 prefix needs), then uploads each file. */
      createLetter(): void {
        const draft = store.draft();
        const plan = store.draftPlan();
        if (store.mode() !== 'create' || !draft || !plan || store.saving() || validateWldaDraft(draft, 'create', plan).length > 0) return;
        patchState(store, { saving: true });

        wldaService
          .createLetter(plan.prewapSeq, toWldaCreatePayload(draft))
          .pipe(switchMap(result => uploadFiles(draft.files, result.prewapSeq, result.ltrId).pipe(map(failed => ({ result, failed })))))
          .subscribe({
            next: ({ result, failed }) => {
              // The draft stays up (now keyed to the saved letter) until the fresh row lands, so the page stays in the editor.
              patchState(store, { saving: false, selectedKey: wldaKey(result), storedAttachments: [], draft: { ...draft, prewapSeq: result.prewapSeq, files: [], removedKeys: [] } });
              notify.info(`${wldaCode(result.ltrId)} created.`);
              if (failed.length > 0) notify.error(`Letter saved, but these files did not upload: ${failed.join(', ')}`);
              wldaService.getLetters().subscribe({
                next: letters => {
                  patchState(store, { letters });
                  refreshLetter(result.prewapSeq, result.ltrId, true);
                },
                // Saved, but the list could not reload: fall back to viewing so a second Create can't duplicate it.
                error: () => patchState(store, { draft: null }),
              });
            },
            error: (err: unknown) => {
              patchState(store, { saving: false });
              notify.error(errorMessage(err, 'Failed to create the WLDA letter'));
            },
          });
      },

      /** Saves remarks (coordinates are not updatable), removes dropped files from S3, uploads new ones. */
      updateLetter(): void {
        const draft = store.draft();
        const letter = store.selectedLetter();
        if (store.mode() !== 'edit' || !draft || !letter || store.saving() || validateWldaDraft(draft, 'edit').length > 0) return;
        patchState(store, { saving: true });
        const names = new Map(store.storedAttachments().map(file => [file.key ?? '', file.name]));

        wldaService
          .updateLetter(letter.prewapSeq, letter.ltrId, toWldaUpdatePayload(draft, letter))
          .pipe(
            switchMap(() =>
              forkJoin({
                removeFailed: deleteStored(draft.removedKeys, names),
                uploadFailed: uploadFiles(draft.files, letter.prewapSeq, letter.ltrId),
              }),
            ),
          )
          .subscribe({
            next: ({ removeFailed, uploadFailed }) => {
              patchState(store, { saving: false, draft: { ...draft, files: [], removedKeys: [] } });
              notify.info(`${wldaCode(letter.ltrId)} updated.`);
              if (removeFailed.length > 0) notify.error(`Could not remove: ${removeFailed.join(', ')}`);
              if (uploadFailed.length > 0) notify.error(`Letter saved, but these files did not upload: ${uploadFailed.join(', ')}`);
              refreshLetter(letter.prewapSeq, letter.ltrId, true);
            },
            error: (err: unknown) => {
              patchState(store, { saving: false });
              notify.error(errorMessage(err, 'Failed to update the WLDA letter'));
            },
          });
      },

      /** Deletes the letter, then its S3 files best-effort (the row is already gone, so a failure only leaves orphans). */
      deleteLetter(): void {
        const letter = store.selectedLetter();
        if (!letter || store.deleting()) return;
        patchState(store, { deleting: true });
        const keys = store.storedAttachments().flatMap(file => (file.key ? [file.key] : []));

        wldaService
          .deleteLetter(letter.prewapSeq, letter.ltrId)
          .pipe(switchMap(() => deleteStored(keys, new Map())))
          .subscribe({
            next: orphaned => {
              const key = wldaKey(letter);
              patchState(store, {
                deleting: false,
                draft: null,
                selectedKey: null,
                storedAttachments: [],
                letters: store.letters().filter(item => wldaKey(item) !== key),
              });
              notify.info(`${wldaCode(letter.ltrId)} deleted.`);
              if (orphaned.length > 0) notify.error(`Letter deleted, but ${orphaned.length} stored file(s) could not be removed.`);
            },
            error: (err: unknown) => {
              patchState(store, { deleting: false });
              notify.error(errorMessage(err, 'Failed to delete the WLDA letter'));
            },
          });
      },
    };
  }),

  withHooks({ onInit: store => store.loadAll() }),
);

import { ChangeDetectionStrategy, Component, ViewEncapsulation, computed, inject, input, output, signal } from '@angular/core';
import { MatIconModule } from '@angular/material/icon';
import { MatSelectChange, MatSelectModule } from '@angular/material/select';
import { MAX_UPLOAD_SIZE_LABEL, isUploadFileAllowed, toAcceptAttribute } from '@core/config/file-upload.config';
import { WldaAttachment, WldaDraft } from '@models/wwell-planning/wlda.model';
import { NotificationService } from '@shared/components/notification/notification.service';
import { WldaFieldErrors } from '../wlda.util';
import { WLDA_WELL_TYPES } from '../wlda.config';

type TextField =
  | 'waterWellName'
  | 'field'
  | 'wellNumber'
  | 'reservoir'
  | 'totalDepth'
  | 'svYutmCord'
  | 'svXutmCord'
  | 'ltrRmks'
  | 'distribution';

/**
 * The WLDA creation form. The pre-well plan a letter is filed under is matched from the typed Field and
 * Well Number (see findPlanForWell); edit mode locks everything the update procedure cannot change.
 */
@Component({
  selector: 'app-wlda-form',
  standalone: true,
  imports: [MatIconModule, MatSelectModule],
  templateUrl: './wlda-form.component.html',
  styleUrl: './wlda-form.component.scss',
  changeDetection: ChangeDetectionStrategy.OnPush,
  // The select's option panel is portaled to <body>, so its styles have to be global.
  encapsulation: ViewEncapsulation.None,
})
export class WldaFormComponent {
  readonly draft = input.required<WldaDraft>();
  /** The values the draft started from; fields that differ are highlighted as unsaved edits. */
  readonly baseline = input<WldaDraft | null>(null);
  /** Validation messages keyed by field; each shows under its own field once that field is touched or filled. */
  readonly errors = input<WldaFieldErrors>({});
  readonly linkOptions = input<readonly { prewapSeq: number; label: string }[]>([]);
  readonly fieldOptions = input<readonly string[]>([]);
  readonly wellNumberOptions = input<readonly string[]>([]);
  readonly mode = input<'create' | 'edit'>('create');
  /** Stored files (minus removed ones) followed by pending picks, as the sheet shows them. */
  readonly attachments = input<readonly WldaAttachment[]>([]);
  readonly patch = output<Partial<WldaDraft>>();
  readonly linkWell = output<number | null>();
  /** Valid picks, handed to the page, which runs each through the upload dialog. */
  readonly filesPicked = output<readonly File[]>();

  private readonly notify = inject(NotificationService);

  protected readonly selectPanelClass = 'wf-select-panel';
  protected readonly attachAccept = toAcceptAttribute();
  protected readonly isCreate = computed(() => this.mode() === 'create');
  protected readonly locked = computed(() => this.mode() === 'edit');

  /** The configured types, plus the draft's own value when a plan supplied one outside the list. */
  protected readonly wellTypes = computed(() => {
    const current = this.draft().wellType;
    return current && !WLDA_WELL_TYPES.includes(current) ? [...WLDA_WELL_TYPES, current] : WLDA_WELL_TYPES;
  });

  /** True when `key` differs from the baseline — an unsaved edit, shown green like the IR editor. */
  protected changed(key: keyof WldaDraft): boolean {
    const base = this.baseline();
    if (!base) return false;
    const draft = this.draft();
    if (key === 'files') return draft.files.length > 0 || draft.removedKeys.length > 0;
    return draft[key] !== base[key];
  }

  private readonly touched = signal<ReadonlySet<keyof WldaDraft>>(new Set());

  /** Marks a field as visited (on blur / dropdown close) so its "required" message can appear. */
  protected touch(key: keyof WldaDraft): void {
    if (!this.touched().has(key)) this.touched.set(new Set([...this.touched(), key]));
  }

  /** The inline message for `key`: shown once the field was visited or already holds a value. */
  protected errorOf(key: keyof WldaDraft): string | null {
    const message = this.errors()[key];
    if (!message) return null;
    const value = this.draft()[key];
    const filled = typeof value === 'string' ? value.trim() !== '' : value !== null;
    return this.touched().has(key) || filled ? message : null;
  }

  protected isRequired(): boolean {
    return this.isCreate();
  }

  protected setText(key: TextField, event: Event): void {
    this.patch.emit({ [key]: (event.target as HTMLInputElement | HTMLTextAreaElement).value });
  }

  protected setLink(change: MatSelectChange): void {
    this.linkWell.emit(change.value === '' ? null : (change.value as number));
  }

  protected setWellType(change: MatSelectChange): void {
    this.patch.emit({ wellType: change.value as string });
  }

  protected setCathodic(required: boolean): void {
    this.patch.emit({ cathodicProtection: required });
  }

  protected addFiles(event: Event): void {
    const input = event.target as HTMLInputElement;
    const selected = Array.from(input.files ?? []);
    input.value = '';
    // `accept` is only a picker hint, so the type and size are re-checked here.
    const rejected = selected.filter(file => !isUploadFileAllowed(file));
    if (rejected.length > 0) {
      this.notify.error(`Skipped (unsupported type or over ${MAX_UPLOAD_SIZE_LABEL}): ${rejected.map(file => file.name).join(', ')}`);
    }
    const accepted = selected.filter(file => isUploadFileAllowed(file));
    if (accepted.length > 0) this.filesPicked.emit(accepted);
  }

  protected removeAttachment(attachment: WldaAttachment): void {
    const draft = this.draft();
    if (attachment.key) {
      this.patch.emit({ removedKeys: [...draft.removedKeys, attachment.key] });
    } else {
      this.patch.emit({ files: draft.files.filter((_, index) => index !== attachment.pendingIndex) });
    }
  }
}

  /** The draft's not-yet-uploaded picks, indexed like `WldaAttachment.pendingIndex`, so they can be opened too. */

  private readonly dialog = inject(MatDialog);
  private readonly notify = inject(NotificationService);
  private readonly destroyRef = inject(DestroyRef);

  /** Attachment (object key) currently being fetched for the viewer — disables just that row's button. */
  protected readonly openingName = signal<string | null>(null);
  /** Object-URL thumbnails of stored image attachments (PDFs keep their icon), keyed by object key. */
  protected readonly thumbnails = signal<Readonly<Record<string, string>>>({});
  private readonly requestedThumbnails = new Set<string>();

  /** Width the sheet was designed for; wider panes scale it up (capped) instead of leaving it a small centred card. */
  private static readonly BASE_WIDTH_PX = 920;
  private static readonly MAX_ZOOM = 3;
  private readonly hostWidth = signal(0);

    effect(() => {
      for (const file of this.view().attachments) {
        if (!file.key || !IMAGE_NAME.test(file.name) || this.requestedThumbnails.has(file.key)) continue;
        const key = file.key;
        this.requestedThumbnails.add(key);
        this.s3
          .download(key, file.name)
          .pipe(takeUntilDestroyed(this.destroyRef))
          .subscribe({
            next: blob => this.thumbnails.update(current => ({ ...current, [key]: URL.createObjectURL(blob) })),
            // A missing thumbnail just leaves the icon; the View button reports real failures.
            error: () => undefined,
          });
      }
    });
    this.destroyRef.onDestroy(() => Object.values(this.thumbnails()).forEach(url => URL.revokeObjectURL(url)));

    if (typeof ResizeObserver === 'undefined') return;
    const observer = new ResizeObserver(([entry]) => this.hostWidth.set(entry.contentRect.width));
    observer.observe(inject<ElementRef<HTMLElement>>(ElementRef).nativeElement);
    inject(DestroyRef).onDestroy(() => observer.disconnect());

  protected readonly formNumber = WLDA_FORM_NUMBER;
  protected readonly toLine = WLDA_TO_LINE;

      panelClass: 'ir-viewer-panel',
      width: '92vw',
      maxWidth: '1500px',
      height: '92vh',
      autoFocus: false,
    });
    if (!attachment.key) {
      const local = attachment.pendingIndex === null ? undefined : this.pendingFiles()[attachment.pendingIndex];
      if (local) open(local);
      return;
    }
    this.openingName.set(attachment.key);
    this.s3
      .download(attachment.key, attachment.name)
      .pipe(takeUntilDestroyed(this.destroyRef))
      .subscribe({
        next: file => {
          this.openingName.set(null);
          open(file);
        },
        error: () => {
          this.openingName.set(null);
          this.notify.error(`Failed to open ${attachment.name}.`);
        },
      });
  }

  protected orDash(value: string): string {
    return value.trim() || '—';
  }
}

import { filter, merge } from 'rxjs';
import { takeUntilDestroyed } from '@angular/core/rxjs-interop';
import { ChangeDetectionStrategy, Component, DestroyRef, effect, inject, signal } from '@angular/core';
import { MatButtonModule } from '@angular/material/button';
import { MAT_DIALOG_DATA, MatDialogModule, MatDialogRef } from '@angular/material/dialog';
import { MatIconModule } from '@angular/material/icon';
import { MatProgressBarModule } from '@angular/material/progress-bar';
import { DomSanitizer, SafeUrl } from '@angular/platform-browser';
import { WldaStore } from '../store/wlda.store';

export interface WldaUploadDialogData {
  /** Passed in rather than injected, like the IR dialog: the store may be provided below the root injector. */
  readonly store: InstanceType<typeof WldaStore>;
  readonly file: File;
  /** True for a saved letter (uploads now); false while creating, where the file waits for Create. */
  readonly canUpload: boolean;
}

/** How long the success state stays up before the dialog closes itself. */
const SUCCESS_LINGER_MS = 900;

/**
 * Opens right after a file is picked: previews it, then either uploads it with live progress (saved letter)
 * or stages it for the Create click (new letter). Cancel, Esc and backdrop discard the pick. Same look and
 * behaviour as the IR attachment dialog, whose stylesheet it shares.
 */
@Component({
  selector: 'app-wlda-upload-dialog',
  standalone: true,
  imports: [MatButtonModule, MatDialogModule, MatIconModule, MatProgressBarModule],
  templateUrl: './wlda-upload-dialog.component.html',
  styleUrl: '../../wwell-ir/wwell-ir-attachment-upload-dialog/wwell-ir-attachment-upload-dialog.component.scss',
  changeDetection: ChangeDetectionStrategy.OnPush,
})
export class WldaUploadDialogComponent {
  protected readonly data = inject<WldaUploadDialogData>(MAT_DIALOG_DATA);
  private readonly dialogRef = inject(MatDialogRef<WldaUploadDialogComponent, boolean>);
  private readonly sanitizer = inject(DomSanitizer);
  private readonly store = this.data.store;

  protected readonly file = this.data.file;
  protected readonly isImage = this.file.type.startsWith('image/');
  protected readonly isPdf = this.file.type === 'application/pdf';
  protected readonly sizeLabel = formatBytes(this.file.size);
  protected readonly typeLabel = this.file.name.split('.').pop()?.toUpperCase() ?? 'FILE';

  private readonly objectUrl = this.isImage ? URL.createObjectURL(this.file) : null;
  protected readonly previewUrl: SafeUrl | null = this.objectUrl ? this.sanitizer.bypassSecurityTrustUrl(this.objectUrl) : null;

  protected readonly canUpload = this.data.canUpload;
  protected readonly busy = this.store.uploadingFile;
  protected readonly percent = this.store.uploadPercent;
  protected readonly error = this.store.uploadError;
  protected readonly done = signal(false);
  private readonly started = signal(false);

  constructor() {
    inject(DestroyRef).onDestroy(() => {
      if (this.objectUrl) URL.revokeObjectURL(this.objectUrl);
    });

    this.dialogRef.disableClose = true;
    merge(this.dialogRef.backdropClick(), this.dialogRef.keydownEvents().pipe(filter(event => event.key === 'Escape')))
      .pipe(takeUntilDestroyed())
      .subscribe(() => this.cancel());

    // The store clears `uploadingFile` once the proxy answers; no error by then means the file is stored.
    effect(() => {
      if (!this.started() || this.busy() || this.error() || this.done()) return;
      this.done.set(true);
      setTimeout(() => this.dialogRef.close(true), SUCCESS_LINGER_MS);
    });
  }

  protected cancel(): void {
    if (this.busy() || this.done()) return;
    this.dialogRef.close(false);
  }

  /** New letter: keep the file for the Create click (the parent adds it to the draft on `true`). */
  protected attachForLater(): void {
    this.dialogRef.close(true);
  }

  protected upload(): void {
    if (this.busy() || this.done()) return;
    this.started.set(true);
    this.store.uploadLetterFile(this.file);
  }
}

function formatBytes(bytes: number): string {
  if (bytes < 1024) return `${bytes} B`;
  if (bytes < 1024 * 1024) return `${(bytes / 1024).toFixed(1)} KB`;
  return `${(bytes / (1024 * 1024)).toFixed(1)} MB`;
}

import { WldaStore } from './store/wlda.store';
import { WldaUploadDialogComponent } from './wlda-upload-dialog/wlda-upload-dialog.component';
import { WldaFormComponent } from './wlda-form/wlda-form.component';

import { WLDA_STATUS_META } from './wlda.config';
import { downloadWldaPdf } from './wlda-pdf.util';
import { WLDA_EMPTY_DRAFT, draftOfLetter, statusOfLetter, viewOfDraft, viewOfLetter, wldaCode, wldaKey, wldaWellName } from './wlda.util';

@Component({
  selector: 'app-wlda',
  standalone: true,
  imports: [FormsModule, MatButtonModule, MatIconModule, WldaFormComponent, WldaSheetComponent],
  templateUrl: './wlda.component.html',
  styleUrl: './wlda.component.scss',
  changeDetection: ChangeDetectionStrategy.OnPush,
})
export class WldaComponent {
  protected readonly store = inject(WldaStore);
  private readonly confirmDialog = inject(ActionConfirmDialogService);
  private readonly notify = inject(NotificationService);
  private readonly dialog = inject(MatDialog);

  protected readonly searchTerm = signal('');
  protected readonly downloading = signal(false);
  protected readonly statusMeta = WLDA_STATUS_META;

  protected readonly rows = computed(() => {
    const term = this.searchTerm().trim().toLowerCase();
    const letters = this.store.letters();
    if (!term) return letters;
    return letters.filter(letter => [wldaCode(letter.ltrId), wldaWellName(letter)].some(value => value.toLowerCase().includes(term)));
  });

  /** What the sheet shows: the draft while creating, the saved letter (with draft overrides) otherwise. */
  protected readonly view = computed<WldaLetterView | null>(() => {
    const draft = this.store.draft();
    const letter = this.store.selectedLetter();
    if (letter) return viewOfLetter(letter, this.store.storedAttachments(), draft);
    return draft ? viewOfDraft(draft, this.store.draftPlan()) : null;
  });

  /** What "changed" is measured against: the saved letter while editing, a blank draft while creating. */
  protected readonly baseline = computed(() => {
    const letter = this.store.selectedLetter();
    return letter ? draftOfLetter(letter) : WLDA_EMPTY_DRAFT;
  });

  /** Options of the "Link to Water Well" dropdown: every pre-well plan, labelled like the IR picker. */
  protected readonly linkOptions = computed(() =>
    this.store.prewaps().map(plan => ({ prewapSeq: plan.prewapSeq, label: wldaWellName(plan) || `Plan #${plan.prewapSeq}` })),
  );

  /** Known plans' field codes / well numbers, offered as suggestions so a typo can't leave Create disabled. */
  protected readonly fieldOptions = computed(() => [...new Set(this.store.prewaps().map(plan => (plan.wPrimHid ?? '').trim()).filter(Boolean))]);
  protected readonly wellNumberOptions = computed(() => {
    const field = this.store.draft()?.field.trim().toLowerCase() ?? '';
    return this.store
      .prewaps()
      .filter(plan => !field || (plan.wPrimHid ?? '').trim().toLowerCase() === field)
      .map(plan => (plan.wNum ?? '').trim())
      .filter(Boolean);
  });

  /** Set by the Back arrow so the auto-select below doesn't immediately reopen what was just closed. */
  private readonly deselected = signal(false);

  constructor() {
    // Like the IR page, opening WLDA lands on the first letter instead of an empty detail pane.
    effect(() => {
      if (this.deselected() || this.store.draft() || this.store.selectedKey() !== null) return;
      const first = this.rows()[0];
      if (first) untracked(() => this.store.select(wldaKey(first)));
    });
  }

  protected readonly canSave = computed(() => this.store.draftErrors().length === 0 && !this.store.saving());

  protected code(letter: WldaLetter): string {
    return wldaCode(letter.ltrId);
  }

  protected wellLabel(letter: WldaLetter): string {
    return wldaWellName(letter) || 'Unnamed well';
  }

  protected status(letter: WldaLetter) {
    return this.statusMeta[statusOfLetter(letter)];
  }

  protected isActive(letter: WldaLetter): boolean {
    return wldaKey(letter) === this.store.selectedKey();
  }

  protected select(letter: WldaLetter): void {
    this.guardUnsaved(() => {
      this.deselected.set(false);
      this.store.select(wldaKey(letter));
    });
  }

  protected newDraft(): void {
    this.guardUnsaved(() => {
      this.deselected.set(false);
      this.store.startDraft();
    });
  }

  /** Back / View: leaves the draft (create → nothing open, edit → the saved letter), confirming first if it changed. */
  protected leaveDraft(): void {
    this.guardUnsaved(() => this.store.discardDraft());
  }

  protected back(): void {
    this.guardUnsaved(() => {
      this.deselected.set(true);
      this.store.select(null);
    });
  }

  protected save(): void {
    if (this.store.mode() === 'create') this.store.createLetter();
    else this.store.updateLetter();
  }

  /** Downloads the letter exactly as the sheet shows it (an unsaved draft downloads as a draft). */
  protected async download(view: WldaLetterView): Promise<void> {
    if (this.downloading()) return;
    this.downloading.set(true);
    try {
      await downloadWldaPdf(view, `${view.status === 'draft' ? 'WLDA-draft' : view.code}.pdf`);
    } catch {
      this.notify.error('Could not generate the WLDA PDF.');
    } finally {
      this.downloading.set(false);
    }
  }

  /** Each picked file gets its own upload dialog, one after another: a saved letter uploads now, a new one stages the file for Create. */
  protected uploadPicked(files: readonly File[]): void {
    const [file, ...rest] = files;
    if (!file) return;
    const canUpload = this.store.mode() === 'edit';
    this.store.resetUpload();
    this.dialog
      .open<WldaUploadDialogComponent, unknown, boolean>(WldaUploadDialogComponent, {
        data: { store: this.store, file, canUpload },
        panelClass: 'ir-upload-panel',
        autoFocus: false,
      })
      .afterClosed()
      .subscribe(accepted => {
        if (accepted && !canUpload) {
          const draft = this.store.draft();
          if (draft) this.store.patchDraft({ files: [...draft.files, file] });
        }
        this.uploadPicked(rest);
      });
  }

  protected confirmDelete(): void {
    const letter = this.store.selectedLetter();
    if (!letter) return;
    this.confirmDialog
      .open({
        title: 'Delete this WLDA letter?',
        rows: [
          { label: 'Letter', value: wldaCode(letter.ltrId) },
          { label: 'Well', value: wldaWellName(letter) || '—' },
        ],
        confirmLabel: 'Delete',
        severity: 'danger',
      })
      .subscribe(confirmed => {
        if (confirmed) this.store.deleteLetter();
      });
  }

  /** Runs `proceed` straight away unless an open draft has unsaved changes, which need a Discard first. */
  private guardUnsaved(proceed: () => void): void {
    if (!this.store.draft() || !this.store.isDirty()) {
      proceed();
      return;
    }
    this.confirmDialog
      .open({
        title: 'Discard unsaved changes?',
        confirmLabel: 'Discard',
        severity: 'warning',
        hint: 'Your changes to this letter have not been saved.',
      })
      .subscribe(confirmed => {
        if (confirmed) proceed();
      });
  }
}

import { HttpClient } from '@angular/common/http';
import { Injectable, inject } from '@angular/core';
import { Observable, map } from 'rxjs';
import { WldaCreatePayload, WldaCreateResult, WldaLetter, WldaPrewap, WldaUpdatePayload } from '@models/wwell-planning/wlda.model';
import { IrEligibleWell } from '../wwell-ir/wwell-ir-form/wwell-ir-payload.model';
import { ApiResponse } from 'src/app/shared/models/wwell/api-response.model';
import { ExternalConfigService } from 'src/app/shared/services/external-config.service';

/** agwa-planning's WLDA letter endpoints. Attachments are not here — they go to the S3 service. */
@Injectable({ providedIn: 'root' })
export class WldaService {
  private readonly http = inject(HttpClient);
  private readonly extConfigService = inject(ExternalConfigService);

  private get apiUrl(): string {
    const serverUrl = this.extConfigService.settings.planningServiceUrl ?? '';
    return `${serverUrl}/planning/api/v1`;
  }

  /**
   * The wells the "Link to Water Well" dropdown offers: the same `GET /wells/eligible-for-ir` list the IR wizard's
   * "Link to Request" picker uses, reshaped into the plan fields a letter is created from. That endpoint has no
   * GOSP column, so GOSP stays blank.
   */
  getPrewaps(): Observable<WldaPrewap[]> {
    return this.http.get<ApiResponse<IrEligibleWell>>(`${this.apiUrl}/wells/eligible-for-ir`).pipe(map(res => res.data.map(toPrewap)));
  }

  /** Every letter of every plan. */
  getLetters(): Observable<WldaLetter[]> {
    return this.http.get<ApiResponse<WldaLetter>>(`${this.apiUrl}/wlda-letters`).pipe(map(res => res.data));
  }

  /** One letter (get_wlda_letters_prc with both prewap and letter id), fresh from the database. */
  getLetter(prewapSeq: number, ltrId: number): Observable<WldaLetter> {
    return this.http
      .get<ApiResponse<WldaLetter>>(`${this.apiUrl}/prewaps/${prewapSeq}/wlda-letters/${ltrId}`)
      .pipe(map(res => res.data[0]));
  }

  createLetter(prewapSeq: number, payload: WldaCreatePayload): Observable<WldaCreateResult> {
    return this.http
      .post<ApiResponse<WldaCreateResult>>(`${this.apiUrl}/prewaps/${prewapSeq}/wlda-letters`, payload)
      .pipe(map(res => res.data[0]));
  }

  updateLetter(prewapSeq: number, ltrId: number, payload: WldaUpdatePayload): Observable<WldaCreateResult> {
    return this.http
      .put<ApiResponse<WldaCreateResult>>(`${this.apiUrl}/prewaps/${prewapSeq}/wlda-letters/${ltrId}`, payload)
      .pipe(map(res => res.data[0]));
  }

  deleteLetter(prewapSeq: number, ltrId: number): Observable<void> {
    return this.http.delete<void>(`${this.apiUrl}/prewaps/${prewapSeq}/wlda-letters/${ltrId}`);
  }
}

function toPrewap(well: IrEligibleWell): WldaPrewap {
  const zone = Number.parseInt(well.utmZone ?? '', 10);
  return {
    prewapSeq: well.prewapSeq,
    wPrimHid: well.primaryHoleId,
    wNum: well.wellNumber,
    wellType: well.wellType,
    tdMd: well.targetDepth,
    targetForm: well.targetAquifer,
    gospAreaPlant: null,
    svUtmZn: Number.isNaN(zone) ? null : zone,
    surfaceNorth: well.surfaceNorth,
    surfaceEast: well.surfaceEast,
  };
}
