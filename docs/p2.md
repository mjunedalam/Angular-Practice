  ChangeDetectionStrategy,
  Component,
  ElementRef,
  OnDestroy,
  QueryList,
  ViewChildren,
  computed,
  input,
} from '@angular/core';
import { AbstractControl, FormArray, FormControl, FormGroup } from '@angular/forms';
import { Subscription } from 'rxjs';
import { DynamicFieldComponent, WwellDynamicFieldMode } from '../dynamic-field/dynamic-field.component';
import { playIfMotionAllowed } from '../gsap-motion.util';
import { RepeatableGroupComponent } from '../repeatable-group/repeatable-group.component';
import { defaultValueFor } from '../wwell-dynamic-form.builder';
import { WwellFieldConfig, WwellFormSectionConfig } from '../wwell-dynamic-form.types';
import {
  buildConditionalFieldEnterTimeline,
  buildConditionalFieldExitTimeline,
  setConditionalFieldVisibility,
} from './conditional-field.animation';

@Component({
  selector: 'app-dynamic-section',
  standalone: true,
  imports: [DynamicFieldComponent, RepeatableGroupComponent],
  templateUrl: './dynamic-section.component.html',
  styleUrl: './dynamic-section.component.scss',
  changeDetection: ChangeDetectionStrategy.OnPush,
})
export class DynamicSectionComponent implements AfterViewInit, OnDestroy {
  readonly section = input.required<WwellFormSectionConfig>();
  readonly control = input.required<AbstractControl>();
  readonly mode = input<WwellDynamicFieldMode>('edit');
  readonly selectPanelClass = input('');

  @ViewChildren('fieldHost') private readonly fieldHosts?: QueryList<ElementRef<HTMLElement>>;

  /** Only fields with `visibleWhen` grow a `#fieldHost` wrapper — this stays index-aligned with that query list. */
  protected readonly conditionalFields = computed(() => (this.section().fields ?? []).filter(field => field.visibleWhen));

  /** Tracks last-applied visibility per field key so we only animate on an actual flip, not every keystroke. */
  private readonly lastVisibility = new Map<string, boolean>();
  private valueChangesSubscription?: Subscription;

  ngAfterViewInit(): void {
    this.syncConditionalFields();
    this.valueChangesSubscription = this.control().valueChanges.subscribe(() => this.syncConditionalFields());
  }

  ngOnDestroy(): void {
    this.valueChangesSubscription?.unsubscribe();
  }

  protected fieldControl(key: string): FormControl {
    return (this.control() as FormGroup).get(key) as FormControl;
  }

  protected asFormArray(): FormArray {
    return this.control() as FormArray;
  }

  private syncConditionalFields(): void {
    this.conditionalFields().forEach((field, index) => {
      const host = this.fieldHosts?.get(index)?.nativeElement;
      if (host) this.applyFieldVisibility(field, host);
    });
  }

  private applyFieldVisibility(field: WwellFieldConfig, host: HTMLElement): void {
    const isVisible = this.isFieldVisible(field);
    const seenBefore = this.lastVisibility.has(field.key);
    const wasVisible = this.lastVisibility.get(field.key);
    this.lastVisibility.set(field.key, isVisible);

    if (!seenBefore) {
      setConditionalFieldVisibility(host, isVisible);
      this.syncControlEnabled(field, isVisible);
      return;
    }
    if (isVisible === wasVisible) return;

    if (isVisible) {
      this.syncControlEnabled(field, true);
      playIfMotionAllowed(buildConditionalFieldEnterTimeline(host));
    } else {
      playIfMotionAllowed(buildConditionalFieldExitTimeline(host, () => this.syncControlEnabled(field, false)));
    }
  }

  private isFieldVisible(field: WwellFieldConfig): boolean {
    const condition = field.visibleWhen;
    if (!condition) return true;

    const sibling = (this.control() as FormGroup).get(condition.field);
    const value = sibling?.value;
    if (condition.equals !== undefined) return value === condition.equals;
    if (condition.oneOf) return condition.oneOf.includes(value);
    if (condition.notOneOf) return !condition.notOneOf.includes(value);
    if (condition.includesLine !== undefined) {
      return typeof value === 'string' && value.split('\n').some(line => line.trim() === condition.includesLine);
    }
    return true;
  }

  /** Hidden fields are disabled (excluded from validity/`.value`) and reset so a stale value from the other mode never submits. */
  private syncControlEnabled(field: WwellFieldConfig, isVisible: boolean): void {
    const control = this.fieldControl(field.key);
    if (isVisible) {
      if (control.disabled) control.enable({ emitEvent: false });
      return;
    }
    if (control.enabled) {
      control.reset(defaultValueFor(field), { emitEvent: false });
      control.disable({ emitEvent: false });
    }
  }
}

  AfterViewInit,
  ChangeDetectionStrategy,
  ChangeDetectorRef,
  Component,
  ElementRef,
  OnDestroy,
  QueryList,
  ViewChildren,
  computed,
  effect,
  inject,
  input,
  signal,
} from '@angular/core';
import { FormArray, FormBuilder, FormControl, FormGroup } from '@angular/forms';
import { MatButtonModule } from '@angular/material/button';
import { MatIconModule } from '@angular/material/icon';
import { Subscription } from 'rxjs';
import {
  buildConditionalFieldEnterTimeline,
  buildConditionalFieldExitTimeline,
  setConditionalFieldVisibility,
} from '../dynamic-section/conditional-field.animation';
import { DynamicFieldComponent, WwellDynamicFieldMode } from '../dynamic-field/dynamic-field.component';
import { playIfMotionAllowed } from '../gsap-motion.util';
import { addRepeatableRow, defaultValueFor } from '../wwell-dynamic-form.builder';
import { WwellFieldConfig, WwellRepeatableRowConfig } from '../wwell-dynamic-form.types';
import { buildRowEnterTimeline, buildRowExitTimeline } from './repeatable-row.animation';

@Component({
  selector: 'app-repeatable-group',
  standalone: true,
  imports: [DynamicFieldComponent, MatButtonModule, MatIconModule],
  templateUrl: './repeatable-group.component.html',
  styleUrl: './repeatable-group.component.scss',
  changeDetection: ChangeDetectionStrategy.OnPush,
})
export class RepeatableGroupComponent implements AfterViewInit, OnDestroy {
  private readonly fb = inject(FormBuilder);
  private readonly cdr = inject(ChangeDetectorRef);

  readonly config = input.required<WwellRepeatableRowConfig>();
  readonly formArray = input.required<FormArray>();
  readonly mode = input<WwellDynamicFieldMode>('edit');
  readonly selectPanelClass = input('');

  @ViewChildren('rowHost') private readonly rowHosts?: QueryList<ElementRef<HTMLElement>>;

  /**
   * Row-level `visibleWhen` support — `app-dynamic-section` implements this for flat
   * `section().fields`, but never for `section().repeatable` rows, so it's ported here
   * rather than duplicated per-feature. Each row gets its own independent visibility
   * state, resolved against that row's own `FormGroup` (not the whole array), because a
   * repeatable row's condition (e.g. "this row's Type is Liner") only ever depends on a
   * sibling field within the *same* row.
   */
  @ViewChildren('rowFieldHost') private readonly rowFieldHosts?: QueryList<ElementRef<HTMLElement>>;

  /** Only fields with `visibleWhen` grow a `#rowFieldHost` wrapper — the template renders
   * exactly this many per row, in this order, so `row index * length + field index` maps
   * a (row, field) pair onto the flat, document-order `rowFieldHosts` query list. */
  protected readonly conditionalRowFields = computed(() => this.config().rowFields.filter(field => field.visibleWhen));

  /** Bumped only on user-triggered `addRow()` so the effect below can play an entrance animation for it. */
  private readonly addedRowTick = signal(0);

  /** Tracks last-applied visibility per (row, field key) so we only animate on an actual flip. Keyed by row
   * object identity (not index) so it survives rows being added/removed/reordered without misattribution. */
  private readonly lastVisibility = new Map<FormGroup, Map<string, boolean>>();
  private valueChangesSubscription?: Subscription;
  private rowFieldHostsSubscription?: Subscription;

  constructor() {
    effect(() => {
      const tick = this.addedRowTick();
      if (tick === 0) return;
      const last = this.rowHosts?.last;
      if (last) playIfMotionAllowed(buildRowEnterTimeline(last.nativeElement));
    });
  }

  ngAfterViewInit(): void {
    this.syncConditionalRowFields();
    this.valueChangesSubscription = this.formArray().valueChanges.subscribe(() => this.onFormArrayChanged());
    // Re-sync whenever the query list's elements change (row added/removed) — a freshly
    // added row's #rowFieldHost isn't queryable until after this fires.
    this.rowFieldHostsSubscription = this.rowFieldHosts?.changes.subscribe(() => this.syncConditionalRowFields());
    // See `onFormArrayChanged`'s doc comment — the very first render can be just as stale as
    // any later one, for the same reason, so it needs the same forced re-check.
    this.cdr.detectChanges();
  }

  ngOnDestroy(): void {
    this.valueChangesSubscription?.unsubscribe();
    this.rowFieldHostsSubscription?.unsubscribe();
  }

  protected rows(): FormGroup[] {
    return this.formArray().controls as FormGroup[];
  }

  /**
   * Pushing/removing a row (or editing any field in one) mutates the `formArray` input's own
   * `FormArray` object in place — the input's own signal reference never changes, so Angular's
   * standard OnPush triggers (changed input, event sourced from this component's own template,
   * a *directly template-read* signal changing) never fire for a `RepeatableGroupComponent`
   * instance that didn't cause the mutation itself. That's invisible for the normal "edit"
   * instance (its own "+ Add"/remove buttons are template-sourced click events, so zone.js
   * re-checks it for free) but left every *other* instance of the same section — concretely,
   * the read-only copy the Attachments & Review step renders for the same FormArray — stuck
   * forever showing whatever row count existed when it first mounted. Confirmed live: filling
   * in an Offset Water Well and opening the review step showed a completely empty section,
   * with the underlying FormArray provably holding the right row the whole time. Forcing a
   * synchronous `detectChanges()` here — rather than the usual `markForCheck()` — is deliberate:
   * this component sits inside a `mat-expansion-panel`, and Material detaches a collapsed
   * panel's content from the ambient change-detection walk, so merely flagging "needs a check"
   * isn't enough when nothing ever walks back down to it on its own.
   */
  private onFormArrayChanged(): void {
    this.syncConditionalRowFields();
    this.cdr.detectChanges();
  }

  protected fieldControl(row: FormGroup, key: string): FormControl {
    return row.get(key) as FormControl;
  }

  protected addRow(): void {
    addRepeatableRow(this.formArray(), this.config().rowFields, this.fb);
    this.addedRowTick.update(tick => tick + 1);
  }

  protected removeRow(index: number): void {
    const row = this.rows()[index];
    const host = this.rowHosts?.get(index);
    if (!host) {
      this.formArray().removeAt(index);
      this.lastVisibility.delete(row);
      return;
    }

    playIfMotionAllowed(buildRowExitTimeline(host.nativeElement, () => {
      this.formArray().removeAt(index);
      this.lastVisibility.delete(row);
    }));
  }

  private syncConditionalRowFields(): void {
    const conditionalFields = this.conditionalRowFields();
    if (!conditionalFields.length) return;

    this.rows().forEach((row, rowIndex) => {
      conditionalFields.forEach((field, fieldIndex) => {
        const host = this.rowFieldHosts?.get(rowIndex * conditionalFields.length + fieldIndex)?.nativeElement;
        if (host) this.applyFieldVisibility(row, field, host);
      });
    });
  }

  private applyFieldVisibility(row: FormGroup, field: WwellFieldConfig, host: HTMLElement): void {
    const rowVisibility = this.lastVisibility.get(row) ?? new Map<string, boolean>();
    this.lastVisibility.set(row, rowVisibility);

    const isVisible = this.isFieldVisible(row, field);
    const seenBefore = rowVisibility.has(field.key);
    const wasVisible = rowVisibility.get(field.key);
    rowVisibility.set(field.key, isVisible);

    if (!seenBefore) {
      setConditionalFieldVisibility(host, isVisible);
      this.syncControlEnabled(row, field, isVisible);
      return;

    if (isVisible === wasVisible) return;

    if (isVisible) {
      this.syncControlEnabled(row, field, true);
      playIfMotionAllowed(buildConditionalFieldEnterTimeline(host));
    } else {
      playIfMotionAllowed(buildConditionalFieldExitTimeline(host, () => this.syncControlEnabled(row, field, false)));
    }
  }

  private isFieldVisible(row: FormGroup, field: WwellFieldConfig): boolean {
    const condition = field.visibleWhen;
    if (!condition) return true;

    const sibling = row.get(condition.field);
    const value = sibling?.value;

export type WwellFieldControlType =

  | 'number'
  | 'textarea'
  | 'select'
  | 'combo'
  | 'rich-textarea'
  | 'toggle-group'
  | 'file';

  readonly label: string;

  readonly icon?: string;

  readonly required?: boolean;
  readonly pattern?: RegExp;
  readonly min?: number;
}

/**
 * Field is only shown (and its control enabled) while a sibling control in the same section
 * matches this condition. Exactly one of `equals` | `oneOf` | `notOneOf` | `includesLine` should
 * be set:
 *  - `equals`: sibling value === this value
 *  - `oneOf`: sibling value is one of this set (e.g. a client tier)
 *  - `notOneOf`: sibling value is absent from this set — the fallback for values outside every
 *    named tier (e.g. a client not yet assigned to a tier keeps the default field set)
 *  - `includesLine`: sibling value, split on `\n` and trimmed, includes this exact line — for a
 *    sibling that's a `type: 'rich-textarea'` checkbox list (see WwellRichTextareaComponent),
 *    whose committed value is every checked option newline-joined rather than a single picked
 *    value, so `equals` can only ever match when exactly one box is checked. `openHoleSizeIn`
 *    (wwell-ir-form.schema.ts) is the motivating case: gated on `completion` (a rich-textarea)
 *    having "Open Hole" checked, which must still hold when other options are checked alongside
 *    it — an `equals` gate there hid (and reset — see DynamicSectionComponent.syncControlEnabled)
 *    this field the instant a second option was checked, silently discarding whatever open hole
 *    size had been typed and leaving `resolveOhHalfWidth` (wellbore-depth.helpers.ts) with no real
 *    diameter to scale Open Hole against — it fell back to a fixed wide default, wider than an
 *    actual entered casing OD.
 */
export interface WwellFieldVisibilityCondition {
  readonly field: string;
  readonly equals?: string;
  readonly oneOf?: readonly string[];
  readonly notOneOf?: readonly string[];
  readonly includesLine?: string;
}

export interface WwellFieldConfig {
  readonly key: string;
  readonly label: string;
  readonly type: WwellFieldControlType;
  readonly placeholder?: string;
  readonly hint?: string;
  readonly suffix?: string;
  /** Used by 'select' | 'combo' | 'toggle-group'. Plain strings are shown as their own label.
   *  A Signal is resolved lazily via `resolveFieldOptions` — for options that come from a store
   *  (e.g. the IR wizard's "Link to Request" picker) rather than being known at schema-definition
   *  time (the schema array itself is a static, module-level constant with no injector). */
  readonly options?: readonly (string | WwellFieldOption)[] | Signal<readonly (string | WwellFieldOption)[]>;
  readonly validators?: WwellFieldValidatorConfig;
  readonly colSpan?: 1 | 2;
  readonly visibleWhen?: WwellFieldVisibilityCondition;
  /** Overrides the type-based empty default (e.g. pre-selecting a toggle-group option). */
  readonly defaultValue?: string | number;
  /** 'select' only — adds a filter input inside the option panel, narrowing `options` as the user
   *  types. For a picker backed by a long or live-fetched list (e.g. the IR wizard's "Link to

  readonly searchable?: boolean;
  /** Disables the control — for values derived elsewhere (e.g. a server-computed metric) rather

export class WwellIrDepthScaleComponent {
  readonly totalDepth = input.required<number>();
  readonly animTrigger = input.required<number>();
  readonly diagramLayout = input<DiagramLayout>(DIAGRAM_LAYOUT);

  private readonly svgRef = viewChild<ElementRef<SVGSVGElement>>('scaleSvg');

  private get layout(): DiagramLayout { return this.diagramLayout(); }

  constructor() {
    effect(() => {
      const svgEl = this.svgRef();
      const depth = this.totalDepth();
      this.animTrigger(); // track trigger so we redraw on every diagram rebuild
      this.diagramLayout(); // track layout changes

      if (svgEl?.nativeElement) {
        this.drawScale(depth, svgEl.nativeElement);
      }
    });
  }

  private drawScale(totalDepth: number, element: SVGSVGElement): void {
    const { depthScaleWidth, svgHeight, marginTop, drawingHeight, depthAxisX } = this.layout;
    const tickHalfLength = 10;
    const tickLabelGap = 6;
    const tickLeftX = depthAxisX - tickHalfLength;
    const tickRightX = depthAxisX + tickHalfLength;
    const tickLabelX = tickLeftX - tickLabelGap;

    const svg = select(element);
    // Interrupt any in-progress/scheduled transitions before tearing down the DOM — this
    // effect re-fires on every live edit that moves `totalDepth` (e.g. switching Completion
    // while Program Details is open re-derives `wellboreDiagramData().displayDepth` on every
    // keystroke/selection change), which used to call `.remove()` on nodes that still had a
    // pending/active D3 transition scheduled on them (see d3-transition's per-node
    // `__transition` slot). That stale scheduled tick/tween callback still fires later against
    // the now-detached node — wasted work at best, and confirmed (via a scratch harness that
    // rapidly re-triggered this effect) to throw when it fires mid-rebuild. Mirrors the
    // `interrupt()`-before-rebuild guard `well-bore-view.component.ts`'s `redraw()` already
    // uses for the exact same "live data keeps changing faster than the animation" reason.
    svg.selectAll('*').interrupt();
    svg.selectAll('*').remove();

    svg
      .attr('width', '100%')
      .attr('height', '100%')
      .attr('viewBox', `0 0 ${depthScaleWidth} ${svgHeight}`)
      .attr('preserveAspectRatio', 'none');

    const g = svg.append('g').attr('transform', `translate(0,${marginTop})`);
    const scale = createDepthScale(totalDepth, drawingHeight);
    const interval = pickTickInterval(totalDepth);
    const ticks = buildDepthTicks(totalDepth, interval);

    const titleX = depthScaleWidth / 2;
    g.append('text')
      .attr('class', 'axis-title')
      .attr('x', titleX)
      .attr('y', -42)
      .attr('text-anchor', 'middle')
      .style('opacity', 0)
      .call((t) => {
        t.append('tspan').attr('x', titleX).attr('dy', 0).text('Depth');
        t.append('tspan').attr('x', titleX).attr('dy', 20).text('(ft bgl)');
      })
      .transition()
      .delay(80)
      .duration(360)
      .ease(easeCubicOut)
      .attr('y', -36)
      .style('opacity', 1);

    const scaleTipY = scale(totalDepth); // === drawingHeight
    const lineLen = scaleTipY;

    g.append('line')
      .attr('class', 'axis-line')
      .attr('x1', depthAxisX).attr('x2', depthAxisX)
      .attr('y1', 0).attr('y2', scaleTipY)
      .attr('stroke-dasharray', lineLen)
      .attr('stroke-dashoffset', lineLen)
      .transition()
      .duration(ANIM.SCALE_DURATION)
      .ease(easeCubicOut)
      .attr('stroke-dashoffset', 0);

    ticks.forEach((depth: number) => {
      const yPx = scale(depth);
      const delay = (depth / totalDepth) * ANIM.SCALE_DURATION + ANIM.TICK_BASE_DELAY;

      const tick = g.append('g')
        .attr('class', 'tick')
        .attr('transform', `translate(0,${yPx - 4})`)
        .style('opacity', 0.01);

      tick.append('line')
        .attr('class', 'tick-line')
        .attr('x1', depthAxisX).attr('x2', depthAxisX)
        .attr('y1', 0).attr('y2', 0)
        .transition()
        .delay(delay)
        .duration(280)
        .ease(easeBackOut.overshoot(1.4))
        .attr('x1', tickLeftX).attr('x2', tickRightX);

      tick.append('text')
        .attr('class', 'tick-label')
        .attr('x', tickLabelX + 8)
        .attr('y', 0)
        .attr('dominant-baseline', 'middle')
        .style('opacity', 0)
        .text(depth === 0 ? '0' : depth.toLocaleString())
        .transition()
        .delay(delay + 110)
        .duration(260)
        .ease(easeCubicOut)
        .attr('x', tickLabelX)
        .style('opacity', 1);

      tick.transition()
        .delay(delay)
        .duration(240)
        .ease(easeCubicOut)
        .attr('transform', `translate(0,${yPx})`)
        .style('opacity', 1);
    });
  }
}

  /**
   * Opens the attachment viewer. A staged file is already a `File`; a stored one is fetched back as
   * one first, so both end up in the same viewer instead of one opening a raw browser tab.
   */
  protected viewAttachment(slot: IrAttachmentSlot): void {
    const config = WWELL_IR_ATTACHMENT_SLOTS.find(entry => entry.slot === slot);
    const label = config?.label ?? 'Attachment';

    const staged = this.pendingAttachmentFor(slot);
    if (staged) {
      this.openViewer(staged, label);
      return;
    }

    const attachment = this.attachmentFor(slot);
    const irSeq = this.ir().irSeq;
    if (!attachment || irSeq == null) return;

    this.wwellIrService
      .getIrAttachmentFile(irSeq, slot, attachment.fileName)
      .pipe(takeUntilDestroyed(this.destroyRef))
      .subscribe({
        next: file => this.openViewer(file, label),
        error: () => this.notify.error(`Failed to open ${attachment.fileName}.`),
      });
  }

  private openViewer(file: File, label: string): void {
    this.dialog.open(WwellIrAttachmentViewerComponent, {
      data: { file, label },
      panelClass: 'ir-viewer-panel',
      width: '92vw',
      maxWidth: '1500px',
      height: '92vh',
      autoFocus: false,
    });
  }

  /** Explicit send of a staged file — attaching alone never uploads. */

    const irSeq = this.ir().irSeq;
    if (irSeq == null) {
      this.notify.info('Save the integrated requirement first — this uploads automatically on save.');

  protected attachmentSizeLabel(bytes: number): string {
    if (bytes < 1024) return `${bytes} B`;
    if (bytes < 1024 * 1024) return `${(bytes / 1024).toFixed(1)} KB`;
    return `${(bytes / (1024 * 1024)).toFixed(1)} MB`;

  protected attachmentIcon(contentType: string): string {
    return contentType === 'application/pdf' ? 'picture_as_pdf' : 'image';

  protected onAttachmentPicked(slot: IrAttachmentSlot, event: Event): void {
    const input = event.target as HTMLInputElement | null;
    const file = input?.files?.[0];
    // Reset immediately so re-picking the same file after a failure still fires a change event.
    if (input) input.value = '';
    if (!file) return;

    // Checked here purely to fail fast without spending the upload — the server enforces the same
    // cap, and is the authority on it (see IrAttachmentStorageService).
    if (file.size > WWELL_IR_ATTACHMENT_MAX_BYTES) {
      this.notify.error('Attachment exceeds the 25 MB limit.');
      return;
    }

    // Attaching only stages the file — the user sends it with the slot's own upload action. On a
    // New IR there's no ir_seq to upload against anyway, and the first save flushes it instead
    // (see WwellIrComponent's uploadPendingAttachments call).
    this.store.setPendingAttachment(slot, file);
  }

  protected removeAttachment(slot: IrAttachmentSlot): void {
    if (this.pendingAttachmentFor(slot)) {
      this.store.removePendingAttachment(slot);
      return;
    }
    const irSeq = this.ir().irSeq;
    if (irSeq == null) return;
    this.store.deleteIrAttachment(irSeq, slot);
  }

  // ── Unsaved-changes tracking — read by WwellIrComponent (via ViewChild) before it lets the user
  // navigate away from this record: switch the selected row, open "New IR", or leave the route
  // entirely (see the wwell-ir CanDeactivate guard). Every open SectionEditState means "the global
  // Save button would submit this section's current live value" (see saveAll() below) — including
  // a row delete or section reset, which open their section and mutate it without saving (see
  // deleteCasingRow/resetProgramSection etc.) — so "any section open" is exactly "something would
  // be lost by leaving now", regardless of whether a control's own `dirty` flag happens to be set
  // (patchValue()/removeAt() don't set it).
  public readonly hasUnsavedChanges = computed(

      this.requestReferenceEdit.isEditing() ||
      this.programEdit.isEditing() ||
      this.geologistContactEdit.isEditing() ||
      this.hydrogeologyEdit.isEditing() ||

  // Resolved once the in-flight save this array was queued for actually settles — see the
  // `creatingIr` effect below and saveUnsavedChangesAndWait()'s own doc comment.
  private saveSettledResolvers: ((succeeded: boolean) => void)[] = [];
  private wasSavingForNavigation = false;

  constructor() {
    // The "Link to Request" select lives inside requestReferenceEdit's own lazily-built
    // FormGroup (a fresh instance every time that section's pencil is clicked — see
    // SectionEditState.open()), so this can't be a one-time subscription set up in the
    // constructor: it has to re-subscribe to whichever FormGroup instance is *currently* open,
    // and stop when that section closes (control() goes back to null) or is reopened (a new
    // FormGroup replaces the old one). `effect()`'s cleanup callback handles both.
    effect(onCleanup => {
      const ctrl = this.requestReferenceEdit.control();
      if (!ctrl) return;
      const linkedRequestIdControl = ctrl.get('linkedRequestId');
      const subscription = linkedRequestIdControl?.valueChanges.subscribe((linkedRequestId: string) => {
        console.log('[IR:link] "Link to Request" select — valueChanges fired', { linkedRequestId });
        if (linkedRequestId) this.applyLinkedRequestWell(ctrl, linkedRequestId);
      });
      onCleanup(() => subscription?.unsubscribe());
    });

    // WwellIrComponent reuses this same editor instance across selections — its `@else if
    // (selectedIr(); as ir)` branch never changes just because the *bound* record does (switching
    // list rows, or opening a blank "New IR" draft), so nothing here gets recreated for free. A
    // section already open at that moment keeps the FormGroup it already built — SectionEditState
    // .open() is a no-op once editing (see its own doc comment) — so without this, an open
    // section keeps showing the *previous* record's values after the identity underneath it
    // changes. Closing everything forces every section back to its read display, which re-derives
    // fresh from whatever `ir()` now is on next open. Compares `id` (not the object reference,
    // which also changes on every in-place update from a save) so an update to the *same* record
    // never closes sections the user still has open mid-edit.
    effect(() => {
      const id = this.ir().id;
      if (this.previousIrId !== null && id !== this.previousIrId) this.closeAllSections();
      this.previousIrId = id;
    });

    // Resolves every promise queued by saveUnsavedChangesAndWait() once the save it kicked off
    // settles — mirrors WwellIrComponent's own `wasCreatingIr` settle-detection effect (see there
    // for why a plain `creatingIr()` read can't be used directly: it's a shared store flag, so only
    // the true→false *transition* means "a save just finished", not any particular read of it).
    effect(() => {
      const saving = this.store.creatingIr();
      const settled = this.wasSavingForNavigation && !saving;
      this.wasSavingForNavigation = saving;
      if (!settled || this.saveSettledResolvers.length === 0) return;

      const succeeded = !this.store.createIrError();
      const resolvers = this.saveSettledResolvers;
      this.saveSettledResolvers = [];
      resolvers.forEach(resolve => resolve(succeeded));
    });
  }

  /** Discards every open section's in-progress edits without saving — reverts the editor back to
   *  its all-read-display state, so the next `ir()` (a different record, or this one refreshed)
   *  renders clean. Used by WwellIrComponent's navigation guard when the user picks "Discard
   *  Changes" on the unsaved-changes prompt. */
  public discardUnsavedChanges(): void {
    this.closeAllSections();
  }

  /** Triggers the same global Save as the footer button and resolves once that save request
   *  settles — `true` on success, `false` on failure (the failure itself is already toasted by
   *  WwellIrComponent's `createIrError` effect, same as any other save). Used by WwellIrComponent's
   *  navigation guard when the user picks "Save & Continue"/"Save & Leave": the guard awaits this
   *  before actually letting the navigation proceed, so a failed save keeps the user on the record
   *  instead of losing it mid-flight. Reuses an already-in-flight save (rather than firing a second
   *  one) if saveAll() was already triggered from elsewhere when this is called. */
  public saveUnsavedChangesAndWait(): Promise<boolean> {
    return new Promise(resolve => {
      this.saveSettledResolvers.push(resolve);
      if (!this.store.creatingIr()) this.saveAll();
    });
  }

  /** Selecting a "Link to Request" option auto-fills whichever Request Reference fields that
   *  eligible-well record actually has data for — BI Number/Supporting Business/Target Aquifer
   *  still have no source there (exad_rcd_prewap doesn't carry them) and stay manually entered.
   *  Target Depth used to be in that same "no source" group; it now comes from the linked well's
   *  own exad_rcd_prewap.target_dpth (`targetDepth` here) — the only real source that field has
   *  anywhere, since exad_gwd_ir_header has no target-depth column of its own to save a
   *  manually-typed value into. That's why `targetDepthFt` is `readonly: true` in the schema
   *  (wwell-ir-form.schema.ts) unconditionally, unlike Surface Elevation/Coordinates below, which
   *  stay editable until a link actually supplies them. Purely a client-side lookup against the
   *  already-loaded eligible-wells list (see store.loadLinkedRequestWells()) — selecting a link
   *  never itself triggers a network call.
   *
   *  Whichever fields the link actually supplies are then locked read-only: they track the source
   *  request record, so letting someone edit or clear them here would silently diverge from (or
   *  desync) that source. Also reused at initial section-open for an already-linked record (see
   *  requestReferenceEdit above) — re-locks the same fields to the same values, so it's a no-op
   *  data-wise there. */
  private applyLinkedRequestWell(referenceGroup: FormGroup, linkedRequestId: string): void {
    console.log('[IR:link] applyLinkedRequestWell — start', { linkedRequestId, eligibleWellsLoaded: this.store.linkedRequestWells().length });
    const well = findLinkedWell(linkedRequestId, this.store.linkedRequestWells());
    if (!well) {
      console.log('[IR:link] applyLinkedRequestWell — no matching eligible well, aborting patch', { linkedRequestId });
      return;
    }

    const hasElevation = linkedWellHasElevation(well);
    const hasCoordinates = linkedWellHasCoordinates(well);
    const hasTargetDepth = linkedWellHasTargetDepth(well);

    const patch: Record<string, string | number> = { wellName: well.wellName };
    if (hasElevation) patch['surfaceElevationFtAsl'] = well.surfaceElevation as number;
    if (hasCoordinates) patch['coordinates'] = formatLinkedWellCoordinates(well);
    // targetDepthFt's control is already permanently disabled (schema `readonly: true`, applied
    // at control-creation time) — patchValue on the parent group still updates a disabled child
    // control's own value, it just stays excluded from the group's aggregated `.value`/validity,
    // same as wellName below is excluded once `.disable()` runs.
    if (hasTargetDepth) patch['targetDepthFt'] = well.targetDepth as number;
    console.log('[IR:link] applyLinkedRequestWell — patching form group', { patch, hasElevation, hasCoordinates, hasTargetDepth });
    referenceGroup.patchValue(patch);

    referenceGroup.get('wellName')?.disable();
    this.setLinkedFieldLock(referenceGroup, 'surfaceElevationFtAsl', hasElevation);
    this.setLinkedFieldLock(referenceGroup, 'coordinates', hasCoordinates);
    console.log('[IR:link] applyLinkedRequestWell — done', { rawValue: referenceGroup.getRawValue() });
  }

  /** Locks/unlocks one Request Reference field depending on whether the currently-linked request
   *  actually supplied it — switching to a different linked request that lacks, say, coordinates
   *  hands that field back for manual entry rather than leaving it stuck disabled and empty. */
  private setLinkedFieldLock(group: FormGroup, key: string, locked: boolean): void {
    const control = group.get(key);
    if (!control) return;
    console.log('[IR:link] setLinkedFieldLock', { key, locked });
    if (locked) control.disable();
    else control.enable();
  }

  /** Per-row delete from the read-only table (Casing/Formation Tops/Offset Wells), without first
   *  clicking the section's own pencil: asks for confirmation (the row's key fields are shown in
   *  the dialog so the user knows exactly what they're removing), then — only once confirmed —
   *  opens the section *silently* (building its FormArray fresh from the currently-persisted rows,
   *  in the same order the read-only table shows them — see buildAndPatchArray — so `index` lines
   *  up 1:1) and immediately removes that row, and confirms the removal with a toast.
   *
   *  `open({ silent: true })` is the whole fix for what used to read as "deleting one row reloads
   *  the whole section": a plain `open()` also flips the section over to its big edit-form UI,
   *  which is jarring for a single-row delete and is exactly what pushed content below it up/down
   *  and felt like the page had reloaded. Silent open still builds the same live FormArray Save
   *  reads from — formationTopsRows/casingRows/offsetWellsRows above fall back to that live value
   *  whenever one exists, so the read-only table simply loses that one row (animated out via
   *  appWwellCollapseAnimate on each `<tr>`, same as any other Angular-control-flow removal) while
   *  staying on screen the whole time. buildWwellIrBulkSavePayload's `existing`-diff (see
   *  wwell-ir-payload.mapper.ts) is what turns this missing row into a real DELETE action, kept
   *  rows into UPDATE/no-op, and any newly-added row into INSERT, all in the same eventual save. */
  protected deleteCasingRow(index: number): void {
    const row = this.ir().casing[index];
    this.confirmThenDelete('Delete casing row?', [
      { label: 'Type', value: row.type },
      { label: 'Size', value: row.sizeIn },
      { label: 'Depth', value: `${row.depthFtBgl} ft bgl` },
    ], () => {
      this.casingEdit.open({ silent: true });
      this.casingEdit.control()?.removeAt(index);
      this.notify.info('Casing row removed — click Save to apply.');
    });
  }

  protected deleteFormationTopRow(index: number): void {
    const row = this.ir().formationTops[index];
    this.confirmThenDelete('Delete formation top?', [
      { label: 'Formation', value: row.name },
      { label: 'Depth', value: `${row.depthFt} ft bgl` },
    ], () => {
      this.formationTopsEdit.open({ silent: true });
      this.formationTopsEdit.control()?.removeAt(index);
      this.notify.info('Formation top removed — click Save to apply.');
    });
  }

  protected deleteOffsetWellRow(index: number): void {
    const row = this.ir().offsetWells[index];
    this.confirmThenDelete('Delete offset well?', [
      { label: 'EP A Num', value: String(row.epANum) },
      { label: 'Aquifer', value: row.aquifer || '—' },
    ], () => {
      this.offsetWellsEdit.open({ silent: true });
      const ctrl = this.offsetWellsEdit.control();
      if (ctrl) this.offsetWellsArrayControl(ctrl).removeAt(index);
      this.notify.info('Offset well removed — click Save to apply.');
    });
  }

  /** Clears every field in Program Details (2.1) back to blank — a flat, non-repeatable section
   *  has no "row" to delete (see this section's own read-only view, which has no Actions column),
   *  so this is its equivalent destructive action. Opens the section for editing (if not already)
   *  with the reset values already applied, leaving the global Save button as the one place that
   *  actually persists it — same "review before Save" pattern as the row deletes above. */
  protected resetProgramSection(): void {
    this.confirmThenAct({ title: 'Reset Program Details?', confirmLabel: 'Reset' }, () =>
      this.resetFlatSection(this.programEdit, this.programSection, 'Program Details'),
    );
  }

  protected resetGeologistSection(): void {
    this.confirmThenAct({ title: 'Reset Geologist Contact?', confirmLabel: 'Reset' }, () =>
      this.resetFlatSection(this.geologistContactEdit, this.geologistSection, 'Geologist Contact'),
    );
  }

  private resetFlatSection(edit: SectionEditState<FormGroup, unknown>, section: WwellFormSectionConfig, label: string): void {
    edit.open();
    const ctrl = edit.control();
    if (!ctrl) return;
    const defaults: Record<string, unknown> = {};
    (section.fields ?? []).forEach(field => {
      defaults[field.key] = defaultValueFor(field);
    });
    ctrl.patchValue(defaults);
    this.notify.info(`${label} cleared — click Save to apply.`);
  }

  /** Shared confirm-before-destructive-action gate for the row deletes above and the section
   *  resets above — `onConfirmed` only ever runs once the user actually confirms. */
  private confirmThenAct(
    config: { title: string; rows?: { label: string; value: string }[]; confirmLabel: string },
    onConfirmed: () => void,
  ): void {
    this.confirmDialog
      .open({ title: config.title, rows: config.rows ?? [], confirmLabel: config.confirmLabel, severity: 'danger' })
      .pipe(takeUntilDestroyed(this.destroyRef))
      .subscribe(confirmed => {
        if (confirmed) onConfirmed();
      });
  }

  private confirmThenDelete(title: string, rows: { label: string; value: string }[], onConfirmed: () => void): void {
    this.confirmThenAct({ title, rows, confirmLabel: 'Delete' }, onConfirmed);
  }

  /** Deletes the whole IR from the backend (header row — casing/tops/hydrogeology/water cascade
   *  server-side via FK, see WwellIrService.deleteIr) after confirmation. Only reachable once the
   *  record is actually persisted (`ir().irSeq !== null`, gated in the template) — a local-only
   *  draft has nothing to delete server-side. WwellIrComponent reacts to store.deletingIr()
   *  settling to notify and fall back the selection once this record disappears from `irs`. */
  protected deleteIr(): void {
    const irSeq = this.ir().irSeq;
    if (irSeq === null || irSeq === undefined) return;
    this.confirmThenAct(
      {
        title: 'Delete this Integrated Requirement?',
        rows: [
          { label: 'IR', value: this.ir().code },
          { label: 'Well', value: this.ir().requestReference.wellName || 'Untitled Well' },
        ],
        confirmLabel: 'Delete',
      },
      () => this.store.deleteIr(irSeq),
    );
  }

  private currentRawFormValue(): WwellIrFormRawValue {
    return mapIntegratedRequirementToRawFormValue(this.ir());
  }

  /** Shared save path: re-derives the payload/bulk-save arrays from the *current* record with the
   *  open sections' slices overlaid, so a section nobody touched this time round is resent
   *  unchanged (and diffed to a DML no-op by buildWwellIrBulkSavePayload's `existing` comparison)
   *  rather than dropped.
   *
   *  Deliberately fires no success toast — the request is still in flight here, so anything shown
   *  now would claim success before the backend has answered (and would still show next to the
   *  error toast when the save actually fails). WwellIrComponent reports the real outcome once
   *  store.creatingIr() settles.
   *
   *  Status handling: the save payload type only ever accepts 'draft' | 'in-progress' (see
   *  CreateIntegratedRequirementPayload) — there's no contract for "edit this field but keep it
   *  Approved/In Review". Editing a draft/in-progress record keeps its current status; editing
   *  anything further along pulls it back to 'in-progress' (not 'draft', since it was already
   *  worked past that point) so it's clear the record needs to move through review again. */
  private saveMergedRaw(raw: WwellIrFormRawValue): void {
    const current = this.ir();
    const status: Extract<IntegratedRequirementStatus, 'draft' | 'in-progress'> =
      current.status === 'draft' || current.status === 'in-progress' ? current.status : 'in-progress';
    const { payload } = mapIrFormValueToSubmission(raw, status);
    const irBulkPayload = buildWwellIrBulkSavePayload(raw, {
      irSeq: current.irSeq ?? null,
      existing: current,
      geologistEmpNumByName: this.store.geologistEmpNumByName(),
    });
    this.store.createIr(payload, irBulkPayload);
  }

  /** Global footer "Save" — merges every currently-open section's live form value (plus
   *  attachments, if that's open too) into a single raw value and saves it in one call, instead of
   *  requiring the user to click each open section's own Save button separately. A section that
   *  isn't open contributes its existing saved value unchanged, exactly as it would for any one
   *  section's own save (see saveMergedRaw / currentRawFormValue). This is also what turns a brand
   *  new local-only draft (ir().irSeq === null) into a real backend record on the very first click,
   *  even if the user never opened a single section — so the no-op guard below only applies once
   *  the record is already persisted; a still-local draft always goes through, even with nothing
   *  open, since that first click is what creates it.
   *
   *  Without the guard, clicking Save again after a save already settled (nothing reopened since)
   *  re-submitted the exact same payload — hasUnsavedChanges() false means no section is open, so
   *  there is nothing new to merge in, but saveMergedRaw() doesn't know that and hits the backend
   *  regardless. */
  protected saveAll(): void {
    if (this.ir().irSeq !== null && !this.hasUnsavedChanges()) {
      this.notify.info('No changes to save.');
      return;
    }

    let raw = this.currentRawFormValue();

    const referenceCtrl = this.requestReferenceEdit.control();
    if (referenceCtrl) raw = { ...raw, reference: { main: referenceCtrl.getRawValue() } };

    const programCtrl = this.programEdit.control();
    if (programCtrl) {
      const flat = programCtrl.getRawValue() as WwellIrRawProgramDetails;
      raw = {
        ...raw,
        programDetails: {
          main: {
            completion: flat.completion,
            openHoleSizeIn: flat.openHoleSizeIn,
            acidization: flat.acidization,
            transmission: flat.transmission,
          },
          developmentTesting: {
            logging: flat.logging,
            mudLogging: flat.mudLogging,
            sampling: flat.sampling,
            flowTest: flat.flowTest,
            remarks: flat.remarks,
          },
        },
      };
    }

    const geologistCtrl = this.geologistContactEdit.control();
    if (geologistCtrl) raw = { ...raw, contacts: { ...raw.contacts, geologist: geologistCtrl.getRawValue() } };

    const hydrogeologyCtrl = this.hydrogeologyEdit.control();
    if (hydrogeologyCtrl) raw = { ...raw, hydrogeology: { main: hydrogeologyCtrl.getRawValue() } };

    const formationTopsCtrl = this.formationTopsEdit.control();
    if (formationTopsCtrl) raw = { ...raw, subsurface: { ...raw.subsurface, formationTops: formationTopsCtrl.getRawValue() } };

    const topsRemarksCtrl = this.topsRemarksEdit.control();
    if (topsRemarksCtrl) raw = { ...raw, subsurface: { ...raw.subsurface, remarks: topsRemarksCtrl.getRawValue() } };

    const casingCtrl = this.casingEdit.control();
    if (casingCtrl) raw = { ...raw, subsurface: { ...raw.subsurface, casing: casingCtrl.getRawValue() } };

    const offsetWellsCtrl = this.offsetWellsEdit.control();
    if (offsetWellsCtrl) {
      raw = {
        ...raw,
        contacts: {
          ...raw.contacts,
          offsetWells: this.offsetWellsArrayControl(offsetWellsCtrl).getRawValue(),
          remarks: this.offsetWellsRemarksControl(offsetWellsCtrl).getRawValue(),
        },
      };
    }

    this.saveMergedRaw(raw);
    this.closeAllSections();
  }

  private closeAllSections(): void {
    this.requestReferenceEdit.close();
    this.programEdit.close();
    this.geologistContactEdit.close();
    this.hydrogeologyEdit.close();
    this.formationTopsEdit.close();
    this.topsRemarksEdit.close();
    this.casingEdit.close();
    this.offsetWellsEdit.close();
  }

  protected submitForReview(): void {
    this.store.submitIrForReview(this.ir());
  }
}

/** Inverse of wwell-ir-payload.mapper.ts's resolveTransmissionCode: exad_gwd_ir_header.trns_type
 *  is VARCHAR2(10), too short for the full label, so a known label is stored as its short code
 *  ('Pipeline' → 'PL'). Decode it back to the label the form's option list expects, otherwise the
 *  select has no matching option and silently renders blank. Anything unrecognised (free-typed
 *  text that was truncated to fit) is passed through as-is. */
function decodeTransmission(code: string | null | undefined): string | null {
  if (!code) return null;
  const label = Object.keys(WWELL_TRANSMISSION_CODE_BY_LABEL).find(key => WWELL_TRANSMISSION_CODE_BY_LABEL[key] === code);
  return label ?? code;
}

/** Every IrXxxRow interface in wwell-ir-payload.model.ts (IrHeaderRow, IrCasingRow, IrTopsRow,
 *  IrHydrogeologyRow, IrWaterRow) uses lowercase snake_case field names mirroring the Oracle
 *  column names — but Oracle's own default column casing (and whatever exad_agwa_queries_pkg's
 *  get_ir_data_prc happens to emit) isn't guaranteed to be lowercase, so a row can come back with
 *  the same fields in UPPERCASE or MixedCase. Since every field this file reads is lowercase by
 *  contract, lowercasing every key up front makes each row match regardless of how the backend
 *  cased it — the same fix as `rowsFor`'s table-name comparison below, applied to row field names. */
function normalizeRowKeys<TRow>(row: Record<string, unknown>): TRow {
  const normalized: Record<string, unknown> = {};
  for (const key of Object.keys(row)) {
    normalized[key.toLowerCase()] = row[key];
  }
  return normalized as TRow;
}

/** Case-insensitive on `candidate.table`: the GET /{irSeq} detail endpoint echoes back
 *  UPPERCASE table names (EXAD_GWD_IR_HEADER...) per exad_agwa_queries_pkg.get_ir_data_prc,
 *  while WwellIrTableName (and the bulk-save write path) uses lowercase — a strict `===` here
 *  never matched, so every block silently resolved to empty and wiped casing/formationTops on
 *  every detail fetch. */
function rowsFor<TRow>(blocks: readonly WwellIrDetailTableBlock[], table: WwellIrTableName): readonly TRow[] {
  const block = blocks.find(candidate => candidate.table.toLowerCase() === table);
  return (block?.data ?? []).map(row => normalizeRowKeys<TRow>(row));
}

/** Inverse of resolveGeologistId in wwell-ir-payload.mapper.ts: cont_geologist/sub_cont_geologist
 *  come back as a raw EMPNUM, not the name the picker showed - resolve it through the same
 *  FULL_NAME/EMPNUM reference data (WwellIrStore.geologistNameByEmpNum) used to build the picker's
 *  own options. Falls back to `fallback` (the caller's `existing` value) when the header has no
 *  value for this column yet, or the EMPNUM isn't found in the current ops-members list (e.g. a
 *  fetch that hasn't completed, or a since-removed employee) - same as every other lookup below,
 *  an unresolved value should keep whatever was last known rather than go blank. */
function resolveGeologistName(
  empNum: number | null | undefined,
  geologistNameByEmpNum: Readonly<Record<number, string>>,
  fallback: string | null,
): string | null {
  if (empNum == null) return fallback;
  return geologistNameByEmpNum[empNum] ?? fallback;
}

/** Inverse of the `h2s: '${ppm} ppm'` formatting in wwell-ir-payload.mapper.ts. */
function parseH2sPpm(h2s: string | null | undefined): number | null {
  if (!h2s) return null;
  const value = parseFloat(h2s);
  return Number.isFinite(value) ? value : null;
}

/** exad_gwd_well_design's flag columns come back as `'Y'|'N'|null` in practice, but the raw block
 *  is read as loosely-typed `Record<string, unknown>` (see rowsFor above) — narrow defensively to
 *  the two real values rather than trust the cast, falling back to 'N' for anything else (an
 *  absent/garbled column reads the same as "not set", matching how deriveCompletionElementFlags
 *  itself never produces a third value). */
function normalizeFlag(value: unknown): 'Y' | 'N' {
  return value === 'Y' ? 'Y' : 'N';
}

function normalizeRemarks(value: unknown): string | null {
  return typeof value === 'string' && value ? value : null;

/** Reads the eight completion-element columns off a fetched exad_gwd_well_design row — see
 *  IntegratedRequirementWellDesignFlags' own doc comment. `null` when no row came back at all
 *  (no exad_gwd_well_design record yet for this well), not `undefined` — `undefined` is reserved
 *  for "never fetched" (a still-local draft), which this function is never called for. */
function mapWellDesignFlags(row: IrWellDesignRow | undefined): IntegratedRequirementWellDesignFlags | null {

  return {
    ohFlg: normalizeFlag(row.oh_flg),
    ohRemarks: normalizeRemarks(row.oh_remarks),
    gpFlg: normalizeFlag(row.gp_flg),

    perforatedFlg: normalizeFlag(row.perforated_flg),
    perforatedRemarks: normalizeRemarks(row.perforated_remarks),

  };

/** `stripLegacyIrHtml` version of the `raw ?? fallback` pattern used throughout this merge: falls

 *  column, same as a bare `??` would — an explicitly-cleared column (`''`) is a real value and

 *  string also yields `''`. */
function resolveLegacyText(raw: string | null | undefined, fallback: string): string {
  return raw != null ? stripLegacyIrHtml(raw) : fallback;

/**
 * Enriches an already-known IntegratedRequirement (e.g. from the GET-all summary list) with the
 * real casing/tops/hydrogeology/program data from a GET /{irSeq} detail fetch. Merges onto
 * `existing` rather than building from scratch because most domain-only fields (client,
 * supportingBusiness, targetAquifer, targetDepthFt, contacts, offsetWells, attachments...) have no
 * backing column on exad_gwd_ir_header — see wwell-ir-summary.mapper.ts for the same limitation on
 * the list side, and docs/ir/EXAD_GWD_IR_HEADER.md for the table's real column set.
 * `linkedRequestId`/`letterNumber` (prewap_seq/ltr_no) and the Development & Testing fields
 * (sampling/logging/mud_rmks/flow_rmks) DO have real columns, so those are read back from `header`
 * below rather than left as `existing` had them. Field mappings mirror buildWwellIrBulkSavePayload
 * in wwell-ir-payload.mapper.ts exactly (this is that function's inverse) — keep the two in sync
 * if either changes.
 *
 * wellName/surfaceElevationFtAsl/coordinates come from the new `exad_rcd_prewap` block
 * (get_ir_data_prc, agwa-planning backend) — a live join on the header's own prewap_seq — rather
 * than from `wwell-ir-linked-well.util.ts`'s eligible-wells lookup: that lookup can only ever
 * resolve a well that does NOT yet have an IR (get_ir_wells_prc's own eligibility filter), so it
 * structurally can never succeed for an IR that's already been saved with a link. This block is
 * the actual source of truth for those three fields once a record's detail has been fetched once.

export function mergeIrDetailBlocksIntoIntegratedRequirement(
  existing: IntegratedRequirement,

  const casingRows = rowsFor<IrCasingRow>(blocks, 'exad_gwd_ir_casing');
  const topsRows = rowsFor<IrTopsRow>(blocks, 'exad_gwd_ir_tops');

  // below for why this, not merely `wellDesignRow` existing, has to gate trusting that row at all.

    requestReference: {

      // backing column there (client, targetAquifer...) stays whatever `existing` already carries.

    },
    // exad_gwd_ir_water has no well-name column — epANum is the only identifier that round-trips

      epANum: row.ep_a_num,
      distanceKm: row.distanct ?? null,
      location: row.direction ?? null,
      aquifer: row.aquifer ?? null,
      totalDepthFt: row.td ?? null,
      testDurationMin: row.test_duration ?? null,
      rateGpm: row.rate ?? null,
      rpm: row.rpm ?? null,
      swlFt: row.swl ?? null,

      h2sPpm: row.h2s ?? null,
    })),
    // tolFtBgl has no column yet (buildWwellIrBulkSavePayload never writes it either) — stays null.
    casing: casingRows.map(row => ({
      type: row.csg_type,
      sizeIn: row.csg_size,
      depthFtBgl: row.csg_depth,
      tolFtBgl: null,
      remarks: row.csg_remarks ?? null,
    })),
    programDetails: {
      ...existing.programDetails,
      // Older records saved these six remarks columns as raw `<HTML>...` markup (a legacy rich-
      // text control's "Save As HTML" dump) instead of plain text — stripLegacyIrHtml() is a no-op
      // on anything that isn't that markup, so this is safe whether the record is old or new. See
      // that util's own doc comment for why running it here (the one place these columns become
      // IntegratedRequirement fields) is also what keeps a later re-save plain-text-only.
      completion: completionText,
      // Both are written by buildWwellIrBulkSavePayload but were never read back here, so
      // Transmission/Acidization rendered blank after any reload even though the values were in
      // the DB the whole time — the same class of gap that hid Logging/Sampling before.
      acidization: resolveLegacyText(header?.acidization, existing.programDetails.acidization),
      transmission: decodeTransmission(header?.trns_type) ?? existing.programDetails.transmission,

 * field wired to `linkedRequestOptions`, and the 'contacts'/'geologist' section's
 * 'geologistPrimary'/'geologistSecondary' fields wired to `geologistOptions` — each replacing its

 * component that actually has one — see WwellIrEditorComponent's `schema` field.

  linkedRequestOptions: Signal<readonly WwellFieldOption[]>,
  geologistOptions: Signal<readonly string[]>,
): readonly WwellFormStepConfig[] {
  return WWELL_IR_FORM_SCHEMA.map(step => {
    if (step.key === 'reference') {
      return {
        ...step,
        sections: step.sections.map(section => {
          if (section.key !== 'main' || !section.fields) return section;
          return {
            ...section,
            fields: section.fields.map(field =>
              field.key === 'linkedRequestId' ? { ...field, options: linkedRequestOptions } : field,
            ),
          };
        }),
      };
    }
    if (step.key === 'contacts') {
      return {
        ...step,
        sections: step.sections.map(section => {
          if (section.key !== 'geologist' || !section.fields) return section;
          return {
            ...section,
            fields: section.fields.map(field =>
              field.key === 'geologistPrimary' || field.key === 'geologistSecondary' ? { ...field, options: geologistOptions } : field,
            ),
          };
        }),
      };
    }
    return step;
  });

import { IntegratedRequirementRequestReference } from '@models/wwell-planning/integrated-requirement.model';
import { IrEligibleWell } from './wwell-ir-payload.model';

/**
 * Resolves a "Link to Request" selection (the raw prewap_seq id) to Water Well Name / Coordinates
 * / Surface Elevation while the user is actively picking a NEW link in the editor — see
 * wwell-ir-editor.component.ts's `applyLinkedRequestWell`. `wells` here is always
 * store.linkedRequestWells() (GET /wells/eligible-for-ir), which is the right source at this
 * exact moment: before the IR is saved, no exad_gwd_ir_header row references this prewap_seq yet,
 * so the well genuinely is still "eligible" and present in that list.
 *
 * NOT used to hydrate an already-saved IR's Request Reference — get_ir_wells_prc's own
 * eligibility filter (NOT EXISTS ... exad_gwd_ir_header) permanently excludes a well the instant
 * any IR is saved against its prewap_seq, so this lookup can never resolve a persisted record's
 * own link again. That case is handled by the new exad_rcd_prewap block get_ir_data_prc returns
 * (a live join on the header's own prewap_seq) — see wwell-ir-detail-response.mapper.ts.
 */

export function findLinkedWell(linkedRequestId: string | null, wells: readonly IrEligibleWell[]): IrEligibleWell | undefined {
  if (!linkedRequestId) return undefined;
  return wells.find(candidate => String(candidate.prewapSeq) === linkedRequestId);
}

export function linkedWellHasElevation(well: IrEligibleWell): boolean {
  return well.surfaceElevation != null;
}

export function linkedWellHasCoordinates(well: IrEligibleWell): boolean {
  return well.surfaceNorth != null && well.surfaceEast != null;
}

export function linkedWellHasTargetDepth(well: IrEligibleWell): boolean {
  return well.targetDepth != null;
}

export function formatLinkedWellCoordinates(well: IrEligibleWell): string {
  return `${well.surfaceNorth}°N, ${well.surfaceEast}°E`;
}

/** Overrides `reference`'s Well Name/Surface Elevation/Coordinates/Target Depth with whatever
 *  `well` actually supplies — BI Number/Supporting Business/Target Aquifer still have no source
 *  on an eligible-well record (exad_rcd_prewap doesn't carry them) and are left untouched. Falls
 *  back to `reference`'s own value for elevation/coordinates/target depth specifically when the
 *  well doesn't have that one field, rather than blanking it. Target Depth is read-only in the
 *  UI (see the field's schema entry, wwell-ir-form.schema.ts) precisely because this is its only
 *  real source — exad_gwd_ir_header has nowhere to save a manually-typed value anyway. */
export function hydrateRequestReferenceFromLinkedWell(
  reference: IntegratedRequirementRequestReference,
  well: IrEligibleWell,
): IntegratedRequirementRequestReference {
  return {
    ...reference,
    wellName: well.wellName,
    surfaceElevationFtAsl: linkedWellHasElevation(well) ? (well.surfaceElevation as number) : reference.surfaceElevationFtAsl,
    coordinates: linkedWellHasCoordinates(well) ? formatLinkedWellCoordinates(well) : reference.coordinates,
    targetDepthFt: linkedWellHasTargetDepth(well) ? (well.targetDepth as number) : reference.targetDepthFt,
  };
}

 * re-INSERTing, plus (when known) the record as it was last loaded/saved — used to diff each
 * repeatable row's OWN action instead of inheriting one blanket action for the whole payload.

 * back to the old whole-payload action (every row shares `resolveRowAction(irSeq)`) — the only
 * safe choice when the caller has no snapshot to diff against, since guessing would otherwise
 * risk re-labeling an already-persisted row as a fresh INSERT (duplicate) or a still-current row
 * as DELETE (data loss).
 */
export interface WwellIrSaveContext {

function resolveRowAction(irSeq: number | null): IrRowAction {
  return irSeq === null ? 'INSERT' : 'UPDATE';

const isBlankCasingRow = (row: WwellIrRawCasingRow): boolean =>
  !row.type && !row.sizeIn && !row.remarks && row.depthFtBgl == null;

const isBlankFormationTop = (row: WwellIrRawFormationTop): boolean =>
  !row.name && row.depthFt == null && row.subSeaFtAsl == null;

/** Natural keys the backend's UPDATE/DELETE branches match repeatable rows by (see
 *  EXAD_AGWA_IUD_PKG.save_casing/save_tops in 05-agwa-iud-package.sql) — a row whose key isn't
 *  already present in `existing` is genuinely new and must be sent as INSERT, not UPDATE, or the
 *  backend's `WHERE <key match>` finds nothing and silently drops it. */
const casingKey = (row: { readonly type: string; readonly sizeIn: string }): string => `${row.type} ${row.sizeIn}`;
const topKey = (row: { readonly name: string }): string => row.name;

/** Only the columns save_casing's/save_tops' UPDATE branch actually writes (type/size are the
 *  match key itself, not a writable column) — comparing anything else could flag a row "changed"
 *  when nothing the backend would touch actually differs. */
const casingRowUnchanged = (existingRow: IntegratedRequirementCasingRow, row: WwellIrRawCasingRow): boolean =>
  existingRow.depthFtBgl === (row.depthFtBgl ?? 0) && (existingRow.remarks ?? null) === (row.remarks || null);

const topRowUnchanged = (existingRow: IntegratedRequirementFormationTop, row: WwellIrRawFormationTop): boolean =>
  existingRow.depthFt === (row.depthFt ?? 0) && (existingRow.subSeaFtAsl ?? null) === (row.subSeaFtAsl ?? null);

function buildCasingRows(
  raw: readonly WwellIrRawCasingRow[],
  irSeq: number | null,
  existing: readonly IntegratedRequirementCasingRow[] | null | undefined,
): IrCasingRow[] {
  const currentRows = raw.filter(row => !isBlankCasingRow(row));

  if (existing == null) {
    const action = resolveRowAction(irSeq);
    return currentRows.map(row => ({
      action,
      ir_seq: irSeq,
      csg_size: row.sizeIn,
      csg_type: row.type,
      csg_depth: row.depthFtBgl ?? 0,
      csg_remarks: row.remarks || null,
    }));
  }

  const existingByKey = new Map(existing.map(row => [casingKey(row), row]));
  const keptKeys = new Set<string>();

  // A row this save didn't actually change is sent as '' (no-op), not a redundant UPDATE — see
  // IrRowAction's doc comment: the backend's per-row action switch then skips the DML entirely
  // instead of re-writing a row with the exact values it already has on every single save.
  const upserts: IrCasingRow[] = currentRows.map(row => {
    const key = casingKey(row);
    const existingRow = existingByKey.get(key);
    if (existingRow) keptKeys.add(key);
    const action: IrRowAction = existingRow ? (casingRowUnchanged(existingRow, row) ? '' : 'UPDATE') : 'INSERT';
    return {
      action,
      ir_seq: existingRow ? irSeq : null,
      csg_size: row.sizeIn,
      csg_type: row.type,
      csg_depth: row.depthFtBgl ?? 0,
      csg_remarks: row.remarks || null,
    };
  });

  const deletes: IrCasingRow[] = existing
    .filter(row => !keptKeys.has(casingKey(row)))
    .map(row => ({
      action: 'DELETE',
      ir_seq: irSeq,
      csg_size: row.sizeIn,
      csg_type: row.type,
      csg_depth: row.depthFtBgl,
      csg_remarks: row.remarks,
    }));

  return [...upserts, ...deletes];
}

      action: 'DELETE',
      ir_seq: irSeq,
      st_long_cd: row.name,
      plan_tvd_dpth: row.depthFt,
      plan_ss_dpth: row.subSeaFtAsl ?? null,
    }));

  return [...upserts, ...deletes];
}

/** True when nothing save_hydrogeology's UPDATE branch would actually write has changed —
 *  est_sta_water_lvl/est_productivity/siwhp come from the form's (and domain model's) testProgram
 *  section, not hydrogeology, so both are needed to tell "unchanged" from "changed". testType is
 *  included because it alone drives the `flowing` column (see buildHydrogeologyRows below) even
 *  though testType itself is persisted on the header row, not here. */
const hydrogeologyRowUnchanged = (
  existingHydro: IntegratedRequirementHydrogeology,
  existingTestProgram: IntegratedRequirementTestProgram,
  raw: WwellIrRawHydrology,
): boolean =>
  existingTestProgram.estimatedWaterLevelFt === raw.estimatedWaterLevelFt &&
  existingHydro.estimatedTdsPpm === raw.estimatedTdsPpm &&
  existingTestProgram.estimatedRateGpm === raw.estimatedRateGpm &&
  existingHydro.h2sPpm === raw.h2sPpm &&
  (existingTestProgram.siwhpPsi ?? null) === raw.siwhpPsi &&
  existingTestProgram.testType === raw.testType;

/** 'flow-test' → 'Y', 'pump-test' → 'N' — exad_gwd_ir_hydrogeology.flowing mirrors the same
 *  Test Type choice the header's own test_type column stores, denormalized onto this row too. */
const flowingFlag = (testType: string): 'Y' | 'N' => (testType === 'flow-test' ? 'Y' : 'N');

/** Only ever 0 or 1 row (the form has a single hydrogeology section, not a repeatable list), keyed
 *  by aquifer (st_long_cd) the same way the backend matches it — so switching to a different
 *  aquifer must DELETE the old row, not silently orphan it under a key nothing will ever match again. */
function buildHydrogeologyRows(
  raw: WwellIrRawHydrology,
  irSeq: number | null,
  existing: Pick<IntegratedRequirement, 'hydrogeology' | 'testProgram'> | null | undefined,
): IrHydrogeologyRow[] {
  const currentAquifer = raw.aquifer || null;

  if (existing == null) {
    if (!currentAquifer) return [];
    const action = resolveRowAction(irSeq);
    return [
      {
        action,
        ir_seq: irSeq,
        st_long_cd: currentAquifer,
        est_sta_water_lvl: raw.estimatedWaterLevelFt,
        est_water_quality: raw.estimatedTdsPpm,
        est_productivity: raw.estimatedRateGpm,
        flowing: flowingFlag(raw.testType),
        siwhp: raw.siwhpPsi,
        h2s: raw.h2sPpm != null ? `${raw.h2sPpm} ppm` : null,
      },
    ];
  }

  const existingAquifer = existing.hydrogeology.aquifer || null;
  const rows: IrHydrogeologyRow[] = [];

  if (existingAquifer && existingAquifer !== currentAquifer) {
    rows.push({
      action: 'DELETE',
      ir_seq: irSeq,
      st_long_cd: existingAquifer,
      est_sta_water_lvl: null,
      est_water_quality: null,
      est_productivity: null,
      flowing: null,
      siwhp: null,
      h2s: null,
    });
  }

  if (currentAquifer) {
    const isExisting = existingAquifer === currentAquifer;
    const action: IrRowAction = isExisting
      ? hydrogeologyRowUnchanged(existing.hydrogeology, existing.testProgram, raw)
        ? ''
        : 'UPDATE'
      : 'INSERT';
    rows.push({
      action,
      ir_seq: isExisting ? irSeq : null,
      st_long_cd: currentAquifer,
      est_sta_water_lvl: raw.estimatedWaterLevelFt,
      est_water_quality: raw.estimatedTdsPpm,
      est_productivity: raw.estimatedRateGpm,
      flowing: flowingFlag(raw.testType),
      siwhp: raw.siwhpPsi,
      h2s: raw.h2sPpm != null ? `${raw.h2sPpm} ppm` : null,
    });
  }

  return rows;
}

const isBlankOffsetWell = (row: WwellIrRawOffsetWell): boolean => row.epANum == null && !row.aquifer;

/** Natural key exad_gwd_ir_water is matched on — (ir_seq, ep_a_num) is its primary key. */
const waterKey = (epANum: number): string => String(epANum);

/** Only the columns save_water's UPDATE branch writes — ep_a_num is the match key itself. */
const waterRowUnchanged = (existingRow: IntegratedRequirementOffsetWell, row: WwellIrRawOffsetWell): boolean =>
  (existingRow.distanceKm ?? null) === row.distanceKm &&
  (existingRow.location ?? null) === (row.location || null) &&
  (existingRow.aquifer ?? null) === (row.aquifer || null) &&
  (existingRow.totalDepthFt ?? null) === row.totalDepthFt &&
  (existingRow.testDurationMin ?? null) === row.testDurationMin &&
  (existingRow.rateGpm ?? null) === row.rateGpm &&
  (existingRow.rpm ?? null) === row.rpm &&
  (existingRow.swlFt ?? null) === row.swlFt &&
  (existingRow.pwlFt ?? null) === row.pwlFt &&
  (existingRow.drawdownFt ?? null) === row.drawdownFt &&
  (existingRow.specificCapacity ?? null) === row.specificCapacity &&
  (existingRow.tdsPpm ?? null) === row.tdsPpm &&
  (existingRow.h2sPpm ?? null) === row.h2sPpm;

/** `temp` is the one exad_gwd_ir_water column the form has no field for — always sent null. */
const toWaterRow = (row: WwellIrRawOffsetWell, action: IrRowAction, irSeq: number | null): IrWaterRow => ({
  action,
  ir_seq: irSeq,
  ep_a_num: row.epANum ?? 0,
  distanct: row.distanceKm,
  direction: row.location || null,
  aquifer: row.aquifer,
  td: row.totalDepthFt,
  rate: row.rateGpm,
  rpm: row.rpm,
  swl: row.swlFt,
  final_pwl: row.pwlFt,
  drawdown: row.drawdownFt,
  spec_capacity: row.specificCapacity,
  temp: null,
  field_tds: row.tdsPpm,
  test_duration: row.testDurationMin,
  h2s: row.h2sPpm,
});

function buildWaterRows(
  raw: readonly WwellIrRawOffsetWell[],
  irSeq: number | null,
  existing: readonly IntegratedRequirementOffsetWell[] | null | undefined,
): IrWaterRow[] {
  const currentRows = raw.filter(row => !isBlankOffsetWell(row));

  if (existing == null) {
    const action = resolveRowAction(irSeq);
    return currentRows.map(row => toWaterRow(row, action, irSeq));
  }

  const existingByKey = new Map(existing.map(row => [waterKey(row.epANum), row]));
  const keptKeys = new Set<string>();

  const upserts: IrWaterRow[] = currentRows.map(row => {
    const key = waterKey(row.epANum ?? 0);
    const existingRow = existingByKey.get(key);
    if (existingRow) keptKeys.add(key);
    const action: IrRowAction = existingRow ? (waterRowUnchanged(existingRow, row) ? '' : 'UPDATE') : 'INSERT';
    return toWaterRow(row, action, existingRow ? irSeq : null);
  });

  const deletes: IrWaterRow[] = existing
    .filter(row => !keptKeys.has(waterKey(row.epANum)))
    .map(row => toWaterRow(offsetWellToRaw(row), 'DELETE', irSeq));

  return [...upserts, ...deletes];
}

/** Domain → raw shape, so a DELETE row is built by the same toWaterRow() as every other action. */

  distanceKm: row.distanceKm ?? null,
  location: row.location ?? '',
  aquifer: row.aquifer ?? '',
  totalDepthFt: row.totalDepthFt ?? null,
  testDurationMin: row.testDurationMin ?? null,
  rateGpm: row.rateGpm ?? null,
  rpm: row.rpm ?? null,
  swlFt: row.swlFt ?? null,
  pwlFt: row.pwlFt ?? null,
  drawdownFt: row.drawdownFt ?? null,
  specificCapacity: row.specificCapacity ?? null,
  tdsPpm: row.tdsPpm ?? null,
  h2sPpm: row.h2sPpm ?? null,
});

/** Resolves a geologist's display name (FULL_NAME, as picked from the Geologist Contact dropdown)
 *  to the real EMPNUM exad_gwd_ir_header.cont_geologist/sub_cont_geologist expects, via the live
 *  map built from GET /gwd-ops-members — see WwellIrSaveContext.geologistEmpNumByName. */
function resolveGeologistId(name: string, empNumByName: Readonly<Record<string, number>> | undefined): number | null {
  return name && empNumByName ? (empNumByName[name] ?? null) : null;
}

/** exad_gwd_ir_header.trns_type is VARCHAR2(10) — too short for the full Transmission label, so a
 *  known option maps to its short code; anything else (free-typed 'combo' text) is truncated to
 *  fit rather than silently dropped by Oracle's lax JSON_TABLE conversion. */
function resolveTransmissionCode(value: string): string | null {
  if (!value) return null;
  return WWELL_TRANSMISSION_CODE_BY_LABEL[value] ?? value.slice(0, 10);

/**
 * Estimated Pump Depth (pump_lvl) plus the eight completion-element columns (derived from
 * Completion/Open Hole Size via deriveCompletionElementFlags — see IrWellDesignRow's own doc
 * comment) are mapped onto exad_gwd_well_design. Unlike every other block above, this table is
 * keyed by wd_seq (not ir_seq): with no wd_seq yet known (no detail fetch has resolved one), a row
 * is only worth INSERTing once there's an actual value to save (a pump depth, or a real Completion
 * selection) and a linked request to hang it off of — otherwise there is nothing to persist yet.
 *
 * DATA-LOSS CAVEAT: update_ir_data_prc's exad_gwd_well_design MERGE (docs/ir/update_ir_data_prc.sql)
 * is a blanket `WHEN MATCHED THEN UPDATE SET <every column the proc's JSON_TABLE declares>` off
 * this same JSON row — a column this app's payload omits (design_rpt, status, design_phase,
 * w_prim_hid, w_num, sta_water_lvl, tds, completion_type, notes, publish_dt/flg/by, curr_flg) comes
 * back NULL from JSON_TABLE and gets written as NULL, not left alone. If a wd_seq this UPDATE
 * targets already carries real data from that other Well Design workflow, this silently wipes it.
 * That risk already existed for pump_lvl-only saves before this change; sending Completion more
 * often (it's now the primary driver, not just an occasional pump-depth edit) makes it fire more
 * often too. Not fixed here — would need either the backend proc to COALESCE(t.col, h.col) instead
 * of overwriting, or this app reading back and re-sending every other column untouched.
 */
function buildWellDesignRow(
  programMain: WwellIrRawProgramMain,
  pumpDepthFt: number | null,
  prewapSeq: number | null,
  existingWellDesignSeq: number | null | undefined,
  existingPumpDepthFt: number | null | undefined,
  existingProgramDetails: Pick<IntegratedRequirementProgramDetails, 'completion' | 'openHoleSizeIn'> | null | undefined,
): IrWellDesignRow | null {
  const wdSeq = existingWellDesignSeq ?? null;
  const flags = deriveCompletionElementFlags(programMain.completion, programMain.openHoleSizeIn || null);
  // IntegratedRequirementWellDesignFlags is camelCase (domain-model shape, shared with the
  // wellbore-diagram mapper); the wire payload below needs update_ir_data_prc's own snake_case
  // column names (see docs/ir/update_ir_data_prc.sql) — translate rather than rename one side.
  const flagColumns = {
    oh_flg: flags.ohFlg,
    oh_remarks: flags.ohRemarks,
    gp_flg: flags.gpFlg,
    gp_remarks: flags.gpRemarks,
    perforated_flg: flags.perforatedFlg,
    perforated_remarks: flags.perforatedRemarks,
    ls_flg: flags.lsFlg,
    ls_remarks: flags.lsRemarks,
  };

  if (wdSeq != null) {
    const unchanged =
      (existingPumpDepthFt ?? null) === pumpDepthFt &&
      (existingProgramDetails?.completion ?? '') === programMain.completion &&
      (existingProgramDetails?.openHoleSizeIn ?? null) === (programMain.openHoleSizeIn || null);
    const action: IrRowAction = unchanged ? '' : 'UPDATE';
    return { action, wd_seq: wdSeq, prewap_seq: prewapSeq, pump_lvl: pumpDepthFt, ...flagColumns };
  }

  if ((pumpDepthFt == null && !programMain.completion) || prewapSeq == null) return null;
  return { action: 'INSERT', wd_seq: null, prewap_seq: prewapSeq, pump_lvl: pumpDepthFt, ...flagColumns };
}

    { table: 'exad_gwd_well_design', data: wellDesign ? [wellDesign] : null },
  ];
}

/**
 * `casing.renderer.ts`'s `getCasingTier` derives each casing's on-screen half-width purely
 * from its position in this array via `indexOf` (tier 0 = narrowest, each later tier wider) —
 * it never reads `csgSize` for width, only for the text label. It also paints casings in this

 * — the standard "wide, shallow casing painted over a narrow, deep one" telescoping look.
 *
 * Sorting by SIZE (ascending — smallest first, so the largest entered size lands last/widest)
 * is what makes width actually track the number the user typed, rather than depth order. For
 * well-formed data (size decreasing with depth, real telescoping construction) this produces
 * the exact same order as depth-descending would — size and depth are inversely correlated
 * there, so there's no behavior change for the common case. It only diverges for physically
 * inverted rows (a deeper casing entered wider than a shallower one, flagged separately by
 * `findInvertedCasingSizeRows` in wwell-ir-casing-warnings.util.ts) — there, the wider-but-
 * deeper entry now paints last and can fully cover the narrower-but-shallower one in their
 * overlapping depth range, since two casings can't actually nest that way in reality either.
 * That's an unavoidable consequence of a genuinely invalid combination, not a rendering bug —
 * see `casing.renderer.ts` (off-limits to edit) for why: `buildCasingPath` always draws a

 * `isRenderableCasingRow` already guarantees `sizeIn` is non-empty text, just not necessarily
 * a `parseInchSize`-recognizable number.

    csgType: row.type,

    csgRemarks: row.remarks || null,

 * that would then get mislabeled as a real measured depth.

 * provided) per liner — never `wLnrBotDepth` or a real `wCsgOdSz` — because `rawLinersFromIR`

 * `wLnrBotDepth`, and gates the casing-actual lookup strictly on `wCsgOdSz`. Leaving both unset
 * keeps every casing and liner silver.
 *

 * the shallowest liner hangs from the deepest structural casing's shoe; each deeper liner hangs
 * from the liner immediately above it (matching the physical assumption already baked into
 * `renderLinerShapes`'s `hangerY`). Casing rows whose type is Conductor, Surface, Intermediate,
 * or Production are all "structural" here — anything that isn't `'Liner'`, exactly matching how
 * casing.renderer.ts itself distinguishes casing from liner (`csgType !== 'Liner'`).

    wCsgBotDpth: 0,

/**
 * Derives completion-element flags/remarks from the "Completion" dropdown (Program
 * Details step) plus, for Open Hole, the conditionally-shown "Open Hole Size" field.
 * `well-bore-view`'s renderers gate Open Hole strictly on a parseable inch size in
 * `ohRemarks`, and Gravel Pack / Liner Screen strictly on non-empty remarks text — the
 * constant fallback labels below exist only to satisfy those gates when the dropdown
 * implies the element but no free-text remarks field exists for it in this form.
 *
 * "Liner + Perforated" only actually renders once a Casing-table Liner row is exactly 7"
 * OD — that 7" convention is hardcoded inside liner.renderer.ts (PRE_PERF_LINER_SIZE_IN)
 * and is intentionally left untouched here.
 *
 * Water level / pump depth (`pump-and-depth.renderer.ts`'s `renderPump`, gated on `pumpLvl`)
 * come from the Estimated Hydrology step instead — see `buildWaterLevelFields` above.
 * `estimatedPumpDepthFt` is only ever populated for a pump test (see the field's own
 * `visibleWhen` in wwell-ir-form.schema.ts), so it's naturally `null` — no pump drawn — on a
 * flow-test record.
 */
/**
 * `programDetails.main.completion`'s field is `type: 'rich-textarea'` (see its own comment in
 * wwell-ir-form.schema.ts), NOT a single-pick select — `WwellRichTextareaComponent` lets any
 * number of `WWELL_COMPLETION_OPTIONS` be checked at once, and each checked option becomes its
 * own `\n`-joined line in the control's plain-text value (see that component's own doc comment
 * and `toggleOption`/`commitFromBox`). Checking both "Open Hole" and "Screened", for instance,
 * commits the control as `"Open Hole\nScreened"`, never the bare string `'Open Hole'` — so
 * `deriveCompletionElementFlags` below must check "is this option one of the checked lines?", not
 * "does the whole value equal this option?". A straight `completion === COMPLETION_OPEN_HOLE`
 * only ever matched the single-selection case; checking two or more options (or an option plus a
 * free-typed line) always fell through to every flag being 'N', which is exactly why the DB never
 * saw any flag flip to 'Y' once more than one box was checked.
 */
function selectedCompletionLabels(completion: string): ReadonlySet<string> {
  return new Set(
    completion
      .split('\n')
      .map(line => line.trim())
      .filter(line => line.length > 0),
  );
}

/**
 * The completion-element flag/remarks derivation itself — pulled out of buildWellDesignFromForm
 * so wwell-ir-payload.mapper.ts's buildWellDesignRow can compute the exact same
 * exad_gwd_well_design.{oh,gp,perforated,ls}_flg/remarks values for the actual save payload,
 * rather than these two call sites drifting apart. See IntegratedRequirementWellDesignFlags'
 * own doc comment for why the real table needs these written, not just this app's own
 * `completion` text.
 */
export function deriveCompletionElementFlags(
  completion: string,
  openHoleSizeIn: string | null,
): IntegratedRequirementWellDesignFlags {
  const selected = selectedCompletionLabels(completion);
  const isOpenHole = selected.has(COMPLETION_OPEN_HOLE);
  const isGravelPack = selected.has(COMPLETION_SCREENED_GRAVEL_PACK);
  const isScreened = selected.has(COMPLETION_SCREENED) || isGravelPack;
  const isLinerPerforated = selected.has(COMPLETION_LINER_PERFORATED);
  const isCasedPerforated = selected.has(COMPLETION_CASED_PERFORATED);
  const isPerforated = isLinerPerforated || isCasedPerforated;

  return {
    ohFlg: isOpenHole ? 'Y' : 'N',
    ohRemarks: isOpenHole ? (openHoleSizeIn || null) : null,
    gpFlg: isGravelPack ? 'Y' : 'N',
    gpRemarks: isGravelPack ? 'Gravel Pack' : null,
    perforatedFlg: isPerforated ? 'Y' : 'N',
    perforatedRemarks: isLinerPerforated ? '7" Pre-Perforated Liner' : isCasedPerforated ? 'Cased & Perforated' : null,
    lsFlg: isScreened ? 'Y' : 'N',
    lsRemarks: isScreened ? 'Screen' : null,
  };
}

/**
 * True once any completion element flag is 'Y' — the same "is a completion element actually
 * selected" test `deriveCompletionElementFlags`' four callers each need, whether the flags came
 * from a live Completion pick or a persisted `wellDesignFlags` row. Used by the emptiness gates
 * below so a record with only a well_design row (no Casing/Formation Top data at all) still shows
 * the diagram instead of the empty state — see `hasWellboreDiagramData` and
 * `hasWellboreDiagramDataForIntegratedRequirement`.
 */
function hasCompletionElementFlags(flags: IntegratedRequirementWellDesignFlags | null | undefined): boolean {
  return !!flags && (flags.ohFlg === 'Y' || flags.gpFlg === 'Y' || flags.perforatedFlg === 'Y' || flags.lsFlg === 'Y');
}

/**
 * `persistedFlags` — `ir.programDetails.wellDesignFlags`, the real exad_gwd_well_design row a GET
 * fetched back (see IntegratedRequirementWellDesignFlags) — wins over re-deriving from
 * `programMain.completion` whenever it's available: `completion` (exad_gwd_ir_header.completion,
 * VARCHAR2(4000) free text) and the well_design flags are two separately-persisted columns/tables
 * that this app always writes together (buildWellDesignRow), but nothing stops them disagreeing —
 * a record saved before this flags round-trip existed, a legacy/imported row, or any write to
 * exad_gwd_well_design from outside this app entirely. Trusting the actual fetched flags over a
 * re-derivation is what makes the diagram match what's really in the DB rather than silently
 * going blank whenever the two drift apart. Only the live-editing path (no fetched record to trust
 * yet, or the user is actively picking a new Completion — see buildWellboreDiagramDataFromIntegratedRequirement)
 * falls back to deriving from `programMain` itself.
 */
function buildWellDesignFromForm(
  programMain: WwellIrRawProgramMain,
  hydrology: WwellIrRawHydrology,
  persistedFlags?: IntegratedRequirementWellDesignFlags | null,
): IWellDesign {
  return {
    name: null,
    prewapSeq: null,
    ...buildWaterLevelFields(hydrology),
    pumpLvl: hydrology.estimatedPumpDepthFt,
    ...(persistedFlags ?? deriveCompletionElementFlags(programMain.completion, programMain.openHoleSizeIn || null)),
  };
}

function computeDisplayDepthFt(
  targetDepthFt: number,
  casingRows: readonly WwellIrRawCasingRow[],
  formationRows: readonly WwellIrRawFormationTop[],
): number {
  const depths = [
    targetDepthFt,
    ...casingRows.map(row => row.depthFtBgl ?? 0),
    ...formationRows.map(row => row.depthFt ?? 0),
  ];
  return Math.max(...depths, MIN_SCALE_DEPTH_FT);

/** Shared tail for both entry points below — takes already-normalized raw rows so the actual
 *  gating/mapping (isRenderable*, mapCasings, buildLinerTopAnchors, mapGeologicTops,
 *  buildWellDesignFromForm) is written once regardless of whether the source was a live,
 *  still-being-edited form value or a submitted `IntegratedRequirement`. */
function assembleWellboreDiagramData(
  wellName: string,
  targetDepthFt: number,
  casingRows: readonly WwellIrRawCasingRow[],
  formationRows: readonly WwellIrRawFormationTop[],
  surfaceElevationFtAsl: number | null,
  programMain: WwellIrRawProgramMain,
  hydrology: WwellIrRawHydrology,
  persistedWellDesignFlags?: IntegratedRequirementWellDesignFlags | null,
): WellboreDiagramData {
  const renderableCasings = casingRows.filter(isRenderableCasingRow);
  const renderableFormationTops = formationRows.filter(isRenderableFormationTop);
  /**
   * `completion.renderer.ts`'s `resolveCompletionBottomDepth` (rule kinds 'none' /
   * 'single-fallback' — i.e. no liner shoe to anchor the bottom to, which is exactly the "just a
   * Completion pick, no Casing rows yet" case this file's own `buildShoeAnchorFallback` exists
   * for) draws Open Hole / Gravel Pack / Liner Screen straight down to `data.totalDepth` — so
   * passing raw `targetDepthFt` through unchanged left the element's bottom pinned to its top
   * (ground level) whenever Target Depth reads as 0, making it technically present in the DOM but
   * zero-height/invisible. Resolving it once here (real value if positive, else
   * `STATIC_TARGET_DEPTH_FT`) and reusing that everywhere below — `totalDepth`, `displayDepth`'s
   * own floor, and `targetDepth` — keeps the whole diagram (scale, ground line, completion
   * elements, total-depth label) consistent instead of some parts anchoring to a real value and
   * others to a fallback.
   */
  const resolvedTargetDepthFt = targetDepthFt > 0 ? targetDepthFt : STATIC_TARGET_DEPTH_FT;
  const displayDepth = computeDisplayDepthFt(resolvedTargetDepthFt, casingRows, formationRows);

  return {
    wellName,
    totalDepth: resolvedTargetDepthFt,
    displayDepth,
    casings: mapCasings(renderableCasings),
    drlgCasings: [...buildLinerTopAnchors(renderableCasings), ...buildShoeAnchorFallback(renderableCasings)],
    geologicTops: mapGeologicTops(renderableFormationTops, surfaceElevationFtAsl),
    hydrogeology: null,
    prewap: null,
    rigActivity: null,
    currentDepth: 0,
    targetDepth: resolvedTargetDepthFt,
    mudCirculation: [],
    wellDesign: buildWellDesignFromForm(programMain, hydrology, persistedWellDesignFlags),
  };
}

/**
 * Gates the *entire* diagram — depth scale, ground level line, and total-depth label included,
 * not just individual casing/formation-top shapes (see `isRenderableCasingRow`/
 * `isRenderableFormationTop` above). Before any row is complete, `well-bore-view` still draws
 * that scaffold around nothing: a bare 0–500ft ruler, a "GROUND LEVEL" line, and a
 * "TOTAL DEPTH: 0 FT" label with no actual construction data behind any of it. Callers should
 * check this first and render an empty state instead of the diagram until it's true.
 */
export function hasWellboreDiagramData(raw: WwellIrFormRawValue): boolean {
  return (
    raw.subsurface.casing.some(isRenderableCasingRow) ||
    raw.subsurface.formationTops.some(isRenderableFormationTop) ||
    hasCompletionElementFlags(
      deriveCompletionElementFlags(raw.programDetails.main.completion, raw.programDetails.main.openHoleSizeIn || null),
    )
  );
}

export function buildWellboreDiagramData(raw: WwellIrFormRawValue): WellboreDiagramData {
  return assembleWellboreDiagramData(
    raw.reference.main.wellName,
    raw.reference.main.targetDepthFt ?? 0,
    raw.subsurface.casing,
    raw.subsurface.formationTops,
    raw.reference.main.surfaceElevationFtAsl,
    raw.programDetails.main,
    raw.hydrogeology.main,
  );
}

/** `IntegratedRequirementCasingRow`/`FormationTop`/`ProgramDetails` are already-resolved
 *  (non-nullable) counterparts of the wizard's raw row types — these just fill in the raw
 *  shape's optional-text fields (`remarks`/`openHoleSizeIn`) with `''` in place
 *  of `null`, since a submitted IR's `null` here means the same "nothing entered" as the
 *  wizard's `''`. */
function toRawCasingRows(rows: readonly IntegratedRequirementCasingRow[]): WwellIrRawCasingRow[] {
  return rows.map(row => ({
    type: row.type,
    sizeIn: row.sizeIn,
    depthFtBgl: row.depthFtBgl,
    tolFtBgl: row.tolFtBgl,
    remarks: row.remarks ?? '',
  }));
}

function toRawFormationRows(rows: readonly IntegratedRequirementFormationTop[]): WwellIrRawFormationTop[] {
  return rows.map(row => ({
    name: row.name,
    depthFt: row.depthFt,
    subSeaFtAsl: row.subSeaFtAsl ?? null,
  }));
}

function toRawProgramMain(ir: IntegratedRequirement): WwellIrRawProgramMain {
  return {
    completion: ir.programDetails.completion,
    openHoleSizeIn: ir.programDetails.openHoleSizeIn ?? '',
    acidization: ir.programDetails.acidization,
    transmission: ir.programDetails.transmission,
  };
}

/** `IntegratedRequirementHydrogeology`/`TestProgram` are two separate domain groups that the
 *  wizard's single "Estimated Hydrology" step (and its one flat `WwellIrRawHydrology` raw shape —
 *  see wwell-ir-form.types.ts) collects together — recombine them here the same way. */
function toRawHydrology(ir: IntegratedRequirement): WwellIrRawHydrology {
  return {
    testType: ir.testProgram.testType,
    aquifer: ir.hydrogeology.aquifer,
    siwhpPsi: ir.testProgram.siwhpPsi ?? null,
    estimatedTdsPpm: ir.hydrogeology.estimatedTdsPpm,
    estimatedRateGpm: ir.testProgram.estimatedRateGpm,
    h2sPpm: ir.hydrogeology.h2sPpm,
    estimatedWaterLevelFt: ir.testProgram.estimatedWaterLevelFt,
    estimatedPumpDepthFt: ir.hydrogeology.estimatedPumpDepthFt,
  };
}

/** Same emptiness gate as `hasWellboreDiagramData`, for the IR editor/detail page. `liveCasingRows`/
 *  `liveFormationRows` let the editor substitute its Casing/Formation Tops sections' in-progress
 *  FormArray values while those sections are open for editing — pass `null`/omit for either one
 *  still reads straight from `ir`. `liveProgramMain` does the same for Program Details: while that
 *  section is open for live edit, the gate reacts to the Completion the user is actively picking
 *  right now, same as `buildWellboreDiagramDataFromIntegratedRequirement` below; otherwise it
 *  trusts the persisted `ir.programDetails.wellDesignFlags` (falling back to re-deriving from
 *  `ir.programDetails.completion` for a record fetched before that round-trip existed) — so a
 *  record carrying only a well_design row (no Casing/Formation Top data at all) still shows the
 *  diagram instead of the empty state. */
export function hasWellboreDiagramDataForIntegratedRequirement(
  ir: IntegratedRequirement,
  liveCasingRows?: readonly WwellIrRawCasingRow[] | null,
  liveFormationRows?: readonly WwellIrRawFormationTop[] | null,
  liveProgramMain?: WwellIrRawProgramMain | null,
): boolean {
  const casingRows = liveCasingRows ?? toRawCasingRows(ir.casing);
  const formationRows = liveFormationRows ?? toRawFormationRows(ir.formationTops);
  const completionFlags = liveProgramMain
    ? deriveCompletionElementFlags(liveProgramMain.completion, liveProgramMain.openHoleSizeIn || null)
    : (ir.programDetails.wellDesignFlags ??
      deriveCompletionElementFlags(ir.programDetails.completion, ir.programDetails.openHoleSizeIn));
  return (
    casingRows.some(isRenderableCasingRow) ||
    formationRows.some(isRenderableFormationTop) ||
    hasCompletionElementFlags(completionFlags)
  );
}

/** Detail/editor-page counterpart of `buildWellboreDiagramData` — same live `app-well-bore-view`
 *  diagram, fed from a submitted `IntegratedRequirement` instead of the in-progress wizard form,
 *  so those pages show the real engineering rendering instead of the lighter
 *  `wwell-ir-design-preview` sketch. See `hasWellboreDiagramDataForIntegratedRequirement` above
 *  for what `liveCasingRows`/`liveFormationRows` are for — the editor passes its Casing/Formation
 *  Tops sections' live FormArray values while they're open, so the preview updates as rows are
 *  added/edited instead of only once that section is saved.
 *
 *  `liveProgramMain`/`liveHydrology` are the same idea for Program Details (Completion — the
 *  element flags in `buildWellDesignFromForm`) and Estimated Hydrology (static water level / SIWHP
 *  / pump depth): without these, picking a different Completion or test type while that section is
 *  open wouldn't move the diagram until the whole record was saved, unlike casing/formation-top
 *  edits, which already updated it live.
 *
 *  While Program Details is open for live edit (`liveProgramMain` set), the diagram must react to
 *  the Completion the user is actively picking right now — so this deliberately does NOT pass
 *  `ir.programDetails.wellDesignFlags` (the last-fetched, possibly-stale real DB flags) through in
 *  that case, only once the section is closed/not being edited. See buildWellDesignFromForm's own
 *  doc comment for why the persisted flags win over re-derivation the rest of the time. */
export function buildWellboreDiagramDataFromIntegratedRequirement(
  ir: IntegratedRequirement,
  liveCasingRows?: readonly WwellIrRawCasingRow[] | null,
  liveFormationRows?: readonly WwellIrRawFormationTop[] | null,
  liveProgramMain?: WwellIrRawProgramMain | null,
  liveHydrology?: WwellIrRawHydrology | null,
): WellboreDiagramData {
  return assembleWellboreDiagramData(
    ir.requestReference.wellName,
    ir.requestReference.targetDepthFt,
    liveCasingRows ?? toRawCasingRows(ir.casing),
    liveFormationRows ?? toRawFormationRows(ir.formationTops),
    ir.requestReference.surfaceElevationFtAsl,
    liveProgramMain ?? toRawProgramMain(ir),
    liveHydrology ?? toRawHydrology(ir),
    liveProgramMain ? null : ir.programDetails.wellDesignFlags,
  );
}
