import { ChangeDetectionStrategy, Component, ElementRef, Signal, computed, input, viewChild } from '@angular/core';
import { MatIconModule } from '@angular/material/icon';
import { DIAGRAM_LAYOUT, WellboreDiagramData } from '@models/well-design/wellbore-diagram.model';
import { ApprovalChainStep } from '@models/wwell-planning/wwell-planning.model';
import { IntegratedRequirement } from '@models/wwell-planning/integrated-requirement.model';
import { WellBoreViewComponent } from '../../../../presentation/well-bore-view/well-bore-view.component';
import { WwellIrDepthScaleComponent } from '../../wwell-ir-depth-scale/wwell-ir-depth-scale.component';
import {
  buildWellboreDiagramDataFromIntegratedRequirement,
  hasWellboreDiagramDataForIntegratedRequirement,
} from '../../wwell-ir-form/wwell-ir-wellbore-diagram.mapper';
import { captureWellboreSnapshot } from './wwell-ir-wellbore-snapshot.util';

const OPEN_HOLE_COMPLETION = 'Open Hole';

/** "Letter" step of the read-only IR view — a printable-looking rendition of the GWD Integrated
 *  Requirements Letter, populated straight from the submitted record (never a separate document
 *  fetched from the backend — there is no such endpoint yet). Section numbering/labels follow
 *  the reference GWD letter template; every value comes directly off `IntegratedRequirement`
 *  fields rather than the editor's `KvRow` builders, since the letter's own labels ("Completion
 *  Design", "Logging Program"…) intentionally read as prose, not as the wizard's field names.
 *  The attachment list is appended at the end, per the same record's `attachmentNames`.
 *
 *  The Well Bore Diagram section reuses the exact same diagram build (
 *  buildWellboreDiagramDataFromIntegratedRequirement) as the "Well Design" step's own preview, so
 *  the letter's picture of the well always agrees with what that tab shows for the same record. */
@Component({
  selector: 'app-wwell-ir-view-letter',
  standalone: true,
  imports: [MatIconModule, WellBoreViewComponent, WwellIrDepthScaleComponent],
  templateUrl: './wwell-ir-view-letter.component.html',
  styleUrl: './wwell-ir-view-letter.component.scss',
  changeDetection: ChangeDetectionStrategy.OnPush,
})
export class WwellIrViewLetterComponent {
  readonly ir = input.required<IntegratedRequirement>();

  private readonly wellboreDiagramRow = viewChild<ElementRef<HTMLElement>>('wellboreDiagramRow');

  /** Public (not protected): WwellIrViewComponent's Download PDF button — hoisted onto the tab bar
   *  so it stays put across tab switches instead of living inside the Letter tab's own scrolling
   *  content — calls this through a `viewChild` reference to get the one piece of this letter it
   *  can't produce itself: a rasterized snapshot of the already-rendered wellbore diagram (see
   *  wwell-ir-wellbore-snapshot.util.ts). The rest of the download (irSeq/loading-state/HTTP call)
   *  lives entirely in the parent now — it already has `ir` as its own input, so there's no risk of
   *  reading a required input before Angular has bound it on a child instance (see git history:
   *  that's exactly what broke when this used to live here and get driven via `viewChild`).
   *  Degrades gracefully — a missing/no-data diagram or a capture failure just means the letter
   *  downloads without that section, not a failed download. */
  async captureWellboreImage(): Promise<string | undefined> {
    if (!this.hasWellboreData()) return undefined;
    const row = this.wellboreDiagramRow()?.nativeElement;
    if (!row) return undefined;
    try {
      return (await captureWellboreSnapshot(row)) ?? undefined;
    } catch {
      return undefined;
    }
  }

  protected readonly isOpenHole = computed(() => this.ir().programDetails.completion === OPEN_HOLE_COMPLETION);
  protected readonly isFlowTest = computed(() => this.ir().testProgram.testType === 'flow-test');

  // The real `well-bore-view` layout (not the wizard's 25%-shrunk preview constant) — the letter
  // card is wide enough to afford it, and the shrunk layout's smaller `drawingHeight` was packing
  // casing shoe labels too close together to read (colliding secondary lines) and clipping the
  // longest labels past the SVG's own right edge (see `.letter__diagram-row` in the stylesheet for
  // the matching container width/aspect-ratio change).
  protected readonly diagramLayout = DIAGRAM_LAYOUT;
  protected readonly wellboreDiagramData: Signal<WellboreDiagramData> = computed(() =>
    buildWellboreDiagramDataFromIntegratedRequirement(this.ir()),
  );
  protected readonly hasWellboreData: Signal<boolean> = computed(() => hasWellboreDiagramDataForIntegratedRequirement(this.ir()));
  // Same gate as the Well Design step's own preview (wwell-ir-editor.component.ts's
  // `hasTargetDepth`) — hides the D3-rendered "TOTAL DEPTH: 0 FT" label when it never got filled in.
  protected readonly hasTargetDepth: Signal<boolean> = computed(() => (this.ir().requestReference.targetDepthFt ?? 0) > 0);

  /** The chain's last (most senior) step stands in for the letter's sign-off line — see
   *  WWELL_IR_CHAIN_ROLES (Hydrogeologist → Lead Hydrogeologist → Manager). Mocked for now, same
   *  as the approval-workflow panel next to this stepper (loadIrApprovalTimeline). */
  protected readonly finalApprover = computed<ApprovalChainStep | null>(() => {
    const chain = this.ir().chain;
    return chain.length ? chain[chain.length - 1] : null;
  });

  protected subSeaFt(depthFt: number): number | null {
    const elevation = this.ir().requestReference.surfaceElevationFtAsl;
    return elevation === null ? null : elevation - depthFt;
  }
}

import { ChangeDetectionStrategy, Component, computed, inject, input, output, signal, viewChild } from '@angular/core';
import { MatButtonModule } from '@angular/material/button';
import { MatIconModule } from '@angular/material/icon';
import { MatProgressSpinnerModule } from '@angular/material/progress-spinner';
import { MatTooltipModule } from '@angular/material/tooltip';
import { IntegratedRequirement } from '@models/wwell-planning/integrated-requirement.model';
import { NotificationService } from '@shared/components/notification/notification.service';
import { WWELL_IR_SECTION_RAIL } from '../wwell-ir-editor/wwell-ir-section-rail.config';
import { wwellIrTimelineStepStatus } from '../wwell-ir-editor/wwell-ir-timeline.util';
import { WWELL_IR_STATUS_META } from '../wwell-ir.config';
import { WwellIrService } from '../wwell-ir.service';
import { WwellIrViewDetailsComponent } from './wwell-ir-view-details/wwell-ir-view-details.component';
import { WwellIrViewLetterComponent } from './wwell-ir-view-letter/wwell-ir-view-letter.component';
import { WwellIrViewWellDesignComponent } from './wwell-ir-view-well-design/wwell-ir-view-well-design.component';

export type WwellIrViewTab = 'details' | 'well-design' | 'letter';

const TABS: readonly { readonly key: WwellIrViewTab; readonly label: string; readonly icon: string }[] = [
  { key: 'details', label: 'Details', icon: 'description' },
  { key: 'well-design', label: 'Well Design', icon: 'architecture' },
  { key: 'letter', label: 'Letter', icon: 'mail_outline' },
];

/**
 * Read-only counterpart to `WwellIrEditorComponent`, swapped into the exact same detail-panel
 * slot in WwellIrComponent's master-detail layout (see that component's `detailMode` signal) —
 * never a separate route. The master list stays visible and clickable the whole time a record is
 * open here, so moving to another IR (or another View) is just another row click, same as edit
 * mode already works.
 *
 * A plain flat tab bar (Details / Well Design / Letter) — not a Material stepper — matches the
 * reference GWD/AGWA "Integrated Requirements" screen this was built to look like. The approval-
 * workflow sidebar lives one level up, in WwellIrComponent's own template — it sits beside this
 * panel (not inside it) so it reads as a property of the whole detail column, not of whichever
 * tab is currently open.
 */
@Component({
  selector: 'app-wwell-ir-view',
  standalone: true,
  imports: [
    MatButtonModule,
    MatIconModule,
    MatProgressSpinnerModule,
    MatTooltipModule,
    WwellIrViewDetailsComponent,
    WwellIrViewWellDesignComponent,
    WwellIrViewLetterComponent,
  ],
  templateUrl: './wwell-ir-view.component.html',
  styleUrl: './wwell-ir-view.component.scss',
  changeDetection: ChangeDetectionStrategy.OnPush,
})
export class WwellIrViewComponent {
  readonly ir = input.required<IntegratedRequirement>();
  // Resolves `requestReference.linkedRequestId` to a well name — see the Details step's own
  // input of the same name / buildRequestReferenceRows' doc comment.
  readonly linkedRequestWellOptions = input<readonly { readonly value: string; readonly label: string }[]>([]);

  /** WwellIrComponent listens and flips its `detailMode` back to 'edit' for this same record —
   *  this component has no editing surface of its own. */
  readonly editRequested = output<void>();

  protected readonly statusMeta = WWELL_IR_STATUS_META;

  protected readonly tabs = TABS;
  protected readonly activeTab = signal<WwellIrViewTab>('details');

  private readonly wwellIrService = inject(WwellIrService);
  private readonly notify = inject(NotificationService);

  /** Only resolves while the `letter` tab is active — the `@switch` in the template below destroys
   *  the child otherwise. Used solely to reach `captureWellboreImage()`, the one piece of the
   *  download this component can't produce itself (rasterizing the diagram SVG the Letter tab
   *  renders). Everything else about the download — state, `ir`, the HTTP call — is handled right
   *  here instead of being driven through this query: an earlier version put `canDownloadPdf` and
   *  `downloadPdf()` on the child and read them from here via this same `viewChild`, which threw
   *  NG0950 ("Input is required but no value is available yet") — reading a query into a
   *  freshly-created child can race Angular's own input-binding for that child's `ir`, since view
   *  queries aren't guaranteed to resolve only after a child's inputs are bound. Reading `this.ir()`
   *  (this component's own long-lived input) instead sidesteps that race entirely. */
  private readonly letterView = viewChild(WwellIrViewLetterComponent);

  protected readonly downloadingPdf = signal(false);
  // Only a persisted record has anything for the backend to render (see IrLetterService, which
  // re-fetches the detail by ir_seq) — a local-only "New IR" draft has never reached the backend.
  protected readonly canDownloadPdf = computed(() => this.ir().irSeq !== null && this.ir().irSeq !== undefined);

  protected async downloadPdf(): Promise<void> {
    const irSeq = this.ir().irSeq;
    if (irSeq === null || irSeq === undefined || this.downloadingPdf()) return;

    this.downloadingPdf.set(true);
    const wellboreImageBase64 = await this.letterView()?.captureWellboreImage();
    this.wwellIrService.downloadLetterPdf(irSeq, wellboreImageBase64).subscribe({
      next: (blob: Blob) => {
        this.downloadingPdf.set(false);
        const url = URL.createObjectURL(blob);
        const link = document.createElement('a');
        link.href = url;
        link.download = `${this.ir().code}-letter.pdf`;
        link.click();
        URL.revokeObjectURL(url);
      },
      error: () => {
        this.downloadingPdf.set(false);
        this.notify.error('Failed to download the integrated requirement letter PDF.');
      },
    });
  }

  /** Header stat: how many of the editor's 7 rail sections (Wellbore, General, Horizons, Casing,
   *  Hydrogeology, Area Water Wells, Attachments) hold meaningful data, per the same
   *  wwellIrTimelineStepStatus completeness rule the editor's Timeline view already uses — so this
   *  view's "N/7 sections" reading always agrees with what Timeline mode shows for the same record. */
  protected readonly sectionProgress = computed(() => {
    const ir = this.ir();
    const total = WWELL_IR_SECTION_RAIL.length;
    const completed = WWELL_IR_SECTION_RAIL.filter(group => wwellIrTimelineStepStatus(ir, group.key) === 'completed').length;
    return { completed, total, percent: Math.round((completed / total) * 100) };
  });

  protected setTab(tab: WwellIrViewTab): void {
    this.activeTab.set(tab);
  }
}
