```ts
  private previewRows<TRow>(
    live: readonly TRow[] | null,
    saved: () => readonly TRow[],
    draftRows: readonly TRow[] | null,
    keyOf: (row: TRow) => string,
  ): readonly TRow[] | null {
    if (!draftRows?.length) return live;
    const rows = live ?? saved();
    const keys = new Set(rows.map(keyOf));
    // "Add" refuses a row whose natural key is already taken (it would overwrite that row) — including by
    // another staged draft row — so a colliding draft is not drawn either.
    const additions = draftRows.filter(draft => {
      const key = keyOf(draft);
      if (keys.has(key)) return false;
      keys.add(key);
      return true;
    });
    return additions.length ? [...rows, ...additions] : live;
  }

  /** Commits every row staged in a "+" add-form's own FormArray into the section's live FormArray — the
   *  same array inline edits and deletes write to, so Save diffs it like any other change. All-or-nothing:
   *  Save matches rows on their natural key, so if ANY staged row's key is already taken (by an existing
   *  row, or by another row in the same batch) the whole commit is refused rather than silently overwriting
   *  that row, or committing some rows and leaving the colliding one stranded in the add-form. */
  private commitNewRows<TRow>(
    add: SectionEditState<FormArray>,
    section: WwellFormSectionConfig,
    commit: {
      readonly existingKeys: readonly string[];
      readonly keyOf: (row: TRow) => string;
      readonly liveArray: () => FormArray | null;
      readonly duplicate: (key: string) => string;
      readonly added: (count: number) => string;
    },
  ): void {
    const form = add.control();
    if (!form || form.invalid || form.length === 0) return;

    const rows = form.getRawValue() as TRow[];
    const keys = new Set(commit.existingKeys);
    for (const row of rows) {
      const key = commit.keyOf(row);
      if (keys.has(key)) {
        this.notify.warn(commit.duplicate(key));
        return;
      }
      keys.add(key);
    }

    const array = commit.liveArray();
    if (!array) return;
    rows.forEach(row => {
      addRepeatableRow(array, section.repeatable?.rowFields ?? [], this.fb);
      array.at(array.length - 1).patchValue(row as object);
    });
    add.close();
    this.notify.info(commit.added(rows.length));
  }

  private saveSettledResolvers: ((succeeded: boolean) => void)[] = [];
  private wasSavingForNavigation = false;

  constructor() {
    effect(onCleanup => {
      const ctrl = this.requestReferenceEdit.control();
      if (!ctrl) return;
      const linkedRequestIdControl = ctrl.get('linkedRequestId');
      const subscription = linkedRequestIdControl?.valueChanges.subscribe((linkedRequestId: string) => {
        console.log('[IR:link] "Link to Request" select — valueChanges fired', { linkedRequestId });
        if (linkedRequestId) this.applyLinkedRequestWell(ctrl, linkedRequestId);
      });
      onCleanup(() => subscription?.unsubscribe());
    });

    effect(() => {
      const id = this.ir().id;
      if (this.previousIrId !== null && id !== this.previousIrId) this.closeAllSections();
      this.previousIrId = id;
    });

    effect(() => {
      if (this.ir().irSeq != null) return;
      untracked(() => this.requestReferenceEdit.open());
    });

    effect(() => {
      const saving = this.store.creatingIr();
      const settled = this.wasSavingForNavigation && !saving;
      this.wasSavingForNavigation = saving;
      if (!settled || this.saveSettledResolvers.length === 0) return;

      const succeeded = !this.store.createIrError();
      const resolvers = this.saveSettledResolvers;
      this.saveSettledResolvers = [];
      resolvers.forEach(resolve => resolve(succeeded));
    });
  }

  public discardUnsavedChanges(): void {
    this.closeAllSections();
  }

  public saveUnsavedChangesAndWait(): Promise<boolean> {
    return new Promise(resolve => {
      if (this.store.creatingIr()) {
        this.saveSettledResolvers.push(resolve);
        return;
      }
      if (this.saveAll()) {
        this.saveSettledResolvers.push(resolve);
      } else {
        resolve(false);
      }
    });
  }

  private applyLinkedRequestWell(referenceGroup: FormGroup, linkedRequestId: string): void {
    console.log('[IR:link] applyLinkedRequestWell — start', { linkedRequestId, eligibleWellsLoaded: this.store.linkedRequestWells().length });
    const well = findLinkedWell(linkedRequestId, this.store.linkedRequestWells());
    if (!well) {
      console.log('[IR:link] applyLinkedRequestWell — no matching eligible well, aborting patch', { linkedRequestId });
      return;
    }

    const hasElevation = linkedWellHasElevation(well);
    const hasCoordinates = linkedWellHasCoordinates(well);
    const hasTargetDepth = linkedWellHasTargetDepth(well);
    const hasTargetAquifer = linkedWellHasTargetAquifer(well);
    const hasSupportingBusiness = linkedWellHasSupportingBusiness(well);

    const patch: Record<string, string | number> = { wellName: well.wellName };
    if (hasElevation) patch['surfaceElevationFtAsl'] = well.surfaceElevation as number;
    patch['coordinates'] = hasCoordinates ? formatLinkedWellCoordinates(well) : '';
    if (hasTargetDepth) patch['targetDepthFt'] = well.targetDepth as number;
    patch['targetAquifer'] = hasTargetAquifer ? (well.targetAquifer as string) : '';
    patch['supportingBusiness'] = hasSupportingBusiness ? (well.supportings as string) : '';
    console.log('[IR:link] applyLinkedRequestWell — patching form group', {
      patch,
      hasElevation,
      hasCoordinates,
      hasTargetDepth,
      hasTargetAquifer,
      hasSupportingBusiness,
    });
    referenceGroup.patchValue(patch);

    referenceGroup.get('wellName')?.disable();
    this.setLinkedFieldLock(referenceGroup, 'surfaceElevationFtAsl', hasElevation);
    console.log('[IR:link] applyLinkedRequestWell — done', { rawValue: referenceGroup.getRawValue() });
  }

  /** Returns whether a save was actually kicked off — `false` on every early-return branch below
   *  (nothing to save, or a blocked validation). `saveUnsavedChangesAndWait` needs this to settle
   *  its own promise immediately in those cases instead of waiting forever for a `creatingIr`
   *  transition that will now never happen. */
  protected saveAll(): boolean {
    // A record whose stored remarks still carry legacy HTML is saved even with nothing edited — the form already
    // holds the cleaned text, and saving is what writes it back.
    if (this.ir().irSeq !== null && !this.hasUnsavedChanges() && !this.ir().hasLegacyMarkup) {
      this.notify.info('No changes to save.');
      return false;
    }

    let raw = this.currentRawFormValue();

    const referenceCtrl = this.requestReferenceEdit.control();
    if (referenceCtrl) raw = { ...raw, reference: { main: referenceCtrl.getRawValue() } };

    // `letterNumber` carries a `Validators.required` in the schema (wwell-ir-form.schema.ts), but
    // nothing elsewhere in this method ever checks form validity before saving — it only gathers
    // raw values. Without this explicit gate, a blank Letter Number saved silently; the asterisk
    // and mat-error only ever showed once the field itself had been touched.
    if (!raw.reference.main.letterNumber?.trim()) {
      this.requestReferenceEdit.open();
      this.requestReferenceEdit.control()?.get('letterNumber')?.markAsTouched();
      this.notify.error('Letter Number is required before saving.');
      return false;
    }

    const programCtrl = this.programEdit.control();
    if (programCtrl) {
      const flat = programCtrl.getRawValue() as WwellIrRawProgramDetails;
      raw = {
        ...raw,
        programDetails: {
          main: {
            completionType: flat.completionType,
            completion: flat.completion,
            openHoleSizeIn: flat.openHoleSizeIn,
            linerSizeIn: flat.linerSizeIn,
            linerDepthFt: flat.linerDepthFt,
            acidization: flat.acidization,
            transmission: flat.transmission,
          },
          developmentTesting: {
            logging: flat.logging,
            mudLogging: flat.mudLogging,
            sampling: flat.sampling,
            flowTest: flat.flowTest,
            remarks: flat.remarks,
          },
        },
      };
    }

    const geologistCtrl = this.geologistContactEdit.control();
    if (geologistCtrl) raw = { ...raw, contacts: { ...raw.contacts, geologist: geologistCtrl.getRawValue() } };

    const hydrogeologyCtrl = this.hydrogeologyEdit.control();
    if (hydrogeologyCtrl) raw = { ...raw, hydrogeology: { main: hydrogeologyCtrl.getRawValue() } };

    const formationTopsCtrl = this.formationTopsEdit.control();
    if (formationTopsCtrl) raw = { ...raw, subsurface: { ...raw.subsurface, formationTops: formationTopsCtrl.getRawValue() } };

    const topsRemarksCtrl = this.topsRemarksEdit.control();
    if (topsRemarksCtrl) raw = { ...raw, subsurface: { ...raw.subsurface, remarks: topsRemarksCtrl.getRawValue() } };

    const casingCtrl = this.casingEdit.control();
    if (casingCtrl) raw = { ...raw, subsurface: { ...raw.subsurface, casing: casingCtrl.getRawValue() } };

    const offsetWellsCtrl = this.offsetWellsEdit.control();
    if (offsetWellsCtrl) raw = { ...raw, contacts: { ...raw.contacts, offsetWells: offsetWellsCtrl.getRawValue() } };

    const offsetRemarksCtrl = this.offsetRemarksEdit.control();
    if (offsetRemarksCtrl) raw = { ...raw, contacts: { ...raw.contacts, remarks: offsetRemarksCtrl.getRawValue() } };

    this.saveMergedRaw(raw);
    this.closeAllSections();
    return true;
  }

  /** Recalling stops a live SAP workflow — uses the same "confirm, then act" pattern as the other
   *  row-delete actions above, since it's not reversible from here (SAP itself would need a fresh
   *  Submit for Review). */
  protected recallIr(): void {
```

/** SAP text that means the workflow was actively killed rather than merely progressing — matched
 *  case-insensitively, the same way wwell-ir-approval.mapper.ts's stepStatus() reads a single
 *  step's own `response`. SAP's exact wording for these isn't part of our contract (the /display
 *  response only documents `wf_status_text` as free text), so this matches on the words themselves

const SAP_APPROVED_STATUS_TEXT = /approve/i;
const SAP_RECALLED_STATUS_TEXT = /recall/i;

 *  list instead gates the one status-transition action that still needs a distinct status ("Submit
 *  for Review" only makes sense while the record hasn't already been submitted). */

 *  (`ir.sapWfStatusText ?? statusMeta[ir.status].label`), but the colour now follows the SAP text

 *  pre-submission — see WwellIrStore.loadIrApprovalTimeline), so without this a rejected/cancelled/
 *  recalled record kept showing its old "In Review"/"Submit for Review" colour with only the label
 *  text changing underneath it. Recognised SAP text is shown red (terminated) or green (approved);
 *  anything else keeps riding on `ir.status`'s own colour, only the label swaps, same as before. */
export function irBadgeMeta(ir: IntegratedRequirement): WwellStatusMeta {
  const text = ir.sapWfStatusText;
  const fallback = WWELL_IR_STATUS_META[ir.status];
  if (!text) return fallback;
  if (SAP_TERMINATED_STATUS_TEXT.test(text)) return wwellStatusMeta(text, WWELL_STATUS_COLORS.red);
  if (SAP_APPROVED_STATUS_TEXT.test(text) && isChainComplete(ir)) return wwellStatusMeta(text, WWELL_STATUS_COLORS.green);
  if (SAP_APPROVED_STATUS_TEXT.test(text)) return wwellStatusMeta(text, WWELL_STATUS_COLORS.amber);
  return { ...fallback, label: text };
}

/** True once SAP's own text says the workflow is done moving — rejected, cancelled, recalled, or
 *  approved — same two patterns irBadgeMeta colours red/green. Gates the editor's "Recall" button
 *  (WwellIrEditorComponent.canRecallIr): a workflow already at a terminal SAP state has nothing left
 *  to recall, same as a real SAP recall would refuse it. */
export function isSapWorkflowTerminal(sapWfStatusText: string | null): boolean {
  return sapWfStatusText !== null && (SAP_TERMINATED_STATUS_TEXT.test(sapWfStatusText) || SAP_APPROVED_STATUS_TEXT.test(sapWfStatusText));
}

 *  SAP workflow id. Deliberately its own predicate rather than reusing `isSapWorkflowTerminal` — the
 *  other three terminal outcomes stay terminal, with no way back from the editor. */
export function isSapWorkflowRecalled(sapWfStatusText: string | null): boolean {
  return sapWfStatusText !== null && SAP_RECALLED_STATUS_TEXT.test(sapWfStatusText);
}

export function isSapWorkflowResubmittable(sapWfStatusText: string | null): boolean {
  return sapWfStatusText !== null && SAP_TERMINATED_STATUS_TEXT.test(sapWfStatusText);
}

/** Classifies the live SAP status text into the IR letter PDF's three stamp states — the exact
 *  same two patterns `irBadgeMeta` uses to colour the badge red/green, reused here so this is the

export function resolveLetterStampStatus(ir: IntegratedRequirement): 'APPROVED' | 'REJECTED' | 'PENDING' | null {

  if (!text) return null;
  if (SAP_TERMINATED_STATUS_TEXT.test(text)) return 'REJECTED';
  if (SAP_APPROVED_STATUS_TEXT.test(text)) return isChainComplete(ir) ? 'APPROVED' : 'PENDING';

}

```ts
import { ChangeDetectionStrategy, Component, ElementRef, computed, input, signal, viewChild } from '@angular/core';
import { toObservable, toSignal } from '@angular/core/rxjs-interop';
import { FormControl, ReactiveFormsModule } from '@angular/forms';
import { MatAutocompleteModule } from '@angular/material/autocomplete';
import { MatButtonToggleModule } from '@angular/material/button-toggle';
import { MatFormFieldModule } from '@angular/material/form-field';
import { MatIconModule } from '@angular/material/icon';
import { MatInputModule } from '@angular/material/input';
import { MatSelectModule } from '@angular/material/select';
import { switchMap, startWith } from 'rxjs';
import { WwellAttachmentDropzoneComponent } from '../../wwell-attachment-dropzone/wwell-attachment-dropzone.component';
import { WwellRichTextareaComponent } from '../rich-textarea/wwell-rich-textarea.component';
import {
  WwellFieldConfig,
  resolveFieldOptions,
  wwellFieldOptionIcon,
  wwellFieldOptionLabel,
  wwellFieldOptionValue,
} from '../wwell-dynamic-form.types';

export type WwellDynamicFieldMode = 'edit' | 'review';

@Component({
  selector: 'app-dynamic-field',
  standalone: true,
  imports: [
    ReactiveFormsModule,
    MatFormFieldModule,
    MatInputModule,
    MatSelectModule,
    MatButtonToggleModule,
    MatAutocompleteModule,
    MatIconModule,
    WwellAttachmentDropzoneComponent,
    WwellRichTextareaComponent,
  ],
  templateUrl: './dynamic-field.component.html',
  styleUrl: './dynamic-field.component.scss',
  changeDetection: ChangeDetectionStrategy.OnPush,
})
export class DynamicFieldComponent {
  readonly config = input.required<WwellFieldConfig>();
  readonly control = input.required<FormControl>();
  readonly mode = input<WwellDynamicFieldMode>('edit');
  /** Class applied to this field's portaled select/autocomplete overlay panel, so a host page can theme it. */
  readonly selectPanelClass = input('');

  protected readonly optionLabel = wwellFieldOptionLabel;
  protected readonly optionValue = wwellFieldOptionValue;
  protected readonly optionIcon = wwellFieldOptionIcon;

  /** Resolves the config's `options` (static array or live Signal) once per change — every
   *  options `@for` loop below reads this instead of `config().options` directly. */
  protected readonly resolvedOptions = computed(() => resolveFieldOptions(this.config()));

  /** Backs the filter input a `searchable: true` 'select' field projects into its own mat-select
   *  panel (see the template) — mat-select has no built-in search. Stays '' for every other
   *  field, since nothing renders an input to change it, so `filteredOptions` below is then just
   *  `resolvedOptions()` unfiltered. */
  protected readonly optionSearchTerm = signal('');

  /** The searchable-select panel's filter `<input>` — queried instead of using the `autofocus`
   *  attribute (flagged by @angular-eslint/template/no-autofocus and unreliable for content
   *  that mounts into a CDK overlay after the panel opens). */
  private readonly searchInput = viewChild<ElementRef<HTMLInputElement>>('searchInput');

  protected readonly filteredOptions = computed(() => {
    const term = this.optionSearchTerm().trim().toLowerCase();
    if (!term) return this.resolvedOptions();
    return this.resolvedOptions().filter(opt => this.optionLabel(opt).toLowerCase().includes(term));
  });

  /** Clears the filter on close so reopening the panel always starts from the full list instead
   *  of whatever was last typed. On open, focuses the filter input once the overlay panel has
   *  attached — a plain template `autofocus` fires before that element exists in the DOM. */
  protected onSelectOpenedChange(opened: boolean): void {
    if (!opened) {
      this.optionSearchTerm.set('');
      return;
    }
    setTimeout(() => this.searchInput()?.nativeElement.focus());
  }

  protected onOptionSearchInput(event: Event): void {
    this.optionSearchTerm.set((event.target as HTMLInputElement).value);
  }

  /**
   * `control()` is a Signal input, but the bound FormControl's own `.value` mutates in place
   * on every keystroke and is never itself signal-tracked — reading it directly inside a
   * `computed()` (as `reviewValue` below used to) only evaluates once, at first read, and then
   * freezes forever: the Attachments & Review step's tiles kept showing '—' even after the
   * matching field was filled in on an earlier step. Bridging through `valueChanges` (and
   * re-subscribing whenever a *different* control is bound) keeps this genuinely reactive.
   */
  // No `initialValue`/`requireSync`: reading `this.control()` synchronously in a field
  // initializer (for an `initialValue`) fails the compiler's NG8118 check — inputs aren't bound
  // yet at that point — and `toObservable` doesn't emit synchronously either (its effect is
  // scheduled, not immediate), so `requireSync` throws NG0601 at construction instead. Left
  // without either, `liveValue()` is simply `undefined` for the first tick, which `reviewValue`
  // below already treats the same as an empty value.
  private readonly liveValue = toSignal(
    toObservable(this.control).pipe(switchMap(control => control.valueChanges.pipe(startWith(control.value)))),
  );

  protected readonly reviewValue = computed(() => {
    const value = this.liveValue();
    const cfg = this.config();

    if (cfg.type === 'file') {
      const files = value as File[] | null;
      return files?.length ? `${files.length} file(s) attached` : '—';
    }
    if (cfg.type === 'select' || cfg.type === 'toggle-group') {
      const match = this.resolvedOptions().find(opt => wwellFieldOptionValue(opt) === value);
      return match ? wwellFieldOptionLabel(match) : (value || '—');
    }
    return value === null || value === undefined || value === '' ? '—' : String(value);
  });

  /** True once the user has changed this control (Angular's `dirty`, set only by user input — the
   *  programmatic patches that seed a form leave it false). Re-evaluated on every value change; stays
   *  true until the owner resets the control (Save/Discard rebuild or `markAsPristine`). */
  protected readonly isModified = computed(() => {
    this.liveValue();
    return this.control().dirty;
  });

  protected readonly isReviewValueEmpty = computed(() => this.reviewValue() === '—');

  /** Non-null only for a field carrying `splitDisplay` — the parts it renders as separate boxes
   *  instead of the type-based `@switch` in the template. */
  protected readonly splitParts = computed(() => {
    const split = this.config().splitDisplay;
    if (!split) return null;
    const value = this.liveValue();
    return split(typeof value === 'string' ? value : '');
  });

  protected onFilesChange(files: File[]): void {
    this.control().setValue(files);
    this.control().markAsTouched();
  }

  protected errorMessage(): string | null {
    const control = this.control();
    if (!control.touched || control.valid) return null;
    if (control.hasError('required')) return `${this.config().label} is required`;
    if (control.hasError('pattern')) return `${this.config().label} format is invalid`;
    if (control.hasError('min')) return `${this.config().label} is below the minimum`;
    return 'Invalid value';
  }
}
```

```ts
import { computed, signal } from '@angular/core';
import { AbstractControl, FormArray, FormBuilder, FormGroup } from '@angular/forms';
import { Subscription } from 'rxjs';
import { addRepeatableRow, buildSectionGroup } from '../../shared/dynamic-form/wwell-dynamic-form.builder';
import { WwellFormSectionConfig } from '../../shared/dynamic-form/wwell-dynamic-form.types';

/**
 * One independent edit toggle for one editor section — the IR editor creates one instance per
 * inline-editable section instead of a single shared "which section is being edited" flag, so any
 * number of sections can be mid-edit at once without stepping on each other.
 *
 * `liveValue` mirrors the control's raw value as a signal, kept in sync via `valueChanges` for as
 * long as the section stays open — this is what lets a consumer (e.g. the Wellbore Preview) react
 * to an in-progress edit instead of only the last-saved record. It's `null` whenever the section
 * isn't being edited, so callers should fall back to their own persisted data in that case.
 *
 * `isEditing` vs `visible`: a row delete from the read-only table (see deleteCasingRow etc. in
 * wwell-ir-editor.component.ts) needs a live control to remove the row from and to source Save's
 * payload/diff from — same mechanism a real edit uses — but must NOT flip the whole section over
 * to its big edit-form UI just to remove one row; that swap is what made a single-row delete read
 * as "the whole section reloaded". `open({ silent: true })` builds the control (or reuses the
 * existing one) without revealing the edit form; `isEditing` still reports true so
 * hasUnsavedChanges (which must catch a silent delete too) is unaffected, while `visible` — what
 * the template actually gates the edit-form-vs-read-table branch on — stays false until an
 * explicit `open()`/`open({ silent: false })` (the pencil buttons) asks for it.
 */
export class SectionEditState<TControl extends FormGroup | FormArray, TValue = unknown> {
  private readonly controlSignal = signal<TControl | null>(null);
  readonly control = this.controlSignal.asReadonly();
  readonly isEditing = computed(() => this.controlSignal() !== null);

  private readonly visibleSignal = signal(false);
  readonly visible = this.visibleSignal.asReadonly();

  private readonly liveValueSignal = signal<TValue | null>(null);
  readonly liveValue = this.liveValueSignal.asReadonly();
  private subscription: Subscription | null = null;

  constructor(private readonly buildControl: () => TControl) {}

  /** No-op on the control itself if already editing — reopening must never discard in-progress
   *  edits with a fresh snapshot. A visible open reveals the form even if it was already open
   *  silently (e.g. the user deletes a row, then clicks the section's own pencil afterward). */
  open(opts?: { silent?: boolean }): void {
    if (!this.isEditing()) {
      const control = this.buildControl();
      this.controlSignal.set(control);
      this.liveValueSignal.set(control.getRawValue());
      this.subscription = control.valueChanges.subscribe(() => this.liveValueSignal.set(control.getRawValue()));
    }
    if (!opts?.silent) this.visibleSignal.set(true);
  }

  close(): void {
    this.subscription?.unsubscribe();
    this.subscription = null;
    this.controlSignal.set(null);
    this.liveValueSignal.set(null);
    this.visibleSignal.set(false);
  }
}

/** Builds a section's plain-fields FormGroup (see buildSectionGroup) and seeds it from an
 *  already-saved record — the editor's counterpart to populateFormForEdit's `.patchValue()`. */
export function buildAndPatchGroup(section: WwellFormSectionConfig, fb: FormBuilder, patch: object): FormGroup {
  const group = buildSectionGroup(section, fb) as FormGroup;
  group.patchValue(patch);
  return group;
}

/** Builds a repeatable section's FormArray and grows+fills it to match existing domain rows. */
export function buildAndPatchArray<TDomainRow>(
  section: WwellFormSectionConfig,
  fb: FormBuilder,
  rows: readonly TDomainRow[],
  toRowPatch: (row: TDomainRow) => object,
): FormArray {
  const array = buildSectionGroup(section, fb) as FormArray;
  const rowFields = section.repeatable?.rowFields ?? [];
  rows.forEach(row => {
    addRepeatableRow(array, rowFields, fb);
    array.at(array.length - 1).patchValue(toRowPatch(row));
  });
  return array;
}

/**
 * Tracks which single row (by index) of a repeatable section's live FormArray is being edited
 * inline, directly inside the read-only table — the row-level counterpart to `SectionEditState`'s
 * `visible`/silent split: a delete already avoids opening the whole section's form (see
 * deleteCasingRow etc. in wwell-ir-editor.component.ts), and this does the same for a row-level
 * edit, so "Edit" on one row never swaps the whole table over to the big multi-row form either.
 *
 * Editing writes straight into the FormArray's own row control (via the same `xEdit.control()`
 * a delete opens silently), so Save's `existing`-diff sees it exactly like any other edit — no
 * separate draft/merge step needed. `cancel()` restores whatever the row held right when `open()`
 * was called, not the persisted record, so it stays correct even if an earlier uncommitted
 * delete/edit already sits in the same FormArray this session (index-based, not identity-based, so
 * it only ever needs the row's own control, not the whole array).
 */
export class InlineRowEditState {
  private readonly indexSignal = signal<number | null>(null);
  readonly index = this.indexSignal.asReadonly();
  private snapshot: unknown = null;
  private wasPristine = true;

  isEditing(rowIndex: number): boolean {
    return this.indexSignal() === rowIndex;
  }

  open(rowIndex: number, control: AbstractControl): void {
    this.snapshot = control.getRawValue();
    this.wasPristine = control.pristine;
    this.indexSignal.set(rowIndex);
  }

  /** No-op if nothing was ever opened (e.g. the row control disappeared from under it). */
  cancel(control: AbstractControl | undefined | null): void {
    if (control && this.snapshot !== null) {
      control.patchValue(this.snapshot);
      // Restored to what it held before this edit began, so the unsaved-change highlight goes with it.
      if (this.wasPristine) control.markAsPristine();
    }
    this.close();
  }

  close(): void {
    this.indexSignal.set(null);
    this.snapshot = null;
  }
}
```
