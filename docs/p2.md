import { easeBackOut, easeBounceOut, easeCubicInOut, easeCubicOut, easeLinear } from 'd3-ease';
import 'd3-transition';

import { WellboreAnimConfig } from '@models/well-design/wellbore-diagram.model';
import { GSel, ResolvedAnimation } from './wellbore-renderer.types';

export function resolveAnimation(config: WellboreAnimConfig): ResolvedAnimation {
  const mult = config.speed === 'slow' ? 1.8 : config.speed === 'fast' ? 0.4 : 1.0;
  const popEase = config.popStyle === 'bounce' ? easeBounceOut
    : config.popStyle === 'elastic' ? easeBackOut
      : easeCubicInOut;

  return {
    lineDur: Math.round(500 * mult),
    fadeDur: Math.round(300 * mult),
    popDur: Math.round(600 * mult),
    lineStyle: config.lineStyle,
    popStyle: config.popStyle,
    popEase,
  };
}

export function animateLabel(group: GSel, delay: number, animation: ResolvedAnimation): void {
  const lineEl = group.select<SVGLineElement>('line');
  if (animation.lineStyle === 'draw') {
    const x1 = parseFloat(lineEl.attr('x1'));
    const x2 = parseFloat(lineEl.attr('x2'));
    lineEl
      .attr('stroke-dasharray', Math.abs(x2 - x1))
      .attr('stroke-dashoffset', Math.abs(x2 - x1))
      .transition().delay(delay).duration(animation.lineDur).ease(easeLinear)
      .attr('stroke-dashoffset', 0);
    group.selectAll<SVGElement, unknown>('path')
      .style('opacity', 0)
      .attr('transform', 'translate(4,0)')
      .transition().delay(delay + animation.lineDur).duration(animation.fadeDur).ease(easeBackOut)
      .attr('transform', 'translate(0,0)')
      .style('opacity', 1);
    group.selectAll<SVGElement, unknown>('text')
      .style('opacity', 0)
      .attr('transform', 'translate(8,0)')
      .transition().delay(delay + animation.lineDur + 80).duration(animation.fadeDur).ease(easeCubicOut)
      .attr('transform', 'translate(0,0)')
      .style('opacity', 1);
  } else {
    group.style('opacity', 0)
      .attr('transform', 'translate(6,0)')
      .transition().delay(delay).duration(animation.lineDur + animation.fadeDur).ease(easeCubicInOut)
      .attr('transform', 'translate(0,0)')
      .style('opacity', 1);
  }
}

export function applyPop(wrap: GSel, cx: number, cy: number, delay: number, animation: ResolvedAnimation): void {
  if (animation.popStyle === 'fade') {
    wrap.attr('transform', `translate(${cx},${cy}) scale(1)`)
      .style('opacity', 0)
      .transition().delay(delay).duration(animation.popDur).ease(easeCubicInOut)
      .style('opacity', 1);
    return;
  }

  wrap.attr('transform', `translate(${cx},${cy}) scale(0)`)
    .transition().delay(delay).duration(animation.popDur).ease(animation.popEase)
    .attr('transform', `translate(${cx},${cy}) scale(1)`);
}

/** Squeezes a right-hand leader label to the room left of the diagram's right edge, so its last characters are never clipped
 *  by the SVG (e.g. `30" Conductor @ 100` losing its final digit). A label that already fits is left untouched. */
export function fitLabelToRoom(
  text: { attr(name: string, value: string | number): unknown; node(): unknown },
  labelX: number,
  viewWidth: number,
): void {
  const room = viewWidth - labelX - RIGHT_EDGE_PADDING;
  const measured = (text.node() as SVGTextElement | null)?.getComputedTextLength?.() ?? 0;
  if (room > 0 && measured > room) {
    text.attr('textLength', room);
    text.attr('lengthAdjust', 'spacingAndGlyphs');
  }
}

const RIGHT_EDGE_PADDING = 6;

function renderCasings({
    rootG,
    defsEl,
    layout,
    casings,
    drlgCasings,
    centerX,
    scale,
    casingsStart,
    gravelStart,
    openHoleHalfWidthPx,

    if (casings.length === 0) {
        renderCasingsFromActual({ rootG, defsEl, layout, drlgCasings, centerX, scale, casingsStart });

    }

    const { baseHalfWidth, halfWidthIncrement, shoeCurveOffset } = layout;
    const completionTopPx = scale(resolveCompletionTopDepth(casings, drlgCasings));

    casings.forEach((csg, i) => {
        if (csg.csgType === 'Liner') return;
        const tier = getCasingTier(csg, casings);
        const hw = computeCasingHalfWidth(tier, baseHalfWidth, halfWidthIncrement);
        const isConductor = csg.csgType.toLowerCase() === 'conductor';
        const isGP = csg.csgType === 'Gravel Pack';
        const actualCsg = drlgCasings.find(d => d.wCsgOdSz === csg.csgSize) ?? null;
        const hasActual = actualCsg !== null && actualCsg.wCsgBotDpth > 0;
        const displayDepth = hasActual ? actualCsg!.wCsgBotDpth : csg.csgDepth;
        const bottomPx = scale(displayDepth);
        const clipId = `dyn-casing-clip-${i}`;
        const animOrder = casings.length - 1 - i;
        const delay = animOrder * ANIM.CASING_STAGGER;
        const gpInnerHW = computeCasingHalfWidth(0, layout.baseHalfWidth, layout.halfWidthIncrement);
        const gpLsHW = Math.min(gpInnerHW, openHoleHalfWidthPx) - layout.linerScreenInset;
        const gpMaxOuter = openHoleHalfWidthPx - layout.gpScreenGap;
        const gpAnnulusWidth = Math.max(0, (gpMaxOuter - gpLsHW) * layout.gpFillRatio);
        const gpOuterEdge = gpLsHW + gpAnnulusWidth;
        const gpAnnulusCenter = gpLsHW + gpAnnulusWidth / 2;
        const clipRadius = isGP ? gpOuterEdge : openHoleHalfWidth(hw) + 20;

        const clipRect = defsEl
            .append('clipPath')
            .attr('class', 'dyn-clip')
            .attr('id', clipId)
            .append('rect')
            .attr('x', centerX - clipRadius - (isGP ? 0 : 15))
            .attr('y', isGP ? completionTopPx : -5)
            .attr('width', clipRadius * 2 + (isGP ? 0 : 30))
            .attr('height', 0);

        if (isGP) {
            const gravelBottomPx = scale(csg.csgDepth);
            rootG.append('path')
                .attr('class', 'gravelHole')
                .attr('d', buildGravelPackUPath(centerX, gpAnnulusCenter, completionTopPx, gravelBottomPx))
                .attr('stroke', 'url(#gravelpattern)')
                .attr('stroke-width', String(gpAnnulusWidth))
                .style('fill', 'none')
                .attr('clip-path', `url(#${clipId})`);
            clipRect.transition()
                .delay(gravelStart)
                .duration(ANIM.GRAVEL_DURATION)
                .ease(easeCubicInOut)
                .attr('height', gravelBottomPx - completionTopPx);
            return;
        }

        const casingD = buildCasingPath(centerX, hw, 0, bottomPx, shoeCurveOffset);

        if (hasActual) {
            rootG.append('path')
                .attr('class', 'casing')
                .attr('d', casingD)
                .attr('clip-path', `url(#${clipId})`)
                .attr('fill', 'url(#actualCasingGradient)')

function uniqueCasingsByOd(drlgCasings: IDrlgCsg[]): IDrlgCsg[] {
    const map = new Map<string, IDrlgCsg>();
    for (const d of drlgCasings) {
        if (d.wCsgBotDpth <= 0) continue;
        const existing = map.get(d.wCsgOdSz);
        if (!existing || d.wCsgBotDpth > existing.wCsgBotDpth) {
            map.set(d.wCsgOdSz, d);
        }
    }
    return [...map.values()].sort((a, b) => b.wCsgBotDpth - a.wCsgBotDpth);
}

/**
 * Implements docs/plan/logic.md Rules 1-3 over a depth-only liner list: which liner (if
 * any) is the 7" completion target, and the depth interval Open Hole / Gravel Pack /
 * Perforated Interval share as a result.
 */
export function resolveLinerRule(liners: readonly RawLinerEntry[], lastCasingShoe: number): LinerRule {
    // Rule 1: No Liner
    if (liners.length === 0) {
        return { kind: 'none', startDepth: lastCasingShoe, structuralEndDepth: null, targetLiner: null, boundaryLiner: null };
    }

    // Rule 2: Single Liner
    if (liners.length === 1) {
        const only = liners[0];
        if (!is7InchLiner(only)) {
            // Not the 7" screen liner: still an ordinary liner with open hole/completion
            // below it — draw from its own shoe down to total/current/target depth, same
            // as a no-match multi-liner well (docs/plan/logic.md leaves this case
            // unaddressed; this mirrors Rule 3's fallback rather than drawing nothing).
            return { kind: 'single-fallback', startDepth: only.botDepth, structuralEndDepth: null, targetLiner: null, boundaryLiner: only };
        }
        return { kind: 'single', startDepth: lastCasingShoe, structuralEndDepth: only.botDepth, targetLiner: only, boundaryLiner: only };
    }

    // Rule 3: Multiple Liners
    const targetIndex = resolveTargetLinerIndex(liners);
    if (targetIndex !== null) {
        const target = liners[targetIndex];
        const startDepth = targetIndex === 0 ? lastCasingShoe : liners[targetIndex - 1].botDepth;
        return {
            kind: 'multi-target',
            startDepth,
            structuralEndDepth: target.botDepth,
            targetLiner: target,
            boundaryLiner: targetIndex === 0 ? null : liners[targetIndex - 1],
        };
    }

    // Fallback: multiple liners, none match 7"
    const deepest = liners[liners.length - 1];
    return { kind: 'multi-fallback', startDepth: deepest.botDepth, structuralEndDepth: null, targetLiner: null, boundaryLiner: deepest };
}

/**
 * Extracts liner entries from drlgCasings when no ICasingIR data is available.
 * Deduplicates by wLnrOdSz, sorts shallowest-first.
 */
function rawLinersFromDrlg(drlgCasings: IDrlgCsg[]): RawLinerEntry[] {
    const dedupMap = new Map<string, RawLinerEntry>();
    for (const d of drlgCasings) {
        if (!d.wLnrOdSz || (d.wLnrBotDepth ?? 0) <= 0) continue;
        const existing = dedupMap.get(d.wLnrOdSz);
        if (!existing || d.wLnrBotDepth! > existing.botDepth) {
            dedupMap.set(d.wLnrOdSz, {
                odSz: d.wLnrOdSz,
                topDepth: d.wLnrTopDepth ?? d.wCsgBotDpth,
                botDepth: d.wLnrBotDepth!,
                hasActual: true,
                hasActualTop: (d.wLnrTopDepth ?? 0) > 0,
            });
        }
    }

    return [...dedupMap.values()].sort((a, b) => a.botDepth - b.botDepth);
}

/**
 * Fallback top-of-liner anchor when no matching drlgCasing liner row exists: the shoe of
 * the last (deepest) non-liner casing. Per casing, prefer its actual (as-drilled)
 * DRLG_CSG shoe over its IR (planned) depth; only use the IR depth when no actual record
 * exists for that casing.
 *
 * When the IR is liner-only (no non-liner casing row at all — e.g. a standalone-liner
 * well), there's no IR casing to read a shoe from, so fall back to the deepest actual
 * non-liner casing record instead of defaulting straight to 0.
 */
export function resolveLastCasingShoe(casings: ICasingIR[], drlgCasings: IDrlgCsg[]): number {
    const nonLinerCasings = casings.filter(c => c.csgType !== 'Liner');

    if (nonLinerCasings.length > 0) {
        return nonLinerCasings.reduce((max, c) => {
            const actualShoe = drlgCasings.find(d => d.wCsgOdSz === c.csgSize)?.wCsgBotDpth ?? 0;
            return Math.max(max, actualShoe > 0 ? actualShoe : c.csgDepth);
        }, 0);
    }

    // Note: a row's wLnrOdSz (a liner hung off THIS casing) doesn't disqualify its own
    // wCsgBotDpth as a real casing shoe — IDrlgCsg legitimately bundles both on one row.
    const actualCasingOnly = drlgCasings.filter(d => d.wCsgBotDpth > 0);
    if (actualCasingOnly.length > 0) {
        return Math.max(...actualCasingOnly.map(d => d.wCsgBotDpth));
    }
    return Math.max(0, ...drlgCasings.map(d => d.wCsgBotDpth).filter(v => v > 0));
}

/**
 * Extracts liner entries from ICasingIR + drlgCasings for actual depth override.
 *
 * Source priority per liner:
 *   - DRLG_CSG (actual/as-drilled) overrides depth when a matching wLnrOdSz exists.
 *   - EXAD_GWD_IR_CASING (planned) is the fallback for depth and the primary OD source.
 *
 * Extra DRLG_CSG liners with no corresponding IR entry are appended at the end.
 */
function rawLinersFromIR(casings: ICasingIR[], drlgCasings: IDrlgCsg[]): RawLinerEntry[] {
    // Sorted shallow-to-deep so the chaining below (each un-anchored liner hangs from the
    // one immediately above it) walks the string in physical order, matching
    // buildLinerTopAnchors' identical convention on the IR wizard side.
    const irLiners = [...casings.filter(c => c.csgType === 'Liner')].sort((a, b) => a.csgDepth - b.csgDepth);

    const lastCasingShoe = resolveLastCasingShoe(casings, drlgCasings);

    const matchedOdSizes = new Set<string>();

    // Only the shallowest liner without its own actual top falls back to lastCasingShoe.
    // Every liner after it hangs from the liner immediately above — carried via chainTop,
    // which always holds that previous liner's own resolved (actual-preferred, IR-fallback)
    // botDepth — otherwise two+ stacked liners with no actual data would all render starting
    // at the same depth (the last casing's shoe) instead of hanging from one another.
    let chainTop = lastCasingShoe;
    const irEntries: RawLinerEntry[] = irLiners.map(csg => {
        // Source priority: use DRLG_CSG depth when available; fall back to IR depth.
        const actual = drlgCasings.find(d => d.wLnrOdSz === csg.csgSize) ?? null;
        const hasActualBot = (actual?.wLnrBotDepth ?? 0) > 0;
        const hasActualTop = (actual?.wLnrTopDepth ?? 0) > 0;
        if (actual?.wLnrOdSz) matchedOdSizes.add(actual.wLnrOdSz);
        const botDepth = hasActualBot ? actual!.wLnrBotDepth! : csg.csgDepth;
        // `actual.wCsgBotDpth` is only a meaningful fallback (a real parent-casing shoe) when
        // it's genuinely positive — the wwell-ir wizard's synthetic anchors always hardcode it
        // to 0 (see buildLinerTopAnchors), so this correctly falls through to chainTop for them
        // instead of collapsing every un-anchored liner to depth 0.
        const fallbackTopDepth = (actual?.wCsgBotDpth ?? 0) > 0 ? actual!.wCsgBotDpth : chainTop;
        const topDepth = hasActualTop ? actual!.wLnrTopDepth! : fallbackTopDepth;
        chainTop = botDepth;
        return {
            odSz: actual?.wLnrOdSz ?? csg.csgSize,
            topDepth,
            botDepth,
            hasActual: hasActualBot,
            hasActualTop,
        };
    });

    const extraEntries: RawLinerEntry[] = drlgCasings
        .filter(d => d.wLnrOdSz != null && (d.wLnrBotDepth ?? 0) > 0 && !matchedOdSizes.has(d.wLnrOdSz!))
        .map(d => ({
            odSz: d.wLnrOdSz!,
            topDepth: d.wLnrTopDepth ?? d.wCsgBotDpth,
            botDepth: d.wLnrBotDepth!,
            hasActual: true,
            hasActualTop: (d.wLnrTopDepth ?? 0) > 0,
        }));

    return [...irEntries, ...extraEntries].sort((a, b) => a.botDepth - b.botDepth);
}

function wrapLabel(value: string, maxLineLength: number, maxLines: number): string[] {
  const words = value.trim().split(/\s+/);
  const lines: string[] = [];
  let current = '';

  for (const word of words) {
    const next = current ? `${current} ${word}` : word;

    if (next.length <= maxLineLength) {
      current = next;
      continue;
    }

    if (current) lines.push(current);
    current = word;

    if (lines.length === maxLines) break;
  }

  if (current && lines.length < maxLines) lines.push(current);

  const rendered = lines.join(' ');
  if (value.length > rendered.length && lines.length > 0) {
    lines[lines.length - 1] = `${lines[lines.length - 1].replace(/\.+$/, '')}...`;
  }

  return lines;
}

/** Raw base64 (no `data:` prefix) of a blob — the shape SAP's `attachment_files[].b64` expects. */
function blobToBase64(blob: Blob): Promise<string> {
  return new Promise((resolve, reject) => {
    const reader = new FileReader();
    reader.onload = () => resolve(String(reader.result).split(',')[1] ?? '');
    reader.onerror = () => reject(reader.error);
    reader.readAsDataURL(blob);
  });
}

/** Prefers the backend's own `{ message }` (see ApiResponse<T> — every exad_agwa endpoint here
 *  returns that envelope on both success and error) over Angular's generic "Http failure
 *  response for ..." text, so a specific backend validation error (e.g. an Oracle constraint
 *  message) reaches the user instead of a vague fallback. */
function errorMessage(err: unknown, fallback: string): string {
  if (err instanceof HttpErrorResponse) {
    // status 0 means the request never reached a server at all (refused connection, DNS failure,
    // CORS rejection, offline). There is no ApiResponse envelope to prefer, and `err.error` holds
    // the transport's own TypeError — surfacing its "Failed to fetch" as though it were a backend
    // message tells the user nothing, so the caller's own copy wins here.
    if (err.status === 0) return extractHttpErrorMessage(err, fallback);
    const body: unknown = err.error;
    const backendMessage = body && typeof body === 'object' ? (body as { message?: unknown }).message : undefined;
    if (typeof backendMessage === 'string' && backendMessage) return backendMessage;
    return err.message || fallback;
  }
  return err instanceof Error ? err.message : fallback;
}

/** Like errorMessage above, but prefers the SAP trigger endpoint's own `{ detail: [{ msg }] }`
 *  422 validation shape (SapValidationError) over the generic `{ message }` envelope every other
 *  exad_agwa endpoint here uses — so a real "field X is required" from SAP reaches the user
 *  instead of a vague fallback. */
function sapErrorMessage(err: unknown, fallback: string): string {
  if (err instanceof HttpErrorResponse) {
    const body: unknown = err.error;
    const detail = body && typeof body === 'object' ? (body as { detail?: unknown }).detail : undefined;
    if (Array.isArray(detail)) {
      const messages = detail
        .map(item => (item && typeof item === 'object' ? (item as { msg?: unknown }).msg : undefined))
        .filter((msg): msg is string => typeof msg === 'string' && msg.length > 0);
      if (messages.length > 0) return messages.join('; ');
    }
  }
  return errorMessage(err, fallback);
}

/** Stamps the just-persisted SAP workflow onto the record: `sapWfId`/`sapWfStatusText` mirror
 *  exactly what `update_ir_sap_wf_prc` just wrote to sap_wf_id/sap_wf_status
 *  (WWELL_IR_SAP_WF_STATUS_IN_PROGRESS — submitting never asks SAP's own GET /display, so this is
 *  a local placeholder, not SAP's real text). `chain` is deliberately left untouched: nothing was
 *  fetched to refresh it with here — the next time this record is *viewed*
 *  (WwellIrStore.loadIrApprovalTimeline, via loadIrDetail or refreshIrApprovalStatus) is what
 *  reads SAP's real current status and chain back over this placeholder, everywhere it's shown
 *  (list badge, detail header badge, Approval Workflow sidebar) at once. */
function markSubmittedForReview(list: readonly IntegratedRequirement[], id: string, sapWfId: number, sapWfStatusText: string): IntegratedRequirement[] {
  // `chain: []` on purpose: a resubmission (after a Rejected/Cancelled/Recalled workflow) mints a brand
  // new SAP workflow, so the previous workflow's steps must not linger on the record — the letter and
  // the sidebar would otherwise keep showing the old table until SAP's own chain is read back.
  return list.map(item => (item.id === id ? { ...item, status: 'submit-for-review' as const, sapWfId, sapWfStatusText, chain: [] } : item));
}

function upsertByIrSeq(list: readonly IntegratedRequirement[], created: IntegratedRequirement): IntegratedRequirement[] {
  const existingIndex = created.irSeq != null ? list.findIndex(item => item.irSeq === created.irSeq) : -1;
  if (existingIndex === -1) return [created, ...list];
  // A brand-new record's `created` (see createIr's own buildIntegratedRequirementFromPayload
  // fallback) has no way to know the record's real approval chain (that's a separate fetch; see
  // loadIrApprovalTimeline), so it always starts a fresh "just initiated" chain. Harmless for a
  // first-time INSERT (nothing to preserve yet), but would otherwise wipe out an already-in-
  // progress/approved record's real chain on every subsequent save. Keep whichever chain this
  // record already had.
  return list.map((item, index) => (index === existingIndex ? { ...created, chain: item.chain } : item));
}

              error: (err: unknown) => notify.error(errorMessage(err, 'Failed to load aquifer list')),
            }),
          ),
        ),
      ),
    );

    const loadDropdownMenus = rxMethod<void>(
      pipe(
        switchMap(() =>
          svc.getDropdownMenus().pipe(
            tapResponse({
              next: dropdownMenus => patchState(store, { dropdownMenus }),
              // Non-fatal: the dropdowns just stay empty (no arrow) until a reload succeeds.
              error: () => undefined,
            }),
          ),
        ),
      ),
    );

    // Keyed by ir_seq (the real primary key once an IR is saved) — switchMap cancels a still-in-
    // flight fetch for a previously-selected row if the user picks another one before it resolves,
    // so a slow first response can never land after (and clobber) a faster later selection.
    const loadIrDetail = rxMethod<number>(
      pipe(
        tap(() => patchState(store, { irDetailLoading: true, irDetailError: null })),
        switchMap(irSeq =>
          svc.getIrDetailBlocks(irSeq).pipe(
            tapResponse({
              next: blocks => {
                const existing = store.irs().find(item => item.irSeq === irSeq);
                // Nothing to merge onto if the summary row isn't loaded (shouldn't happen from the
                // normal select-a-row flow) — just clear the loading flag and leave state as-is.
                if (!existing) {
                  patchState(store, { irDetailLoading: false });
                  return;
                }
                const merged = mergeIrDetailBlocksIntoIntegratedRequirement(existing, blocks, store.geologistNameByEmpNum());
                patchState(store, { irs: upsertByIrSeq(store.irs(), merged), irDetailLoading: false });
                // Only the detail fetch's exad_gwd_ir_header block carries sap_wf_id (the list
                // endpoint doesn't — see wwell-ir-summary.mapper.ts), so this is the first point a
                // real SAP lookup becomes possible for a freshly-selected record.
                loadIrApprovalTimeline({ irSeq, sapWfId: merged.sapWfId });
              },
              error: (err: unknown) =>
                patchState(store, { irDetailLoading: false, irDetailError: errorMessage(err, 'Failed to load integrated requirement detail') }),
            }),
          ),
        ),
      ),
    );

    // Feeds the Approval Workflow sidebar straight from SAP's own GET /display, looked up by the
    // IR's sapWfId (== wf_obj_id, the raw id minted by Submit for Review — see
    // wwell-ir-sap.config.ts). An IR never submitted has no sapWfId and therefore no SAP workflow
    // to look up — that case clears the chain locally instead of calling SAP with nothing to ask
    // about. Same switchMap-cancels-a-stale-fetch reasoning as loadIrDetail applies when the
    // selection changes quickly.
    const loadIrApprovalTimeline = rxMethod<{ irSeq: number; sapWfId: number | null }>(
      pipe(
        tap(() => patchState(store, { irApprovalTimelineLoading: true, irApprovalTimelineError: null })),
        switchMap(({ irSeq, sapWfId }) => {
          // Nothing to look up yet — leave `chain` exactly as it was (the initial local preview;
          // see buildIntegratedRequirementFromPayload) rather than blank it, since chainBorderColor
          // reads it for the detail panel's accent strip in both view and edit mode, not only the
          // view-mode sidebar this gates (wwell-ir.component.html).
          if (sapWfId == null) {
            patchState(store, { irApprovalTimelineLoading: false });
            return EMPTY;
          }
          return sap.getWorkflowStatus(WWELL_IR_SAP_PROCESS_ID, sapWfId).pipe(
            tapResponse({
              next: status => {
                patchState(store, {
                  // header.wf_status_text is the real workflow status SAP is showing right now —
                  // surfaced on the list badge (wwell-ir.component.html) in place of the local
                  // "Submit for Review" stand-in markSubmittedForReview sets optimistically, so a
                  // freshly-submitted record doesn't keep showing that stale label once SAP's own
                  // status is known.
                  irs: store.irs().map(item =>
                    item.irSeq === irSeq
                      ? { ...item, chain: mapSapWorkflowStatusToApprovalChain(status), sapWfStatusText: status.header.wf_status_text }
                      : item,
                  ),
                  irApprovalTimelineLoading: false,
                });
              },
              error: (err: unknown) =>
                patchState(store, {
                  irApprovalTimelineLoading: false,
                  irApprovalTimelineError: sapErrorMessage(err, 'Failed to load the SAP approval workflow status'),
                }),
            }),
          );
        }),
      ),
    );

    /** Every object stored under this IR's prefix (trailing slash so IR 900's query can never match
     *  IR 9006's objects — see S3Service.search), newest-first from the proxy and reversed here to
     *  oldest-uploaded-first (matches `irAttachmentsFor`'s doc comment). A key that doesn't parse as
     *  one of this app's own slots (see `slotFromObjectKey`) is silently skipped rather than shown
     *  with a broken slot. */
    const loadIrAttachments = rxMethod<number>(
      pipe(
        tap(() => patchState(store, { attachmentsLoading: true, attachmentError: null })),
        switchMap(irSeq =>
          s3.search(`ir-attachments/ir-${irSeq}/`).pipe(
            tapResponse({
              next: ({ objects }) => {
                const attachments: IrAttachment[] = [...objects]
                  .reverse()
                  .flatMap(object => {
                    const slot = slotFromObjectKey(object.OBJ_KEY);
                    if (!slot) return [];
                    return [
                      {
                        slot,
                        objectKey: object.OBJ_KEY,
                        label: labelForSlot(slot),
                        fileName: object.ORIGINAL_FILENAME,
                        sizeBytes: object.OBJ_SIZE,
                        contentType: object.CONTENT_TYPE_CD,
                      },
                    ];
                  });
                patchState(store, {
                  irs: store.irs().map(item => (item.irSeq === irSeq ? { ...item, attachments } : item)),
                  attachmentsLoading: false,
                });
              },
              error: (err: unknown) =>
                patchState(store, { attachmentsLoading: false, attachmentError: errorMessage(err, 'Failed to load attachments') }),
            }),
          ),
        ),
      ),
    );

    /** Appends a newly-uploaded attachment onto the record, leaving every existing one (in this
     *  slot or any other) untouched — a slot accumulates attachments rather than holding just one. */
    const appendAttachment = (irSeq: number, attachment: IrAttachment) =>
      store.irs().map(item => (item.irSeq === irSeq ? { ...item, attachments: [...item.attachments, attachment] } : item));

    /** Drops one specific stored attachment (matched by its object key, its only identity) off the
     *  record, leaving every other attachment in place. */
    const withoutAttachment = (irSeq: number, objectKey: string) =>
      store.irs().map(item =>
        item.irSeq === irSeq ? { ...item, attachments: item.attachments.filter(existing => existing.objectKey !== objectKey) } : item,
      );

    /** Marks one pending file as mid-upload (or clears that marking) — shared by the single-item
     *  and sweep upload paths below so both track busy/progress identically per file id. */
    function setPendingUploadBusy(id: string, percent: number | null): void {
      patchState(store, {
        uploadingPendingIds: new Set(store.uploadingPendingIds()).add(id),
        uploadProgressByPendingId: { ...store.uploadProgressByPendingId(), [id]: percent },
      });
    }

    function clearPendingUploadBusy(id: string): void {
      const uploadingPendingIds = new Set(store.uploadingPendingIds());
      uploadingPendingIds.delete(id);
      const uploadProgressByPendingId = { ...store.uploadProgressByPendingId() };
      delete uploadProgressByPendingId[id];
      patchState(store, { uploadingPendingIds, uploadProgressByPendingId });
    }

    /** Drops one staged file (by id) from its slot's pending list, leaving any others staged there untouched. */
    function withoutPendingAttachment(slot: IrAttachmentSlot, id: string): Partial<Record<IrAttachmentSlot, readonly IrPendingAttachment[]>> {
      const remaining = (store.pendingAttachments()[slot] ?? []).filter(item => item.id !== id);
      return { ...store.pendingAttachments(), [slot]: remaining };
    }

    /**
     * Uploads one staged file straight to the S3 object-store proxy — the sole upload call now,
     * with no agwa-planning involvement at all (see ir-attachment.model.ts's class doc comment).
     * Independent of every other file in flight — each call tracks its own busy/progress by the
     * pending item's id (see `uploadingPendingIds`/`uploadProgressByPendingId`), so clicking several
     * rows' Upload buttons in a row fires that many concurrent requests rather than queuing behind
     * one another. Clears the staging only after the proxy confirms, so a failed upload leaves the
     * file in its slot to retry rather than silently dropping the user's pick.
     */
    function uploadOne(irSeq: number, slot: IrAttachmentSlot, item: IrPendingAttachment): void {
      setPendingUploadBusy(item.id, null);
      patchState(store, { attachmentError: null });
      const networkId = authStore.networkId() ?? '';
      const prefix = `ir-attachments/ir-${irSeq}/${slot}`;
      s3.store(item.file, prefix, networkId, item.file.name).subscribe({
        next: ({ percent, response }) => {
          if (!response) {
            setPendingUploadBusy(item.id, percent);
            return;
          }
          const attachment: IrAttachment = {
            slot,
            objectKey: response.key,
            label: labelForSlot(slot),
            fileName: item.file.name,
            sizeBytes: item.file.size,
            contentType: response.content_type,
          };
          clearPendingUploadBusy(item.id);
          patchState(store, {
            irs: appendAttachment(irSeq, attachment),
            pendingAttachments: withoutPendingAttachment(slot, item.id),
          });
          notify.info(`${attachment.label} uploaded.`);
        },
        error: (err: unknown) => {
          clearPendingUploadBusy(item.id);
          patchState(store, { attachmentError: errorMessage(err, `Failed to upload the ${slot} attachment`) });
        },
      });
    }

    return {
      loadIrs(): void {
        loadIrs();
      },

      loadIrDetail(irSeq: number): void {
        loadIrDetail(irSeq);
      },

      loadIrAttachments(irSeq: number): void {
        loadIrAttachments(irSeq);
      },

      /**
       * Stages a picked file alongside whatever else is already staged for that slot — attaching
       * never replaces, only adds. Attaching never uploads on its own either: the user sends it with
       * the file's own upload action (`uploadPendingAttachment`), or, for a New IR with no ir_seq to
       * upload against yet, the first save flushes everything staged via `uploadPendingAttachments`.
       */
      addPendingAttachment(slot: IrAttachmentSlot, file: File): string {
        const item: IrPendingAttachment = { id: crypto.randomUUID(), file };
        const existing = store.pendingAttachments()[slot] ?? [];
        patchState(store, {
          pendingAttachments: { ...store.pendingAttachments(), [slot]: [...existing, item] },
          attachmentError: null,
        });
        return item.id;
      },

      /** Sends one staged file. See `uploadOne` above for why this is safe to call for several
       *  files at once without them fighting over shared progress/busy state. */
      uploadPendingAttachment(irSeq: number, slot: IrAttachmentSlot, id: string): void {
        const item = (store.pendingAttachments()[slot] ?? []).find(candidate => candidate.id === id);
        if (!item) return;
        uploadOne(irSeq, slot, item);
      },

      /** Un-stages one file without sending it — the user changed their mind before uploading. */
      removePendingAttachment(slot: IrAttachmentSlot, id: string): void {
        patchState(store, { pendingAttachments: withoutPendingAttachment(slot, id) });
      },

      /** Drops every held file — the draft was abandoned, or a different record was selected. */
      clearPendingAttachments(): void {
        patchState(store, { pendingAttachments: {} });
      },

      /**
       * Uploads everything held for a draft that has just been assigned `irSeq` — every staged file
       * in every slot, each independently (see `uploadOne`). State is cleared up front so a second
       * save can't re-upload the same files, and a failure on one file doesn't abort the others:
       * each reports its own error and the rest continue.
       */
      uploadPendingAttachments(irSeq: number): void {
        const bySlot = store.pendingAttachments();
        patchState(store, { pendingAttachments: {} });
        (Object.entries(bySlot) as [IrAttachmentSlot, readonly IrPendingAttachment[]][]).forEach(([slot, items]) =>
          items.forEach(item => uploadOne(irSeq, slot, item)),
        );
      },

      /** Deletes one specific stored attachment by its object key — its only identity now. Tracked
       *  by that same key in `deletingAttachmentKeys` so only this one row shows busy, not the rest
       *  of the slot's list. Calls the S3 object-store proxy directly — there is no backend
       *  round-trip for attachments at all anymore (see ir-attachment.model.ts's class doc comment). */
      deleteIrAttachment(irSeq: number, objectKey: string): void {
        const networkId = authStore.networkId() ?? '';
        patchState(store, { deletingAttachmentKeys: new Set(store.deletingAttachmentKeys()).add(objectKey), attachmentError: null });
        s3.delete(objectKey, networkId).subscribe({
          next: () => {
            const deletingAttachmentKeys = new Set(store.deletingAttachmentKeys());
            deletingAttachmentKeys.delete(objectKey);
            patchState(store, { irs: withoutAttachment(irSeq, objectKey), deletingAttachmentKeys });
          },
          error: (err: unknown) => {
            const deletingAttachmentKeys = new Set(store.deletingAttachmentKeys());
            deletingAttachmentKeys.delete(objectKey);
            patchState(store, { deletingAttachmentKeys, attachmentError: errorMessage(err, 'Failed to remove the attachment') });
          },
        });
      },

      loadLinkedRequestWells(): void {
        loadLinkedRequestWells();
      },

      loadGwdOpsMembers(): void {
        loadGwdOpsMembers();
      },

      loadAquifers(): void {
        loadAquifers();
      },

      loadStratInfos(): void {
        loadStratInfos();
      },

      loadOffsetWellSources(): void {
        loadOffsetWellSources();
      },

      loadDropdownMenus(): void {
        loadDropdownMenus();
      },

      /**
       * No optimistic echo: the backend only ever confirms an `ir_seq` (see WwellIrSaveResult),
       * so this waits out a real follow-up GET (getIrDetailBlocks) and merges *that* onto the
       * record's last known-real state before `irs` ever changes — never onto whatever was just
       * typed. A field with no backing column at all (client, targetAquifer, targetDepthFt... see
       * mergeIrDetailBlocksIntoIntegratedRequirement's own doc comment) simply won't reflect the
       * edit once this settles, since there is nowhere real to read it back from — better an
       * honest revert than a value that only looks saved until the next reload.
       *
       * `creatingIr` stays true for both round-trips (POST, then GET), so the spinner is what
       * covers the extra latency — nothing renders in between.
       */
      createIr(payload: CreateIntegratedRequirementPayload, irBulkPayload: WwellIrBulkSavePayload): void {
        patchState(store, { creatingIr: true, createIrError: null });
        svc
          .createIr(irBulkPayload)
          .pipe(
            delay(WWELL_SAVE_DELAY_MS),
            switchMap(irSeq => {
              if (irSeq === null) return of({ irSeq, blocks: null });
              return svc.getIrDetailBlocks(irSeq).pipe(
                map(blocks => ({ irSeq, blocks })),
                // The save itself already succeeded here — a hiccup on this second, read-only
                // call must not read as "the save failed" (see the `blocks === null` branch below).
                catchError(() => of({ irSeq, blocks: null })),
              );
            }),
          )
          .subscribe({
            next: ({ irSeq, blocks }) => {
              if (irSeq === null) {
                // No identity to file this under at all — shouldn't happen for a genuine backend
                // success, so there's nothing to merge or select; just stop spinning.
                patchState(store, { creatingIr: false });
                return;
              }
              if (blocks === null) {
                // Confirmed saved (a real ir_seq came back) but the follow-up detail fetch failed
                // — record the real identity so a resave correctly UPDATEs instead of re-INSERTing,
                // but leave `irs` untouched rather than show anything unconfirmed.
                patchState(store, {
                  draftIrSeq: irSeq,
                  creatingIr: false,
                  createIrError: 'Saved, but the latest data failed to reload — refresh to see it.',
                });
                return;
              }
              const existing = store.irs().find(item => item.irSeq === irSeq);
              const base =
                existing ?? buildIntegratedRequirementFromPayload(payload, { id: `IR-${irSeq}`, code: `IR-${irSeq}`, irSeq });
              const merged = mergeIrDetailBlocksIntoIntegratedRequirement(base, blocks, store.geologistNameByEmpNum());
              patchState(store, {
                irs: upsertByIrSeq(store.irs(), merged),
                draftIrSeq: irSeq,
                creatingIr: false,
              });
              // A brand-new record's local draft never had a real sapWfId — this GET's own
              // exad_gwd_ir_header block is the first point one could exist (a resave of an
              // already-submitted IR).
              loadIrApprovalTimeline({ irSeq, sapWfId: merged.sapWfId });
            },
            error: (err: unknown) => patchState(store, { creatingIr: false, createIrError: errorMessage(err, 'Failed to create integrated requirement') }),
          });
      },

      /**
       * "Submit for Review" is the SAP workflow submission, in three steps against two backends:
       * mint a workflow id (agwa-planning's get_wf_next_id_fnc) → push the workflow to SAP AAE
       * under that id (SapService.trigger) → record the id and a local "in progress" placeholder
       * on the IR header (update_ir_sap_wf_prc). Deliberately does not also call SAP's own GET
       * /display to ask for a real status — see WWELL_IR_SAP_WF_STATUS_IN_PROGRESS's own doc
       * comment — so submitting stays fast and only ever costs one round-trip to SAP itself.
       *
       * There is no separate IR status-transition call: exad_gwd_ir_header has no status column of
       * its own, so the SAP workflow columns this writes ARE the record of having submitted.
       * `updatingIrStatus` stays true across all three so the editor's Submit button spins for the
       * whole action rather than flickering between steps.
       */
      submitIrForReview(ir: IntegratedRequirement): void {
        // A draft that was never saved has no ir_seq to file the workflow under, so there would be
        // nothing to record the outcome against — refuse before calling SAP rather than after.
        if (ir.irSeq == null) {
          patchState(store, { sapTriggerError: 'Save the integrated requirement before submitting it for review.' });
          return;
        }
        if (!ir.attachments.some(attachment => attachment.slot === 'location-map')) {
          patchState(store, { sapTriggerError: 'Attach a Location Map before submitting for review.' });
          return;
        }
        patchState(store, { updatingIrStatus: true, sapTriggerError: null });
        triggerSapWorkflow(ir, ir.irSeq);
      },

      /** "Recall" pulls back an already-submitted workflow — the editor only shows the button once
       *  `sapWfId` is set (see WwellIrEditorComponent.canRecallIr), but this guards it again since
       *  the store method is the one place that actually matters: there is no workflow to recall
       *  without a real `irSeq`/`sapWfId` pair to act on. */
      recallIr(ir: IntegratedRequirement): void {
        if (ir.irSeq == null || ir.sapWfId == null) {
          patchState(store, { sapRecallError: 'This integrated requirement has not been submitted for review.' });
          return;
        }
        patchState(store, { recallingIr: true, sapRecallError: null });
        recallSapWorkflow(ir.irSeq, ir.sapWfId);
      },

      /** On-demand counterpart to the automatic loadIrDetail → loadIrApprovalTimeline refresh: lets
       *  the UI (the Approval Workflow sidebar's refresh button, the detail header's status badge)
       *  pull SAP's real current status/chain for an already-selected record without needing a
       *  reselect. A no-op for a record with no `sapWfId` — there is no SAP workflow to ask about
       *  yet, same guard `loadIrApprovalTimeline` itself already applies. */
      refreshIrApprovalStatus(irSeq: number, sapWfId: number | null): void {
        if (sapWfId == null) return;
        loadIrApprovalTimeline({ irSeq, sapWfId });
      },

      /** Reads SAP's live status/chain for the record's CURRENT workflow, writes it into the store
       *  (so the screen matches) and returns the record with it applied — what the letter/PDF must
       *  print, rather than whatever was cached when the record was last opened. Resolves with the
       *  record unchanged when it has no `sapWfId` or SAP can't be reached: a SAP outage must not
       *  block a download, it just falls back to the last known workflow. */
      async syncApprovalStatus(ir: IntegratedRequirement): Promise<IntegratedRequirement> {
        if (ir.irSeq == null || ir.sapWfId == null) return ir;
        try {
          const status = await firstValueFrom(sap.getWorkflowStatus(WWELL_IR_SAP_PROCESS_ID, ir.sapWfId));
          const fresh = { chain: mapSapWorkflowStatusToApprovalChain(status), sapWfStatusText: status.header.wf_status_text };
          patchState(store, { irs: store.irs().map(item => (item.irSeq === ir.irSeq ? { ...item, ...fresh } : item)) });
          return { ...ir, ...fresh };
        } catch {
          return ir;
        }
      },

      /** Deletes the whole IR (header row, cascading to casing/tops/hydrogeology/water — see
       *  WwellIrService.deleteIr) and drops it from `irs` on success. `irSeq` must already be a
       *  persisted record's — there's nothing to delete server-side for a local-only draft. */
      deleteIr(irSeq: number): void {
        patchState(store, { deletingIr: true, deleteIrError: null });
        svc
          .deleteIr(irSeq)
          .pipe(delay(WWELL_DELETE_DELAY_MS))
          .subscribe({
            next: () => patchState(store, { irs: store.irs().filter(item => item.irSeq !== irSeq), deletingIr: false }),
            error: (err: unknown) => patchState(store, { deletingIr: false, deleteIrError: errorMessage(err, 'Failed to delete integrated requirement') }),
          });
      },
    };
  }),
);

import { MAX_UPLOAD_SIZE_BYTES, hasAllowedExtension, toAcceptAttribute } from '@core/config/file-upload.config';
import { IrAttachment, IrAttachmentSlot } from '@models/wwell-planning/ir-attachment.model';

export interface WwellIrAttachmentSlotConfig {
  readonly slot: IrAttachmentSlot;
  /** Matches the IR SECTIONS rail numbering (see wwell-ir-section-rail.config.ts). */
  readonly number: string;
  readonly label: string;
  /** Material icon ligature — topic-specific, the same way the rail's group icons are chosen. */
  readonly icon: string;
  /** Mirrors the schema's own validator on each slot's file field (wwell-ir-form.schema.ts). */
  readonly required: boolean;
  readonly hint: string;
}

/**
 * The IR's attachment slots, in the order they're appended to the letter PDF as trailing pages.
 * Single source of truth for both the editor cards and the letter view, so the two can't drift.
 */
export const WWELL_IR_ATTACHMENT_SLOTS: readonly WwellIrAttachmentSlotConfig[] = [
  {
    slot: 'location-map',
    number: '7.1',
    label: 'Location Map',
    icon: 'map',
    required: true,
    hint: 'Site location map — appended to the letter as its own page.',
  },
  {
    slot: 'initiation-request',
    number: '7.2',
    label: 'Initiation Request',
    icon: 'assignment',
    required: true,
    hint: 'Signed initiation request — appended to the letter after the location map.',
  },
  {
    slot: 'wlda',
    number: '7.3',
    label: 'Drilling Approval Letter',
    icon: 'verified',
    required: true,
    hint: 'WLDA document — appended to the letter after the initiation request and sent to SAP on submit.',
  },
];

/**
 * Restricted to what the backend can actually embed as PDF pages (see agwa-planning's
 * IrAttachmentPdfAppender). Office formats would need a headless converter, so they're refused at
 * upload rather than stored as something that silently never appears in the letter.
 */
/** True once every slot has at least one stored file — the gate for Submit for Review. */
export function hasAllRequiredAttachments(attachments: readonly IrAttachment[]): boolean {
  return WWELL_IR_ATTACHMENT_SLOTS.filter(config => config.required).every(config => irAttachmentsFor(attachments, config.slot).length > 0);
}

export const WWELL_IR_ATTACHMENT_ACCEPT = toAcceptAttribute();

/** The `accept` attribute is only a picker hint, so handlers re-check against the same list. */
export function hasAcceptedIrAttachmentExtension(fileName: string): boolean {
  return hasAllowedExtension(fileName);
}

/** Purely a client-side guard now — the S3 object-store proxy enforces no size cap of its own, so
 *  this exists only to fail fast before spending time on an upload that's almost certainly wrong. */
export const WWELL_IR_ATTACHMENT_MAX_BYTES = MAX_UPLOAD_SIZE_BYTES;

/** Every file occupying a slot, oldest-uploaded first — shared by the editor, letter and details
 *  views. A slot can hold any number of attachments, not just one, so callers render this as a list
 *  rather than picking a single result. */
export function irAttachmentsFor(attachments: readonly IrAttachment[], slot: IrAttachmentSlot): readonly IrAttachment[] {
  return attachments.filter(attachment => attachment.slot === slot);
}

/**
 * Recovers the slot from one of this app's own object keys
 * (`ir-attachments/ir-<irSeq>/<slot>/<uuid>`). Anchored on the literal `ir-attachments` segment
 * rather than the key's start, because the real S3 proxy prefixes every key with its own
 * environment segment (e.g. `dev/`) that this app never generates and must simply tolerate.
 * Returns null for a key that doesn't look like one of this app's own.
 */
export function slotFromObjectKey(objectKey: string): IrAttachmentSlot | null {
  const segments = objectKey.split('/');
  const anchor = segments.indexOf('ir-attachments');
  const candidate = anchor >= 0 ? segments[anchor + 2] : undefined;
  const config = WWELL_IR_ATTACHMENT_SLOTS.find(entry => entry.slot === candidate);
  return config ? config.slot : null;
}

/** Display label for a slot ("Location Map") — used to build an `IrAttachment` straight from a
 *  search result, since the proxy itself has no notion of this app's slot labels. */
export function labelForSlot(slot: IrAttachmentSlot): string {
  return WWELL_IR_ATTACHMENT_SLOTS.find(entry => entry.slot === slot)?.label ?? slot;
}

/** Flags a value that's actually meaningful, not just informational — e.g. a nonzero H2S
 *  reading should read as amber/danger regardless of which section it sits in. Omitted (or
 *  `'default'`) renders as a plain field. */
export type RowTone = 'default' | 'warning' | 'danger' | 'success';

export interface KvRow {
  readonly label: string;
  readonly value: string;
  readonly tone?: RowTone;
  /** Marks a value picked from a fixed set of options (a completion type, a test type, ...).
   *  A multi-select's picks arrive newline-separated in `value` and are shown one per line. */
  readonly options?: boolean;
  /** An identifier or a measurement (an IR number, a depth, a reading): set in the mono face, like the figures elsewhere. */
  readonly mono?: boolean;
  /** Long free text (a remark): the editor's read display gives it a full-width card instead of a short spec cell. */
  readonly wide?: boolean;
  /** Material icon name the editor's card display shows beside the label. */
  readonly icon?: string;
  /** Listed in the Program Details read display even with nothing entered ("Not specified"); every other long-form row is hidden when empty. */
  readonly alwaysShow?: boolean;
}

export const EMPTY_VALUE = '—';

export function fmtNum(value: number | null | undefined, unit?: string): string {
  if (value === null || value === undefined) return EMPTY_VALUE;
  return unit ? `${value} ${unit}` : `${value}`;
}

export function fmtStr(value: string | null | undefined): string {
  return value ? value : EMPTY_VALUE;
}

/** `linkedRequestId` is the raw prewap_seq (e.g. "5008") — resolve it to the well name. Prefers
 *  `r.wellName` (populated from exad_rcd_prewap on a saved IR, or by the picker on a fresh one)
 *  over the "eligible for a new IR" wellOptions lookup: get_ir_wells_prc.sql's eligibility filter
 *  excludes any well that already has an IR header, so the instant this IR is saved its own
 *  linked well permanently drops out of that list and the lookup can never succeed again — the
 *  wellOptions param only still helps a brand-new, not-yet-saved draft. Falls back to the raw id
 *  if neither source has a name yet, rather than showing nothing. */
function resolveLinkedRequestLabel(
  linkedRequestId: string | null,
  wellName: string,
  wellOptions: readonly { readonly value: string; readonly label: string }[],
): string {
  if (!linkedRequestId) return EMPTY_VALUE;
  return wellName || wellOptions.find(option => option.value === linkedRequestId)?.label || linkedRequestId;
}

/** Splits a "Northing X, Easting Y, Zone Z" coordinate string into its own Northing/Easting/UTM
 *  Zone rows — falls back to a single "Coordinates" row when the text doesn't read as that shape
 *  (unset, or hand-typed free text), same fallback `toUtmAxes` uses for the wellbore panel's
 *  Location card. Shown exactly as the raw survey values, no unit conversion. */
function coordinateKvRows(coordinates: string | null | undefined): readonly KvRow[] {
  if (!coordinates) return [{ label: 'Coordinates', value: EMPTY_VALUE, mono: true }];
  return toUtmAxes(coordinates).map(axis => ({
    label: axis.label,
    value: axis.value,
    mono: true,
  }));
}

/** Mirrors the IR wizard's "Request Reference" step (wwell-ir-form.schema.ts) field-for-field,
 *  so the detail page's sections line up 1:1 with the stepper the data was captured through. */
export function buildRequestReferenceRows(
  ir: IntegratedRequirement,
  linkedRequestWellOptions: readonly { readonly value: string; readonly label: string }[] = [],
  /** "Link to Water Well" is only a picker for creating/editing an IR — read-only displays leave it out (the
   *  Water Well Name row below already says which well it is). */
  includeLinkedWell = false,
): readonly KvRow[] {
  const r = ir.requestReference;
  return [
    ...(includeLinkedWell
      ? [{ label: 'Link to Water Well', value: resolveLinkedRequestLabel(r.linkedRequestId, r.wellName, linkedRequestWellOptions) }]
      : []),
    { label: 'Water Well Name', value: fmtStr(r.wellName) },
    { label: 'Letter Number', value: fmtStr(r.letterNumber), mono: true },
    { label: 'Supporting Business', value: fmtStr(r.supportingBusiness) },
    { label: 'Target Aquifer', value: fmtStr(r.targetAquifer), options: true },
    // A brand-new draft is created through the required-number fallback in
    // mapIrFormValueToSubmission (targetDepthFt ?? 0) before the author has entered a real value —
    // treat a literal 0 as "not yet entered" rather than a misleading real reading (see the
    // matching fallback in mapIntegratedRequirementToFormPatch, wwell-ir-edit.mapper.ts).
    { label: 'Target Depth', value: r.targetDepthFt ? fmtNum(r.targetDepthFt, 'ft') : EMPTY_VALUE, mono: true },
    { label: 'Surface Elevation', value: `${fmtNum(r.surfaceElevationFtAsl)} ft ASL`, mono: true },
    ...coordinateKvRows(r.coordinates),
  ];
}

/** Mirrors the schema's "Program Details" section (2.1) — its own rail leaf/edit unit, separate
 *  from Pumping Test (2.2) below. `openHoleSizeIn` only ever means something when the
 *  completion type includes 'Open Hole' (see the field's `visibleWhen` gate in the schema), so
 *  it's only included then rather than showing a dangling "—" the rest of the time.
 *
 *  `completion` is a multi-select — any number of `WWELL_COMPLETION_OPTIONS` can be checked at
 *  once, each becoming its own `\n`-joined line (see wwell-ir-wellbore-diagram.mapper.ts's own
 *  doc comment on this same shape) — so this must check "is 'Open Hole' one of the checked
 *  lines?", the same `includesLine` semantics the schema's `visibleWhen` gate uses
 *  (dynamic-section.component.ts), not a straight equality that only ever matched the
 *  single-selection case and otherwise silently hid an already-entered size. */
export function buildProgramMainRows(ir: IntegratedRequirement): readonly KvRow[] {
  return [...buildProgramCompletionRows(ir), ...buildProgramTreatmentRows(ir)];
}

/** The completion half of `buildProgramMainRows`: type, the sizes that go with it, and the remarks picks (one line). */
export function buildProgramCompletionRows(ir: IntegratedRequirement): readonly KvRow[] {
  const p = ir.programDetails;
  const rows: KvRow[] = [{ label: 'Completion', value: fmtStr(p.completionType) }];

  // Every pick is drawn inside the open hole, so its size is listed whatever the pick.
  if (p.completionType) {
    rows.push({ label: 'Open Hole Size', value: fmtStr(formatInchSize(p.openHoleSizeIn)), mono: true });
  }
  if (isLinerCompletionType(p.completionType)) {
    rows.push({ label: 'Liner Screen Size', value: fmtStr(formatInchSize(p.linerSizeIn)), mono: true });
    rows.push({ label: 'Liner Depth (ft)', value: fmtNum(p.linerDepthFt), mono: true });
  }
  rows.push({ label: 'Completion Remarks', value: fmtStr(normalizeInchMarks(p.completion)), options: true, wide: true, icon: 'construction' });
  return rows;
}

/** The acidization / transmission half of `buildProgramMainRows`. */
export function buildProgramTreatmentRows(ir: IntegratedRequirement): readonly KvRow[] {
  const p = ir.programDetails;
  return [
    { label: 'Acidization', value: fmtStr(normalizeInchMarks(p.acidization)), options: true, icon: 'science' },
    { label: 'Transmission', value: fmtStr(normalizeInchMarks(p.transmission)), options: true, icon: 'settings_input_antenna' },
  ];
}

/** Mirrors the schema's "Pumping Test" section (2.2) — see buildProgramMainRows' doc

export function buildProgramDevRows(ir: IntegratedRequirement): readonly KvRow[] {

  return [

    { label: 'Sampling', value: fmtStr(d.sampling), options: true, icon: 'colorize' },
    // Both are always listed here; the IR letter / PDF / SAP copy print only the one matching the Hydrology Test Type.
    { label: 'Flow Test', value: fmtStr(d.flowTest), options: true, icon: 'speed', alwaysShow: true },
    { label: 'Pumping Test', value: fmtStr(d.remarks), wide: true, icon: 'engineering', alwaysShow: true },

}

/** Mirrors the wizard's "Estimated Hydrology" step, including its test-type-conditional
 *  fields (SIWHP for a flow test; water level + pump depth for a pump test) — see the
 *  matching `visibleWhen` gates on `siwhpPsi`/`estimatedWaterLevelFt`/`estimatedPumpDepthFt`
 *  in wwell-ir-form.schema.ts. Any nonzero H2S gets `tone: 'warning'` — a toxic-gas reading is
 *  worth flagging on sight, not just listed at the same visual weight as TDS/rate. */
export function buildHydrogeologyRows(ir: IntegratedRequirement): readonly KvRow[] {
  const isFlowTest = ir.testProgram.testType === 'flow-test';
  const h2sPpm = ir.hydrogeology.h2sPpm;
  const rows: KvRow[] = [
    { label: 'Aquifer', value: fmtStr(ir.hydrogeology.aquifer), options: true },
    { label: 'Test Type', value: isFlowTest ? 'Flow Test' : 'Pump Test', options: true },
    { label: 'Estimated TDS', value: fmtNum(ir.hydrogeology.estimatedTdsPpm, 'ppm'), mono: true },
    { label: 'Estimated Rate', value: fmtNum(ir.testProgram.estimatedRateGpm, 'GPM'), mono: true },
    {
      label: 'H2S',
      value: fmtNum(h2sPpm, 'ppm'),
      tone: h2sPpm ? 'warning' : 'default',
      mono: true,
    },
  ];

  if (isFlowTest) {
    rows.push({ label: 'SIWHP', value: fmtNum(ir.testProgram.siwhpPsi, 'psi'), mono: true });
  } else {
    rows.push(
      { label: 'Estimated Water Level', value: fmtNum(ir.testProgram.estimatedWaterLevelFt, 'ft'), mono: true },
      { label: 'Estimated Pump Depth', value: fmtNum(ir.hydrogeology.estimatedPumpDepthFt, 'ft'), mono: true },
    );
  }
  return rows;
}

function normalizeRowKeys<TRow>(row: Record<string, unknown>): TRow {
  const normalized: Record<string, unknown> = {};
  for (const key of Object.keys(row)) {
    normalized[key.toLowerCase()] = row[key];
  }
  return normalized as TRow;

/** The row the record's own saves write to: the one it already holds (so a save UPDATEs it), else the newest
 *  — GET returns every row for the well unordered, and reading `[0]` picked an old, empty one over the
 *  completion the user had just saved, which then made each later save INSERT yet another row. */
function pickWellDesignRow(rows: readonly IrWellDesignRow[], heldWdSeq: number | null | undefined): IrWellDesignRow | undefined {
  const held = rows.find(row => row.wd_seq === heldWdSeq);
  if (held) return held;
  return rows.reduce<IrWellDesignRow | undefined>((newest, row) => (!newest || (row.wd_seq ?? 0) > (newest.wd_seq ?? 0) ? row : newest), undefined);
}

/** A remark column the backend returns as null/absent on a header row that *is* there means the text was
 *  deleted — so it reads as blank, not as "keep what the client last had" (which left a deleted
 *  Mud Logging / Sampling / Flow Test... showing until a full reload). The fallback only applies when the
 *  response carried no header row at all. */
function resolveLegacyText(raw: string | null | undefined, fallback: string, headerPresent: boolean): string {
  if (raw != null) return stripLegacyIrHtml(raw);
  return headerPresent ? '' : fallback;

/** Rows shallowest first (stable; a row with no depth goes last) — the order the casing and formation-tops grids list them. */
export function sortByDepthAscending<T>(rows: readonly T[], depthOf: (row: T) => number | null | undefined): T[] {
  const depth = (row: T): number => depthOf(row) ?? Number.POSITIVE_INFINITY;
  return [...rows].sort((a, b) => depth(a) - depth(b));
}

  const headerCompletionType = completionTypeFromCode(header?.cmpl_type);
  const completionRaw = header?.completion ?? (headerCompletionType ? null : header?.cmpl_type);
  const completionText = resolveLegacyText(completionRaw, existing.programDetails.completion, headerPresent);
  const persistedFlags = wellDesignRow ? mapWellDesignFlags(wellDesignRow) : null;
  // exad_gwd_well_design is keyed by prewap_seq alone, so a well can hold several rows (every INSERT adds
  // one). Whatever row is picked is this record's design — including one whose flags are all 'N': that is
  // a completion that was *deleted*, and it must read back as blank, not fall back to the stale value.
  const wellDesignRowBelongsToThisIr = !!wellDesignRow;
  const flagsType = wellDesignRowBelongsToThisIr ? resolveCompletionTypeFromFlags(persistedFlags) : null;
  // The header code fills in the pick when the design row's flags don't decode to one (a size is still read from the remarks if there).
  const resolvedType =
    flagsType ?? (headerCompletionType ? { completionType: headerCompletionType, linerSizeIn: null, linerDepthFt: null } : null);
  const hasLegacyMarkup = [
    completionRaw,
    header?.acidization,
    header?.dt_rmks,
    header?.logging,
    header?.mud_rmks,
    header?.sampling,
    header?.flow_rmks,
    header?.dev_pump_test,
    header?.tops_remarks,
    header?.water_remarks,
  ].some(isLegacyIrHtmlText);

  return {
    ...existing,
    hasLegacyMarkup,
    wellDesignSeq: wellDesignRowBelongsToThisIr ? wellDesignRow.wd_seq : existing.wellDesignSeq,
    sapWfId: header?.sap_wf_id ?? existing.sapWfId,

      // wwell-ir-coordinates.util.ts's doc comment.

        linkedWell?.surfacen != null && linkedWell?.surfacee != null
          ? (formatUtmCoordinates(linkedWell.surfacen, linkedWell.surfacee, linkedWell.sv_utm_zn) ??
            existing.requestReference.coordinates)
          : existing.requestReference.coordinates,
      targetDepthFt: linkedWell?.target_dpth ?? existing.requestReference.targetDepthFt,
      targetAquifer: linkedWell?.target_form ?? existing.requestReference.targetAquifer,
      supportingBusiness: linkedWell?.expl_wb_name ?? existing.requestReference.supportingBusiness,

    contacts: {

        existing.contacts.geologistPrimary,
      geologistSecondary: resolveGeologistName(header?.sub_cont_geologist, geologistNameByEmpNum, existing.contacts.geologistSecondary),
    },
    offsetWellsRemarks: header?.water_remarks != null ? stripLegacyIrHtml(header.water_remarks) : existing.offsetWellsRemarks,

      epANum: row.ep_a_num,

      location: row.direction ?? null,
      aquifer: row.aquifer ?? null,
      totalDepthFt: row.td ?? null,

      rateGpm: row.rate ?? null,

      swlFt: row.swl ?? null,

      drawdownFt: row.drawdown ?? null,
      specificCapacity: row.spec_capacity ?? null,

      linerDepthFt: wellDesignRowBelongsToThisIr ? (resolvedType?.linerDepthFt ?? null) : existing.programDetails.linerDepthFt,
      // oh_remarks is the only backend column open hole size ever reached (see

      // — without parsing it back out here, the Open Hole Size input silently reverted to blank

      openHoleSizeIn: wellDesignRowBelongsToThisIr

        logging: resolveLegacyText(header?.logging, existing.programDetails.developmentTesting.logging, headerPresent),
        mudLogging: resolveLegacyText(header?.mud_rmks, existing.programDetails.developmentTesting.mudLogging, headerPresent),

        remarks: resolveLegacyText(header?.dev_pump_test, existing.programDetails.developmentTesting.remarks, headerPresent),

      wellDesignFlags: wellDesignRowBelongsToThisIr ? persistedFlags : existing.programDetails.wellDesignFlags,

    hydrogeology: hydroRow
      ? {
          aquifer: hydroRow.st_long_cd,
          estimatedTdsPpm: hydroRow.est_water_quality,
          h2sPpm: parseH2sPpm(hydroRow.h2s),
          estimatedPumpDepthFt: wellDesignRow?.pump_lvl ?? existing.hydrogeology.estimatedPumpDepthFt,

    testProgram: {
      ...existing.testProgram,

      estimatedRateGpm: hydroRow?.est_productivity ?? existing.testProgram.estimatedRateGpm,
      estimatedWaterLevelFt: hydroRow?.est_sta_water_lvl ?? existing.testProgram.estimatedWaterLevelFt,
      siwhpPsi: hydroRow?.siwhp ?? existing.testProgram.siwhpPsi,
    },
  };
}

/** `SHYB_1031(SHAYBAH)` — the short well name plus the primary hole's full name in brackets, dropped when missing or just repeating the code. */
export function buildWellFullName(primaryHoleId: string | null, wellNumber: string | null, primaryHoleName: string | null | undefined): string {
  const base = `${primaryHoleId ?? ''}_${wellNumber ?? ''}`;
  const name = primaryHoleName?.trim();
  return name && name.toLowerCase() !== (primaryHoleId ?? '').toLowerCase() ? `${base}(${name})` : base;
}

export function buildWwellIrFormSchema(
  linkedRequestOptions: Signal<readonly WwellFieldOption[]>,
  geologistOptions: Signal<readonly string[]>,
  dropdownMenus: Signal<Readonly<Record<string, readonly string[]>>> = signal({}),
  aquiferOptions: Signal<readonly WwellFieldOption[]> = signal([]),
  formationOptions: Signal<readonly WwellFieldOption[]> = signal([]),
  offsetWellAquiferOptions: Signal<readonly WwellFieldOption[]> = signal([]),
): readonly WwellFormStepConfig[] {
  return WWELL_IR_FORM_SCHEMA.map(step => {
    if (step.key === 'subsurface') {
      return {
        ...step,
        sections: step.sections.map(section =>
          section.key === 'formationTops' && section.repeatable
            ? {
                ...section,
                repeatable: {
                  ...section.repeatable,
                  rowFields: section.repeatable.rowFields.map(field => (field.key === 'name' ? { ...field, options: formationOptions } : field)),
                },
              }
            : section.key === 'casing' && section.repeatable
              ? {
                  ...section,
                  repeatable: {
                    ...section.repeatable,
                    rowFields: section.repeatable.rowFields.map(field => {
                      if (field.key === 'type') return { ...field, options: computed(() => dropdownMenus()[WWELL_IR_CASING_TYPE_MENU] ?? []) };
                      if (field.key === 'sizeIn') return { ...field, options: computed(() => dropdownMenus()[WWELL_IR_CASING_SIZE_MENU] ?? []) };
                      return field;
                    }),
                  },
                }
              : section,
        ),
      };
    }
    if (step.key === 'hydrogeology') {
      return {
        ...step,
        sections: step.sections.map(section => ({
          ...section,
          fields: section.fields?.map(field => (field.key === 'aquifer' ? { ...field, options: aquiferOptions } : field)),
        })),
      };
    }
    if (step.key === 'programDetails') {
      return {
        ...step,
        sections: step.sections.map(section => ({
          ...section,
          fields: section.fields?.map(field => withDropdownMenuOptions(field, dropdownMenus)),
        })),
      };
    }
    if (step.key === 'reference') {
      return {
        ...step,
        sections: step.sections.map(section => {
          if (section.key !== 'main' || !section.fields) return section;
          return {
            ...section,
            fields: section.fields.map(field =>
              field.key === 'linkedRequestId' ? { ...field, options: linkedRequestOptions } : field,
            ),
          };
        }),
      };
    }
    if (step.key === 'contacts') {
      return {
        ...step,
        sections: step.sections.map(section => {
          if (section.key === 'offsetWells' && section.repeatable) {
            return {
              ...section,
              repeatable: {
                ...section.repeatable,
                rowFields: section.repeatable.rowFields.map(field => (field.key === 'aquifer' ? { ...field, options: offsetWellAquiferOptions } : field)),
              },
            };
          }
          if (section.key !== 'geologist' || !section.fields) return section;
          return {
            ...section,
            fields: section.fields.map(field =>
              field.key === 'geologistPrimary' || field.key === 'geologistSecondary' ? { ...field, options: geologistOptions } : field,
            ),
          };
        }),
      };
    }
    return step;
  });
}

/** Drives a Program Details dropdown purely from its EXAD_DHUB_LKUP_DATA menu — empty until loaded,

  const menu = WWELL_IR_DROPDOWN_MENU_BY_FIELD[field.key];
  if (!menu) return field;

function buildLinerTopAnchors(rows: readonly WwellIrRawCasingRow[]): IDrlgCsg[] {
  const linersShallowToDeep = rows
    .filter(row => row.type === LINER_TYPE)
    .slice()
    .sort((a, b) => (a.depthFtBgl ?? 0) - (b.depthFtBgl ?? 0));

  return linersShallowToDeep.map(row => ({
    wCsgOdSz: LINER_ANCHOR_SENTINEL,
    wCsgBotDpth: 0,
    wLnrOdSz: row.sizeIn,
    wLnrTopDepth: row.tolFtBgl ?? null,
    wLnrBotDepth: null,
  }));
}

/**
 * Lets Open Hole / Screened / Screened + Gravel Pack draw immediately from Program Details, even
 * before a single Casing row has been entered — otherwise `completion.renderer.ts`'s
 * `renderOpenHoleAndScreen`/`renderGravelPackDesign` both bail out on `!casings.length &&
 * !drlgCasings.length`, since with zero casing rows `casings` is `[]` and `buildLinerTopAnchors`
 * above also returns `[]` (no Liner rows either). Only fires when there are no renderable Casing
 * rows at all — once one exists, `casings.length > 0` already satisfies that guard on its own.
 *
 * Deliberately NOT added to `casings` — it must never itself look like a declared casing. Kept to
 * `drlgCasings`, with `wCsgBotDpth: 0`, so it stays invisible everywhere that array is read:
 * - `uniqueCasingsByOd`/`resolveLastCasingShoe` (casing.renderer.ts / liner.renderer.ts) both skip
 *   any entry with `wCsgBotDpth <= 0`, so `renderCasingsFromActual`'s "no IR casing plan" fallback
 *   path never draws a phantom casing pipe from it.
 * - `rawLinersFromDrlg` (liner.renderer.ts) requires a truthy `wLnrOdSz`, left unset here, so it's
 *   never mistaken for a liner either.
 * - `resolveShoeDepth`'s `drlgCasings[0]?.wCsgBotDpth ?? 0` fallback reads `0` from this row —
 *   the exact same value it already returns when `drlgCasings` is empty — so the completion
 *   interval anchors at ground level (`resolveLinerRule`'s `kind: 'none'` case) with no other
 *   change in behavior. The element then spans ground level → target depth, the same placeholder
 *   interval already used whenever no liner qualifies, until real Casing rows give it a real shoe.
 */
function buildShoeAnchorFallback(renderableCasings: readonly WwellIrRawCasingRow[]): IDrlgCsg[] {
  if (renderableCasings.length > 0) return [];
  return [{ wCsgOdSz: SHOE_ANCHOR_SENTINEL, wCsgBotDpth: 0, wLnrOdSz: null, wLnrTopDepth: null, wLnrBotDepth: null }];
}

/**
 * The Completion dropdown's Liner picks (Pre-Perforated / Perforated Liner + Liner Size) describe a
 * liner string exactly like a `Liner` row in the Casing table does — and the shared diagram draws a
 * liner purely from such a row (liner.renderer.ts's `rawLinersFromIR`: a shape hanging from the last
 * casing shoe, with its size label). Setting the completion alone only flipped the perforated flag,
 * so no liner appeared. This adds the liner the pick implies, run down to the well's target depth, as
 * a synthetic Casing row for the drawing only — it is never saved or shown in the Casing table.
 * Skipped when the Casing table already holds a Liner of that size, so a liner the user entered
 * themselves (with its own depth/TOL) is never doubled up.
 */
function withCompletionLiner(
  rows: readonly WwellIrRawCasingRow[],
  programMain: WwellIrRawProgramMain,
  targetDepthFt: number,
): WwellIrRawCasingRow[] {
  const linerSize = programMain.linerSizeIn?.trim();
  if (!isLinerCompletionType(programMain.completionType) || !linerSize) return [...rows];

  const wantedSize = parseInchSize(linerSize);
  const alreadyDeclared = rows.some(row => row.type === LINER_TYPE && parseInchSize(row.sizeIn) === wantedSize);
  if (alreadyDeclared) return [...rows];

  // No depth entered → run to the target depth; its top is left null so the renderer hangs it from the last casing shoe.
  return [...rows, { type: LINER_TYPE, sizeIn: linerSize, depthFtBgl: programMain.linerDepthFt ?? targetDepthFt, tolFtBgl: null, remarks: '' }];
}

function mapGeologicTops(
  rows: readonly WwellIrRawFormationTop[],
  surfaceElevationFtAsl: number | null,
): ITopsIR[] {
  return rows.map(row => {
    const planTvdDepth = row.depthFt ?? 0;
    return {
      stLongCd: row.name,
      planTvdDepth,
      planSsDepth: surfaceElevationFtAsl !== null ? surfaceElevationFtAsl - planTvdDepth : 0,
    };
  });
}

const FLOW_TEST_TYPE = 'flow-test';

/**
 * Derives the water-level fields `water-level.renderer.ts`'s `renderWaterLevel` reads —
 * `wellDesign.staWaterLvl` (a bare number, the actual depth it draws the line at) and
 * `wellDesign.swLvlTxt` (free text: presence of "flow" anywhere in it is that renderer's own
 * flag for "this is a flowing well, not a static level" — see its `isFlowing` check — and for a
 * flowing well it also re-parses the psi value straight back out of this same string via
 * `parseSiwhpPsi`, rather than taking a separate numeric field).
 *
 * Mirrors the Estimated Hydrology step's own test-type split (see `buildHydrogeologyRows` in
 * wwell-ir-detail-rows.util.ts): a flow test's wellhead pressure (SIWHP) draws as an above-ground
 * riser, a pump test's estimated water level draws as a below-ground line — the renderer can only
 * ever draw one of the two per well, same as the form only ever collects one.
 */
function buildWaterLevelFields(hydrology: WwellIrRawHydrology): Pick<IWellDesign, 'staWaterLvl' | 'swLvlTxt'> {
  if (hydrology.testType === FLOW_TEST_TYPE) {
    return hydrology.siwhpPsi !== null
      ? { staWaterLvl: null, swLvlTxt: `Flowing — SIWHP: ${hydrology.siwhpPsi} psi` }
      : { staWaterLvl: null, swLvlTxt: null };
  }

  return hydrology.estimatedWaterLevelFt !== null
    ? { staWaterLvl: hydrology.estimatedWaterLevelFt, swLvlTxt: `Static WL: ${hydrology.estimatedWaterLevelFt} ft` }
    : { staWaterLvl: null, swLvlTxt: null };
}

/**
 * Derives completion-element flags/remarks from the "Completion" dropdown (Program
 * Details step) plus, for Open Hole, the conditionally-shown "Open Hole Size" field.
 * `well-bore-view`'s renderers gate Open Hole strictly on a parseable inch size in
 * `ohRemarks`, and Gravel Pack / Liner Screen strictly on non-empty remarks text — the
 * constant fallback labels below exist only to satisfy those gates when the dropdown
 * implies the element but no free-text remarks field exists for it in this form.
 *
 * "Liner + Perforated" only actually renders once a Casing-table Liner row is exactly 7"
 * OD — that 7" convention is hardcoded inside liner.renderer.ts (PRE_PERF_LINER_SIZE_IN)
 * and is intentionally left untouched here.
 *
 * Water level / pump depth (`pump-and-depth.renderer.ts`'s `renderPump`, gated on `pumpLvl`)
 * come from the Estimated Hydrology step instead — see `buildWaterLevelFields` above.
 * `estimatedPumpDepthFt` is only ever populated for a pump test (see the field's own
 * `visibleWhen` in wwell-ir-form.schema.ts), so it's naturally `null` — no pump drawn — on a
 * flow-test record.
 */
/**
 * `programDetails.main.completion`'s field is `type: 'rich-textarea'` (see its own comment in
 * wwell-ir-form.schema.ts), NOT a single-pick select — `WwellRichTextareaComponent` lets any
 * number of `WWELL_COMPLETION_OPTIONS` be checked at once, and each checked option becomes its
 * own `\n`-joined line in the control's plain-text value (see that component's own doc comment
 * and `toggleOption`/`commitFromBox`). Checking both "Open Hole" and "Screened", for instance,
 * commits the control as `"Open Hole\nScreened"`, never the bare string `'Open Hole'` — so
 * `deriveCompletionElementFlags` below must check "is this option one of the checked lines?", not
 * "does the whole value equal this option?". A straight `completion === COMPLETION_OPEN_HOLE`
 * only ever matched the single-selection case; checking two or more options (or an option plus a
 * free-typed line) always fell through to every flag being 'N', which is exactly why the DB never
 * saw any flag flip to 'Y' once more than one box was checked.
 */
function selectedCompletionLabels(completion: string): ReadonlySet<string> {
  return new Set(
    completion
      .split('\n')
      .map(line => line.trim())
      .filter(line => line.length > 0),
  );
}

/**
 * The completion-element flag/remarks derivation itself — pulled out of buildWellDesignFromForm
 * so wwell-ir-payload.mapper.ts's buildWellDesignRow can compute the exact same
 * exad_gwd_well_design.{oh,gp,perforated,ls}_flg/remarks values for the actual save payload,
 * rather than these two call sites drifting apart. See IntegratedRequirementWellDesignFlags'
 * own doc comment for why the real table needs these written, not just this app's own
 * `completion` text.
 */
export function deriveCompletionElementFlags(
  completion: string,
  openHoleSizeIn: string | null,
  completionType = '',
  linerSizeIn: string | null = null,
  linerDepthFt: number | null = null,
): IntegratedRequirementWellDesignFlags {
  const typedFlags = flagsForCompletionType(completionType, openHoleSizeIn, linerSizeIn, { depthFt: linerDepthFt });
  if (typedFlags) return typedFlags;

  const selected = selectedCompletionLabels(completion);
  const isGravelPack = selected.has(COMPLETION_SCREENED_GRAVEL_PACK);
  const isScreened = selected.has(COMPLETION_SCREENED) || isGravelPack;
  const isLinerPerforated = selected.has(COMPLETION_LINER_PERFORATED);
  const isCasedPerforated = selected.has(COMPLETION_CASED_PERFORATED);
  const isPerforated = isLinerPerforated || isCasedPerforated;
  // Open hole is always drawn: a screen, gravel pack or liner completion sits inside it. Only a
  // cased & perforated completion has no open hole around it.
  const isOpenHole = selected.has(COMPLETION_OPEN_HOLE) || isScreened || isLinerPerforated;
  const ohSizeText = openHoleSizeIn?.trim() || null;

  return {
    ohFlg: isOpenHole ? 'Y' : 'N',
    // exad_gwd_well_design has no column of its own for the open hole size — oh_remarks is the
    // only place it can travel to the backend, so it's baked in as "<size> Open Hole" (matching
    // this app's other size-prefixed labels, e.g. perforatedRemarks' '7" Pre-Perforated Liner',
    // and completion.renderer.ts's own hardcoded '8 1/2" Open Hole' fallback). See
    // `parseOpenHoleSizeFromRemarks` for the reverse: pulling the size back out on load.
    ohRemarks: isOpenHole && ohSizeText ? `${ohSizeText} ${COMPLETION_OPEN_HOLE}` : null,
    gpFlg: isGravelPack ? 'Y' : 'N',
    gpRemarks: isGravelPack ? 'Gravel Pack' : null,
    perforatedFlg: isPerforated ? 'Y' : 'N',
    perforatedRemarks: isLinerPerforated ? '7" Pre-Perforated Liner' : isCasedPerforated ? 'Cased & Perforated' : null,
    lsFlg: isScreened ? 'Y' : 'N',
    lsRemarks: isScreened ? 'Screen' : null,
  };
}

const NO_COMPLETION_FLAGS: IntegratedRequirementWellDesignFlags = {
  ohFlg: 'N',
  ohRemarks: null,
  gpFlg: 'N',
  gpRemarks: null,
  perforatedFlg: 'N',
  perforatedRemarks: null,
  lsFlg: 'N',
  lsRemarks: null,
};

const PRE_PERFORATED_SUFFIX = WWELL_COMPLETION_TYPE.PRE_PERFORATED_LINER;
const LINER_SCREEN_LABEL = WWELL_COMPLETION_TYPE.SCREEN;

/** Remarks are compared case-insensitively, and the screen's old bare 'Screen' label equals
 *  'Liner Screen' — records saved before the Completion menu was renamed must still resolve to their pick. */
function canonicalRemark(remark: string | null): string | null {
  const text = remark?.trim().toLowerCase() ?? null;
  return text === 'screen' ? LINER_SCREEN_LABEL.toLowerCase() : text;
}

function sizedLabel(size: string | null, label: string): string {
  return size ? `${size} ${label}` : label;
}

/** Where the picked liner ends — optional: no depth means the target depth. (Its top always hangs from
 *  the last casing shoe, the renderer's own fallback.) */
interface LinerExtent {
  readonly depthFt: number | null;
}

const NO_LINER_EXTENT: LinerExtent = { depthFt: null };

/** `7" Liner Screen (depth 2500 ft)` — the depth rides in the remarks, like the size does,
 *  since exad_gwd_well_design has no column for it. Omitted when blank. */
function linerRemarks(size: string | null, label: string, extent: LinerExtent): string {
  const base = sizedLabel(size, label);
  return extent.depthFt != null ? `${base} (depth ${extent.depthFt} ft)` : base;
}

const LINER_EXTENT_SUFFIX = /\s*\(depth (\d+(?:\.\d+)?) ft\)\s*$/;

/** Reverses `linerRemarks`: the remarks without the extent suffix, plus the extent it carried. */
function splitLinerRemarks(remarks: string | null): { readonly base: string; readonly extent: LinerExtent } {
  const text = remarks ?? '';
  const match = LINER_EXTENT_SUFFIX.exec(text);
  if (!match) return { base: text, extent: NO_LINER_EXTENT };
  return {
    base: text.slice(0, match.index),
    extent: { depthFt: Number(match[1]) },
  };
}

/** Flags for the single-pick Completion dropdown, or `null` when nothing (or an unknown value) is
 *  picked — the caller then falls back to deriving from the legacy Completion Remarks text. Sizes
 *  ride in the remarks, same encoding as the open hole's (see `parseOpenHoleSizeFromRemarks`):
 *  exad_gwd_well_design has no column for either size. The perforated remarks double as the
 *  diagram's visible label (completion.renderer.ts prefers them over its generic text), which is
 *  how the entered liner size reaches the drawing. */
function flagsForCompletionType(
  completionType: string,
  openHoleSizeIn: string | null,
  linerSizeIn: string | null,
  extent: LinerExtent = NO_LINER_EXTENT,
): IntegratedRequirementWellDesignFlags | null {
  const ohSize = openHoleSizeIn?.trim() || null;
  const linerSize = linerSizeIn?.trim() || null;

  switch (completionType) {
    case WWELL_COMPLETION_TYPE.OPEN_HOLE:
      return { ...NO_COMPLETION_FLAGS, ohFlg: 'Y', ohRemarks: ohSize ? `${ohSize} ${COMPLETION_OPEN_HOLE}` : null };
    case WWELL_COMPLETION_TYPE.SCREEN:
      return { ...openHoleFlags(ohSize), lsFlg: 'Y', lsRemarks: linerRemarks(linerSize, LINER_SCREEN_LABEL, extent) };
    case WWELL_COMPLETION_TYPE.SCREEN_GRAVEL_PACK:
      return { ...openHoleFlags(ohSize), gpFlg: 'Y', gpRemarks: 'Gravel Pack', lsFlg: 'Y', lsRemarks: linerRemarks(linerSize, LINER_SCREEN_LABEL, extent) };
    case WWELL_COMPLETION_TYPE.PRE_PERFORATED_LINER:
      return { ...openHoleFlags(ohSize), perforatedFlg: 'Y', perforatedRemarks: linerRemarks(linerSize, PRE_PERFORATED_SUFFIX, extent) };
    default:
      return null;
  }
}

/** The open hole every completion pick is drawn inside — Screen, Screen with Gravel Pack and the
 *  liners all run in it, so their picks carry `ohFlg` too (the renderer keeps each of them inside
 *  the hole's width). */
function openHoleFlags(ohSize: string | null): IntegratedRequirementWellDesignFlags {
  return { ...NO_COMPLETION_FLAGS, ohFlg: 'Y', ohRemarks: ohSize ? `${ohSize} ${COMPLETION_OPEN_HOLE}` : null };
}

/** The same picks as saved before the open hole was implied (no `ohFlg` beside the completion
 *  element) — still recognised on load so an older record keeps its dropdown value. */
function legacyFlagsForCompletionType(
  completionType: string,
  linerSize: string | null,
): IntegratedRequirementWellDesignFlags | null {
  switch (completionType) {
    case WWELL_COMPLETION_TYPE.SCREEN:
      return { ...NO_COMPLETION_FLAGS, lsFlg: 'Y', lsRemarks: sizedLabel(linerSize, LINER_SCREEN_LABEL) };
    case WWELL_COMPLETION_TYPE.SCREEN_GRAVEL_PACK:
      return { ...NO_COMPLETION_FLAGS, gpFlg: 'Y', gpRemarks: 'Gravel Pack', lsFlg: 'Y', lsRemarks: sizedLabel(linerSize, LINER_SCREEN_LABEL) };
    case WWELL_COMPLETION_TYPE.PRE_PERFORATED_LINER:
      return { ...NO_COMPLETION_FLAGS, perforatedFlg: 'Y', perforatedRemarks: sizedLabel(linerSize, PRE_PERFORATED_SUFFIX) };
    default:
      return null;
  }
}

/** The liner size ridden in `lsRemarks` as `"<size> Liner Screen"` (a bare 'Screen' / 'Liner Screen' has none). */
function linerSizeFromScreenRemarks(base: string): string | null {
  if (!base || !base.toLowerCase().endsWith(LINER_SCREEN_LABEL.toLowerCase())) return null;
  return base.slice(0, -LINER_SCREEN_LABEL.length).trim() || null;
}

/** Header `cmpl_type` short codes (VARCHAR2(10)) that name a Completion pick outright. */
const COMPLETION_TYPE_BY_CODE: Readonly<Record<string, string>> = { PPL: WWELL_COMPLETION_TYPE.PRE_PERFORATED_LINER };

/** The Completion pick a header `cmpl_type` code stands for (`PPL` → Pre-perforated Liner), or `null` for anything else. */
export function completionTypeFromCode(code: string | null | undefined): string | null {
  return COMPLETION_TYPE_BY_CODE[code?.trim().toUpperCase() ?? ''] ?? null;
}

// Hand-written / legacy perforated remarks: "+/- 1570 ft of 7" Pre-Perforated Liner", "7" Pre Perforated Liner", "PPL 7in".
const PRE_PERFORATED_REMARK = /pre[\s-]*perforated|\bppl\b/i;
const LINER_SIZE_IN_REMARK = /(\d+(?:\s+\d+\/\d+|\.\d+|\/\d+)?)\s*(?:"|''|in\b)/i;

/** Reads a free-typed Pre-Perforated Liner remark: its size (`7"`), with the leading "+/- N ft of" length ignored.
 *  `null` when the text isn't a pre-perforated liner remark at all. */
function parsePrePerforatedRemark(remarks: string | null): { readonly linerSizeIn: string | null } | null {
  if (!remarks || !PRE_PERFORATED_REMARK.test(remarks)) return null;
  const withoutLength = remarks.replace(/^\s*(?:\+\/-|±)?\s*[\d,.]+\s*(?:ft|feet|')\s+of\s+/i, '');
  const size = LINER_SIZE_IN_REMARK.exec(withoutLength)?.[1]?.replace(/\s+/g, ' ').trim();
  return { linerSizeIn: size ? `${size}"` : null };
}

/**
 * Reverses `flagsForCompletionType` for a persisted exad_gwd_well_design row — the Completion pick
 * itself has no column (cmpl_type is VARCHAR2(10), too short), so it round-trips through the flags.
 * Returns `null` unless the flags are *exactly* what one dropdown pick would have produced: a
 * legacy record (several options checked, "Cased & Perforated", …) stays unresolved and keeps
 * deriving from its Completion Remarks text, instead of being silently collapsed onto one pick the
 * next time it is saved.
 */
export function resolveCompletionTypeFromFlags(
  flags: IntegratedRequirementWellDesignFlags | null | undefined,
): { readonly completionType: string; readonly linerSizeIn: string | null; readonly linerDepthFt: number | null } | null {
  if (!flags) return null;

  let completionType: string | null = null;
  let linerSizeIn: string | null = null;
  let extent = NO_LINER_EXTENT;
  if (flags.perforatedFlg === 'Y') {
    const split = splitLinerRemarks(flags.perforatedRemarks);
    const ownSize = split.base.toLowerCase().endsWith(PRE_PERFORATED_SUFFIX.toLowerCase())
      ? split.base.slice(0, -PRE_PERFORATED_SUFFIX.length).trim() || null
      : undefined;
    // A prefix like "+/- 1570 ft of 7\"" is someone else's wording, not our bare size — the lenient parse below handles it.
    if (ownSize !== undefined && !/\bft\b|\bof\b/i.test(ownSize ?? '')) {
      completionType = PRE_PERFORATED_SUFFIX;
      linerSizeIn = ownSize;
      extent = split.extent;
    } else {
      // Not our own encoding (e.g. "+/- 1570 ft of 7" Pre-Perforated Liner" from another tool): read the size out of it.
      const parsed = parsePrePerforatedRemark(flags.perforatedRemarks);
      if (parsed) {
        return { completionType: PRE_PERFORATED_SUFFIX, linerSizeIn: parsed.linerSizeIn, linerDepthFt: null };
      }
    }
  } else if (flags.gpFlg === 'Y') {
    completionType = WWELL_COMPLETION_TYPE.SCREEN_GRAVEL_PACK;
    const split = splitLinerRemarks(flags.lsRemarks);
    linerSizeIn = linerSizeFromScreenRemarks(split.base);
    extent = split.extent;
  } else if (flags.lsFlg === 'Y') {
    completionType = WWELL_COMPLETION_TYPE.SCREEN;
    const split = splitLinerRemarks(flags.lsRemarks);
    linerSizeIn = linerSizeFromScreenRemarks(split.base);
    extent = split.extent;
  } else if (flags.ohFlg === 'Y') {
    completionType = WWELL_COMPLETION_TYPE.OPEN_HOLE;
  }
  if (!completionType) return null;

  const candidates = [
    flagsForCompletionType(completionType, parseOpenHoleSizeFromRemarks(flags.ohRemarks), linerSizeIn, extent),
    legacyFlagsForCompletionType(completionType, linerSizeIn),
  ];
  const matches = candidates.some(
    candidate =>
      !!candidate &&
      (Object.keys(candidate) as (keyof IntegratedRequirementWellDesignFlags)[]).every(key =>
        key.endsWith('Remarks') && key !== 'ohRemarks' ? canonicalRemark(candidate[key]) === canonicalRemark(flags[key]) : candidate[key] === flags[key],
      ),
  );
  return matches ? { completionType, linerSizeIn, linerDepthFt: extent.depthFt } : null;
}

/** Reverses the `"<size> Open Hole"` encoding `deriveCompletionElementFlags` bakes into
 *  `ohRemarks` above — the only way the open hole size round-trips through a save/reload, since
 *  there's no dedicated backend column for it. Anything that doesn't end in the expected suffix
 *  (a legacy record saved before this encoding existed, or genuinely unrelated free text) is left
 *  blank rather than guessed at. */
export function parseOpenHoleSizeFromRemarks(ohRemarks: string | null | undefined): string | null {
  if (!ohRemarks) return null;
  const suffix = ` ${COMPLETION_OPEN_HOLE}`;
  if (!ohRemarks.endsWith(suffix)) return null;
  return ohRemarks.slice(0, -suffix.length).trim() || null;
}

/**
 * True once any completion element flag is 'Y' — the same "is a completion element actually
 * selected" test `deriveCompletionElementFlags`' four callers each need, whether the flags came
 * from a live Completion pick or a persisted `wellDesignFlags` row. Used by the emptiness gates
 * below so a record with only a well_design row (no Casing/Formation Top data at all) still shows
 * the diagram instead of the empty state — see `hasWellboreDiagramData` and
 * `hasWellboreDiagramDataForIntegratedRequirement`.
 */
function hasCompletionElementFlags(flags: IntegratedRequirementWellDesignFlags | null | undefined): boolean {
  return !!flags && (flags.ohFlg === 'Y' || flags.gpFlg === 'Y' || flags.perforatedFlg === 'Y' || flags.lsFlg === 'Y');
}

/**
 * `persistedFlags` — `ir.programDetails.wellDesignFlags`, the real exad_gwd_well_design row a GET
 * fetched back (see IntegratedRequirementWellDesignFlags) — wins over re-deriving from
 * `programMain.completion` whenever it's available: `completion` (exad_gwd_ir_header.completion,
 * VARCHAR2(4000) free text) and the well_design flags are two separately-persisted columns/tables
 * that this app always writes together (buildWellDesignRow), but nothing stops them disagreeing —
 * a record saved before this flags round-trip existed, a legacy/imported row, or any write to
 * exad_gwd_well_design from outside this app entirely. Trusting the actual fetched flags over a
 * re-derivation is what makes the diagram match what's really in the DB rather than silently
 * going blank whenever the two drift apart. Only the live-editing path (no fetched record to trust
 * yet, or the user is actively picking a new Completion — see buildWellboreDiagramDataFromIntegratedRequirement)
 * falls back to deriving from `programMain` itself.
 */
function buildWellDesignFromForm(
  programMain: WwellIrRawProgramMain,
  hydrology: WwellIrRawHydrology,
  persistedFlags?: IntegratedRequirementWellDesignFlags | null,
): IWellDesign {
  const flags =
    persistedFlags ??
    deriveCompletionElementFlags(programMain.completion, programMain.openHoleSizeIn || null, programMain.completionType, programMain.linerSizeIn || null, programMain.linerDepthFt);
  return {
    name: null,
    prewapSeq: null,
    ...buildWaterLevelFields(hydrology),
    // A flow test has no pump: the pump-depth field is hidden for it, but a value saved while the IR was a pump test must not still draw one.
    pumpLvl: hydrology.testType === FLOW_TEST_TYPE ? null : hydrology.estimatedPumpDepthFt,
    ...flags,
    // The pre-perforated liner's own label already says "(Pre Perf Liner)": the remark ("7" Pre-Perforated Liner",
    // "+/- 1570 ft of 7" Pre-Perforated Liner") would only repeat it as a second line under that label. Drawing-only —
    // the saved flags (derived separately for the payload) keep their remarks.
    perforatedRemarks: flags.perforatedRemarks && PRE_PERFORATED_REMARK.test(flags.perforatedRemarks) ? null : flags.perforatedRemarks,
  };
}

/** Shared tail for both entry points below — takes already-normalized raw rows so the actual
 *  gating/mapping (isRenderable*, mapCasings, buildLinerTopAnchors, mapGeologicTops,
 *  buildWellDesignFromForm) is written once regardless of whether the source was a live,
 *  still-being-edited form value or a submitted `IntegratedRequirement`. */
function assembleWellboreDiagramData(
  wellName: string,
  targetDepthFt: number,
  casingRows: readonly WwellIrRawCasingRow[],
  formationRows: readonly WwellIrRawFormationTop[],
  surfaceElevationFtAsl: number | null,
  programMain: WwellIrRawProgramMain,
  hydrology: WwellIrRawHydrology,
  persistedWellDesignFlags?: IntegratedRequirementWellDesignFlags | null,
): WellboreDiagramData {
  const renderableFormationTops = formationRows.filter(isRenderableFormationTop);
  /**
   * `completion.renderer.ts`'s `resolveCompletionBottomDepth` (rule kinds 'none' /
   * 'single-fallback' — i.e. no liner shoe to anchor the bottom to, which is exactly the "just a
   * Completion pick, no Casing rows yet" case this file's own `buildShoeAnchorFallback` exists
   * for) draws Open Hole / Gravel Pack / Liner Screen straight down to `data.totalDepth` — so
   * passing raw `targetDepthFt` through unchanged left the element's bottom pinned to its top
   * (ground level) whenever Target Depth reads as 0, making it technically present in the DOM but
   * zero-height/invisible. Resolving it once here (real value if positive, else
   * `STATIC_TARGET_DEPTH_FT`) and reusing that everywhere below — `totalDepth` and `targetDepth`
   * (and `displayDepth`, until a formation top goes past it) — keeps the whole diagram (scale,
   * ground line, completion elements, total-depth label) consistent instead of some parts anchoring
   * to a real value and others to a fallback.
   */
  const resolvedTargetDepthFt = targetDepthFt > 0 ? targetDepthFt : STATIC_TARGET_DEPTH_FT;
  const renderableCasings = withCompletionLiner(casingRows.filter(isRenderableCasingRow), programMain, resolvedTargetDepthFt);
  /**
   * Presentation draws its depth scale around whichever is deeper, the target depth or the deepest
   * formation top (`selectDisplayDepth` in well-data.selectors.ts) — this keeps that rule, minus its
   * padding: the scale is drawn to the target depth, which is what an IR is planned against, and is
   * only redrawn once a top is added (or edited) below it. Only tops count, as there; a casing row
   * deeper than the target still runs off the end of the scale.
   */
  const deepestTopFt = Math.max(0, ...renderableFormationTops.map(row => row.depthFt ?? 0));

  return {
    wellName,
    totalDepth: resolvedTargetDepthFt,
    // The shared diagram's own vertical domain is `max(totalDepth, displayDepth)`, so this moves
    // the diagram and (through `resolveScaleDepthFt`) the ruler and gridlines together.
    displayDepth: Math.max(resolvedTargetDepthFt, deepestTopFt),
    casings: mapCasings(renderableCasings),
    drlgCasings: [...buildLinerTopAnchors(renderableCasings), ...buildShoeAnchorFallback(renderableCasings)],
    geologicTops: mapGeologicTops(renderableFormationTops, surfaceElevationFtAsl),
    hydrogeology: null,
    prewap: null,
    rigActivity: null,
    currentDepth: 0,
    targetDepth: resolvedTargetDepthFt,
    mudCirculation: [],
    wellDesign: buildWellDesignFromForm(programMain, hydrology, persistedWellDesignFlags),
  };
}

/** Same emptiness gate as `hasWellboreDiagramData`, for the IR editor/detail page. `liveCasingRows`/
 *  `liveFormationRows` let the editor substitute its Casing/Formation Tops sections' in-progress
 *  FormArray values while those sections are open for editing — pass `null`/omit for either one
 *  still reads straight from `ir`. `liveProgramMain` does the same for Program Details: while that
 *  section is open for live edit, the gate reacts to the Completion the user is actively picking
 *  right now, same as `buildWellboreDiagramDataFromIntegratedRequirement` below; otherwise it
 *  trusts the persisted `ir.programDetails.wellDesignFlags` (falling back to re-deriving from
 *  `ir.programDetails.completion` for a record fetched before that round-trip existed) — so a
 *  record carrying only a well_design row (no Casing/Formation Top data at all) still shows the
 *  diagram instead of the empty state. */
export function hasWellboreDiagramDataForIntegratedRequirement(
  ir: IntegratedRequirement,
  liveCasingRows?: readonly WwellIrRawCasingRow[] | null,
  liveFormationRows?: readonly WwellIrRawFormationTop[] | null,
  liveProgramMain?: WwellIrRawProgramMain | null,
): boolean {
  const casingRows = liveCasingRows ?? toRawCasingRows(ir.casing);
  const formationRows = liveFormationRows ?? toRawFormationRows(ir.formationTops);
  const completionFlags = liveProgramMain
    ? deriveCompletionElementFlags(
        liveProgramMain.completion,
        liveProgramMain.openHoleSizeIn || null,
        liveProgramMain.completionType,
        liveProgramMain.linerSizeIn || null,
        liveProgramMain.linerDepthFt,
      )
    : (ir.programDetails.wellDesignFlags ??
      deriveCompletionElementFlags(
        ir.programDetails.completion,
        ir.programDetails.openHoleSizeIn,
        ir.programDetails.completionType,
        ir.programDetails.linerSizeIn,
        ir.programDetails.linerDepthFt,
      ));
  return (
    casingRows.some(isRenderableCasingRow) ||
    formationRows.some(isRenderableFormationTop) ||
    hasCompletionElementFlags(completionFlags)
  );
}

/**
 * Structural equality for `WellboreDiagramData`.
 *
 * The editor rebuilds the diagram's data on every keystroke in any section it live-edits, because the
 * builders below are pure functions returning a fresh object each call. Angular's signals compare with
 * `Object.is`, so a fresh-but-identical object still counts as a change — and `well-bore-view`'s redraw
 * effect then fades the whole drawing out over 120ms and re-animates it from scratch. Typing a client
 * name or a geologist, neither of which the diagram draws, restarted the animation.
 *
 * Handing this to `computed(..., { equal })` makes the signal keep the previous object whenever the
 * newly-built one is equivalent, so nothing downstream sees a change and the diagram holds still.
 *
 * Deliberately a whole-object comparison rather than a list of the fields that "matter": getting this
 * wrong in the conservative direction costs one redundant redraw, whereas a hand-maintained field list
 * silently stops updating the diagram the day someone adds a field to the model.
 */
export function wellboreDiagramDataEqual(
  a: WellboreDiagramData | null,
  b: WellboreDiagramData | null,
): boolean {
  return deepEquals(a, b);
}

/** Plain-data structural comparison. The diagram data is JSON-shaped — numbers, strings, nulls, arrays
 *  and object literals — so there are no Dates, Maps or cycles to account for here. */
function deepEquals(a: unknown, b: unknown): boolean {
  if (Object.is(a, b)) return true;
  if (a === null || b === null || typeof a !== 'object' || typeof b !== 'object') return false;

  if (Array.isArray(a) || Array.isArray(b)) {
    if (!Array.isArray(a) || !Array.isArray(b) || a.length !== b.length) return false;
    return a.every((item, i) => deepEquals(item, b[i]));
  }

  const aRecord = a as Record<string, unknown>;
  const bRecord = b as Record<string, unknown>;
  const aKeys = Object.keys(aRecord);
  if (aKeys.length !== Object.keys(bRecord).length) return false;
  // `in` rather than a truthiness check, so a key explicitly set to undefined on one side and absent
  // on the other is still treated as a difference.
  return aKeys.every(key => key in bRecord && deepEquals(aRecord[key], bRecord[key]));
}

/** Detail/editor-page counterpart of `buildWellboreDiagramData` — same live `app-well-bore-view`
 *  diagram, fed from a submitted `IntegratedRequirement` instead of the in-progress wizard form,
 *  so those pages show the real engineering rendering instead of the lighter
 *  `wwell-ir-design-preview` sketch. See `hasWellboreDiagramDataForIntegratedRequirement` above
 *  for what `liveCasingRows`/`liveFormationRows` are for — the editor passes its Casing/Formation
 *  Tops sections' live FormArray values while they're open, so the preview updates as rows are
 *  added/edited instead of only once that section is saved.
 *
 *  `liveProgramMain`/`liveHydrology` are the same idea for Program Details (Completion — the
 *  element flags in `buildWellDesignFromForm`) and Estimated Hydrology (static water level / SIWHP
 *  / pump depth): without these, picking a different Completion or test type while that section is
 *  open wouldn't move the diagram until the whole record was saved, unlike casing/formation-top
 *  edits, which already updated it live.
 *
 *  `liveReference` completes that set for the Request Reference section, which owns Target Depth,
 *  Surface Elevation and Well Name. Without it those three alone kept reading the last-saved record while
 *  every other section updated live, so editing Target Depth moved nothing until the whole IR was saved —
 *  and since Target Depth is what `totalDepth`, `displayDepth` and the depth scale are all derived from,
 *  that one gap froze the entire vertical extent of the drawing mid-edit.
 *
 *  While Program Details is open for live edit (`liveProgramMain` set), the diagram must react to
 *  the Completion the user is actively picking right now — so this deliberately does NOT pass
 *  `ir.programDetails.wellDesignFlags` (the last-fetched, possibly-stale real DB flags) through in
 *  that case, only once the section is closed/not being edited. See buildWellDesignFromForm's own
 *  doc comment for why the persisted flags win over re-derivation the rest of the time. */
export function buildWellboreDiagramDataFromIntegratedRequirement(
  ir: IntegratedRequirement,
  liveCasingRows?: readonly WwellIrRawCasingRow[] | null,
  liveFormationRows?: readonly WwellIrRawFormationTop[] | null,
  liveProgramMain?: WwellIrRawProgramMain | null,
  liveHydrology?: WwellIrRawHydrology | null,
  liveReference?: WwellIrRawReference | null,
): WellboreDiagramData {
  return assembleWellboreDiagramData(
    liveReference?.wellName ?? ir.requestReference.wellName,
    liveReference ? (liveReference.targetDepthFt ?? 0) : ir.requestReference.targetDepthFt,
    liveCasingRows ?? toRawCasingRows(ir.casing),
    liveFormationRows ?? toRawFormationRows(ir.formationTops),
    liveReference ? liveReference.surfaceElevationFtAsl : ir.requestReference.surfaceElevationFtAsl,
    liveProgramMain ?? toRawProgramMain(ir),
    liveHydrology ?? toRawHydrology(ir),
    liveProgramMain ? null : ir.programDetails.wellDesignFlags,
  );
}

const INCH_MARKED = /["″”]/;

/** "24" → 24", "9 5/8" → 9 5/8"; a size that already carries an inch mark is left alone, blank stays blank. */
export function formatInchSize(size: string | null | undefined): string {
  const trimmed = size?.trim() ?? '';
  if (!trimmed) return '';
  return INCH_MARKED.test(trimmed) ? trimmed : `${trimmed}"`;
}

const FRACTION_INCH_MARK = /(\d+(?:[-\s]\d+\/\d+|\/\d+))\s?;(?!\d)/g;
const WHOLE_INCH_MARK = /(?<![\d./-])(\d+(?:\.\d+)?);(?=\s+(?:pre-?perforated|perforated|slotted|screens?|screened|liner|casing|csg|tubing|pipe|hole|bit|shoe|blank|conductor|surface|production|open)\b)/gi;

/** Legacy remarks store an inch mark as ";" ("7; liner", "9-5/8; liner shoe"): a fraction size before ";" is always an inch size, a
 *  whole/decimal one only when a casing/hole word follows, so "3000 ft; 4000 ft" is left alone. Same rule as the letter PDF (LegacyIrTextUtil). */
export function normalizeInchMarks(text: string | null | undefined): string {
  return (text ?? '').replace(FRACTION_INCH_MARK, '$1"').replace(WHOLE_INCH_MARK, '$1"');
}
