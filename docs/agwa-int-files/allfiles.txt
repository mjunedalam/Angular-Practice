private EmailApiPayload buildPlainPayload(EmailRequest request, java.util.Map<String, Object> data) {
    var payload = new EmailApiPayload();
    payload.setFromAddress(request.getFrom());
    payload.setTo(request.getTo());
    payload.setCc(request.getCc() != null ? request.getCc() : List.of());
    payload.setSubject(request.getSubject());
    payload.setContent((String) data.get("content"));
    var assets = new java.util.ArrayList<InlineAssets>();
    var attachments = (List<java.util.Map<String, String>>) data.getOrDefault("attachments", List.of());
    for (int i = 0; i < attachments.size(); i++) {
        var a = attachments.get(i);
        assets.add(createAsset("attachment" + i, a.get("filename"), a.get("contentType"), a.get("b64")));
    }
    payload.setInlineAssets(assets);
    return payload;
}

import com.fasterxml.jackson.annotation.JsonCreator;
import com.fasterxml.jackson.annotation.JsonProperty;

@RegisterForReflection
public class EmailRequest {
    private final String subject;
    private final String from;
    private final List<String> to;
    private final List<String> cc;
    private final List<String> bcc;
    private final String replyTo;
    private final Map<String, Object> templateData;
    private final boolean generatePdf;
    private final String templateName;

    // Private constructor used internally
    private EmailRequest(Builder builder) {
        this.subject = builder.subject;
        this.from = builder.from;
        this.to = builder.to;
        this.cc = builder.cc;
        this.bcc = builder.bcc;
        this.replyTo = builder.replyTo;
        this.templateData = builder.templateData;
        this.generatePdf = builder.generatePdf;
        this.templateName = builder.templateName;
    }

    // Jackson needs a way to create the object — this is it
    @JsonCreator
    public static EmailRequest fromJson(
            @JsonProperty("subject") String subject,
            @JsonProperty("from") String from,
            @JsonProperty("to") List<String> to,
            @JsonProperty("cc") List<String> cc,
            @JsonProperty("bcc") List<String> bcc,
            @JsonProperty("replyTo") String replyTo,
            @JsonProperty("templateData") Map<String, Object> templateData,
            @JsonProperty("generatePdf") boolean generatePdf,
            @JsonProperty("templateName") String templateName) {

        Builder builder = new Builder()
                .subject(subject)
                .from(from)
                .to(to)
                .cc(cc)
                .bcc(bcc)
                .replyTo(replyTo)
                .templateData(templateData)
                .generatePdf(generatePdf)
                .templateName(templateName);

        return builder.build();
    }

    public String getSubject() { return subject; }
    public String getFrom() { return from; }
    public List<String> getTo() { return to; }
    public List<String> getCc() { return cc; }
    public List<String> getBcc() { return bcc; }
    public String getReplyTo() { return replyTo; }
    public Map<String, Object> getTemplateData() { return templateData; }
    public boolean isGeneratePdf() { return generatePdf; }
    public String getTemplateName() { return templateName; }

    public static class Builder {
        private String subject;
        private String from;
        private List<String> to;
        private List<String> cc;
        private List<String> bcc;
        private String replyTo;
        private Map<String, Object> templateData;
        private boolean generatePdf;
        private String templateName;

        public Builder subject(String subject) {

        }

        public Builder from(String from) {
            this.from = from;
            return this;
        }

        public Builder to(List<String> to) {
            this.to = to;
            return this;
        }

        public Builder cc(List<String> cc) {
            this.cc = cc;
            return this;

            this.bcc = bcc;

package com.aramco.easd.agwa.integrations.service;

import io.quarkus.runtime.annotations.RegisterForReflection;
import jakarta.enterprise.context.ApplicationScoped;
import jakarta.inject.Inject;
import jakarta.inject.Named;
import org.apache.camel.Exchange;

import java.util.HashMap;
import java.util.List;

@ApplicationScoped
@RegisterForReflection
@Named("emailPayloadBuilder")
public class EmailPayloadBuilder {

    @Inject
    MorningReportTemplate morningReportTemplate;

    @Inject
    CurrentDateProvider currentDateProvider;

    @Inject
    EmailTemplateGlobals emailTemplateGlobals;

    public EmailApiPayload buildPayload(Exchange exchange) throws Exception {
        EmailRequest request = exchange.getIn().getBody(EmailRequest.class);
        var tmplData = new HashMap<>(request.getTemplateData());
        tmplData.put("currentDateProvider", currentDateProvider);

        var payload = new EmailApiPayload();
        var assetsList = List.of(
                new InlineAssets(
                        AssetType.HEADER.getId(),
                        AssetType.HEADER.getFilename(),
                        AssetType.HEADER.getMimeType(),
                        emailTemplateGlobals.getHeaderImageData()
                ),
                new InlineAssets(
                        AssetType.FOOTER.getId(),
                        AssetType.FOOTER.getFilename(),
                        AssetType.FOOTER.getMimeType(),
                        emailTemplateGlobals.getFooterImageData()
                ),
                new InlineAssets(
                        AssetType.MAP.getId(),
                        AssetType.MAP.getFilename(),
                        AssetType.MAP.getMimeType(),
                        (String) request.getTemplateData().get("mapImageData")
                )
        );
        var htmlContent = morningReportTemplate.render(tmplData).render();
        payload.setFromAddress(request.getFrom());
        payload.setTo(request.getTo());
        payload.setCc(request.getCc() != null ? request.getCc() : List.of());
        payload.setSubject(request.getSubject());
        payload.setContent(htmlContent);
        payload.setInlineAssets(assetsList);

        return payload;
    }
}
package com.aramco.easd.agwa.integrations.service;

import com.aramco.easd.agwa.integrations.dto.EmailRequest;
import com.aramco.easd.agwa.integrations.template.MorningReportTemplate;
import com.aramco.easd.agwa.integrations.utill.EmailTemplateGlobals;
import io.quarkus.qute.TemplateInstance;
import jakarta.enterprise.context.ApplicationScoped;
import jakarta.inject.Inject;
import org.apache.camel.Exchange;
import org.apache.camel.Message;
import org.apache.hc.client5.http.entity.mime.MultipartEntityBuilder;

@ApplicationScoped
public class EmailService {

    @Inject
    MorningReportTemplate morningReportTemplate;

    @Inject
    EmailTemplateGlobals emailTemplateGlobals;

    public void prepareEmail(Exchange exchange) throws Exception {
        EmailRequest request = exchange.getIn().getBody(EmailRequest.class);
        var tmplData = request.getTemplateData();
        tmplData.put("headerImageData", emailTemplateGlobals.getHeaderImageData());
        tmplData.put("footerImageData", emailTemplateGlobals.getFooterImageData());
        TemplateInstance template = morningReportTemplate.render(tmplData);
        String htmlContent = template.render();

        Message message = exchange.getIn();
        message.setBody(htmlContent);
        message.setHeader("Subject", request.getSubject());
        message.setHeader("From", request.getFrom());
        message.setHeader("To", String.join(",", request.getTo()));
        if (request.getCc() != null && !request.getCc().isEmpty()) {
            message.setHeader("Cc", String.join(",", request.getCc()));
        }
    }
}

            return footerImageData;
        }

        private String loadImageAsBase64(String resourceName) {
            try (var is = Thread.currentThread().getContextClassLoader().getResourceAsStream(resourceName)) {
                if (is == null) {
                    LOG.warnf("Email image not found in classpath: %s", resourceName);
                    return "";
                }
                return Base64.getEncoder().encodeToString(is.readAllBytes());
            } catch (Exception e) {
                LOG.errorf("Failed to load email image: %s — %s", resourceName, e.getMessage());
                return "";
            }
        }
